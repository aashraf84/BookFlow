   130→   130→export default function BookCatalogModule() {
   131→   131→  const [books, setBooks] = useState<Book[]>([])
   132→   132→  const [totalPages, setTotalPages] = useState(1)
   133→   133→  const [total, setTotal] = useState(0)
   134→   134→  const [page, setPage] = useState(1)
   135→   135→  const [loading, setLoading] = useState(true)
   136→   136→  const [search, setSearch] = useState('')
   137→   137→  const [filterCategory, setFilterCategory] = useState('')
   138→   138→  const [filterAuthor, setFilterAuthor] = useState('')
   139→   139→  const [filterPublisher, setFilterPublisher] = useState('')
   140→   140→  const [dialogOpen, setDialogOpen] = useState(false)
   141→   141→  const [editing, setEditing] = useState<Book | null>(null)
   142→   142→  const [form, setForm] = useState<BookFormData>(emptyForm)
   143→   143→  const [saving, setSaving] = useState(false)
   144→   144→  const [deleting, setDeleting] = useState<string | null>(null)
   145→   145→
   146→   146→  // Dropdown data
   147→   147→  const [authors, setAuthors] = useState<Author[]>([])
   148→   148→  const [publishers, setPublishers] = useState<Publisher[]>([])
   149→   149→  const [categories, setCategories] = useState<Category[]>([])
   150→   150→
   151→   151→  const limit = 10
   152→   152→
   153→   153→  // ── Fetch dropdown data ────────────────────────────────────
   154→   154→  useEffect(() => {
   155→   155→    Promise.all([
   156→   156→      fetch('/api/authors?limit=200').then((r) => r.json()),
   157→   157→      fetch('/api/publishers?limit=200').then((r) => r.json()),
   158→   158→      fetch('/api/categories').then((r) => r.json()),

   160→   160→      setAuthors(authorsJson.data || [])
   161→   161→      setPublishers(publishersJson.data || [])
   162→   162→      setCategories(categoriesJson.data || [])
   163→   163→    })
   164→   164→  }, [])
   165→   165→
   166→   166→  // ── Fetch books ────────────────────────────────────────────
   167→   167→  const fetchBooks = useCallback(async () => {
   168→   168→    setLoading(true)
   169→   169→    try {
   170→   170→      const params = new URLSearchParams({
   171→   171→        page: String(page),
   172→   172→        limit: String(limit),
   173→   173→      })
   174→   174→      if (search) params.set('search', search)
   175→   175→      if (filterCategory && filterCategory !== 'all') params.set('categoryId', filterCategory)
   176→   176→      if (filterAuthor && filterAuthor !== 'all') params.set('authorId', filterAuthor)
   177→   177→      if (filterPublisher && filterPublisher !== 'all') params.set('publisherId', filterPublisher)
   178→   178→
   179→   179→      const res = await fetch(`/api/books?${params}`)
   180→   180→      if (!res.ok) throw new Error('Failed to fetch books')
   181→   181→      const json = await res.json()
   182→   182→      setBooks(json.data || [])
   183→   183→      setTotalPages(json.pagination?.totalPages || 1)
   184→   184→      setTotal(json.pagination?.total || 0)
   185→   185→    } catch {
   186→   186→      setBooks([])

   188→   188→      setLoading(false)
   189→   189→    }
   190→   190→  }, [page, search, filterCategory, filterAuthor, filterPublisher])
   191→   191→
   192→   192→  useEffect(() => {
   193→   193→    fetchBooks()

   195→   195→
   196→   196→  // Reset to page 1 when filters change
   197→   197→  useEffect(() => {

   199→   199→  }, [search, filterCategory, filterAuthor, filterPublisher])
   200→   200→
   201→   201→  // ── Save book (create or update) ───────────────────────────

   203→   203→    if (!form.title.trim()) return
   204→   204→    setSaving(true)
   205→   205→    try {
   206→   206→      const payload: Record<string, unknown> = {
   207→   207→        title: form.title.trim(),
   208→   208→        subtitle: form.subtitle || null,
   209→   209→        isbn: form.isbn || null,
   210→   210→        authorId: form.authorId || null,
   211→   211→        publisherId: form.publisherId || null,
   212→   212→        categoryId: form.categoryId || null,
   213→   213→        edition: form.edition || null,
   214→   214→        language: form.language || 'English',
   215→   215→        pages: form.pages ? parseInt(form.pages) : null,
   216→   216→        publishYear: form.publishYear ? parseInt(form.publishYear) : null,
   217→   217→        description: form.description || null,
   218→   218→        price: form.price ? parseFloat(form.price) : 0,
   219→   219→        costPrice: form.costPrice ? parseFloat(form.costPrice) : 0,
   220→   220→        discount: form.discount ? parseFloat(form.discount) : 0,
   221→   221→        academicLevel: form.academicLevel || null,
   222→   222→        subject: form.subject || null,
   223→   223→        barcode: form.barcode || null,
   224→   224→      }
   225→   225→
   226→   226→      let res: Response
   227→   227→      if (editing) {
   228→   228→        payload.id = editing.id
   229→   229→        res = await fetch('/api/books', {
   230→   230→          method: 'PUT',
   231→   231→          headers: { 'Content-Type': 'application/json' },
   232→   232→          body: JSON.stringify(payload),
   233→   233→        })
   234→   234→      } else {
   235→   235→        res = await fetch('/api/books', {
   236→   236→          method: 'POST',
   237→   237→          headers: { 'Content-Type': 'application/json' },
   238→   238→          body: JSON.stringify(payload),
   239→   239→        })
   240→   240→      }
   241→   241→
   242→   242→      if (!res.ok) {
   243→   243→        const err = await res.json()
   244→   244→        throw new Error(err.error || 'Failed to save book')
   245→   245→      }
   246→   246→
   247→   247→      setDialogOpen(false)
   248→   248→      setEditing(null)
   249→   249→      setForm(emptyForm)
   250→   250→      fetchBooks()
   251→   251→    } catch (err) {
   252→   252→      alert(err instanceof Error ? err.message : 'Failed to save book')
   253→   253→    } finally {
   254→   254→      setSaving(false)
   255→   255→    }
   256→   256→  }
   257→   257→
   258→   258→  // ── Delete book ────────────────────────────────────────────

   263→   263→      const res = await fetch(`/api/books?id=${id}`, { method: 'DELETE' })

   265→   265→        const err = await res.json()
   266→   266→        throw new Error(err.error || 'Failed to delete')
   267→   267→      }
   268→   268→      fetchBooks()
   269→   269→    } catch (err) {
   270→   270→      alert(err instanceof Error ? err.message : 'Failed to delete book')
   271→   271→    } finally {
   272→   272→      setDeleting(null)
   273→   273→    }
   274→   274→  }
   275→   275→
   276→   276→  // ── Open dialog for edit ───────────────────────────────────

   280→   280→      title: book.title,
   281→   281→      subtitle: book.subtitle || '',

   283→   283→      authorId: book.authorId || '',
   284→   284→      publisherId: book.publisherId || '',
   285→   285→      categoryId: book.categoryId || '',
   286→   286→      edition: book.edition || '',
   287→   287→      language: book.language || 'English',
   288→   288→      pages: book.pages?.toString() || '',
   289→   289→      publishYear: book.publishYear?.toString() || '',
   290→   290→      description: book.description || '',
   291→   291→      price: book.price?.toString() || '',
   292→   292→      costPrice: book.costPrice?.toString() || '',
   293→   293→      discount: book.discount?.toString() || '',
   294→   294→      academicLevel: book.academicLevel || '',
   295→   295→      subject: book.subject || '',
   296→   296→      barcode: book.barcode || '',
   297→   297→    })
   298→   298→    setDialogOpen(true)
   299→   299→  }
   300→   300→
   301→   301→  // ── Open dialog for add ────────────────────────────────────
   302→   302→  const openAdd = () => {

   304→   304→    setForm(emptyForm)
   305→   305→    setDialogOpen(true)
   306→   306→  }
   307→   307→
   308→   308→  const formatCurrency = (value: number) =>
   309→   309→    new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(value)
   310→   310→
   311→   311→  // ── Helpers for inventory stock display ──────────────────────
   312→   312→  // We fetch stock from inventory for each book or show N/A
   313→   313→  // For now, we'll just show the price and status
   314→   314→
   315→   315→  return (
   316→   316→    <div className="space-y-4">
   317→   317→      {/* ── Header + Add Button ─────────────────────────────── */}
   318→   318→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   319→   319→        <div>
   320→   320→          <h2 className="text-xl font-bold tracking-tight">Book Catalog</h2>

   322→   322→        </div>
   323→   323→        <Button onClick={openAdd} className="bg-emerald-600 hover:bg-emerald-700">
   324→   324→          <Plus className="size-4 mr-2" /> Add Book
   325→   325→        </Button>
   326→   326→      </div>
   327→   327→
   328→   328→      {/* ── Search & Filters ────────────────────────────────── */}
   329→   329→      <Card>
   330→   330→        <CardContent className="p-4">
   331→   331→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
   332→   332→            <div className="relative">
   333→   333→              <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   334→   334→              <Input
   335→   335→                placeholder="Search by title, ISBN, author..."
   336→   336→                value={search}
   337→   337→                onChange={(e) => setSearch(e.target.value)}
   338→   338→                className="pl-8"
   339→   339→              />
   340→   340→            </div>
   341→   341→            <Select value={filterCategory} onValueChange={setFilterCategory}>
   342→   342→              <SelectTrigger>
   343→   343→                <SelectValue placeholder="All Categories" />
   344→   344→              </SelectTrigger>
   345→   345→              <SelectContent>
   346→   346→                <SelectItem value="all">All Categories</SelectItem>
   347→   347→                {categories.map((cat) => (
   348→   348→                  <SelectItem key={cat.id} value={cat.id}>
   349→   349→                    {cat.name}
   350→   350→                  </SelectItem>
   351→   351→                ))}
   352→   352→              </SelectContent>
   353→   353→            </Select>
   354→   354→            <Select value={filterAuthor} onValueChange={setFilterAuthor}>
   355→   355→              <SelectTrigger>
   356→   356→                <SelectValue placeholder="All Authors" />
   357→   357→              </SelectTrigger>
   358→   358→              <SelectContent>
   359→   359→                <SelectItem value="all">All Authors</SelectItem>
   360→   360→                {authors.map((a) => (
   361→   361→                  <SelectItem key={a.id} value={a.id}>
   362→   362→                    {a.name}
   363→   363→                  </SelectItem>

   365→   365→              </SelectContent>
   366→   366→            </Select>
   367→   367→            <Select value={filterPublisher} onValueChange={setFilterPublisher}>

   381→   381→          {(search || (filterCategory && filterCategory !== 'all') || (filterAuthor && filterAuthor !== 'all') || (filterPublisher && filterPublisher !== 'all')) && (
   382→   382→            <div className="mt-2 flex items-center gap-2">
   383→   383→              <Button
   384→   384→                variant="ghost"
   385→   385→                size="sm"
   386→   386→                onClick={() => {
   387→   387→                  setSearch('')
   388→   388→                  setFilterCategory('')
   389→   389→                  setFilterAuthor('')
   390→   390→                  setFilterPublisher('')
   391→   391→                }}
   392→   392→              >
   393→   393→                <X className="size-3 mr-1" /> Clear filters

   406→   406→                <Skeleton key={i} className="h-12 w-full" />
   407→   407→              ))}
   408→   408→            </div>
   409→   409→          ) : books.length === 0 ? (
   410→   410→            <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
