   114→// ─── Component ────────────────────────────────────────────────────
   115→export function WholesaleModule() {
   116→  const [activeTab, setActiveTab] = React.useState('orders')
   117→  const [orders, setOrders] = React.useState<WholesaleOrder[]>([])
   118→  const [dealers, setDealers] = React.useState<Dealer[]>([])
   119→  const [loading, setLoading] = React.useState(true)
   120→  const [error, setError] = React.useState<string | null>(null)
   121→
   122→  // Status filter
   123→  const [statusFilter, setStatusFilter] = React.useState('all')
   124→
   125→  // New order dialog
   126→  const [orderDialogOpen, setOrderDialogOpen] = React.useState(false)
   127→  const [saving, setSaving] = React.useState(false)
   128→  const [selectedDealerId, setSelectedDealerId] = React.useState('')
   129→  const [dealerName, setDealerName] = React.useState('')
   130→  const [orderDiscount, setOrderDiscount] = React.useState('')
   131→  const [orderTax, setOrderTax] = React.useState('')
   132→  const [deliveryAddress, setDeliveryAddress] = React.useState('')
   133→  const [orderNotes, setOrderNotes] = React.useState('')
   134→  const [orderItems, setOrderItems] = React.useState<OrderFormItem[]>([])
   135→
   136→  // Product search for adding items
   137→  const [productSearch, setProductSearch] = React.useState('')
   138→  const [searchResults, setSearchResults] = React.useState<Array<{ id: string; type: 'book' | 'stationery'; name: string; price: number }>>([])
   139→  const [searchLoading, setSearchLoading] = React.useState(false)
   140→
   141→  // Order detail dialog
   142→  const [detailOrder, setDetailOrder] = React.useState<WholesaleOrder | null>(null)
   143→  const [detailDialogOpen, setDetailDialogOpen] = React.useState(false)
   144→
   145→  // Updating status
   146→  const [updatingStatus, setUpdatingStatus] = React.useState<string | null>(null)
   147→
   148→  // Fetch orders
   149→  const fetchOrders = React.useCallback(async () => {
   150→    setLoading(true)
   151→    setError(null)
   152→    try {
   153→      const params = new URLSearchParams()
   154→      params.set('limit', '50')
   155→      if (statusFilter && statusFilter !== 'all') params.set('status', statusFilter)
   156→      const res = await fetch(`/api/wholesale?${params.toString()}`)
   157→      if (!res.ok) throw new Error('Failed to fetch orders')
   158→      const json = await res.json()
   159→      setOrders(json.data || [])
   160→    } catch (err) {
   161→      setError(err instanceof Error ? err.message : 'Unknown error')
   162→    } finally {
   163→      setLoading(false)
   164→    }
   165→  }, [statusFilter])
   166→
   167→  // Fetch dealers (customers of type wholesale/dealer/distributor)
   168→  const fetchDealers = React.useCallback(async () => {
   169→    try {
   170→      const res = await fetch('/api/customers?type=wholesale&limit=100')
   171→      const res2 = await fetch('/api/customers?type=dealer&limit=100')
   172→      const json1 = res.ok ? await res.json() : { data: [] }
   173→      const json2 = res2.ok ? await res2.json() : { data: [] }
   174→      // Also get distributors
   175→      const res3 = await fetch('/api/customers?type=distributor&limit=100')
   176→      const json3 = res3.ok ? await res3.json() : { data: [] }
   177→      setDealers([...(json1.data || []), ...(json2.data || []), ...(json3.data || [])])
   178→    } catch {
   179→      // silent
   180→    }
   181→  }, [])
   182→
   183→  React.useEffect(() => {
   184→    fetchOrders()
   185→  }, [fetchOrders])
   186→
   187→  React.useEffect(() => {
   188→    fetchDealers()
   189→  }, [fetchDealers])
   190→
   191→  // Product search for order items
   192→  React.useEffect(() => {
   193→    if (!productSearch) {
   194→      setSearchResults([])
   195→      return
   196→    }
   197→    const searchProducts = async () => {
   198→      setSearchLoading(true)
   199→      try {
   200→        const params = new URLSearchParams()
   201→        params.set('search', productSearch)
   202→        params.set('limit', '10')
   203→
   204→        const [booksRes, stationeryRes] = await Promise.all([
   205→          fetch(`/api/books?${params.toString()}`),
   206→          fetch(`/api/stationery?${params.toString()}`),
   207→        ])
   208→
   209→        const bookData = booksRes.ok ? await booksRes.json() : { data: [] }
   210→        const stationeryData = stationeryRes.ok ? await stationeryRes.json() : { data: [] }
   211→
   212→        const results: Array<{ id: string; type: 'book' | 'stationery'; name: string; price: number }> = [
   213→          ...(bookData.data || []).map((b: Record<string, unknown>) => ({
   214→            id: b.id as string,
   215→            type: 'book' as const,
   216→            name: b.title as string,
   217→            price: (b.price as number) || 0,
   218→          })),
   219→          ...(stationeryData.data || []).map((s: Record<string, unknown>) => ({
   220→            id: s.id as string,
   221→            type: 'stationery' as const,
   222→            name: s.name as string,
   223→            price: (s.price as number) || 0,
   224→          })),
   225→        ]
   226→
   227→        setSearchResults(results)
   228→      } catch {
   229→        // silent
   230→      } finally {
   231→        setSearchLoading(false)
   232→      }
   233→    }
   234→    const debounce = setTimeout(searchProducts, 300)
   235→    return () => clearTimeout(debounce)
   236→  }, [productSearch])
   237→
   238→  // Add product to order items
   239→  const addItemToOrder = (product: { id: string; type: 'book' | 'stationery'; name: string; price: number }) => {
   240→    const existing = orderItems.find((i) => i.productId === product.id && i.productType === product.type)
   241→    if (existing) {
   242→      setOrderItems((prev) =>
   243→        prev.map((i) =>
   244→          i.productId === product.id && i.productType === product.type
   245→            ? { ...i, quantity: i.quantity + 1 }
   246→            : i
   247→        )
   248→      )
   249→    } else {
   250→      setOrderItems((prev) => [
   251→        ...prev,
   252→        {
   253→          productType: product.type,
   254→          productId: product.id,
   255→          productName: product.name,
   256→          quantity: 1,
   257→          unitPrice: product.price,
   258→          discount: 0,
   259→        },
   260→      ])
   261→    }
   262→    setProductSearch('')
   263→    setSearchResults([])
   264→  }
   265→
   266→  // Remove order item
   267→  const removeOrderItem = (index: number) => {
   268→    setOrderItems((prev) => prev.filter((_, i) => i !== index))
   269→  }
   270→
   271→  // Update order item
   272→  const updateOrderItem = (index: number, field: keyof OrderFormItem, value: number | string) => {
   273→    setOrderItems((prev) =>
   274→      prev.map((item, i) => (i === index ? { ...item, [field]: value } : item))
   275→    )
   276→  }
   277→
   278→  // Status badges
   279→  const getStatusBadge = (status: string) => {
   280→    const map: Record<string, { className: string; label: string }> = {
   281→      pending: { className: 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400 border-amber-200 dark:border-amber-800', label: 'Pending' },
   282→      confirmed: { className: 'bg-sky-100 text-sky-700 dark:bg-sky-950/40 dark:text-sky-400 border-sky-200 dark:border-sky-800', label: 'Confirmed' },
   283→      shipped: { className: 'bg-teal-100 text-teal-700 dark:bg-teal-950/40 dark:text-teal-400 border-teal-200 dark:border-teal-800', label: 'Shipped' },
   284→      delivered: { className: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800', label: 'Delivered' },
   285→      cancelled: { className: 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400 border-red-200 dark:border-red-800', label: 'Cancelled' },
   286→    }
   287→    const s = map[status] || { className: '', label: status }
   288→    return <Badge className={`text-xs ${s.className}`}>{s.label}</Badge>
   289→  }
   290→
   291→  const getPaymentBadge = (status: string) => {
   292→    const map: Record<string, { className: string; label: string }> = {
   293→      unpaid: { className: 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400 border-red-200 dark:border-red-800', label: 'Unpaid' },
   294→      partial: { className: 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400 border-amber-200 dark:border-amber-800', label: 'Partial' },
   295→      paid: { className: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800', label: 'Paid' },
   296→    }
   297→    const s = map[status] || { className: '', label: status }
   298→    return <Badge className={`text-xs ${s.className}`}>{s.label}</Badge>
   299→  }
   300→
   301→  // Compute order total
   302→  const orderSubtotal = orderItems.reduce((s, i) => s + i.quantity * i.unitPrice * (1 - i.discount / 100), 0)
   303→  const orderDiscountAmt = parseFloat(orderDiscount) || 0
   304→  const orderTaxAmt = (orderSubtotal - orderDiscountAmt) * ((parseFloat(orderTax) || 0) / 100)
   305→  const orderTotal = orderSubtotal - orderDiscountAmt + orderTaxAmt
   306→
   307→  // Create wholesale order
   308→  const handleCreateOrder = async () => {
   309→    if (orderItems.length === 0) return
   310→    setSaving(true)
   311→    setError(null)
   312→    try {
   313→      const items = orderItems.map((item) => ({
   314→        bookId: item.productType === 'book' ? item.productId : undefined,
   315→        stationeryId: item.productType === 'stationery' ? item.productId : undefined,
   316→        quantity: item.quantity,
   317→        unitPrice: item.unitPrice,
   318→        discount: item.discount,
   319→      }))
   320→
   321→      const res = await fetch('/api/wholesale', {
   322→        method: 'POST',
   323→        headers: { 'Content-Type': 'application/json' },
   324→        body: JSON.stringify({
   325→          customerId: selectedDealerId || undefined,
   326→          dealerName: dealerName || undefined,
   327→          discount: orderDiscountAmt,
   328→          tax: parseFloat(orderTax) || 0,
   329→          paymentStatus: 'unpaid',
   330→          deliveryAddress: deliveryAddress || undefined,
   331→          notes: orderNotes || undefined,
   332→          items,
   333→        }),
   334→      })
   335→
   336→      if (!res.ok) {
   337→        const errData = await res.json()
   338→        throw new Error(errData.error || 'Failed to create order')
   339→      }
   340→
   341→      // Reset form
   342→      setOrderDialogOpen(false)
   343→      setSelectedDealerId('')
   344→      setDealerName('')
   345→      setOrderDiscount('')
   346→      setOrderTax('')
   347→      setDeliveryAddress('')
   348→      setOrderNotes('')
   349→      setOrderItems([])
   350→      fetchOrders()
   351→    } catch (err) {
   352→      setError(err instanceof Error ? err.message : 'Failed to create order')
   353→    } finally {
   354→      setSaving(false)
   355→    }
   356→  }
   357→
   358→  // Update order status
   359→  const handleStatusUpdate = async (orderId: string, newStatus: string) => {
   360→    setUpdatingStatus(orderId)
   361→    try {
   362→      const res = await fetch('/api/wholesale', {
   363→        method: 'PUT',
   364→        headers: { 'Content-Type': 'application/json' },
   365→        body: JSON.stringify({ id: orderId, status: newStatus }),
   366→      })
   367→      if (!res.ok) throw new Error('Failed to update status')
   368→      fetchOrders()
   369→      if (detailOrder?.id === orderId) {
   370→        setDetailOrder((prev) => prev ? { ...prev, status: newStatus } : null)
   371→      }
   372→    } catch (err) {
   373→      setError(err instanceof Error ? err.message : 'Failed to update status')
   374→    } finally {
   375→      setUpdatingStatus(null)
   376→    }
   377→  }
   378→
   379→  // View order detail
   380→  const handleViewOrder = (order: WholesaleOrder) => {
   381→    setDetailOrder(order)
   382→    setDetailDialogOpen(true)
   383→  }
   384→
   385→  // Analytics
   386→  const totalRevenue = orders.filter((o) => o.status === 'delivered').reduce((s, o) => s + o.total, 0)
   387→  const pendingAmount = orders.filter((o) => o.paymentStatus === 'unpaid' || o.paymentStatus === 'partial').reduce((s, o) => s + o.total, 0)
   388→  const topDealers = dealers.sort((a, b) => b.totalSpent - a.totalSpent).slice(0, 5)
   389→  const outstandingReceivables = dealers.reduce((s, d) => s + d.outstanding, 0)
   390→
   391→  // Get item name
   392→  const getItemName = (item: OrderItem): string => {
   393→    if (item.book) return item.book.title
   394→    if (item.stationery) return item.stationery.name
   395→    return 'Unknown Item'
   396→  }
   397→
   398→  return (
   399→    <div className="space-y-6">
   400→      {/* Header */}
   401→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   402→        <div>
   403→          <h2 className="text-2xl font-bold tracking-tight">Wholesale Management</h2>
   404→          <p className="text-muted-foreground text-sm">Manage bulk orders, dealers, and distributor relationships</p>
   405→        </div>
   406→        <Button onClick={() => setOrderDialogOpen(true)} className="bg-emerald-600 hover:bg-emerald-700 shrink-0">
   407→          <Plus className="size-4 mr-1.5" />
   408→          New Order
   409→        </Button>
   410→      </div>
   411→
   412→      {/* Error Alert */}
   413→      {error && (
   414→        <div className="flex items-center gap-2 rounded-lg border border-red-200 bg-red-50 dark:border-red-900 dark:bg-red-950/30 p-3 text-sm text-red-700 dark:text-red-400">
   415→          <AlertCircle className="size-4 shrink-0" />
   416→          {error}
   417→          <Button variant="ghost" size="sm" className="ml-auto" onClick={() => setError(null)}>
   418→            Dismiss
   419→          </Button>
   420→        </div>
   421→      )}
   422→
   423→      {/* Tabs */}
   424→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   425→        <TabsList className="grid w-full grid-cols-3">
   426→          <TabsTrigger value="orders" className="text-xs sm:text-sm">
   427→            <Package className="size-4 mr-1.5 hidden sm:inline" />
   428→            Orders
   429→          </TabsTrigger>
   430→          <TabsTrigger value="dealers" className="text-xs sm:text-sm">
   431→            <Users className="size-4 mr-1.5 hidden sm:inline" />
   432→            Dealers
   433→          </TabsTrigger>
   434→          <TabsTrigger value="analytics" className="text-xs sm:text-sm">
   435→            <BarChart3 className="size-4 mr-1.5 hidden sm:inline" />
   436→            Analytics
   437→          </TabsTrigger>
   438→        </TabsList>
   439→
   440→        {/* Orders Tab */}
   441→        <TabsContent value="orders" className="space-y-4">
   442→          {/* Status Filter */}
   443→          <Card>
   444→            <CardContent className="p-4">
   445→              <div className="flex items-center gap-2 flex-wrap">
   446→                <span className="text-sm text-muted-foreground">Filter:</span>
   447→                {['all', 'pending', 'confirmed', 'shipped', 'delivered', 'cancelled'].map((status) => (
   448→                  <Button
   449→                    key={status}
   450→                    variant={statusFilter === status ? 'default' : 'outline'}
   451→                    size="sm"
   452→                    className={statusFilter === status ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   453→                    onClick={() => setStatusFilter(status)}
   454→                  >
   455→                    {status === 'all' ? 'All' : status.charAt(0).toUpperCase() + status.slice(1)}
   456→                  </Button>
   457→                ))}
   458→              </div>
   459→            </CardContent>
   460→          </Card>
   461→
   462→          {/* Orders Table */}
   463→          <Card>
   464→            <CardContent className="p-0">
   465→              {loading ? (
   466→                <div className="p-4 space-y-3">
   467→                  {Array.from({ length: 5 }).map((_, i) => (
   468→                    <Skeleton key={i} className="h-12 w-full" />
   469→                  ))}
   470→                </div>
   471→              ) : orders.length === 0 ? (
   472→                <div className="flex flex-col items-center justify-center py-12 text-center">
   473→                  <Package className="size-12 text-muted-foreground/40 mb-3" />
   474→                  <p className="text-muted-foreground">No wholesale orders found</p>
   475→                  <p className="text-xs text-muted-foreground mt-1">Create your first wholesale order</p>
   476→                </div>
   477→              ) : (
   478→                <div className="overflow-x-auto">
   479→                  <Table>
   480→                    <TableHeader>
   481→                      <TableRow>
   482→                        <TableHead>Order #</TableHead>
   483→                        <TableHead>Dealer</TableHead>
   484→                        <TableHead>Status</TableHead>
   485→                        <TableHead className="text-right">Total</TableHead>
   486→                        <TableHead>Payment</TableHead>
   487→                        <TableHead className="hidden sm:table-cell">Date</TableHead>
   488→                        <TableHead className="text-right">Actions</TableHead>
   489→                      </TableRow>
   490→                    </TableHeader>
   491→                    <TableBody>
   492→                      {orders.map((order) => (
   493→                        <TableRow key={order.id}>
   494→                          <TableCell className="font-mono text-sm font-medium">{order.orderNumber}</TableCell>
   495→                          <TableCell className="text-sm">
   496→                            {order.customer?.name || order.dealerName || '—'}
   497→                          </TableCell>
   498→                          <TableCell>{getStatusBadge(order.status)}</TableCell>
   499→                          <TableCell className="text-right font-medium">${order.total.toFixed(2)}</TableCell>
   500→                          <TableCell>{getPaymentBadge(order.paymentStatus)}</TableCell>
   501→                          <TableCell className="hidden sm:table-cell text-muted-foreground text-sm">
   502→                            {new Date(order.createdAt).toLocaleDateString()}
   503→                          </TableCell>
   504→                          <TableCell className="text-right">
   505→                            <div className="flex items-center justify-end gap-1">
   506→                              <Button variant="ghost" size="sm" onClick={() => handleViewOrder(order)}>
   507→                                <Eye className="size-3.5 mr-1" />
   508→                                View
   509→                              </Button>
   510→                              {order.status === 'pending' && (
   511→                                <Button
   512→                                  variant="ghost"
   513→                                  size="sm"
   514→                                  className="text-sky-600 hover:text-sky-700"
   515→                                  onClick={() => handleStatusUpdate(order.id, 'confirmed')}
   516→                                  disabled={updatingStatus === order.id}
   517→                                >
   518→                                  <CheckCircle2 className="size-3.5 mr-1" />
   519→                                  Confirm
   520→                                </Button>
   521→                              )}
   522→                              {order.status === 'confirmed' && (
   523→                                <Button
   524→                                  variant="ghost"
   525→                                  size="sm"
   526→                                  className="text-teal-600 hover:text-teal-700"
   527→                                  onClick={() => handleStatusUpdate(order.id, 'shipped')}
   528→                                  disabled={updatingStatus === order.id}
   529→                                >
   530→                                  <Ship className="size-3.5 mr-1" />
   531→                                  Ship
   532→                                </Button>
   533→                              )}
   534→                              {order.status === 'shipped' && (
   535→                                <Button
   536→                                  variant="ghost"
   537→                                  size="sm"
