    99→    99→function getNextStatus(current: string): string[] {
   100→   100→  if (current === 'rejected') return []
   101→   101→  const idx = statusWorkflow.indexOf(current)
   102→   102→  if (idx === -1) return []
   103→   103→  if (idx === statusWorkflow.length - 1) return []
   104→   104→  return [statusWorkflow[idx + 1]]
   105→   105→}
