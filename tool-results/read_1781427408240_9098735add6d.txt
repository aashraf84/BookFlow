    78→  IT: { bg: 'bg-macos-teal/10', text: 'text-macos-teal', icon: 'bg-macos-teal/10' },

    83→function getDeptBadgeVariant(dept: string): 'default' | 'success' | 'warning' | 'danger' | 'info' | 'purple' {
    84→  const map: Record<string, 'default' | 'success' | 'warning' | 'danger' | 'info' | 'purple'> = {
    85→    Management: 'purple',
    86→    Sales: 'info',
    87→    Finance: 'success',
    88→    Warehouse: 'warning',
    89→    Marketing: 'danger',
    90→    IT: 'info',
    91→    HR: 'warning',
    92→    Operations: 'purple',
    93→  }
    94→  return map[dept] || 'default'
    95→}
    96→
    97→function getDeptInitialsColor(dept: string) {
    98→  const colors = departmentColors[dept]
    99→  if (colors) return colors
   100→  return { bg: 'bg-secondary', text: 'text-muted-foreground', icon: 'bg-secondary' }
   101→}

   105→export function EmployeesModule() {
   106→  const authUser = useAppStore((s) => s.authUser)
   107→  const isViewOnly = authUser?.isViewOnly ?? false
   108→
   109→  const [employees, setEmployees] = React.useState<Employee[]>([])
   110→  const [branches, setBranches] = React.useState<Branch[]>([])
   111→  const [loading, setLoading] = React.useState(true)
   112→  const [search, setSearch] = React.useState('')

   114→  const [statusFilter, setStatusFilter] = React.useState<'all' | 'active' | 'inactive'>('all')

   119→  const fetchEmployees = React.useCallback(async () => {

   130→      console.error('Failed to fetch employees:', err)
   131→    } finally {
   132→      setLoading(false)
   133→    }
   134→  }, [search, departmentFilter])
   135→
   136→  const fetchBranches = React.useCallback(async () => {
   137→    try {
   138→      const res = await fetch('/api/branches')
   139→      const json = await res.json()
   140→      if (json.data) setBranches(json.data)

   142→      console.error('Failed to fetch branches:', err)

   146→  React.useEffect(() => {
   147→    fetchEmployees()

   174→      if (emp.isActive) acc[dept].active++

   178→    {} as Record<string, { total: number; active: number; inactive: number }>
   179→  )

   182→  const filteredEmployees = employees.filter((e) => {

   184→    if (statusFilter === 'inactive') return !e.isActive
   185→    return true
   186→  })
   187→

   189→    { id: 'all', label: 'All', count: totalEmployees },
   190→    { id: 'active', label: 'Active', count: activeEmployees },
   191→    { id: 'inactive', label: 'Inactive', count: inactiveEmployees },
   192→  ]

   197→      <MacosPageHeader
   198→        title="Employees"
   199→        subtitle="Manage your team, departments & payroll"
   200→        icon={Contact}
   201→        iconColor="text-gg-bridge"
   202→        iconBg="bg-gg-bridge/10"
   203→      >

   206→            Add Employee
   207→          </MacosButton>
   208→        )}
   209→      </MacosPageHeader>
   210→
   211→      {/* Stats Row */}
   212→      <MacosStatGrid>
   213→        <MacosStatCard
   214→          label="Total"
   215→          value={totalEmployees}
   216→          icon={Users}
   217→          iconColor="text-macos-blue"
   218→          iconBg="bg-macos-blue/10"
   219→        />
   220→        <MacosStatCard
   221→          label="Active"
   222→          value={activeEmployees}
   223→          icon={UserCheck}
   224→          iconColor="text-macos-green"
   225→          iconBg="bg-macos-green/10"
   226→        />
   227→        <MacosStatCard
   228→          label="Inactive"
   229→          value={inactiveEmployees}
   230→          icon={UserX}
   231→          iconColor="text-macos-red"
   232→          iconBg="bg-macos-red/10"
   233→        />
   234→        <MacosStatCard
   235→          label="Monthly Payroll"
   236→          value={`$${totalPayroll.toLocaleString()}`}
   237→          icon={DollarSign}
   238→          iconColor="text-gg-golden"
   239→          iconBg="bg-gg-golden/10"
   240→        />

   242→
   243→      {/* macOS Contacts-style layout: Sidebar departments + Contact list */}
   244→      <div className="flex gap-4 min-h-[480px]">

   246→        <div className="w-56 shrink-0 hidden md:block">
   247→          <MacosCard noPadding>
   248→            <div className="macos-section-header px-3 pt-3 pb-1">
   249→              <h3 className="macos-section-title">Groups</h3>
   250→            </div>
   251→            <div className="max-h-[400px] overflow-y-auto custom-scrollbar px-1.5 pb-2">
   252→              <div className="space-y-0.5">
   253→                <button
   254→                  className={`w-full flex items-center justify-between rounded-lg px-2.5 py-1.5 text-[13px] font-medium cursor-pointer ${
   255→                    departmentFilter === 'all'
   256→                      ? 'macos-filter-item-active'
   257→                      : 'text-foreground/70 hover:bg-secondary/50'
   258→                  }`}
   259→                  onClick={() => setDepartmentFilter('all')}
   260→                >
   261→                  <span className="flex items-center gap-2">
   262→                    <Users className="size-3.5" strokeWidth={1.5} />
   263→                    All Contacts
   264→                  </span>
   265→                  <span className="text-[10px] font-semibold text-muted-foreground">{totalEmployees}</span>
   266→                </button>
   267→                {Object.entries(departmentStats).map(([dept, stats]) => {
   268→                  const colors = getDeptInitialsColor(dept)
   269→                  return (
   270→                    <button
   271→                      key={dept}
   272→                      className={`w-full flex items-center justify-between rounded-lg px-2.5 py-1.5 text-[13px] font-medium cursor-pointer ${
   273→                        departmentFilter === dept
   274→                          ? `${colors.bg} ${colors.text}`
   275→                          : 'text-foreground/70 hover:bg-secondary/50'
   276→                      }`}
   277→                      onClick={() => setDepartmentFilter(dept === departmentFilter ? 'all' : dept)}
   278→                    >
   279→                      <span className="flex items-center gap-2">
   280→                        <Briefcase className="size-3.5" strokeWidth={1.5} />
   281→                        {dept}
   282→                      </span>

   292→        {/* Right: Contact List */}

   295→            {/* Search + Filter Bar */}
   296→            <div className="p-3 border-b border-border/50">
   297→              <div className="flex items-center gap-3">
   298→                <MacosSearch
   299→                  value={search}
   300→                  onChange={setSearch}
   301→                  placeholder="Search employees..."
   302→                  className="flex-1"
   303→                />
   304→                <MacosFilterBar
   305→                  items={statusFilterItems}
   306→                  activeId={statusFilter}
   307→                  onChange={(id) => setStatusFilter(id as 'all' | 'active' | 'inactive')}
   308→                />
   309→              </div>
   310→              {/* Mobile department pills */}
   311→              <div className="flex flex-wrap gap-1.5 mt-2 md:hidden">
   312→                <button
   313→                  className={`rounded-md px-2.5 py-1 text-[10px] font-semibold tracking-wider uppercase cursor-pointer ${
   314→                    departmentFilter === 'all'
   315→                      ? 'macos-badge bg-macos-blue/10 text-macos-blue'
   316→                      : 'macos-badge bg-secondary/50 text-muted-foreground hover:text-foreground'
   317→                  }`}
   318→                  onClick={() => setDepartmentFilter('all')}
   319→                >
   320→                  All
   321→                </button>
   322→                {Object.entries(departmentStats).map(([dept, stats]) => (
   323→                  <button
   324→                    key={dept}
   325→                    className={`rounded-md px-2.5 py-1 text-[10px] font-semibold tracking-wider uppercase cursor-pointer ${
   326→                      departmentFilter === dept
   327→                        ? `macos-badge ${getDeptInitialsColor(dept).bg} ${getDeptInitialsColor(dept).text}`
   328→                        : 'macos-badge bg-secondary/50 text-muted-foreground hover:text-foreground'
   329→                    }`}
   330→                    onClick={() => setDepartmentFilter(dept === departmentFilter ? 'all' : dept)}
   331→                  >
   332→                    {dept} ({stats.active})
   333→                  </button>
   334→                ))}
   335→              </div>
   336→            </div>
   337→
   338→            {/* Contact Cards List */}
   339→            {loading ? (
   340→              <div className="p-6 space-y-3">
   341→                {[1, 2, 3, 4, 5].map((i) => (
   342→                  <div key={i} className="h-14 bg-muted rounded-xl animate-pulse" />
   343→                ))}
   344→              </div>
   345→            ) : filteredEmployees.length === 0 ? (
   346→              <MacosEmptyState
   347→                icon={Contact}
   348→                title="No contacts found"
   349→                description="Try adjusting your search or filters"
   350→              />
   351→            ) : (
   352→              <div className="max-h-[400px] overflow-y-auto custom-scrollbar">
   353→                <div className="divide-y divide-border/30">
   354→                  {filteredEmployees.map((employee) => {
   355→                    const deptColors = getDeptInitialsColor(employee.department || '')
   356→                    return (
   357→                      <div
   358→                        key={employee.id}
   359→                        className="flex items-center gap-3 px-4 py-2.5 cursor-pointer macos-table-row group"
   360→                        onClick={() => setSelectedEmployee(employee)}
   361→                      >
   362→                        {/* Avatar */}
   363→                        <Avatar className="size-9 ring-1 ring-border/30">
   364→                          <AvatarFallback className={`${deptColors.bg} ${deptColors.text} text-[11px] font-semibold`}>
   365→                            {employee.name.split(' ').map((n) => n[0]).join('').slice(0, 2).toUpperCase()}
   366→                          </AvatarFallback>
   367→                        </Avatar>
   368→
   369→                        {/* Name & Email */}
   370→                        <div className="flex-1 min-w-0">
   371→                          <div className="flex items-center gap-2">
   372→                            <p className="text-[13px] font-medium truncate">{employee.name}</p>
   373→                            <div className={`size-1.5 rounded-full shrink-0 ${employee.isActive ? 'bg-macos-green' : 'bg-macos-red'}`} />
   374→                          </div>
   375→                          <p className="text-[11px] text-muted-foreground truncate">{employee.email}</p>
   376→                        </div>
   377→
   378→                        {/* Department Badge */}
   379→                        <MacosBadge variant={getDeptBadgeVariant(employee.department || '')}>
   380→                          {employee.department || '—'}
   381→                        </MacosBadge>
   382→
   383→                        {/* Position */}
   384→                        <span className="text-[13px] text-muted-foreground truncate max-w-[120px] hidden lg:block">
   385→                          {employee.position || '—'}
   386→                        </span>
   387→
   388→                        {/* Chevron */}
   389→                        <ChevronRight className="size-3.5 text-muted-foreground/50 shrink-0 group-hover:text-muted-foreground" strokeWidth={1.5} />
   390→
   391→                        {/* Actions (shown on hover, hidden for view-only) */}
   392→                        {!isViewOnly && (
   393→                          <div className="flex items-center gap-0.5 opacity-0 group-hover:opacity-100" onClick={(e) => e.stopPropagation()}>
   394→                            <MacosButton
   395→                              variant="ghost"
   396→                              icon={Pencil}
   397→                              onClick={() => setEditEmployee(employee)}
   398→                              className="h-7 w-7 p-0 text-macos-blue hover:text-macos-blue hover:bg-macos-blue/10 rounded-lg"
   399→                            />
   400→                            <MacosButton
   401→                              variant="ghost"
   402→                              icon={Trash2}
   403→                              onClick={() => handleDelete(employee.id)}
   404→                              className="h-7 w-7 p-0 text-macos-red hover:text-macos-red hover:bg-macos-red/10 rounded-lg"
   405→                            />
   406→                          </div>
   407→                        )}
   408→                      </div>
   409→                    )
   410→                  })}
   411→                </div>
   412→              </div>
   413→            )}
   414→
   415→            {/* Footer count */}
   416→            <div className="px-4 py-2 border-t border-border/30">
   417→              <p className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">
   418→                {filteredEmployees.length} contact{filteredEmployees.length !== 1 ? 's' : ''}
   419→                {departmentFilter !== 'all' ? ` in ${departmentFilter}` : ''}
   420→              </p>
   421→            </div>
   422→          </MacosCard>
   423→        </div>
   424→      </div>
   425→
   426→      <AddEmployeeDialog
   427→        open={showAddDialog}
   428→        onOpenChange={setShowAddDialog}
   429→        branches={branches}
   430→        onSuccess={fetchEmployees}
   431→      />
   432→      <EditEmployeeDialog
   433→        employee={editEmployee}
   434→        onOpenChange={(open) => !open && setEditEmployee(null)}
   435→        branches={branches}
   436→        onSuccess={fetchEmployees}
   437→      />
   438→      <EmployeeDetailDialog
   439→        employee={selectedEmployee}
   440→        onOpenChange={(open) => !open && setSelectedEmployee(null)}
   441→      />
   442→    </div>
   443→  )
   444→}

   446→// ==================== ADD EMPLOYEE DIALOG ====================
   447→
   448→function AddEmployeeDialog({
   449→  open,
   450→  onOpenChange,
   451→  branches,
   452→  onSuccess,
   453→}: {
   454→  open: boolean
   455→  onOpenChange: (open: boolean) => void
   456→  branches: Branch[]
   457→  onSuccess: () => void
   458→}) {
   459→  const [saving, setSaving] = React.useState(false)
   460→  const [form, setForm] = React.useState({
   461→    name: '',
   462→    email: '',
   463→    phone: '',
   464→    department: '',
   465→    position: '',
   466→    salary: '',
   467→    hireDate: '',
   468→    branchId: '',
   469→  })
   470→
   471→  const handleSubmit = async (e: React.FormEvent) => {
   472→    e.preventDefault()
   473→    if (!form.name.trim() || !form.email.trim()) return
   474→    setSaving(true)
   475→    try {
   476→      const res = await fetch('/api/employees', {
   477→        method: 'POST',
   478→        headers: { 'Content-Type': 'application/json' },
   479→        body: JSON.stringify({
   480→          ...form,
   481→          salary: parseFloat(form.salary) || 0,
   482→          hireDate: form.hireDate || null,
   483→          branchId: form.branchId || null,
   484→        }),
   485→      })
   486→      if (res.ok) {
   487→        setForm({ name: '', email: '', phone: '', department: '', position: '', salary: '', hireDate: '', branchId: '' })
   488→        onOpenChange(false)
   489→        onSuccess()
   490→      } else {
   491→        const json = await res.json()
   492→        alert(json.error || 'Failed to create employee')
   493→      }
   494→    } catch (err) {
   495→      console.error('Failed to create employee:', err)
   496→      alert('Failed to create employee')
   497→    } finally {
   498→      setSaving(false)
   499→    }
   500→  }
   501→
   502→  return (
   503→    <MacosDialog
   504→      open={open}
   505→      onOpenChange={onOpenChange}
   506→      title="Add Employee"
   507→      description="Register a new team member"
   508→      footer={
   509→        <>
   510→          <MacosButton variant="secondary" onClick={() => onOpenChange(false)}>
   511→            Cancel
   512→          </MacosButton>
   513→          <MacosButton
   514→            variant="primary"
   515→            onClick={handleSubmit as any}
   516→            disabled={saving || !form.name.trim() || !form.email.trim()}
   517→          >
   518→            {saving ? 'Saving...' : 'Add Employee'}
   519→          </MacosButton>
   520→        </>
   521→      }
   522→    >
   523→      <form onSubmit={handleSubmit} className="space-y-4">
   524→        <MacosFormField label="Full Name *">
   525→          <Input
   526→            value={form.name}
   527→            onChange={(e) => setForm({ ...form, name: e.target.value })}
   528→            placeholder="John Smith"
   529→            className="macos-search rounded-lg text-[13px] h-8 border-0"
   530→            required
   531→          />
   532→        </MacosFormField>
   533→        <div className="grid grid-cols-2 gap-3">
   534→          <MacosFormField label="Email *">
   535→            <Input
   536→              type="email"
   537→              value={form.email}
   538→              onChange={(e) => setForm({ ...form, email: e.target.value })}
   539→              placeholder="john@bookflow.com"
   540→              className="macos-search rounded-lg text-[13px] h-8 border-0"
   541→              required
   542→            />
   543→          </MacosFormField>
   544→          <MacosFormField label="Phone">
   545→            <Input
   546→              value={form.phone}
   547→              onChange={(e) => setForm({ ...form, phone: e.target.value })}
   548→              placeholder="+1-555-0100"
   549→              className="macos-search rounded-lg text-[13px] h-8 border-0"
   550→            />
   551→          </MacosFormField>
   552→        </div>
   553→        <div className="grid grid-cols-2 gap-3">
   554→          <MacosFormField label="Department">
   555→            <Select value={form.department} onValueChange={(v) => setForm({ ...form, department: v })}>
   556→              <SelectTrigger className="macos-search rounded-lg text-[13px] h-8 border-0">
   557→                <SelectValue placeholder="Select..." />
   558→              </SelectTrigger>
   559→              <SelectContent>

   622→function EditEmployeeDialog({
   623→  employee,
   624→  onOpenChange,
   625→  branches,
   626→  onSuccess,
   627→}: {
   628→  employee: Employee | null
   629→  onOpenChange: (open: boolean) => void
   630→  branches: Branch[]
   631→  onSuccess: () => void
   632→}) {
   633→  const [saving, setSaving] = React.useState(false)
   634→  const [form, setForm] = React.useState({
   635→    name: '',
   636→    email: '',
   637→    phone: '',
   638→    department: '',
   639→    position: '',
   640→    salary: '',
   641→    hireDate: '',
   642→    branchId: '',
   643→    isActive: true,
   644→  })
   645→
   646→  React.useEffect(() => {
   647→    if (employee) {
   648→      setForm({
   649→        name: employee.name || '',
   650→        email: employee.email || '',
   651→        phone: employee.phone || '',
   652→        department: employee.department || '',
   653→        position: employee.position || '',
   654→        salary: employee.salary?.toString() || '',
   655→        hireDate: employee.hireDate ? new Date(employee.hireDate).toISOString().split('T')[0] : '',
   656→        branchId: employee.branchId || '',
   657→        isActive: employee.isActive,
   658→      })
   659→    }
   660→  }, [employee])
   661→
   662→  const handleSubmit = async (e: React.FormEvent) => {
   663→    e.preventDefault()
   664→    if (!employee) return
   665→    setSaving(true)
   666→    try {
   667→      const res = await fetch('/api/employees', {
   668→        method: 'PUT',
   669→        headers: { 'Content-Type': 'application/json' },
   670→        body: JSON.stringify({
   671→          id: employee.id,
   672→          name: form.name,
   673→          email: form.email,
   674→          phone: form.phone,
   675→          department: form.department,
   676→          position: form.position,
   677→          salary: parseFloat(form.salary) || 0,
   678→          hireDate: form.hireDate || null,
   679→          branchId: form.branchId || null,
   680→          isActive: form.isActive,
   681→        }),
   682→      })
   683→      if (res.ok) {
   684→        onOpenChange(false)
   685→        onSuccess()
   686→      } else {
   687→        const json = await res.json()
   688→        alert(json.error || 'Failed to update employee')
   689→      }
   690→    } catch (err) {
   691→      console.error('Failed to update employee:', err)
   692→      alert('Failed to update employee')
   693→    } finally {
   694→      setSaving(false)
   695→    }
   696→  }
   697→
   698→  if (!employee) return null
   699→
   700→  return (
   701→    <MacosDialog
   702→      open={!!employee}
   703→      onOpenChange={onOpenChange}
   704→      title="Edit Employee"
   705→      description="Update employee information"
   706→      footer={
   707→        <>
   708→          <MacosButton variant="secondary" onClick={() => onOpenChange(false)}>
   709→            Cancel
   710→          </MacosButton>
   711→          <MacosButton variant="primary" onClick={handleSubmit as any} disabled={saving}>
   712→            {saving ? 'Saving...' : 'Update Employee'}
   713→          </MacosButton>
   714→        </>
   715→      }
   716→    >
   717→      <form onSubmit={handleSubmit} className="space-y-4">
   718→        <MacosFormField label="Full Name *">
   719→          <Input
   720→            value={form.name}
   721→            onChange={(e) => setForm({ ...form, name: e.target.value })}
   722→            className="macos-search rounded-lg text-[13px] h-8 border-0"
   723→            required
   724→          />
   725→        </MacosFormField>
   726→        <div className="grid grid-cols-2 gap-3">
   727→          <MacosFormField label="Email *">
   728→            <Input
   729→              type="email"
   730→              value={form.email}
   731→              onChange={(e) => setForm({ ...form, email: e.target.value })}
   732→              className="macos-search rounded-lg text-[13px] h-8 border-0"
   733→              required
   734→            />
   735→          </MacosFormField>
   736→          <MacosFormField label="Phone">
   737→            <Input
   738→              value={form.phone}
   739→              onChange={(e) => setForm({ ...form, phone: e.target.value })}
   740→              className="macos-search rounded-lg text-[13px] h-8 border-0"
   741→            />
   742→          </MacosFormField>

   747→              <SelectTrigger className="macos-search rounded-lg text-[13px] h-8 border-0">
   748→                <SelectValue placeholder="Select..." />
