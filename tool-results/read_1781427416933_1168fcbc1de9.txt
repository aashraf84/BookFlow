   104→// ─── Helpers ──────────────────────────────────────────────────────
   105→function getProductName(item: InventoryItem): string {
   106→  if (item.book) return item.book.title
   107→  if (item.stationery) return item.stationery.name
   108→  return 'Unknown Product'
   109→}
   110→
   111→type StockLevel = 'healthy' | 'moderate' | 'low' | 'critical'
   112→
   113→function getStockLevel(item: InventoryItem): StockLevel {
   114→  if (item.quantity === 0) return 'critical'
   115→  const ratio = item.maxStockLevel > 0 ? item.quantity / item.maxStockLevel : item.quantity / (item.reorderLevel * 3)
   116→  if (item.isLowStock || item.quantity <= item.reorderLevel) return 'low'
   117→  if (ratio < 0.5) return 'moderate'
   118→  return 'healthy'
   119→}
   120→
   121→function getStockLevelColors(level: StockLevel) {
   122→  const map: Record<StockLevel, { bar: string; bg: string; text: string; label: string }> = {
   123→    healthy: { bar: 'bg-macos-green', bg: 'bg-macos-green/10', text: 'text-macos-green', label: 'In Stock' },
   124→    moderate: { bar: 'bg-macos-yellow', bg: 'bg-macos-yellow/10', text: 'text-macos-yellow', label: 'Moderate' },
   125→    low: { bar: 'bg-macos-orange', bg: 'bg-macos-orange/10', text: 'text-macos-orange', label: 'Low Stock' },
   126→    critical: { bar: 'bg-macos-red', bg: 'bg-macos-red/10', text: 'text-macos-red', label: 'Out of Stock' },
   127→  }
   128→  return map[level]
   129→}
   130→
   131→function getTransferStatusBadge(status: string) {
   132→  const map: Record<string, { variant: 'warning' | 'info' | 'success' | 'danger'; label: string }> = {
   133→    pending: { variant: 'warning', label: 'Pending' },
   134→    'in-transit': { variant: 'info', label: 'In Transit' },
   135→    completed: { variant: 'success', label: 'Completed' },
   136→    cancelled: { variant: 'danger', label: 'Cancelled' },
   137→  }
   138→  const s = map[status] || { variant: 'default' as const, label: status }
   139→  return <MacosBadge variant={s.variant}>{s.label}</MacosBadge>
   140→}
   141→
   142→// ─── Stock Level Progress Bar ─────────────────────────────────────
   143→function StockLevelBar({ quantity, maxStockLevel, reorderLevel }: { quantity: number; maxStockLevel: number; reorderLevel: number }) {
   144→  const max = maxStockLevel || reorderLevel * 3
   145→  const pct = max > 0 ? Math.min(100, (quantity / max) * 100) : 0
   146→  const item = { quantity, reorderLevel, maxStockLevel, isLowStock: quantity <= reorderLevel } as InventoryItem
   147→  const level = getStockLevel(item)
   148→  const colors = getStockLevelColors(level)
   149→
   150→  return (
   151→    <div className="flex items-center gap-2.5 min-w-[120px]">
   152→      <div className="flex-1 h-1.5 rounded-full bg-secondary/80 overflow-hidden">
   153→        <div
   154→          className={`h-full rounded-full transition-all duration-500 ease-out ${colors.bar}`}
   155→          style={{ width: `${pct}%` }}
   156→        />
   157→      </div>
   158→      <span className="text-[13px] font-medium tabular-nums tracking-tight w-8 text-right">{quantity}</span>
   159→    </div>
   160→  )
   161→}
