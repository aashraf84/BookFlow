    97→const getAvatarColor = (name: string) => {
    98→  let hash = 0
    99→  for (let i = 0; i < name.length; i++) {
   100→    hash = name.charCodeAt(i) + ((hash << 5) - hash)
   101→  }
   102→  return avatarColors[Math.abs(hash) % avatarColors.length]
   103→}

   142→// ── Component ────────────────────────────────────────────────────
   143→export function AuthorsModule() {
   144→  const { authUser } = useAppStore()
   145→  const isViewOnly = authUser?.isViewOnly ?? false
   146→
   147→  const [authors, setAuthors] = useState<AuthorData[]>([])
   148→  const [total, setTotal] = useState(0)
   149→  const [page, setPage] = useState(1)
   150→  const [loading, setLoading] = useState(true)
   151→  const [search, setSearch] = useState('')
   152→  const [dialogOpen, setDialogOpen] = useState(false)
   153→  const [editing, setEditing] = useState<AuthorData | null>(null)
   154→  const [form, setForm] = useState<AuthorFormData>(emptyForm)
   155→  const [saving, setSaving] = useState(false)
   156→  const [deleting, setDeleting] = useState<string | null>(null)
   157→
   158→  // Detail view state
   159→  const [selectedAuthor, setSelectedAuthor] = useState<AuthorData | null>(null)
   160→  const [authorBooks, setAuthorBooks] = useState<BookData[]>([])
   161→  const [loadingBooks, setLoadingBooks] = useState(false)
   162→
   163→  const limit = 12
   164→
   165→  // ── Fetch authors ──────────────────────────────────────────
   166→  const fetchAuthors = useCallback(async () => {
   167→    setLoading(true)
   168→    try {
   169→      const params = new URLSearchParams({
   170→        page: String(page),
   171→        limit: String(limit),
   172→      })
   173→      if (search) params.set('search', search)
   174→
   175→      const res = await fetch(`/api/authors?${params}`)
   176→      if (!res.ok) throw new Error('Failed to fetch authors')
   177→      const json = await res.json()
   178→      setAuthors(json.data || [])
   179→      setTotal(json.pagination?.total || 0)
   180→    } catch {
   181→      setAuthors([])
   182→    } finally {
   183→      setLoading(false)
   184→    }
   185→  }, [page, search])
   186→
   187→  useEffect(() => {
   188→    fetchAuthors()
   189→  }, [fetchAuthors])
   190→
   191→  useEffect(() => {
   192→    setPage(1)
   193→  }, [search])
   194→
   195→  // ── Fetch author books ─────────────────────────────────────
   196→  const fetchAuthorBooks = async (authorId: string) => {
   197→    setLoadingBooks(true)
   198→    try {
   199→      const res = await fetch(`/api/books?authorId=${authorId}&limit=50`)
   200→      if (!res.ok) throw new Error('Failed to fetch books')
   201→      const json = await res.json()
   202→      setAuthorBooks(json.data || [])
   203→    } catch {
   204→      setAuthorBooks([])
   205→    } finally {
   206→      setLoadingBooks(false)
   207→    }
   208→  }
   209→
   210→  // ── Open detail view ───────────────────────────────────────
   211→  const openDetail = (author: AuthorData) => {
   212→    setSelectedAuthor(author)
   213→    fetchAuthorBooks(author.id)
   214→  }
   215→
   216→  // ── Save author ────────────────────────────────────────────
   217→  const handleSave = async () => {
   218→    if (!form.name.trim()) return
   219→    setSaving(true)
   220→    try {
   221→      const payload = {
   222→        name: form.name.trim(),
   223→        bio: form.bio || null,
   224→        nationality: form.nationality || null,
   225→        website: form.website || null,
   226→        email: form.email || null,
   227→        phone: form.phone || null,
   228→      }
   229→
   230→      let res: Response
   231→      if (editing) {
   232→        res = await fetch('/api/authors', {
   233→          method: 'PUT',
   234→          headers: { 'Content-Type': 'application/json' },
   235→          body: JSON.stringify({ id: editing.id, ...payload }),
   236→        })
   237→      } else {
   238→        res = await fetch('/api/authors', {
   239→          method: 'POST',
   240→          headers: { 'Content-Type': 'application/json' },
   241→          body: JSON.stringify(payload),
   242→        })
   243→      }
   244→
   245→      if (!res.ok) {
   246→        const err = await res.json()
   247→        throw new Error(err.error || 'Failed to save author')
   248→      }
   249→
   250→      setDialogOpen(false)
   251→      setEditing(null)
   252→      setForm(emptyForm)
   253→      fetchAuthors()
   254→    } catch (err) {
   255→      alert(err instanceof Error ? err.message : 'Failed to save author')
   256→    } finally {
   257→      setSaving(false)
   258→    }
   259→  }
   260→
   261→  // ── Delete author ──────────────────────────────────────────
   262→  const handleDelete = async (id: string) => {
   263→    if (!confirm('Are you sure you want to delete this author?')) return
   264→    setDeleting(id)
   265→    try {
   266→      const res = await fetch(`/api/authors?id=${id}`, { method: 'DELETE' })
   267→      if (!res.ok) {
   268→        const err = await res.json()
   269→        throw new Error(err.error || 'Failed to delete')
   270→      }
   271→      fetchAuthors()
   272→    } catch (err) {
   273→      alert(err instanceof Error ? err.message : 'Failed to delete author')
   274→    } finally {
   275→      setDeleting(null)
   276→    }
   277→  }
   278→
   279→  // ── Open dialog for edit ───────────────────────────────────
   280→  const openEdit = (author: AuthorData) => {
   281→    setEditing(author)
   282→    setForm({
   283→      name: author.name,
   284→      bio: author.bio || '',
   285→      nationality: author.nationality || '',
   286→      website: author.website || '',
   287→      email: author.email || '',
   288→      phone: author.phone || '',
   289→    })
   290→    setDialogOpen(true)
   291→  }
   292→
   293→  // ── Open dialog for add ────────────────────────────────────
   294→  const openAdd = () => {
   295→    setEditing(null)
   296→    setForm(emptyForm)
   297→    setDialogOpen(true)
   298→  }
   299→
   300→  const getInitials = (name: string) =>
   301→    name
   302→      .split(' ')
   303→      .map((w) => w[0])
   304→      .join('')
   305→      .toUpperCase()
   306→      .slice(0, 2)
   307→
   308→  const totalPages = Math.ceil(total / limit)
   309→
   310→  // ── Detail View ────────────────────────────────────────────
   311→  if (selectedAuthor) {
   312→    const avatarColor = getAvatarColor(selectedAuthor.name)
   313→    return (
   314→      <div className="space-y-5">
   315→        {/* Back button */}
   316→        <button
   317→          onClick={() => setSelectedAuthor(null)}
   318→          className="flex items-center gap-1.5 text-[13px] font-medium text-macos-blue hover:text-macos-blue/80 transition-colors"
   319→        >
   320→          <ArrowLeft className="size-4" strokeWidth={1.5} /> Back to Authors
   321→        </button>
   322→
   323→        {/* Author Profile Card */}
   324→        <div className="macos-card macos-shadow-card p-6">
   325→          <div className="flex flex-col sm:flex-row gap-5">
   326→            <div className={`size-16 rounded-2xl ${avatarColor.bg} flex items-center justify-center shrink-0 ring-2 ${avatarColor.ring} ring-offset-2 ring-offset-card`}>
   327→              <span className={`text-xl font-bold ${avatarColor.text}`}>
   328→                {getInitials(selectedAuthor.name)}
   329→              </span>
   330→            </div>
   331→            <div className="flex-1 min-w-0">
   332→              <h1 className="macos-page-title text-[20px]">{selectedAuthor.name}</h1>
   333→              <div className="flex flex-wrap items-center gap-3 mt-2">
   334→                {selectedAuthor.nationality && (
   335→                  <span className="flex items-center gap-1.5 text-[12px] text-muted-foreground font-medium">
   336→                    <MapPin className="size-3" strokeWidth={1.5} />
   337→                    {getFlag(selectedAuthor.nationality)} {selectedAuthor.nationality}
   338→                  </span>
   339→                )}
   340→                <span className={`macos-badge bg-macos-green/10 text-macos-green`}>
   341→                  {selectedAuthor._count.books} {selectedAuthor._count.books === 1 ? 'book' : 'books'}
   342→                </span>
   343→              </div>
   344→              <div className="flex flex-wrap gap-4 mt-3">
   345→                {selectedAuthor.email && (
   346→                  <a
   347→                    href={`mailto:${selectedAuthor.email}`}
   348→                    className="flex items-center gap-1.5 text-[12px] font-medium text-macos-blue hover:text-macos-blue/80 transition-colors"
   349→                  >
   350→                    <Mail className="size-3" strokeWidth={1.5} /> {selectedAuthor.email}
   351→                  </a>
   352→                )}
   353→                {selectedAuthor.phone && (
   354→                  <span className="flex items-center gap-1.5 text-[12px] text-muted-foreground font-medium">
   355→                    <Phone className="size-3" strokeWidth={1.5} /> {selectedAuthor.phone}
   356→                  </span>
   357→                )}
   358→                {selectedAuthor.website && (
   359→                  <a
   360→                    href={selectedAuthor.website}
   361→                    target="_blank"
   362→                    rel="noopener noreferrer"
   363→                    className="flex items-center gap-1.5 text-[12px] font-medium text-macos-blue hover:text-macos-blue/80 transition-colors"
   364→                  >
   365→                    <ExternalLink className="size-3" strokeWidth={1.5} /> Website
   366→                  </a>
   367→                )}
   368→              </div>
   369→              {selectedAuthor.bio && (
   370→                <p className="text-[13px] text-muted-foreground mt-3 leading-relaxed max-w-2xl">
   371→                  {selectedAuthor.bio}
   372→                </p>
   373→              )}
   374→            </div>
   375→          </div>
   376→        </div>
   377→
   378→        {/* Books by Author Card */}
   379→        <div className="macos-card macos-shadow-card overflow-hidden">
   380→          <div className="px-5 py-4 border-b border-border/30">
   381→            <div className="flex items-center gap-2">
   382→              <BookOpen className="size-4 text-macos-purple" strokeWidth={1.5} />
   383→              <h3 className="macos-section-title">
   384→                Books by {selectedAuthor.name}
   385→              </h3>
   386→              <span className="macos-badge bg-macos-purple/10 text-macos-purple ml-1">
   387→                {authorBooks.length}
   388→              </span>
   389→            </div>
   390→          </div>
   391→          {loadingBooks ? (
   392→            <div className="p-5 space-y-3">
   393→              {Array.from({ length: 3 }).map((_, i) => (
   394→                <div key={i} className="flex items-center gap-3">
   395→                  <Skeleton className="size-8 rounded-lg" />
   396→                  <div className="flex-1 space-y-1.5">
   397→                    <Skeleton className="h-4 w-2/3" />
   398→                    <Skeleton className="h-3 w-1/3" />
   399→                  </div>
   400→                </div>
   401→              ))}
   402→            </div>
   403→          ) : authorBooks.length === 0 ? (
   404→            <div className="macos-empty py-12">
   405→              <div className="size-12 rounded-2xl bg-macos-orange/10 flex items-center justify-center mb-3">
   406→                <BookOpen className="size-6 text-macos-orange" strokeWidth={1.5} />
   407→              </div>
   408→              <p className="text-[13px] font-semibold text-foreground">No books yet</p>
   409→              <p className="text-[11px] text-muted-foreground mt-1">
   410→                This author hasn&apos;t been linked to any books
   411→              </p>
   412→            </div>
   413→          ) : (
   414→            <div className="overflow-x-auto">
   415→              <Table>
   416→                <TableHeader>
   417→                  <TableRow className="border-b border-border/50 hover:bg-transparent">
   418→                    <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Title</TableHead>
   419→                    <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">ISBN</TableHead>
   420→                    <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Category</TableHead>
   421→                    <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Publisher</TableHead>
   422→                    <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Price</TableHead>
   423→                  </TableRow>
   424→                </TableHeader>
   425→                <TableBody>
   426→                  {authorBooks.map((book) => {
   427→                    const catColor = book.category ? getCategoryColor(book.category.name) : null
   428→                    return (
   429→                      <TableRow key={book.id} className="macos-table-row border-b border-border/30">
   430→                        <TableCell className="text-[13px] font-medium">{book.title}</TableCell>
   431→                        <TableCell className="text-[11px] text-muted-foreground font-mono">
   432→                          {book.isbn || <span className="text-muted-foreground/50">—</span>}
   433→                        </TableCell>
   434→                        <TableCell>
   435→                          {book.category && catColor ? (
   436→                            <span className={`macos-badge ${catColor.bg} ${catColor.text}`}>
   437→                              {book.category.name}
   438→                            </span>
   439→                          ) : (
   440→                            <span className="text-[13px] text-muted-foreground/50">—</span>
   441→                          )}
   442→                        </TableCell>
   443→                        <TableCell className="text-[13px] text-muted-foreground font-medium">
   444→                          {book.publisher?.name || <span className="text-muted-foreground/50">—</span>}
   445→                        </TableCell>
   446→                        <TableCell className="text-[13px] text-right font-semibold tracking-tight">
   447→                          ${book.price.toFixed(2)}
   448→                        </TableCell>
   449→                      </TableRow>
   450→                    )
   451→                  })}
   452→                </TableBody>
   453→              </Table>
   454→            </div>
   455→          )}
   456→        </div>
   457→      </div>
   458→    )
   459→  }
   460→
   461→  // ── Main List View ─────────────────────────────────────────
   462→  return (
   463→    <div className="space-y-5">
   464→      {/* macOS Page Header */}
   465→      <div className="macos-page-header flex-col gap-3 sm:flex-row">
   466→        <div className="flex items-center gap-3">
   467→          <div className="flex items-center justify-center size-9 rounded-xl bg-gg-sunset/10">
   468→            <Users className="size-[18px] text-gg-sunset" strokeWidth={1.5} />
   469→          </div>
   470→          <div>
   471→            <h1 className="macos-page-title">Authors</h1>
   472→            <p className="macos-page-subtitle">
   473→              {total} {total === 1 ? 'author' : 'authors'} in your catalog
   474→            </p>
   475→          </div>
   476→        </div>
   477→        {!isViewOnly && (
   478→          <Button
   479→            onClick={openAdd}
   480→            className="bg-primary text-primary-foreground rounded-lg font-medium text-[13px] h-8 gap-1.5"
   481→          >
   482→            <Plus className="size-4" strokeWidth={1.5} /> Add Author
   483→          </Button>
   484→        )}
   485→      </div>
   486→
   487→      {/* Spotlight Search */}
   488→      <div className="macos-card macos-shadow-card p-4">
   489→        <div className="macos-search flex items-center gap-2 px-3 py-1.5">
   490→          <Search className="size-[14px] text-muted-foreground shrink-0" strokeWidth={1.5} />
   491→          <input
   492→            type="text"
   493→            placeholder="Search by name, nationality, email..."
   494→            value={search}
   495→            onChange={(e) => setSearch(e.target.value)}
   496→            className="flex-1 bg-transparent border-0 outline-none text-[13px] font-medium placeholder:text-muted-foreground/60 focus:ring-0 focus:outline-none"
   497→          />
   498→          {search && (
   499→            <button
   500→              onClick={() => setSearch('')}
   501→              className="size-4 rounded-full bg-muted-foreground/20 flex items-center justify-center hover:bg-muted-foreground/30 transition-colors"
   502→            >
   503→              <X className="size-2.5 text-background" strokeWidth={2} />
   504→            </button>
   505→          )}
   506→        </div>
   507→      </div>
   508→
   509→      {/* Authors Grid */}
   510→      {loading ? (
