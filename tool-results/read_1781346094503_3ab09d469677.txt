   145→  { bg: 'bg-macos-teal/10', text: 'text-macos-teal', accent: 'bg-macos-teal' },
   146→  { bg: 'bg-gg-bay/10', text: 'text-gg-bay', accent: 'bg-gg-bay' },
   147→  { bg: 'bg-gg-sunset/10', text: 'text-gg-sunset', accent: 'bg-gg-sunset' },

   149→
   150→const getCoverColor = (title: string) => {
   151→  let hash = 0
   152→  for (let i = 0; i < title.length; i++) {
   153→    hash = title.charCodeAt(i) + ((hash << 5) - hash)
   154→  }
   155→  return coverColors[Math.abs(hash) % coverColors.length]
   156→}

   191→  const [editing, setEditing] = useState<Book | null>(null)
   192→  const [form, setForm] = useState<BookFormData>(emptyForm)
   193→  const [saving, setSaving] = useState(false)
   194→  const [deleting, setDeleting] = useState<string | null>(null)
   195→
   196→  // Dropdown data
   197→  const [authors, setAuthors] = useState<Author[]>([])
   198→  const [publishers, setPublishers] = useState<Publisher[]>([])
   199→  const [categories, setCategories] = useState<Category[]>([])
   200→
   201→  const limit = 10
   202→
   203→  // ── Fetch dropdown data ────────────────────────────────────
   204→  useEffect(() => {
   205→    Promise.all([
   206→      fetch('/api/authors?limit=200').then((r) => r.json()),
   207→      fetch('/api/publishers?limit=200').then((r) => r.json()),
   208→      fetch('/api/categories').then((r) => r.json()),

   211→      setPublishers(publishersJson.data || [])

   213→    })
   214→  }, [])
   215→
   216→  // ── Fetch books ────────────────────────────────────────────
   217→  const fetchBooks = useCallback(async () => {
   218→    setLoading(true)
   219→    try {
   220→      const params = new URLSearchParams({
   221→        page: String(page),
   222→        limit: String(limit),
   223→      })
   224→      if (search) params.set('search', search)
   225→      if (filterCategory && filterCategory !== 'all') params.set('categoryId', filterCategory)
   226→      if (filterAuthor && filterAuthor !== 'all') params.set('authorId', filterAuthor)
   227→      if (filterPublisher && filterPublisher !== 'all') params.set('publisherId', filterPublisher)
   228→
   229→      const res = await fetch(`/api/books?${params}`)
   230→      if (!res.ok) throw new Error('Failed to fetch books')
   231→      const json = await res.json()
   232→      setBooks(json.data || [])
   233→      setTotalPages(json.pagination?.totalPages || 1)
   234→      setTotal(json.pagination?.total || 0)
   235→    } catch {
   236→      setBooks([])
   237→    } finally {
   238→      setLoading(false)
   239→    }
   240→  }, [page, search, filterCategory, filterAuthor, filterPublisher])
   241→
   242→  useEffect(() => {
   243→    fetchBooks()
   244→  }, [fetchBooks])
   245→
   246→  // Reset to page 1 when filters change
   247→  useEffect(() => {
   248→    setPage(1)
   249→  }, [search, filterCategory, filterAuthor, filterPublisher])
   250→
   251→  // ── Save book (create or update) ───────────────────────────
   252→  const handleSave = async () => {
   253→    if (!form.title.trim()) return
   254→    setSaving(true)
   255→    try {
   256→      const payload: Record<string, unknown> = {
   257→        title: form.title.trim(),
   258→        subtitle: form.subtitle || null,
   259→        isbn: form.isbn || null,
   260→        authorId: form.authorId || null,
   261→        publisherId: form.publisherId || null,
   262→        categoryId: form.categoryId || null,
   263→        edition: form.edition || null,
   264→        language: form.language || 'English',
   265→        pages: form.pages ? parseInt(form.pages) : null,
   266→        publishYear: form.publishYear ? parseInt(form.publishYear) : null,
   267→        description: form.description || null,
   268→        price: form.price ? parseFloat(form.price) : 0,
   269→        costPrice: form.costPrice ? parseFloat(form.costPrice) : 0,
   270→        discount: form.discount ? parseFloat(form.discount) : 0,
   271→        academicLevel: form.academicLevel || null,
   272→        subject: form.subject || null,
   273→        barcode: form.barcode || null,
   274→      }
   275→
   276→      let res: Response
   277→      if (editing) {
   278→        payload.id = editing.id
   279→        res = await fetch('/api/books', {
   280→          method: 'PUT',
   281→          headers: { 'Content-Type': 'application/json' },
   282→          body: JSON.stringify(payload),
   283→        })
   284→      } else {
   285→        res = await fetch('/api/books', {
   286→          method: 'POST',
   287→          headers: { 'Content-Type': 'application/json' },
   288→          body: JSON.stringify(payload),
   289→        })
   290→      }
   291→
   292→      if (!res.ok) {
   293→        const err = await res.json()
   294→        throw new Error(err.error || 'Failed to save book')
   295→      }
   296→
   297→      setDialogOpen(false)
   298→      setEditing(null)
   299→      setForm(emptyForm)
   300→      fetchBooks()
   301→    } catch (err) {
   302→      alert(err instanceof Error ? err.message : 'Failed to save book')
   303→    } finally {
   304→      setSaving(false)
   305→    }
   306→  }
   307→
   308→  // ── Delete book ────────────────────────────────────────────

   315→        const err = await res.json()
   316→        throw new Error(err.error || 'Failed to delete')
   317→      }
   318→      fetchBooks()
   319→    } catch (err) {
   320→      alert(err instanceof Error ? err.message : 'Failed to delete book')
   321→    } finally {
   322→      setDeleting(null)
   323→    }
   324→  }

   333→      authorId: book.authorId || '',
   334→      publisherId: book.publisherId || '',
   335→      categoryId: book.categoryId || '',
   336→      edition: book.edition || '',
   337→      language: book.language || 'English',
   338→      pages: book.pages?.toString() || '',
   339→      publishYear: book.publishYear?.toString() || '',
   340→      description: book.description || '',
   341→      price: book.price?.toString() || '',
   342→      costPrice: book.costPrice?.toString() || '',
   343→      discount: book.discount?.toString() || '',
   344→      academicLevel: book.academicLevel || '',
   345→      subject: book.subject || '',
   346→      barcode: book.barcode || '',

   354→    setForm(emptyForm)

   358→  const formatCurrency = (value: number) =>

   360→
   361→  const hasActiveFilters = search || (filterCategory && filterCategory !== 'all') || (filterAuthor && filterAuthor !== 'all') || (filterPublisher && filterPublisher !== 'all')
   362→
   363→  return (
