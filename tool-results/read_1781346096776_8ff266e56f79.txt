   113→   113→// ─── Component ────────────────────────────────────────────────────
   114→   114→export function StationeryModule() {
   115→   115→  const authUser = useAppStore((s) => s.authUser)
   116→   116→  const isViewOnly = authUser?.isViewOnly ?? false
   117→   117→
   118→   118→  const [products, setProducts] = React.useState<StationeryProduct[]>([])
   119→   119→  const [categories, setCategories] = React.useState<StationeryCategory[]>([])
   120→   120→  const [loading, setLoading] = React.useState(true)
   121→   121→  const [error, setError] = React.useState<string | null>(null)
   122→   122→  const [search, setSearch] = React.useState('')
   123→   123→  const [categoryFilter, setCategoryFilter] = React.useState('all')
   124→   124→  const [dialogOpen, setDialogOpen] = React.useState(false)
   125→   125→  const [editingProduct, setEditingProduct] = React.useState<StationeryProduct | null>(null)
   126→   126→  const [form, setForm] = React.useState<ProductFormData>(emptyForm)
   127→   127→  const [saving, setSaving] = React.useState(false)
   128→   128→  const [deleting, setDeleting] = React.useState<string | null>(null)
   129→   129→
   130→   130→  // Fetch products
   131→   131→  const fetchProducts = React.useCallback(async () => {
   132→   132→    setLoading(true)
   133→   133→    setError(null)
   134→   134→    try {
   135→   135→      const params = new URLSearchParams()
   136→   136→      if (search) params.set('search', search)
   137→   137→      if (categoryFilter && categoryFilter !== 'all') params.set('category', categoryFilter)
   138→   138→      params.set('limit', '50')
   139→   139→      const res = await fetch(`/api/stationery?${params.toString()}`)
   140→   140→      if (!res.ok) throw new Error('Failed to fetch products')
   141→   141→      const json = await res.json()
   142→   142→      setProducts(json.data || [])
   143→   143→    } catch (err) {
   144→   144→      setError(err instanceof Error ? err.message : 'Unknown error')
   145→   145→    } finally {
   146→   146→      setLoading(false)
   147→   147→    }
   148→   148→  }, [search, categoryFilter])
   149→   149→
   150→   150→  // Fetch categories
   151→   151→  const fetchCategories = React.useCallback(async () => {
   152→   152→    try {
   153→   153→      const res = await fetch('/api/stationery?categories=true&limit=100')
   154→   154→      if (!res.ok) throw new Error('Failed to fetch categories')
   155→   155→      const json = await res.json()
   156→   156→      setCategories(json.categories || [])
   157→   157→    } catch {
   158→   158→      // Silent fail for categories
   159→   159→    }
   160→   160→  }, [])
   161→   161→
   162→   162→  React.useEffect(() => {
   163→   163→    fetchProducts()
   164→   164→  }, [fetchProducts])
   165→   165→
   166→   166→  React.useEffect(() => {
   167→   167→    fetchCategories()
   168→   168→  }, [fetchCategories])
   169→   169→
   170→   170→  // Stock status helper
   171→   171→  const getStockStatus = (product: StationeryProduct): { label: string; dotClass: string; textClass: string; badgeClass: string } => {
   172→   172→    const inv = product.inventory
   173→   173→    if (!inv || inv.length === 0) {
   174→   174→      return { label: 'No Stock', dotClass: 'bg-macos-red', textClass: 'text-macos-red', badgeClass: 'bg-macos-red/10 text-macos-red' }
   175→   175→    }
   176→   176→    const totalQty = inv.reduce((s, i) => s + i.quantity, 0)
   177→   177→    const reorder = inv[0]?.reorderLevel || 10
   178→   178→    if (totalQty === 0) return { label: 'Out of Stock', dotClass: 'bg-macos-red', textClass: 'text-macos-red', badgeClass: 'bg-macos-red/10 text-macos-red' }
   179→   179→    if (totalQty <= reorder) return { label: 'Low Stock', dotClass: 'bg-macos-orange', textClass: 'text-macos-orange', badgeClass: 'bg-macos-orange/10 text-macos-orange' }
   180→   180→    return { label: 'In Stock', dotClass: 'bg-macos-green', textClass: 'text-macos-green', badgeClass: 'bg-macos-green/10 text-macos-green' }
   181→   181→  }
   182→   182→
   183→   183→  // Computed stats
   184→   184→  const inStockCount = products.filter((p) => {
   185→   185→    const inv = p.inventory
   186→   186→    if (!inv || inv.length === 0) return false
   187→   187→    const qty = inv.reduce((s, i) => s + i.quantity, 0)
   188→   188→    const reorder = inv[0]?.reorderLevel || 10
   189→   189→    return qty > reorder
   190→   190→  }).length
   191→   191→
   192→   192→  const lowStockCount = products.filter((p) => {
   193→   193→    const inv = p.inventory
   194→   194→    if (!inv || inv.length === 0) return false
   195→   195→    const qty = inv.reduce((s, i) => s + i.quantity, 0)
   196→   196→    const reorder = inv[0]?.reorderLevel || 10
   197→   197→    return qty > 0 && qty <= reorder
   198→   198→  }).length
   199→   199→
   200→   200→  const statValues = [products.length, inStockCount, lowStockCount, categories.length]
   201→   201→
   202→   202→  // Open add dialog
   203→   203→  const handleAdd = () => {
   204→   204→    setEditingProduct(null)
   205→   205→    setForm(emptyForm)
   206→   206→    setDialogOpen(true)
   207→   207→  }
   208→   208→
   209→   209→  // Open edit dialog
   210→   210→  const handleEdit = (product: StationeryProduct) => {
   211→   211→    setEditingProduct(product)
   212→   212→    setForm({
   213→   213→      name: product.name,
   214→   214→      sku: product.sku || '',
   215→   215→      categoryId: product.categoryId || '',
   216→   216→      brand: product.brand || '',
   217→   217→      description: product.description || '',
   218→   218→      price: product.price.toString(),
   219→   219→      costPrice: product.costPrice.toString(),
   220→   220→      discount: product.discount.toString(),
   221→   221→      barcode: product.barcode || '',
   222→   222→      unit: product.unit,
   223→   223→      minOrderQty: product.minOrderQty.toString(),
   224→   224→    })
   225→   225→    setDialogOpen(true)
   226→   226→  }
   227→   227→
   228→   228→  // Save product (create or update)
   229→   229→  const handleSave = async () => {
   230→   230→    if (!form.name.trim()) return
   231→   231→    setSaving(true)
   232→   232→    try {
   233→   233→      const payload = {
   234→   234→        ...(editingProduct ? { id: editingProduct.id } : {}),
   235→   235→        name: form.name.trim(),
   236→   236→        sku: form.sku.trim() || null,
   237→   237→        categoryId: form.categoryId || null,
   238→   238→        brand: form.brand.trim() || null,
   239→   239→        description: form.description.trim() || null,
   240→   240→        price: parseFloat(form.price) || 0,
   241→   241→        costPrice: parseFloat(form.costPrice) || 0,
   242→   242→        discount: parseFloat(form.discount) || 0,
   243→   243→        barcode: form.barcode.trim() || null,
   244→   244→        unit: form.unit || 'piece',
   245→   245→        minOrderQty: parseInt(form.minOrderQty) || 1,
   246→   246→      }
   247→   247→
   248→   248→      const res = await fetch('/api/stationery', {
   249→   249→        method: editingProduct ? 'PUT' : 'POST',
   250→   250→        headers: { 'Content-Type': 'application/json' },
   251→   251→        body: JSON.stringify(payload),
   252→   252→      })
   253→   253→
   254→   254→      if (!res.ok) {
   255→   255→        const errData = await res.json()
   256→   256→        throw new Error(errData.error || 'Failed to save product')
   257→   257→      }
   258→   258→
   259→   259→      setDialogOpen(false)
   260→   260→      fetchProducts()
   261→   261→    } catch (err) {
   262→   262→      setError(err instanceof Error ? err.message : 'Failed to save product')
   263→   263→    } finally {
   264→   264→      setSaving(false)
   265→   265→    }
   266→   266→  }
   267→   267→
   268→   268→  // Delete product
   269→   269→  const handleDelete = async (id: string) => {
   270→   270→    if (!confirm('Are you sure you want to delete this product?')) return
   271→   271→    setDeleting(id)
   272→   272→    try {
   273→   273→      const res = await fetch(`/api/stationery?id=${id}`, { method: 'DELETE' })
   274→   274→      if (!res.ok) throw new Error('Failed to delete product')
   275→   275→      fetchProducts()
   276→   276→    } catch (err) {
   277→   277→      setError(err instanceof Error ? err.message : 'Failed to delete product')
   278→   278→    } finally {
   279→   279→      setDeleting(null)
   280→   280→    }
   281→   281→  }
   282→   282→
   283→   283→  // Update form field
   284→   284→  const updateField = (field: keyof ProductFormData, value: string) => {
   285→   285→    setForm((prev) => ({ ...prev, [field]: value }))
   286→   286→  }
   287→   287→
   288→   288→  // Category tabs for filter bar
   289→   289→  const categoryTabs = [
   290→   290→    { id: 'all', label: 'All' },
   291→   291→    ...categories.map((c) => ({ id: c.name, label: c.name })),
   292→   292→  ]
   293→   293→
   294→   294→  return (
   295→   295→    <div className="space-y-5">
   296→   296→      {/* Header */}
   297→   297→      <div className="macos-page-header flex-col gap-3 sm:flex-row">
   298→   298→        <div>
   299→   299→          <h2 className="macos-page-title">Stationery</h2>
   300→   300→          <p className="macos-page-subtitle">{products.length} products in catalog</p>
   301→   301→        </div>
   302→   302→        {!isViewOnly && (
   303→   303→          <Button
   304→   304→            onClick={handleAdd}
   305→   305→            className="bg-primary text-primary-foreground rounded-lg font-medium text-[13px] h-8 shrink-0"
   306→   306→          >
   307→   307→            <Plus className="size-3.5 mr-1.5" strokeWidth={1.5} />
   308→   308→            Add Product
   309→   309→          </Button>
   310→   310→        )}
   311→   311→      </div>
   312→   312→
   313→   313→      {/* Stat Cards */}
   314→   314→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   315→   315→        {statColors.map((stat, i) => {
   316→   316→          const Icon = stat.icon
   317→   317→          return (
   318→   318→            <div key={stat.label} className="macos-card macos-shadow-card p-4 flex items-center gap-3">
   319→   319→              <div className={`size-9 rounded-xl ${stat.bg} flex items-center justify-center shrink-0`}>
   320→   320→                <Icon className={`size-4 ${stat.text}`} strokeWidth={1.5} />
   321→   321→              </div>
   322→   322→              <div className="min-w-0">
   323→   323→                <p className="text-[22px] font-bold tracking-[-0.03em] leading-tight">{statValues[i]}</p>
   324→   324→                <p className="text-[11px] text-muted-foreground font-medium truncate">{stat.label}</p>
   325→   325→              </div>
   326→   326→            </div>
   327→   327→          )
   328→   328→        })}
   329→   329→      </div>
   330→   330→
   331→   331→      {/* Search + Filter Bar */}
   332→   332→      <div className="macos-card macos-shadow-card p-4 space-y-3">
   333→   333→        <div className="macos-search flex items-center px-3 h-9">
   334→   334→          <Search className="size-3.5 text-muted-foreground shrink-0" strokeWidth={1.5} />
   335→   335→          <input
   336→   336→            placeholder="Search by name, SKU, brand, or barcode..."
   337→   337→            value={search}
   338→   338→            onChange={(e) => setSearch(e.target.value)}
   339→   339→            className="flex-1 bg-transparent border-0 text-[13px] font-medium placeholder:text-muted-foreground/60 focus:outline-none ml-2"
   340→   340→          />
   341→   341→        </div>
   342→   342→        <div className="flex items-center gap-3">
   343→   343→          <Filter className="size-3.5 text-muted-foreground shrink-0" strokeWidth={1.5} />
   344→   344→          <div className="macos-filter-bar overflow-x-auto">
   345→   345→            {categoryTabs.slice(0, 7).map((tab) => (
   346→   346→              <button
   347→   347→                key={tab.id}
   348→   348→                className={`macos-filter-item whitespace-nowrap ${categoryFilter === tab.id ? 'macos-filter-item-active' : ''}`}
   349→   349→                onClick={() => setCategoryFilter(tab.id)}
   350→   350→              >
   351→   351→                {tab.label}
   352→   352→              </button>
   353→   353→            ))}
   354→   354→            {categoryTabs.length > 7 && (
   355→   355→              <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   356→   356→                <SelectTrigger className="h-6 w-auto border-0 bg-transparent text-[12px] font-medium text-muted-foreground hover:text-foreground p-0 pl-2">
   357→   357→                  <SelectValue placeholder="More..." />
   358→   358→                </SelectTrigger>
   359→   359→                <SelectContent>
   360→   360→                  {categoryTabs.slice(7).map((tab) => (
   361→   361→                    <SelectItem key={tab.id} value={tab.id}>{tab.label}</SelectItem>
   362→   362→                  ))}
   363→   363→                </SelectContent>
   364→   364→              </Select>
   365→   365→            )}
   366→   366→          </div>
   367→   367→        </div>
   368→   368→      </div>
   369→   369→
   370→   370→      {/* Error Alert */}
   371→   371→      {error && (
   372→   372→        <div className="flex items-center gap-2.5 macos-card macos-shadow-card p-4 bg-macos-red/5 border-macos-red/20">
   373→   373→          <div className="size-7 rounded-lg bg-macos-red/10 flex items-center justify-center shrink-0">
   374→   374→            <AlertCircle className="size-3.5 text-macos-red" />
   375→   375→          </div>
   376→   376→          <span className="text-[13px] font-medium text-macos-red flex-1">{error}</span>
   377→   377→          <Button
   378→   378→            variant="ghost"
   379→   379→            size="sm"
   380→   380→            className="h-7 text-[11px] text-macos-red hover:text-macos-red hover:bg-macos-red/10 rounded-lg"
   381→   381→            onClick={() => setError(null)}
   382→   382→          >
   383→   383→            Dismiss
   384→   384→          </Button>
   385→   385→        </div>
   386→   386→      )}
   387→   387→
   388→   388→      {/* Products Table */}
   389→   389→      <div className="macos-card macos-shadow-card overflow-hidden">
   390→   390→        {loading ? (
   391→   391→          <div className="p-5 space-y-3">
   392→   392→            {Array.from({ length: 5 }).map((_, i) => (
   393→   393→              <div key={i} className="flex items-center gap-3">
   394→   394→                <Skeleton className="size-8 rounded-xl shrink-0" />
   395→   395→                <Skeleton className="h-4 flex-1 rounded-lg" />
   396→   396→              </div>
   397→   397→            ))}
   398→   398→          </div>
   399→   399→        ) : products.length === 0 ? (
   400→   400→          <div className="macos-empty py-16">
   401→   401→            <Package className="size-12 text-muted-foreground/25 mb-3" strokeWidth={1.5} />
   402→   402→            <p className="text-[13px] font-semibold text-muted-foreground">No stationery products found</p>
   403→   403→            <p className="text-[11px] text-muted-foreground mt-1">Add your first product to get started</p>
   404→   404→          </div>
   405→   405→        ) : (
   406→   406→          <div className="overflow-x-auto">
   407→   407→            <Table>
   408→   408→              <TableHeader>
   409→   409→                <TableRow className="border-b border-border/50 hover:bg-transparent">
   410→   410→                  <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Product</TableHead>
   411→   411→                  <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground hidden md:table-cell">SKU</TableHead>
   412→   412→                  <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground hidden sm:table-cell">Category</TableHead>
   413→   413→                  <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground hidden lg:table-cell">Brand</TableHead>
   414→   414→                  <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Price</TableHead>
   415→   415→                  <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground hidden sm:table-cell">Stock</TableHead>
   416→   416→                  {!isViewOnly && (
   417→   417→                    <TableHead className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Actions</TableHead>
   418→   418→                  )}
   419→   419→                </TableRow>
   420→   420→              </TableHeader>
   421→   421→              <TableBody>
   422→   422→                {products.map((product) => {
   423→   423→                  const stock = getStockStatus(product)
   424→   424→                  return (
   425→   425→                    <TableRow key={product.id} className="macos-table-row border-b border-border/30">
   426→   426→                      <TableCell>
   427→   427→                        <div className="flex items-center gap-2.5">
   428→   428→                          <div className="size-8 rounded-xl bg-macos-teal/10 flex items-center justify-center shrink-0">
   429→   429→                            <Package className="size-4 text-macos-teal" strokeWidth={1.5} />
   430→   430→                          </div>
   431→   431→                          <div className="min-w-0">
   432→   432→                            <p className="text-[13px] font-medium truncate max-w-[180px]">{product.name}</p>
   433→   433→                            {product.brand && (
   434→   434→                              <p className="text-[11px] text-muted-foreground truncate">{product.brand}</p>
   435→   435→                            )}
   436→   436→                          </div>
   437→   437→                        </div>
   438→   438→                      </TableCell>
   439→   439→                      <TableCell className="hidden md:table-cell text-muted-foreground text-[13px] font-mono">
   440→   440→                        {product.sku || '\u2014'}
   441→   441→                      </TableCell>
   442→   442→                      <TableCell className="hidden sm:table-cell">
   443→   443→                        {product.category ? (
