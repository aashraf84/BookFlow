   167→   167→// Order detail pipeline for a single order in the detail dialog
   168→   168→function OrderDeliveryPipeline({ status }: { status: string }) {
   169→   169→  const currentIdx = deliveryPipelineStages.indexOf(status)
   170→   170→  const isCancelled = status === 'cancelled'
   171→   171→
   172→   172→  return (
   173→   173→    <div className="py-3">
   174→   174→      {isCancelled ? (
   175→   175→        <div className="flex items-center gap-3 px-4 py-3 rounded-xl bg-macos-red/5 border border-macos-red/10">
   176→   176→          <XCircle className="size-5 text-macos-red" />
   177→   177→          <div>
   178→   178→            <p className="text-[13px] font-semibold text-macos-red">Order Cancelled</p>
   179→   179→            <p className="text-[11px] text-muted-foreground">This order has been cancelled</p>
   180→   180→          </div>
   181→   181→        </div>
   182→   182→      ) : (
   183→   183→        <div className="flex items-start gap-2">
   184→   184→          {deliveryPipelineStages.map((stage, idx) => {
   185→   185→            const config = statusConfig[stage]
   186→   186→            const isLast = idx === deliveryPipelineStages.length - 1
   187→   187→            const isCompleted = idx < currentIdx
   188→   188→            const isCurrent = idx === currentIdx
   189→   189→            return (
   190→   190→              <DeliveryPipelineStep
   191→   191→                key={stage}
   192→   192→                label={config.label}
   193→   193→                icon={config.icon}
   194→   194→                isActive={isCompleted || isCurrent}
   195→   195→                isCompleted={isCompleted}
   196→   196→                isCurrent={isCurrent}
   197→   197→                color={config.color}
   198→   198→                bgColor={config.bgColor}
   199→   199→                isLast={isLast}
   200→   200→              />
   201→   201→            )
   202→   202→          })}
   203→   203→        </div>
   204→   204→      )}
   205→   205→    </div>
   206→   206→  )
   207→   207→}
