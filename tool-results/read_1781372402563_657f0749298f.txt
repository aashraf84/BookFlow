    53→// ─── Types ────────────────────────────────────────────────────────
    54→type ProductType = 'book' | 'stationery'
    55→
    56→interface POSProduct {
    57→  id: string
    58→  type: ProductType
    59→  name: string
    60→  sku: string | null
    61→  isbn: string | null
    62→  barcode: string | null
    63→  price: number
    64→  costPrice: number
    65→  discount: number
    66→  author?: string | null
    67→  brand?: string | null
    68→  category?: string | null
    69→}
    70→
    71→interface CartItem {
    72→  product: POSProduct
    73→  quantity: number
    74→  unitPrice: number
    75→  discount: number
    76→  totalPrice: number
    77→}
    78→
    79→interface Customer {
    80→  id: string
    81→  name: string
    82→  email?: string | null
    83→  phone?: string | null
    84→  loyaltyPoints: number
    85→}
    86→
    87→interface CompletedSale {
    88→  invoiceNumber: string
    89→  total: number
    90→  subtotal: number
    91→  discount: number
    92→  tax: number
    93→  paymentMethod: string
    94→  customerName: string | null
    95→  items: Array<{ name: string; quantity: number; unitPrice: number; totalPrice: number }>
    96→}

    98→// ─── Component ────────────────────────────────────────────────────
    99→export function RetailPOSModule() {
   100→  const { authUser } = useAppStore()
   101→  const isViewOnly = authUser?.isViewOnly ?? false
   102→
   103→  // Products
   104→  const [products, setProducts] = React.useState<POSProduct[]>([])
   105→  const [searchTerm, setSearchTerm] = React.useState('')
   106→  const [categoryFilter, setCategoryFilter] = React.useState<'all' | 'books' | 'stationery' | 'school'>('all')
   107→  const [searchLoading, setSearchLoading] = React.useState(false)
   108→
   109→  // Cart
   110→  const [cart, setCart] = React.useState<CartItem[]>([])
   111→  const [cartDiscount, setCartDiscount] = React.useState(0)
   112→  const [discountType, setDiscountType] = React.useState<'percentage' | 'fixed'>('percentage')
   113→  const [taxRate, setTaxRate] = React.useState(0)
   114→
   115→  // Customer
   116→  const [customerSearch, setCustomerSearch] = React.useState('')
   117→  const [customers, setCustomers] = React.useState<Customer[]>([])
   118→  const [selectedCustomer, setSelectedCustomer] = React.useState<Customer | null>(null)
   119→  const [customerLookupOpen, setCustomerLookupOpen] = React.useState(false)
   120→
   121→  // Payment
   122→  const [paymentMethod, setPaymentMethod] = React.useState('cash')
   123→
   124→  // Sale
   125→  const [completing, setCompleting] = React.useState(false)
   126→  const [completedSale, setCompletedSale] = React.useState<CompletedSale | null>(null)
   127→  const [error, setError] = React.useState<string | null>(null)
   128→
   129→  // Search products
   130→  React.useEffect(() => {
   131→    const searchProducts = async () => {
   132→      setSearchLoading(true)
   133→      try {
   134→        const params = new URLSearchParams()
   135→        params.set('limit', '30')
   136→        if (searchTerm) params.set('search', searchTerm)
   137→
   138→        const [booksRes, stationeryRes] = await Promise.all([
   139→          fetch(`/api/books?${params.toString()}`),
   140→          fetch(`/api/stationery?${params.toString()}`),
   141→        ])
   142→
   143→        const bookData = booksRes.ok ? await booksRes.json() : { data: [] }
   144→        const stationeryData = stationeryRes.ok ? await stationeryRes.json() : { data: [] }
   145→
   146→        const bookProducts: POSProduct[] = (bookData.data || []).map((b: Record<string, unknown>) => ({
   147→          id: b.id as string,
   148→          type: 'book' as ProductType,
   149→          name: b.title as string,
   150→          sku: null,
   151→          isbn: (b.isbn as string) || null,
   152→          barcode: (b.barcode as string) || null,
   153→          price: (b.price as number) || 0,
   154→          costPrice: (b.costPrice as number) || 0,
   155→          discount: (b.discount as number) || 0,
   156→          author: (b.author as { name: string } | null)?.name || null,
   157→          category: (b.category as { name: string } | null)?.name || null,
   158→        }))
   159→
   160→        const stationeryProducts: POSProduct[] = (stationeryData.data || []).map((s: Record<string, unknown>) => ({
   161→          id: s.id as string,
   162→          type: 'stationery' as ProductType,
   163→          name: s.name as string,
   164→          sku: (s.sku as string) || null,
   165→          isbn: null,
   166→          barcode: (s.barcode as string) || null,
   167→          price: (s.price as number) || 0,
   168→          costPrice: (s.costPrice as number) || 0,
   169→          discount: (s.discount as number) || 0,
   170→          brand: (s.brand as string) || null,
   171→          category: (s.category as { name: string } | null)?.name || null,
   172→        }))
   173→
   174→        let all = [...bookProducts, ...stationeryProducts]
   175→
   176→        // Apply category filter
   177→        if (categoryFilter === 'books') {
   178→          all = all.filter((p) => p.type === 'book')
   179→        } else if (categoryFilter === 'stationery') {
   180→          all = all.filter((p) => p.type === 'stationery')
   181→        } else if (categoryFilter === 'school') {
   182→          all = all.filter((p) => p.category?.toLowerCase().includes('school') || p.category?.toLowerCase().includes('educational'))
   183→        }
   184→
   185→        setProducts(all)
   186→      } catch {
   187→        // silent
   188→      } finally {
   189→        setSearchLoading(false)
   190→      }
   191→    }
   192→
   193→    const debounce = setTimeout(searchProducts, 300)
   194→    return () => clearTimeout(debounce)
   195→  }, [searchTerm, categoryFilter])
   196→
   197→  // Search customers
   198→  React.useEffect(() => {
   199→    if (!customerSearch) {
   200→      setCustomers([])
   201→      return
   202→    }
   203→    const searchCustomers = async () => {
   204→      try {
   205→        const res = await fetch(`/api/customers?search=${encodeURIComponent(customerSearch)}&type=retail&limit=10`)
   206→        if (res.ok) {
   207→          const json = await res.json()
   208→          setCustomers(json.data || [])
   209→        }
   210→      } catch {
   211→        // silent
   212→      }
   213→    }
   214→    const debounce = setTimeout(searchCustomers, 300)
   215→    return () => clearTimeout(debounce)
   216→  }, [customerSearch])
   217→
   218→  // Add to cart
   219→  const addToCart = (product: POSProduct) => {
   220→    if (isViewOnly) return
   221→    setCart((prev) => {
   222→      const existing = prev.find((item) => item.product.id === product.id && item.product.type === product.type)
   223→      if (existing) {
   224→        return prev.map((item) =>
   225→          item.product.id === product.id && item.product.type === product.type
   226→            ? {
   227→                ...item,
   228→                quantity: item.quantity + 1,
   229→                totalPrice: (item.quantity + 1) * item.unitPrice * (1 - item.discount / 100),
   230→              }
   231→            : item
   232→        )
   233→      }
   234→      const unitPrice = product.price * (1 - product.discount / 100)
   235→      return [
   236→        ...prev,
   237→        {
   238→          product,
   239→          quantity: 1,
   240→          unitPrice: product.price,
   241→          discount: product.discount,
   242→          totalPrice: unitPrice,
   243→        },
   244→      ]
   245→    })
   246→  }
   247→
   248→  // Update cart quantity
   249→  const updateQuantity = (index: number, newQty: number) => {
   250→    if (isViewOnly) return
   251→    if (newQty < 1) {
   252→      removeFromCart(index)
   253→      return
   254→    }
   255→    setCart((prev) =>
   256→      prev.map((item, i) =>
   257→        i === index
   258→          ? { ...item, quantity: newQty, totalPrice: newQty * item.unitPrice * (1 - item.discount / 100) }
   259→          : item
   260→      )
   261→    )
   262→  }
   263→
   264→  // Remove from cart
   265→  const removeFromCart = (index: number) => {
   266→    if (isViewOnly) return
   267→    setCart((prev) => prev.filter((_, i) => i !== index))
   268→  }
   269→
   270→  // Clear cart
   271→  const clearCart = () => {
   272→    if (isViewOnly) return
   273→    setCart([])
   274→    setCartDiscount(0)
   275→    setSelectedCustomer(null)
   276→    setCompletedSale(null)
   277→  }
   278→
   279→  // Compute totals
   280→  const subtotal = cart.reduce((s, item) => s + item.totalPrice, 0)
   281→  const discountAmount = discountType === 'percentage' ? subtotal * (cartDiscount / 100) : cartDiscount
   282→  const afterDiscount = Math.max(0, subtotal - discountAmount)
   283→  const taxAmount = afterDiscount * (taxRate / 100)
   284→  const total = afterDiscount + taxAmount
   285→
   286→  // Complete sale
   287→  const handleCompleteSale = async () => {
   288→    if (isViewOnly || cart.length === 0) return
   289→    setCompleting(true)
   290→    setError(null)
   291→    try {
   292→      const saleItems = cart.map((item) => ({
   293→        bookId: item.product.type === 'book' ? item.product.id : undefined,
   294→        stationeryId: item.product.type === 'stationery' ? item.product.id : undefined,
   295→        quantity: item.quantity,
   296→        unitPrice: item.unitPrice,
   297→        discount: item.discount,
   298→      }))
   299→
   300→      const res = await fetch('/api/sales', {
   301→        method: 'POST',
   302→        headers: { 'Content-Type': 'application/json' },
   303→        body: JSON.stringify({
   304→          customerId: selectedCustomer?.id || null,
   305→          discount: discountAmount,
   306→          tax: taxRate,
   307→          paymentMethod,
   308→          paymentStatus: 'paid',
   309→          saleType: 'retail',
   310→          items: saleItems,
   311→        }),
   312→      })
   313→
   314→      if (!res.ok) {
   315→        const errData = await res.json()
   316→        throw new Error(errData.error || 'Failed to complete sale')
   317→      }
   318→
   319→      const json = await res.json()
   320→      const sale = json.data
   321→
   322→      setCompletedSale({
   323→        invoiceNumber: sale.invoiceNumber,
   324→        total: sale.total,
   325→        subtotal: sale.subtotal,
   326→        discount: sale.discount,
   327→        tax: sale.tax,
   328→        paymentMethod: sale.paymentMethod,
   329→        customerName: selectedCustomer?.name || null,
   330→        items: cart.map((item) => ({
   331→          name: item.product.name,
   332→          quantity: item.quantity,
   333→          unitPrice: item.unitPrice,
   334→          totalPrice: item.totalPrice,
   335→        })),
   336→      })
   337→
   338→      // Clear cart
   339→      setCart([])
   340→      setCartDiscount(0)
   341→      setSelectedCustomer(null)
   342→    } catch (err) {
   343→      setError(err instanceof Error ? err.message : 'Failed to complete sale')
   344→    } finally {
   345→      setCompleting(false)
   346→    }
   347→  }
   348→
   349→  // Payment method icons
   350→  const paymentMethods = [
   351→    { value: 'cash', label: 'Cash', icon: Banknote, color: 'macos-green' as const },
   352→    { value: 'card', label: 'Card', icon: CreditCard, color: 'macos-blue' as const },
   353→    { value: 'bank', label: 'Bank', icon: Building, color: 'macos-purple' as const },
   354→    { value: 'wallet', label: 'Wallet', icon: Smartphone, color: 'macos-orange' as const },
   355→  ]
   356→
   357→  // Category filters for MacosFilterBar
   358→  const categoryFilterItems = [
   359→    { id: 'all', label: 'All' },
   360→    { id: 'books', label: 'Books' },
   361→    { id: 'stationery', label: 'Stationery' },
   362→    { id: 'school', label: 'School' },
   363→  ]
   364→
   365→  return (
   366→    <div className="flex flex-col h-full min-h-0">
   367→      {/* ─── Page Header ─── */}
   368→      <MacosPageHeader
   369→        title="Point of Sale"
   370→        subtitle="Process retail transactions — select products, add to cart, and complete sales"
   371→        icon={ShoppingCart}
   372→        iconColor="text-gg-bridge"
   373→        iconBg="bg-gg-bridge/10"
   374→      >
   375→        {isViewOnly && (
   376→          <MacosBadge variant="warning" className="flex items-center gap-1.5">
   377→            <Eye className="size-3" />
   378→            View Only
   379→          </MacosBadge>
   380→        )}
   381→      </MacosPageHeader>
   382→
   383→      {/* ─── Error Alert ─── */}
   384→      {error && (
   385→        <div className="macos-card macos-shadow-card flex items-center gap-2.5 p-3 mb-4 border-macos-red/20 bg-macos-red/5 flex-shrink-0">
   386→          <div className="size-6 rounded-md bg-macos-red/10 flex items-center justify-center flex-shrink-0">
   387→            <AlertCircle className="size-3.5 text-macos-red" />
   388→          </div>
   389→          <span className="text-[13px] font-medium text-macos-red flex-1">{error}</span>
   390→          <button
   391→            className="size-6 rounded-md hover:bg-macos-red/10 flex items-center justify-center transition-colors"
   392→            onClick={() => setError(null)}
   393→          >
   394→            <X className="size-3.5 text-macos-red" />
   395→          </button>
   396→        </div>
   397→      )}
   398→
   399→      {/* ─── Receipt Dialog ─── */}
   400→      <MacosDialog
   401→        open={!!completedSale}
   402→        onOpenChange={(open) => !open && setCompletedSale(null)}
   403→        title="Sale Completed"
   404→        description="Transaction processed successfully"
   405→        className="glass-card"
   406→        footer={
   407→          <Button
   408→            onClick={() => setCompletedSale(null)}
   409→            className="w-full bg-macos-green text-white rounded-lg font-semibold text-[13px] h-10 hover:bg-macos-green/90"
   410→          >
   411→            <Check className="size-4 mr-1.5" />
   412→            Done
   413→          </Button>
   414→        }
   415→      >
   416→        {completedSale && (
   417→          <div className="space-y-4">
   418→            <div className="text-center space-y-1.5 py-3">
   419→              <div className="flex items-center justify-center gap-1.5">
   420→                <Sparkles className="size-3.5 text-gg-golden" />
   421→                <p className="text-[15px] font-bold tracking-tight">BookFlow OS</p>
   422→                <Sparkles className="size-3.5 text-gg-golden" />
   423→              </div>
   424→              <p className="text-[11px] text-muted-foreground font-medium">Invoice {completedSale.invoiceNumber}</p>
   425→              <p className="text-[10px] text-muted-foreground">{new Date().toLocaleString()}</p>
   426→            </div>
   427→
   428→            <div className="rounded-xl bg-secondary/50 p-3 space-y-2">
   429→              {completedSale.items.map((item, i) => (
   430→                <div key={i} className="flex justify-between text-[13px] tracking-tight">
   431→                  <span className="flex-1 truncate mr-3 font-medium">{item.name} <span className="text-muted-foreground">×{item.quantity}</span></span>
   432→                  <span className="font-semibold tabular-nums">${item.totalPrice.toFixed(2)}</span>
   433→                </div>
   434→              ))}
   435→            </div>
   436→
   437→            <Separator className="opacity-50" />
   438→
   439→            <div className="space-y-2 text-[13px] tracking-tight">
   440→              <div className="flex justify-between">
   441→                <span className="text-muted-foreground">Subtotal</span>
   442→                <span className="tabular-nums">${completedSale.subtotal.toFixed(2)}</span>
   443→              </div>
   444→              {completedSale.discount > 0 && (
   445→                <div className="flex justify-between text-macos-green">
   446→                  <span>Discount</span>
   447→                  <span className="tabular-nums">-${completedSale.discount.toFixed(2)}</span>
   448→                </div>
   449→              )}
   450→              {completedSale.tax > 0 && (
   451→                <div className="flex justify-between">
   452→                  <span className="text-muted-foreground">Tax</span>
   453→                  <span className="tabular-nums">${completedSale.tax.toFixed(2)}</span>
   454→                </div>
   455→              )}
   456→              <Separator className="opacity-50" />
   457→              <div className="flex justify-between font-bold text-[15px] tracking-tight">
   458→                <span>Total</span>
   459→                <span className="tabular-nums">${completedSale.total.toFixed(2)}</span>
   460→              </div>
   461→            </div>
   462→
   463→            <div className="flex justify-between text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">
   464→              <span>Paid: {completedSale.paymentMethod}</span>
   465→              {completedSale.customerName && <span>Customer: {completedSale.customerName}</span>}
   466→            </div>
   467→          </div>
   468→        )}
   469→      </MacosDialog>
   470→
   471→      {/* ─── Main POS Split Layout ─── */}
   472→      <div className="flex-1 grid grid-cols-1 lg:grid-cols-5 gap-4 min-h-0 overflow-hidden">
   473→        {/* ─── Left: Products ─── */}
   474→        <div className="lg:col-span-3 flex flex-col min-h-0 gap-4">
   475→          {/* Search + Category Filter Bar */}
   476→          <div className="macos-card macos-shadow-card p-4 flex-shrink-0">
   477→            <MacosSearch
   478→              value={searchTerm}
   479→              onChange={setSearchTerm}
   480→              placeholder="Search by name, ISBN, or barcode…"
   481→              className="h-9"
   482→            />
   483→
   484→            <div className="mt-3">
   485→              <MacosFilterBar
   486→                items={categoryFilterItems}
   487→                activeId={categoryFilter}
   488→                onChange={(id) => setCategoryFilter(id as 'all' | 'books' | 'stationery' | 'school')}
   489→              />
   490→            </div>
   491→          </div>
   492→
   493→          {/* Product Grid */}
   494→          <div className="flex-1 min-h-0 overflow-y-auto custom-scrollbar pr-1">
   495→            {searchLoading ? (
   496→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   497→                {Array.from({ length: 6 }).map((_, i) => (
   498→                  <div key={i} className="macos-card macos-shadow-card p-4 animate-pulse">
   499→                    <div className="h-3 bg-secondary rounded-md w-3/4 mb-3" />
   500→                    <div className="h-2.5 bg-secondary rounded-md w-1/2 mb-3" />
   501→                    <div className="h-4 bg-secondary rounded-md w-1/3" />
   502→                  </div>
   503→                ))}
   504→              </div>
   505→            ) : products.length === 0 ? (
   506→              <MacosEmptyState
   507→                icon={Package}
   508→                title="No products found"
   509→                description="Try a different search or category"
   510→                className="macos-card macos-shadow-card"
   511→              />
   512→            ) : (
   513→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   514→                {products.map((product) => {
   515→                  const inCart = cart.find((c) => c.product.id === product.id && c.product.type === product.type)
   516→                  return (
   517→                    <button
   518→                      key={`${product.type}-${product.id}`}
   519→                      className={`macos-card macos-shadow-card macos-card-hover p-4 text-left w-full cursor-pointer group ${inCart ? 'ring-1 ring-macos-green/30 bg-macos-green/[0.02]' : 'bg-card'} ${isViewOnly ? 'cursor-default' : ''}`}
   520→                      onClick={() => addToCart(product)}
   521→                      disabled={isViewOnly}
   522→                    >
   523→                      <div className="space-y-2">
   524→                        <div className="flex items-start justify-between gap-1.5">
   525→                          <p className="text-[13px] font-semibold leading-snug line-clamp-2 tracking-tight group-hover:text-macos-blue transition-colors">{product.name}</p>
   526→                          <MacosBadge variant={product.type === 'book' ? 'danger' : 'warning'}>
   527→                            {product.type === 'book' ? 'Book' : 'Item'}
   528→                          </MacosBadge>
   529→                        </div>
   530→                        {product.author && (
   531→                          <p className="text-[11px] text-muted-foreground truncate tracking-tight">by {product.author}</p>
   532→                        )}
   533→                        {product.brand && (
   534→                          <p className="text-[11px] text-muted-foreground truncate tracking-tight">{product.brand}</p>
   535→                        )}
   536→                        {product.category && (
   537→                          <MacosBadge>{product.category}</MacosBadge>
   538→                        )}
   539→                        <div className="flex items-center justify-between pt-1">
   540→                          <div>
   541→                            {product.discount > 0 ? (
   542→                              <div className="flex items-center gap-1.5">
   543→                                <span className="text-[11px] text-muted-foreground line-through tracking-tight">${product.price.toFixed(2)}</span>
   544→                                <span className="text-[14px] font-bold text-macos-green tracking-tight">${(product.price * (1 - product.discount / 100)).toFixed(2)}</span>
   545→                              </div>
   546→                            ) : (
   547→                              <span className="text-[14px] font-bold tracking-tight">${product.price.toFixed(2)}</span>
   548→                            )}
   549→                          </div>
   550→                          {inCart ? (
   551→                            <MacosBadge variant="success">×{inCart.quantity}</MacosBadge>
   552→                          ) : !isViewOnly ? (
   553→                            <div className="size-6 rounded-lg bg-macos-blue/10 flex items-center justify-center group-hover:bg-macos-blue/20 transition-colors">
