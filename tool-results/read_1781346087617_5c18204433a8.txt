   150→   150→const getCoverColor = (title: string) => {
   151→   151→  let hash = 0
   152→   152→  for (let i = 0; i < title.length; i++) {
   153→   153→    hash = title.charCodeAt(i) + ((hash << 5) - hash)
   154→   154→  }
   155→   155→  return coverColors[Math.abs(hash) % coverColors.length]
   156→   156→}
