    94→function getDeptBadge(dept: string) {
    95→  const colors = departmentColors[dept]
    96→  if (colors) return `${colors.bg} ${colors.text}`
    97→  return 'bg-secondary text-secondary-foreground'
    98→}
    99→
   100→function getDeptInitialsColor(dept: string) {
   101→  const colors = departmentColors[dept]
   102→  if (colors) return colors
   103→  return { bg: 'bg-secondary', text: 'text-muted-foreground', icon: 'bg-secondary' }
   104→}

   485→
   486→function AddEmployeeDialog({
   487→  open,
   488→  onOpenChange,
   489→  branches,
   490→  onSuccess,
   491→}: {
   492→  open: boolean
   493→  onOpenChange: (open: boolean) => void
   494→  branches: Branch[]
   495→  onSuccess: () => void
   496→}) {
   497→  const [saving, setSaving] = React.useState(false)
   498→  const [form, setForm] = React.useState({
   499→    name: '',
   500→    email: '',
   501→    phone: '',
   502→    department: '',
   503→    position: '',
   504→    salary: '',
   505→    hireDate: '',
   506→    branchId: '',
   507→  })
   508→
   509→  const handleSubmit = async (e: React.FormEvent) => {
   510→    e.preventDefault()
   511→    if (!form.name.trim() || !form.email.trim()) return
   512→    setSaving(true)
   513→    try {
   514→      const res = await fetch('/api/employees', {
   515→        method: 'POST',
   516→        headers: { 'Content-Type': 'application/json' },
   517→        body: JSON.stringify({
