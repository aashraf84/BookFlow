   753→                    className="w-20 rounded-lg bg-secondary/60 border-0 text-[13px] h-7 font-medium focus-visible:ring-1 focus-visible:ring-macos-blue/30"
   754→                  />
   755→                </div>
   756→              </div>
   757→
   758→              {/* Tax */}
   759→              <div className="flex items-center justify-between">
   760→                <Label className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Tax Rate (%)</Label>
   761→                <Input
   762→                  type="number"
   763→                  min="0"
   764→                  step="0.5"
   765→                  value={taxRate || ''}
   766→                  onChange={(e) => setTaxRate(parseFloat(e.target.value) || 0)}
   767→                  placeholder="0"
   768→                  disabled={isViewOnly}
   769→                  className="w-16 text-right rounded-lg bg-secondary/60 border-0 text-[13px] h-7 font-medium focus-visible:ring-1 focus-visible:ring-macos-blue/30"
   770→                />
   771→              </div>
   772→
   773→              <Separator className="opacity-40" />
   774→
   775→              {/* Totals */}
   776→              <div className="space-y-2 text-[13px] tracking-tight">
   777→                <div className="flex justify-between">
   778→                  <span className="text-muted-foreground font-medium">Subtotal</span>
   779→                  <span className="tabular-nums font-medium">${subtotal.toFixed(2)}</span>
   780→                </div>
   781→                {discountAmount > 0 && (
   782→                  <div className="flex justify-between text-macos-green">
   783→                    <span className="font-medium">Discount</span>
   784→                    <span className="tabular-nums font-medium">-${discountAmount.toFixed(2)}</span>
   785→                  </div>
   786→                )}
   787→                {taxAmount > 0 && (
   788→                  <div className="flex justify-between">
   789→                    <span className="text-muted-foreground font-medium">Tax ({taxRate}%)</span>
   790→                    <span className="tabular-nums font-medium">${taxAmount.toFixed(2)}</span>
   791→                  </div>
   792→                )}
   793→                <Separator className="opacity-40" />
   794→                <div className="flex justify-between items-baseline">
   795→                  <span className="font-bold text-[15px] tracking-tight">Total</span>
   796→                  <span className="font-bold text-[20px] tracking-tight tabular-nums">${total.toFixed(2)}</span>
   797→                </div>
   798→              </div>
   799→
   800→              {/* Payment Method */}
   801→              <div className="space-y-2.5">
   802→                <Label className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Payment Method</Label>
   803→                <RadioGroup value={paymentMethod} onValueChange={setPaymentMethod} className="grid grid-cols-4 gap-2">

   806→                    const isSelected = paymentMethod === pm.value
   807→                    const colorMap: Record<string, string> = {
   808→                      'macos-green': 'text-macos-green bg-macos-green/10 ring-macos-green/20',

   812→                    }
   813→                    return (
   814→                      <Label
   815→                        key={pm.value}
   816→                        className={`flex flex-col items-center gap-1.5 rounded-xl p-3 cursor-pointer transition-all ${isSelected ? `ring-1 ${colorMap[pm.color]}` : 'bg-secondary/40 hover:bg-secondary/60 ring-0'}`}
   817→                      >
   818→                        <RadioGroupItem value={pm.value} className="sr-only" />
   819→                        <Icon className={`size-4 ${isSelected ? '' : 'text-muted-foreground'}`} />
   820→                        <span className="text-[10px] font-semibold tracking-wide">{pm.label}</span>
   821→                      </Label>
   822→                    )
   823→                  })}
   824→                </RadioGroup>
   825→              </div>
   826→
   827→              {/* Complete Sale */}
   828→              <Button
   829→                className="w-full bg-gg-bridge hover:bg-gg-bridge/90 text-white rounded-lg font-semibold text-[13px] h-11 shadow-sm disabled:opacity-50"
   830→                onClick={handleCompleteSale}
   831→                disabled={cart.length === 0 || completing || isViewOnly}
   832→              >
   833→                {completing ? (
   834→                  <>
   835→                    <Loader2 className="size-4 mr-2 animate-spin" />
   836→                    Processing…
   837→                  </>
   838→                ) : isViewOnly ? (
   839→                  <>
   840→                    <Eye className="size-4 mr-2" />
   841→                    View Only — Checkout Disabled
   842→                  </>
   843→                ) : (
   844→                  <>
   845→                    <Check className="size-4 mr-2" />
   846→                    Complete Sale — ${total.toFixed(2)}
   847→                  </>
   848→                )}
   849→              </Button>
   850→            </div>
   851→          </div>
   852→        </div>
   853→      </div>
   854→
   855→      {/* ─── Customer Lookup Dialog ─── */}
   856→      <Dialog open={customerLookupOpen} onOpenChange={setCustomerLookupOpen}>
   857→        <DialogContent className="sm:max-w-[420px] rounded-xl macos-shadow-float border-0 bg-card p-0 overflow-hidden">
   858→          <DialogHeader className="p-6 pb-0">
   859→            <DialogTitle className="text-[13px] font-semibold tracking-tight">Select Customer</DialogTitle>
   860→            <DialogDescription className="text-[11px] text-muted-foreground">Search for an existing customer</DialogDescription>
   861→          </DialogHeader>
   862→          <div className="p-6 pt-4 space-y-4">
   863→            <div className="macos-search flex items-center gap-2 px-3 h-9">
   864→              <Search className="size-3.5 text-muted-foreground flex-shrink-0" />
   865→              <input
   866→                type="text"
   867→                placeholder="Search by name, email, or phone…"
   868→                value={customerSearch}
   869→                onChange={(e) => setCustomerSearch(e.target.value)}
   870→                className="flex-1 bg-transparent text-[13px] font-medium placeholder:text-muted-foreground/60 outline-none"
   871→              />
   872→              {customerSearch && (
   873→                <button onClick={() => setCustomerSearch('')} className="size-4 rounded-full bg-muted-foreground/20 flex items-center justify-center hover:bg-muted-foreground/30">
   874→                  <X className="size-2.5 text-background" />
   875→                </button>
   876→              )}
   877→            </div>
   878→            <div className="max-h-64 overflow-y-auto custom-scrollbar space-y-0.5">
   879→              {customers.length === 0 && customerSearch ? (
   880→                <div className="macos-empty py-10">
   881→                  <p className="text-[13px] text-muted-foreground font-medium tracking-tight">No customers found</p>
   882→                </div>
   883→              ) : (
   884→                customers.map((c) => (
   885→                  <button
   886→                    key={c.id}
   887→                    className="w-full flex items-center justify-between rounded-lg p-3 hover:bg-secondary/50 transition-colors text-left group"
   888→                    onClick={() => {
   889→                      setSelectedCustomer(c)
   890→                      setCustomerLookupOpen(false)
   891→                      setCustomerSearch('')
   892→                    }}
   893→                  >
   894→                    <div>
   895→                      <p className="text-[13px] font-semibold tracking-tight">{c.name}</p>
   896→                      <p className="text-[11px] text-muted-foreground tracking-tight">{c.email || c.phone || 'No contact info'}</p>
   897→                    </div>
   898→                    <span className="rounded-md text-[10px] font-bold tracking-wide px-2 py-0.5 bg-gg-golden/10 text-gg-golden">{c.loyaltyPoints} pts</span>
   899→                  </button>
   900→                ))
   901→              )}
   902→            </div>
   903→          </div>
   904→          <DialogFooter className="p-6 pt-0">
   905→            <Button
   906→              variant="ghost"
   907→              className="bg-secondary text-secondary-foreground rounded-lg text-[13px] h-9 font-medium hover:bg-secondary/80"
   908→              onClick={() => {
   909→                setCustomerLookupOpen(false)
   910→                setCustomerSearch('')
   911→                setSelectedCustomer(null)
   912→              }}
   913→            >
   914→              Walk-in Customer
   915→            </Button>
   916→          </DialogFooter>
   917→        </DialogContent>
   918→      </Dialog>
   919→    </div>
