   215→            type: 'book' as const,
   216→            name: b.title as string,
   217→            price: (b.price as number) || 0,
   218→          })),

   220→            id: s.id as string,
   221→            type: 'stationery' as const,
   222→            name: s.name as string,
   223→            price: (s.price as number) || 0,
   224→          })),
   225→        ]
   226→
   227→        setSearchResults(results)
   228→      } catch {
   229→        // silent
   230→      } finally {
   231→        setSearchLoading(false)
   232→      }

   235→    return () => clearTimeout(debounce)

   242→      setOrderItems((prev) =>
   243→        prev.map((i) =>
   244→          i.productId === product.id && i.productType === product.type
   245→            ? { ...i, quantity: i.quantity + 1 }
   246→            : i
   247→        )
   248→      )

   251→        ...prev,
   252→        {
   253→          productType: product.type,

   255→          productName: product.name,
   256→          quantity: 1,
   257→          unitPrice: product.price,
   258→          discount: 0,
   259→        },
   260→      ])
   261→    }
   262→    setProductSearch('')
   263→    setSearchResults([])
   264→  }

   268→    setOrderItems((prev) => prev.filter((_, i) => i !== index))

   289→  }

   292→    const map: Record<string, { className: string; label: string }> = {
   293→      unpaid: { className: 'bg-macos-red/10 text-macos-red', label: 'Unpaid' },
   294→      partial: { className: 'bg-macos-orange/10 text-macos-orange', label: 'Partial' },
   295→      paid: { className: 'bg-macos-green/10 text-macos-green', label: 'Paid' },
   296→    }

   305→  const orderTotal = orderSubtotal - orderDiscountAmt + orderTaxAmt

   308→  const handleCreateOrder = async () => {
   309→    if (orderItems.length === 0) return
   310→    setSaving(true)
   311→    setError(null)
   312→    try {
   313→      const items = orderItems.map((item) => ({
   314→        bookId: item.productType === 'book' ? item.productId : undefined,
   315→        stationeryId: item.productType === 'stationery' ? item.productId : undefined,
   316→        quantity: item.quantity,
   317→        unitPrice: item.unitPrice,
   318→        discount: item.discount,
   319→      }))
   320→
   321→      const res = await fetch('/api/wholesale', {
   322→        method: 'POST',
   323→        headers: { 'Content-Type': 'application/json' },
   324→        body: JSON.stringify({
   325→          customerId: selectedDealerId || undefined,
   326→          dealerName: dealerName || undefined,
   327→          discount: orderDiscountAmt,
   328→          tax: parseFloat(orderTax) || 0,
   329→          paymentStatus: 'unpaid',
   330→          deliveryAddress: deliveryAddress || undefined,
   331→          notes: orderNotes || undefined,
   332→          items,
   333→        }),
   334→      })
   335→
   336→      if (!res.ok) {
   337→        const errData = await res.json()

   339→      }
   340→
   341→      // Reset form
   342→      setOrderDialogOpen(false)
   343→      setSelectedDealerId('')
   344→      setDealerName('')
   345→      setOrderDiscount('')
   346→      setOrderTax('')
   347→      setDeliveryAddress('')
   348→      setOrderNotes('')
   349→      setOrderItems([])
   350→      fetchOrders()
   351→    } catch (err) {
   352→      setError(err instanceof Error ? err.message : 'Failed to create order')
   353→    } finally {
   354→      setSaving(false)
   355→    }
   356→  }
   357→

   359→  const handleStatusUpdate = async (orderId: string, newStatus: string) => {
   360→    setUpdatingStatus(orderId)
   361→    try {
   362→      const res = await fetch('/api/wholesale', {
   363→        method: 'PUT',
   364→        headers: { 'Content-Type': 'application/json' },
   365→        body: JSON.stringify({ id: orderId, status: newStatus }),

   368→      fetchOrders()
   369→      if (detailOrder?.id === orderId) {
   370→        setDetailOrder((prev) => prev ? { ...prev, status: newStatus } : null)
   371→      }
   372→    } catch (err) {
   373→      setError(err instanceof Error ? err.message : 'Failed to update status')
   374→    } finally {
   375→      setUpdatingStatus(null)
   376→    }
   377→  }
   378→
   379→  // View order detail
   380→  const handleViewOrder = (order: WholesaleOrder) => {

   382→    setDetailDialogOpen(true)
   383→  }
   384→

   395→    return 'Unknown Item'
   396→  }
   397→

   399→  const pendingOrders = orders.filter((o) => o.status === 'pending').length
   400→  const shippedOrders = orders.filter((o) => o.status === 'shipped').length
   401→

   405→    { id: 'dealers', label: 'Dealers' },
   406→    { id: 'analytics', label: 'Analytics' },
   407→  ]
   408→

   410→  const statusTabs = [
   411→    { id: 'all', label: 'All' },
   412→    { id: 'pending', label: 'Pending' },

   432→        )}

   439→          {error}
   440→          <Button variant="ghost" size="sm" className="ml-auto h-6 text-[10px] text-macos-red hover:text-macos-red" onClick={() => setError(null)}>

   443→        </div>

   446→      {/* Stats Row */}

   450→            <div className="size-9 rounded-xl bg-gg-bay/10 flex items-center justify-center">
   451→              <Package className="size-4 text-gg-bay" strokeWidth={1.5} />
   452→            </div>
   453→            <div>
   454→              <p className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Total Orders</p>

   457→          </div>
   458→        </div>

   465→              <p className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Pending</p>

   474→            </div>
   475→            <div>
   476→              <p className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Shipped</p>

   478→            </div>
   479→          </div>
   480→        </div>
   481→        <div className="macos-stat macos-shadow-card">

   483→            <div className="size-9 rounded-xl bg-macos-red/10 flex items-center justify-center">

   486→            <div>
   487→              <p className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Outstanding</p>

   506→
   507→      {/* Orders Tab */}
   508→      {activeTab === 'orders' && (
   509→        <div className="space-y-4">
   510→          {/* Status Filter */}

   514→                key={tab.id}
   515→                className={`macos-filter-item ${statusFilter === tab.id ? 'macos-filter-item-active' : ''}`}

   519→              </button>

   522→
   523→          {/* Orders Table */}
   524→          <div className="macos-card macos-shadow-card overflow-hidden">
