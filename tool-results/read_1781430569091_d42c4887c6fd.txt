   469→// ─── Component ────────────────────────────────────────────────────
   470→export function SettingsModule() {
   471→  const { authUser } = useAppStore()
   472→  const isViewOnly = authUser?.isViewOnly ?? false
   473→
   474→  const [activeSection, setActiveSection] = React.useState('general')
   475→  const [saving, setSaving] = React.useState(false)
   476→  const [saved, setSaved] = React.useState(false)
   477→  const [error, setError] = React.useState<string | null>(null)
   478→  const [hasChanges, setHasChanges] = React.useState(false)
   479→
   480→  // Settings state
   481→  const [general, setGeneral] = React.useState<GeneralSettings>(defaultGeneral)
   482→  const [pos, setPos] = React.useState<PosSettings>(defaultPos)
   483→  const [inventory, setInventory] = React.useState<InventorySettings>(defaultInventory)
   484→  const [tax, setTax] = React.useState<TaxSettings>(defaultTax)
   485→  const [sales, setSales] = React.useState<SalesSettings>(defaultSales)
   486→  const [email, setEmail] = React.useState<EmailSettings>(defaultEmail)
   487→  const [notifications, setNotifications] = React.useState<NotificationSettings>(defaultNotification)
   488→  const [security, setSecurity] = React.useState<SecuritySettings>(defaultSecurity)
   489→  const [printing, setPrinting] = React.useState<PrintingSettings>(defaultPrinting)
   490→  const [appearance, setAppearance] = React.useState<AppearanceSettings>(defaultAppearance)
   491→  const [backup, setBackup] = React.useState<BackupSettings>(defaultBackup)
   492→  const [integrations, setIntegrations] = React.useState<IntegrationSettings>(defaultIntegration)
   493→
   494→  // Load settings from API
   495→  React.useEffect(() => {
   496→    async function loadSettings() {
   497→      try {
   498→        const res = await fetch('/api/settings')
   499→        if (res.ok) {
   500→          const data = await res.json()
   501→          if (data.success && data.settings) {
   502→            const s = data.settings
   503→            if (s.general) setGeneral({ ...defaultGeneral, ...s.general })
   504→            if (s.pos) setPos({ ...defaultPos, ...s.pos })
   505→            if (s.inventory) setInventory({ ...defaultInventory, ...s.inventory })
   506→            if (s.tax) setTax({ ...defaultTax, ...s.tax })
   507→            if (s.sales) setSales({ ...defaultSales, ...s.sales })
   508→            if (s.email) setEmail({ ...defaultEmail, ...s.email })
   509→            if (s.notifications) setNotifications({ ...defaultNotification, ...s.notifications })
   510→            if (s.security) setSecurity({ ...defaultSecurity, ...s.security })
   511→            if (s.printing) setPrinting({ ...defaultPrinting, ...s.printing })
   512→            if (s.appearance) setAppearance({ ...defaultAppearance, ...s.appearance })
   513→            if (s.backup) setBackup({ ...defaultBackup, ...s.backup })
   514→            if (s.integrations) setIntegrations({ ...defaultIntegration, ...s.integrations })
   515→          }
   516→        }
   517→      } catch {
   518→        // Settings will use defaults
   519→      }
   520→    }
   521→    loadSettings()
   522→  }, [])
   523→
   524→  // Mark changes
   525→  const markChanged = React.useCallback(() => {
   526→    if (!hasChanges) setHasChanges(true)
   527→    setSaved(false)
   528→  }, [hasChanges])
   529→
   530→  // Save all settings
   531→  const handleSave = async () => {
   532→    setSaving(true)
   533→    setError(null)
   534→    try {
   535→      const res = await fetch('/api/settings', {
   536→        method: 'PUT',
   537→        headers: { 'Content-Type': 'application/json' },
   538→        body: JSON.stringify({
   539→          general, pos, inventory, tax, sales, email,
   540→          notifications, security, printing, appearance, backup, integrations,
   541→        }),
   542→      })
   543→      if (!res.ok) throw new Error('Failed to save settings')
   544→      setSaved(true)
   545→      setHasChanges(false)
   546→      setTimeout(() => setSaved(false), 3000)
   547→    } catch (err) {
   548→      setError(err instanceof Error ? err.message : 'Save failed')
   549→    } finally {
   550→      setSaving(false)
   551→    }
   552→  }
   553→
   554→  // Reset current section
   555→  const handleResetSection = () => {
   556→    switch (activeSection) {
   557→      case 'general': setGeneral(defaultGeneral); break
   558→      case 'pos': setPos(defaultPos); break
   559→      case 'inventory': setInventory(defaultInventory); break
   560→      case 'tax': setTax(defaultTax); break
   561→      case 'sales': setSales(defaultSales); break
   562→      case 'email': setEmail(defaultEmail); break
   563→      case 'notifications': setNotifications(defaultNotification); break
   564→      case 'security': setSecurity(defaultSecurity); break
   565→      case 'printing': setPrinting(defaultPrinting); break
   566→      case 'appearance': setAppearance(defaultAppearance); break
   567→      case 'backup': setBackup(defaultBackup); break
   568→      case 'integrations': setIntegrations(defaultIntegration); break
   569→    }
   570→    markChanged()
   571→  }
   572→
   573→  const currentSectionInfo = settingsSections.find(s => s.id === activeSection)!
   574→  const CurrentIcon = currentSectionInfo.icon
   575→
   576→  return (
   577→    <div className="flex flex-col gap-5">
   578→      {/* Header */}
   579→      <MacosPageHeader
   580→        title="Settings"
   581→        subtitle="Configure your ERP system — manage company, POS, inventory, tax, security, and integrations"
   582→        icon={Settings}
   583→        iconColor="text-gg-bridge"
   584→        iconBg="bg-gg-bridge/10"
   585→      >
   586→        {isViewOnly && (
   587→          <MacosBadge variant="warning" className="flex items-center gap-1.5">
   588→            <Eye className="size-3" />
   589→            View Only
   590→          </MacosBadge>
   591→        )}
   592→        <div className="flex items-center gap-2">
   593→          {hasChanges && (
   594→            <MacosBadge variant="warning">Unsaved changes</MacosBadge>
   595→          )}
   596→          {saved && (
   597→            <MacosBadge variant="success" className="flex items-center gap-1">
   598→              <Check className="size-3" />
   599→              Saved
   600→            </MacosBadge>
   601→          )}
   602→        </div>
   603→      </MacosPageHeader>
   604→
   605→      {/* Error Alert */}
   606→      {error && (
   607→        <div className="macos-card macos-shadow-card flex items-center gap-2.5 p-3 border-macos-red/20 bg-macos-red/5 flex-shrink-0">
   608→          <div className="size-6 rounded-md bg-macos-red/10 flex items-center justify-center flex-shrink-0">
   609→            <AlertCircle className="size-3.5 text-macos-red" />
   610→          </div>
   611→          <span className="text-[13px] font-medium text-macos-red flex-1">{error}</span>
   612→          <button className="size-6 rounded-md hover:bg-macos-red/10 flex items-center justify-center" onClick={() => setError(null)}>
   613→            <X className="size-3.5 text-macos-red" />
   614→          </button>
   615→        </div>
   616→      )}
   617→
   618→      {/* Main Layout */}
   619→      <div className="grid grid-cols-1 lg:grid-cols-4 gap-4">
   620→        {/* Sidebar Navigation */}
   621→        <div className="lg:col-span-1">
   622→          <div className="macos-card macos-shadow-card p-2 sticky top-4">
   623→            <nav className="space-y-0.5">
   624→              {settingsSections.map((section) => {
   625→                const Icon = section.icon
   626→                const isActive = activeSection === section.id
   627→                return (
   628→                  <button
   629→                    key={section.id}
   630→                    onClick={() => setActiveSection(section.id)}
   631→                    className={`w-full flex items-center gap-2.5 rounded-xl px-3 py-2.5 text-left transition-all ${
