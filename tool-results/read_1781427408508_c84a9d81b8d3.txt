   116→// ─── Component ────────────────────────────────────────────────────
   117→export function WholesaleModule() {
   118→  const { authUser } = useAppStore()
   119→  const isViewOnly = authUser?.isViewOnly ?? false
   120→
   121→  const [activeTab, setActiveTab] = React.useState('orders')
   122→  const [orders, setOrders] = React.useState<WholesaleOrder[]>([])
   123→  const [dealers, setDealers] = React.useState<Dealer[]>([])
   124→  const [loading, setLoading] = React.useState(true)
   125→  const [error, setError] = React.useState<string | null>(null)
   126→
   127→  // Status filter
   128→  const [statusFilter, setStatusFilter] = React.useState('all')
   129→
   130→  // New order dialog
   131→  const [orderDialogOpen, setOrderDialogOpen] = React.useState(false)
   132→  const [saving, setSaving] = React.useState(false)
   133→  const [selectedDealerId, setSelectedDealerId] = React.useState('')
   134→  const [dealerName, setDealerName] = React.useState('')
   135→  const [orderDiscount, setOrderDiscount] = React.useState('')
   136→  const [orderTax, setOrderTax] = React.useState('')
   137→  const [deliveryAddress, setDeliveryAddress] = React.useState('')
   138→  const [orderNotes, setOrderNotes] = React.useState('')
   139→  const [orderItems, setOrderItems] = React.useState<OrderFormItem[]>([])
   140→
   141→  // Product search for adding items
   142→  const [productSearch, setProductSearch] = React.useState('')
   143→  const [searchResults, setSearchResults] = React.useState<Array<{ id: string; type: 'book' | 'stationery'; name: string; price: number }>>([])
   144→  const [searchLoading, setSearchLoading] = React.useState(false)
   145→
   146→  // Order detail dialog
   147→  const [detailOrder, setDetailOrder] = React.useState<WholesaleOrder | null>(null)
   148→  const [detailDialogOpen, setDetailDialogOpen] = React.useState(false)
   149→
   150→  // Updating status
   151→  const [updatingStatus, setUpdatingStatus] = React.useState<string | null>(null)
   152→
   153→  // Fetch orders
   154→  const fetchOrders = React.useCallback(async () => {
   155→    setLoading(true)
   156→    setError(null)
   157→    try {
   158→      const params = new URLSearchParams()
   159→      params.set('limit', '50')
   160→      if (statusFilter && statusFilter !== 'all') params.set('status', statusFilter)
   161→      const res = await fetch(`/api/wholesale?${params.toString()}`)
   162→      if (!res.ok) throw new Error('Failed to fetch orders')
   163→      const json = await res.json()
   164→      setOrders(json.data || [])
   165→    } catch (err) {
   166→      setError(err instanceof Error ? err.message : 'Unknown error')
   167→    } finally {
   168→      setLoading(false)
   169→    }
   170→  }, [statusFilter])
   171→
   172→  // Fetch dealers (customers of type wholesale/dealer/distributor)
   173→  const fetchDealers = React.useCallback(async () => {
   174→    try {
   175→      const res = await fetch('/api/customers?type=wholesale&limit=100')
   176→      const res2 = await fetch('/api/customers?type=dealer&limit=100')
   177→      const json1 = res.ok ? await res.json() : { data: [] }
   178→      const json2 = res2.ok ? await res2.json() : { data: [] }
   179→      // Also get distributors
   180→      const res3 = await fetch('/api/customers?type=distributor&limit=100')
   181→      const json3 = res3.ok ? await res3.json() : { data: [] }
   182→      setDealers([...(json1.data || []), ...(json2.data || []), ...(json3.data || [])])
   183→    } catch {
   184→      // silent
   185→    }
   186→  }, [])
   187→
   188→  React.useEffect(() => {
   189→    fetchOrders()
   190→  }, [fetchOrders])
   191→
   192→  React.useEffect(() => {
   193→    fetchDealers()
   194→  }, [fetchDealers])
   195→
   196→  // Product search for order items
   197→  React.useEffect(() => {
   198→    if (!productSearch) {
   199→      setSearchResults([])
   200→      return
   201→    }
   202→    const searchProducts = async () => {
   203→      setSearchLoading(true)
   204→      try {
   205→        const params = new URLSearchParams()
   206→        params.set('search', productSearch)
   207→        params.set('limit', '10')
   208→
   209→        const [booksRes, stationeryRes] = await Promise.all([
   210→          fetch(`/api/books?${params.toString()}`),
   211→          fetch(`/api/stationery?${params.toString()}`),
   212→        ])
   213→
   214→        const bookData = booksRes.ok ? await booksRes.json() : { data: [] }
   215→        const stationeryData = stationeryRes.ok ? await stationeryRes.json() : { data: [] }
   216→
   217→        const results: Array<{ id: string; type: 'book' | 'stationery'; name: string; price: number }> = [
   218→          ...(bookData.data || []).map((b: Record<string, unknown>) => ({
   219→            id: b.id as string,
   220→            type: 'book' as const,
   221→            name: b.title as string,
   222→            price: (b.price as number) || 0,
   223→          })),
   224→          ...(stationeryData.data || []).map((s: Record<string, unknown>) => ({
   225→            id: s.id as string,
   226→            type: 'stationery' as const,
   227→            name: s.name as string,
   228→            price: (s.price as number) || 0,
   229→          })),
   230→        ]
   231→
   232→        setSearchResults(results)
   233→      } catch {
   234→        // silent
   235→      } finally {
   236→        setSearchLoading(false)
   237→      }
   238→    }
   239→    const debounce = setTimeout(searchProducts, 300)
   240→    return () => clearTimeout(debounce)
   241→  }, [productSearch])
   242→
   243→  // Add product to order items
   244→  const addItemToOrder = (product: { id: string; type: 'book' | 'stationery'; name: string; price: number }) => {
   245→    const existing = orderItems.find((i) => i.productId === product.id && i.productType === product.type)
   246→    if (existing) {
   247→      setOrderItems((prev) =>
   248→        prev.map((i) =>
   249→          i.productId === product.id && i.productType === product.type
   250→            ? { ...i, quantity: i.quantity + 1 }
   251→            : i
   252→        )
   253→      )
   254→    } else {
   255→      setOrderItems((prev) => [
   256→        ...prev,
   257→        {
   258→          productType: product.type,
   259→          productId: product.id,
   260→          productName: product.name,
   261→          quantity: 1,
   262→          unitPrice: product.price,
   263→          discount: 0,
   264→        },
   265→      ])
   266→    }
   267→    setProductSearch('')
   268→    setSearchResults([])
   269→  }
   270→
   271→  // Remove order item
   272→  const removeOrderItem = (index: number) => {
   273→    setOrderItems((prev) => prev.filter((_, i) => i !== index))
   274→  }
   275→
   276→  // Update order item
   277→  const updateOrderItem = (index: number, field: keyof OrderFormItem, value: number | string) => {
   278→    setOrderItems((prev) =>
   279→      prev.map((item, i) => (i === index ? { ...item, [field]: value } : item))
   280→    )
   281→  }
   282→
   283→  // Status badges using MacosBadge
   284→  const getStatusBadge = (status: string) => {
   285→    const map: Record<string, { variant: 'warning' | 'info' | 'purple' | 'success' | 'danger'; label: string }> = {
   286→      pending: { variant: 'warning', label: 'Pending' },
   287→      confirmed: { variant: 'info', label: 'Confirmed' },
   288→      shipped: { variant: 'purple', label: 'Shipped' },
   289→      delivered: { variant: 'success', label: 'Delivered' },
   290→      cancelled: { variant: 'danger', label: 'Cancelled' },
   291→    }
   292→    const s = map[status] || { variant: 'default' as const, label: status }
   293→    return <MacosBadge variant={s.variant}>{s.label}</MacosBadge>
   294→  }
   295→
   296→  const getPaymentBadge = (status: string) => {
   297→    const map: Record<string, { variant: 'danger' | 'warning' | 'success'; label: string }> = {
   298→      unpaid: { variant: 'danger', label: 'Unpaid' },
   299→      partial: { variant: 'warning', label: 'Partial' },
   300→      paid: { variant: 'success', label: 'Paid' },
   301→    }
   302→    const s = map[status] || { variant: 'default' as const, label: status }
   303→    return <MacosBadge variant={s.variant}>{s.label}</MacosBadge>
   304→  }
   305→
   306→  // Compute order total
   307→  const orderSubtotal = orderItems.reduce((s, i) => s + i.quantity * i.unitPrice * (1 - i.discount / 100), 0)
   308→  const orderDiscountAmt = parseFloat(orderDiscount) || 0
   309→  const orderTaxAmt = (orderSubtotal - orderDiscountAmt) * ((parseFloat(orderTax) || 0) / 100)
   310→  const orderTotal = orderSubtotal - orderDiscountAmt + orderTaxAmt
   311→
   312→  // Create wholesale order
   313→  const handleCreateOrder = async () => {
   314→    if (orderItems.length === 0) return
   315→    setSaving(true)
   316→    setError(null)
   317→    try {
   318→      const items = orderItems.map((item) => ({
   319→        bookId: item.productType === 'book' ? item.productId : undefined,
   320→        stationeryId: item.productType === 'stationery' ? item.productId : undefined,
   321→        quantity: item.quantity,
   322→        unitPrice: item.unitPrice,
   323→        discount: item.discount,
   324→      }))
   325→
   326→      const res = await fetch('/api/wholesale', {
   327→        method: 'POST',
   328→        headers: { 'Content-Type': 'application/json' },
   329→        body: JSON.stringify({
   330→          customerId: selectedDealerId || undefined,
   331→          dealerName: dealerName || undefined,
   332→          discount: orderDiscountAmt,
   333→          tax: parseFloat(orderTax) || 0,
   334→          paymentStatus: 'unpaid',
   335→          deliveryAddress: deliveryAddress || undefined,
   336→          notes: orderNotes || undefined,
   337→          items,
   338→        }),
   339→      })
   340→
   341→      if (!res.ok) {
   342→        const errData = await res.json()
   343→        throw new Error(errData.error || 'Failed to create order')
   344→      }
   345→
   346→      // Reset form
   347→      setOrderDialogOpen(false)
   348→      setSelectedDealerId('')
   349→      setDealerName('')
   350→      setOrderDiscount('')
   351→      setOrderTax('')
   352→      setDeliveryAddress('')
   353→      setOrderNotes('')
   354→      setOrderItems([])
   355→      fetchOrders()
   356→    } catch (err) {
   357→      setError(err instanceof Error ? err.message : 'Failed to create order')
   358→    } finally {
   359→      setSaving(false)
   360→    }
   361→  }
   362→
   363→  // Update order status
   364→  const handleStatusUpdate = async (orderId: string, newStatus: string) => {
   365→    setUpdatingStatus(orderId)
   366→    try {
   367→      const res = await fetch('/api/wholesale', {
   368→        method: 'PUT',
   369→        headers: { 'Content-Type': 'application/json' },
   370→        body: JSON.stringify({ id: orderId, status: newStatus }),
   371→      })
   372→      if (!res.ok) throw new Error('Failed to update status')
   373→      fetchOrders()
   374→      if (detailOrder?.id === orderId) {
   375→        setDetailOrder((prev) => prev ? { ...prev, status: newStatus } : null)
   376→      }
   377→    } catch (err) {
   378→      setError(err instanceof Error ? err.message : 'Failed to update status')
   379→    } finally {
   380→      setUpdatingStatus(null)
   381→    }
   382→  }
   383→
   384→  // View order detail
   385→  const handleViewOrder = (order: WholesaleOrder) => {
   386→    setDetailOrder(order)
   387→    setDetailDialogOpen(true)
   388→  }
   389→
   390→  // Analytics
   391→  const totalRevenue = orders.filter((o) => o.status === 'delivered').reduce((s, o) => s + o.total, 0)
   392→  const pendingAmount = orders.filter((o) => o.paymentStatus === 'unpaid' || o.paymentStatus === 'partial').reduce((s, o) => s + o.total, 0)
   393→  const topDealers = dealers.sort((a, b) => b.totalSpent - a.totalSpent).slice(0, 5)
   394→  const outstandingReceivables = dealers.reduce((s, d) => s + d.outstanding, 0)
   395→
   396→  // Get item name
   397→  const getItemName = (item: OrderItem): string => {
   398→    if (item.book) return item.book.title
   399→    if (item.stationery) return item.stationery.name
   400→    return 'Unknown Item'
   401→  }
   402→
   403→  // Stats
   404→  const pendingOrders = orders.filter((o) => o.status === 'pending').length
   405→  const shippedOrders = orders.filter((o) => o.status === 'shipped').length
   406→
   407→  // Tabs
   408→  const tabs = [
   409→    { id: 'orders', label: 'Orders' },
   410→    { id: 'dealers', label: 'Dealers' },
   411→    { id: 'analytics', label: 'Analytics' },
   412→  ]
   413→
   414→  // Status filter tabs
   415→  const statusTabs = [
   416→    { id: 'all', label: 'All' },
   417→    { id: 'pending', label: 'Pending' },
   418→    { id: 'confirmed', label: 'Confirmed' },
   419→    { id: 'shipped', label: 'Shipped' },
   420→    { id: 'delivered', label: 'Delivered' },
   421→    { id: 'cancelled', label: 'Cancelled' },
   422→  ]
   423→
   424→  return (
   425→    <div className="space-y-5">
   426→      {/* Page Header */}
   427→      <MacosPageHeader
   428→        title="Wholesale"
   429→        subtitle="Manage bulk orders, dealers & distributor relationships"
   430→        icon={Truck}
   431→        iconColor="text-gg-bay"
   432→        iconBg="bg-gg-bay/10"
   433→      >
   434→        {!isViewOnly && (
   435→          <MacosButton
   436→            variant="primary"
   437→            icon={Plus}
   438→            onClick={() => setOrderDialogOpen(true)}
   439→          >
   440→            New Order
   441→          </MacosButton>
   442→        )}
   443→      </MacosPageHeader>
   444→
   445→      {/* Error Alert */}
   446→      {error && (
   447→        <div className="flex items-center gap-2 macos-card macos-shadow-card p-3 text-[13px] text-macos-red bg-macos-red/5">
   448→          <AlertCircle className="size-3.5 shrink-0" />
   449→          {error}
   450→          <MacosButton variant="ghost" className="ml-auto text-macos-red hover:text-macos-red" onClick={() => setError(null)}>
   451→            Dismiss
   452→          </MacosButton>
   453→        </div>
   454→      )}
   455→
   456→      {/* Stats Row */}
   457→      <MacosStatGrid>
   458→        <MacosStatCard
   459→          label="Total Orders"
   460→          value={orders.length}
   461→          icon={Package}
   462→          iconColor="text-gg-bay"
   463→          iconBg="bg-gg-bay/10"
   464→        />
   465→        <MacosStatCard
   466→          label="Pending"
   467→          value={pendingOrders}
   468→          icon={DollarSign}
   469→          iconColor="text-macos-orange"
   470→          iconBg="bg-macos-orange/10"
   471→        />
   472→        <MacosStatCard
   473→          label="Shipped"
   474→          value={shippedOrders}
   475→          icon={Truck}
   476→          iconColor="text-macos-purple"
   477→          iconBg="bg-macos-purple/10"
   478→        />
   479→        <MacosStatCard
   480→          label="Outstanding"
   481→          value={`$${outstandingReceivables.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`}
   482→          icon={TrendingUp}
   483→          iconColor="text-macos-red"
   484→          iconBg="bg-macos-red/10"
   485→        />
   486→      </MacosStatGrid>
   487→
   488→      {/* Filter Bar — Main Tabs */}
   489→      <MacosFilterBar
   490→        items={tabs}
   491→        activeId={activeTab}
   492→        onChange={setActiveTab}
   493→      />
   494→
   495→      {/* Orders Tab */}
   496→      {activeTab === 'orders' && (
   497→        <div className="space-y-4">
   498→          {/* Status Filter */}
   499→          <MacosFilterBar
   500→            items={statusTabs}
   501→            activeId={statusFilter}
   502→            onChange={setStatusFilter}
   503→          />
   504→
   505→          {/* Orders Table */}
   506→          <div className="macos-card macos-shadow-card overflow-hidden">
   507→            {loading ? (
   508→              <div className="p-4 space-y-3">
   509→                {Array.from({ length: 5 }).map((_, i) => (
   510→                  <Skeleton key={i} className="h-10 w-full" />
   511→                ))}
   512→              </div>
   513→            ) : orders.length === 0 ? (
   514→              <MacosEmptyState
   515→                icon={Package}
   516→                title="No wholesale orders found"
   517→                description="Create your first wholesale order"
   518→              />
   519→            ) : (
   520→              <div className="overflow-x-auto custom-scrollbar">
   521→                <Table>
   522→                  <TableHeader>
   523→                    <TableRow className="border-b border-border/50 hover:bg-transparent">
   524→                      <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9">Order #</TableHead>
   525→                      <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9">Dealer</TableHead>
   526→                      <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9">Status</TableHead>
   527→                      <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9 text-right">Total</TableHead>
   528→                      <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9">Payment</TableHead>
   529→                      <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9 hidden sm:table-cell">Date</TableHead>
   530→                      <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9 text-right">Actions</TableHead>
   531→                    </TableRow>
   532→                  </TableHeader>
   533→                  <TableBody>
   534→                    {orders.map((order) => (
   535→                      <TableRow key={order.id} className="macos-table-row cursor-default border-b border-border/30 last:border-0">
   536→                        <TableCell className="font-mono text-[13px] font-medium tracking-tight py-3">{order.orderNumber}</TableCell>
   537→                        <TableCell className="text-[13px] font-medium tracking-tight py-3">
   538→                          {order.customer?.name || order.dealerName || '—'}
   539→                        </TableCell>
   540→                        <TableCell className="py-3">{getStatusBadge(order.status)}</TableCell>
   541→                        <TableCell className="text-right text-[13px] font-medium tracking-tight py-3">${order.total.toFixed(2)}</TableCell>
   542→                        <TableCell className="py-3">{getPaymentBadge(order.paymentStatus)}</TableCell>
   543→                        <TableCell className="hidden sm:table-cell text-muted-foreground text-[13px] tracking-tight py-3">
   544→                          {new Date(order.createdAt).toLocaleDateString()}
   545→                        </TableCell>
   546→                        <TableCell className="text-right py-3">
   547→                          <div className="flex items-center justify-end gap-0.5">
   548→                            <MacosButton variant="ghost" onClick={() => handleViewOrder(order)} className="h-7">
   549→                              <Eye className="size-3" />
   550→                              View
   551→                            </MacosButton>
   552→                            {!isViewOnly && order.status === 'pending' && (
   553→                              <MacosButton
   554→                                variant="ghost"
   555→                                className="h-7 text-macos-blue hover:text-macos-blue hover:bg-macos-blue/10"
   556→                                onClick={() => handleStatusUpdate(order.id, 'confirmed')}
   557→                                disabled={updatingStatus === order.id}
   558→                              >
   559→                                <CheckCircle2 className="size-3" />
   560→                                Confirm
