   103→   103→function getProductName(item: InventoryItem): string {
   104→   104→  if (item.book) return item.book.title
   105→   105→  if (item.stationery) return item.stationery.name
   106→   106→  return 'Unknown Product'
   107→   107→}
   108→   108→
   109→   109→type StockLevel = 'healthy' | 'moderate' | 'low' | 'critical'
   110→   110→
   111→   111→function getStockLevel(item: InventoryItem): StockLevel {
   112→   112→  if (item.quantity === 0) return 'critical'
   113→   113→  const ratio = item.maxStockLevel > 0 ? item.quantity / item.maxStockLevel : item.quantity / (item.reorderLevel * 3)
   114→   114→  if (item.isLowStock || item.quantity <= item.reorderLevel) return 'low'
   115→   115→  if (ratio < 0.5) return 'moderate'
   116→   116→  return 'healthy'
   117→   117→}
   118→   118→
   119→   119→function getStockLevelColors(level: StockLevel) {
   120→   120→  const map: Record<StockLevel, { bar: string; bg: string; text: string; label: string }> = {
   121→   121→    healthy: { bar: 'bg-macos-green', bg: 'bg-macos-green/10', text: 'text-macos-green', label: 'In Stock' },
   122→   122→    moderate: { bar: 'bg-macos-yellow', bg: 'bg-macos-yellow/10', text: 'text-macos-yellow', label: 'Moderate' },
   123→   123→    low: { bar: 'bg-macos-orange', bg: 'bg-macos-orange/10', text: 'text-macos-orange', label: 'Low Stock' },
   124→   124→    critical: { bar: 'bg-macos-red', bg: 'bg-macos-red/10', text: 'text-macos-red', label: 'Out of Stock' },
   125→   125→  }
   126→   126→  return map[level]
   127→   127→}

   133→   133→    completed: { className: 'bg-macos-green/10 text-macos-green', label: 'Completed' },
   134→   134→    cancelled: { className: 'bg-macos-red/10 text-macos-red', label: 'Cancelled' },
   135→   135→  }

   137→   137→  return (
   138→   138→    <span className={`rounded-md bg-opacity-100 text-[10px] font-semibold tracking-wide px-2 py-0.5 ${s.className}`}>

   141→   141→  )
   142→   142→}

   145→   145→function StockLevelBar({ quantity, maxStockLevel, reorderLevel }: { quantity: number; maxStockLevel: number; reorderLevel: number }) {
   146→   146→  const max = maxStockLevel || reorderLevel * 3
   147→   147→  const pct = max > 0 ? Math.min(100, (quantity / max) * 100) : 0
   148→   148→  const item = { quantity, reorderLevel, maxStockLevel, isLowStock: quantity <= reorderLevel } as InventoryItem
   149→   149→  const level = getStockLevel(item)
   150→   150→  const colors = getStockLevelColors(level)
   151→   151→
   152→   152→  return (
   153→   153→    <div className="flex items-center gap-2.5 min-w-[120px]">
   154→   154→      <div className="flex-1 h-1.5 rounded-full bg-secondary/80 overflow-hidden">
   155→   155→        <div
   156→   156→          className={`h-full rounded-full transition-all duration-500 ease-out ${colors.bar}`}
   157→   157→          style={{ width: `${pct}%` }}
   158→   158→        />
   159→   159→      </div>
   160→   160→      <span className="text-[13px] font-medium tabular-nums tracking-tight w-8 text-right">{quantity}</span>
   161→   161→    </div>
   162→   162→  )
   163→   163→}
