   416→   416→function AccountsTab() {
   417→   417→  const [accounts, setAccounts] = React.useState<Account[]>([])
   418→   418→  const [loading, setLoading] = React.useState(true)
   419→   419→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   420→   420→
   421→   421→  const fetchAccounts = React.useCallback(async () => {
   422→   422→    setLoading(true)
   423→   423→    try {
   424→   424→      const res = await fetch('/api/finance?type=accounts&limit=100')
   425→   425→      const json = await res.json()
   426→   426→      if (json.data) setAccounts(json.data)
   427→   427→    } catch (err) {
   428→   428→      console.error('Failed to fetch accounts:', err)
   429→   429→    } finally {
   430→   430→      setLoading(false)
   431→   431→    }
   432→   432→  }, [])
   433→   433→
   434→   434→  React.useEffect(() => {
   435→   435→    fetchAccounts()
   436→   436→  }, [fetchAccounts])
   437→   437→
   438→   438→  const getTypeColor = (type: string) => {
   439→   439→    switch (type) {
   440→   440→      case 'asset':
   441→   441→        return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400'
   442→   442→      case 'liability':
   443→   443→        return 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400'
   444→   444→      case 'equity':
   445→   445→        return 'bg-teal-100 text-teal-700 dark:bg-teal-950/40 dark:text-teal-400'
   446→   446→      case 'revenue':
   447→   447→        return 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400'
   448→   448→      case 'expense':
   449→   449→        return 'bg-purple-100 text-purple-700 dark:bg-purple-950/40 dark:text-purple-400'
   450→   450→      default:
   451→   451→        return ''
   452→   452→    }
   453→   453→  }
   454→   454→
   455→   455→  const groupedAccounts = accounts.reduce(
   456→   456→    (groups, account) => {
   457→   457→      if (!groups[account.type]) groups[account.type] = []
   458→   458→      groups[account.type].push(account)
   459→   459→      return groups
   460→   460→    },
   461→   461→    {} as Record<string, Account[]>
   462→   462→  )
   463→   463→
   464→   464→  return (
   465→   465→    <div className="space-y-4">
   466→   466→      <div className="flex items-center justify-between">
