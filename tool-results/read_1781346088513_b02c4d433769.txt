   101→function parseItems(items: OrderItem[] | string): OrderItem[] {
   102→  if (typeof items === 'string') {
   103→    try {
   104→      return JSON.parse(items)
   105→    } catch {
   106→      return []
   107→    }
   108→  }
   109→  return items
   110→}

   112→// Delivery Pipeline Step Component
   113→function DeliveryPipelineStep({
   114→  label,
   115→  icon: Icon,
   116→  isActive,
   117→  isCompleted,
   118→  isCurrent,
   119→  color,
   120→  bgColor,
   121→  isLast,
   122→}: {
   123→  label: string
   124→  icon: React.ElementType
   125→  isActive: boolean
   126→  isCompleted: boolean
   127→  isCurrent: boolean
   128→  color: string
   129→  bgColor: string
   130→  isLast: boolean
   131→}) {
   132→  return (
   133→    <div className="flex items-center gap-0 flex-1 min-w-0">
   134→      <div className="flex flex-col items-center gap-1.5 min-w-0 flex-1">
   135→        <div
   136→          className={`relative flex items-center justify-center size-10 rounded-xl transition-all duration-300 ${
   137→            isCurrent
   138→              ? `${bgColor} ring-2 ring-offset-1 ring-offset-background ${color.replace('text-', 'ring-')}`
   139→              : isCompleted
   140→              ? bgColor
   141→              : 'bg-secondary/60'
   142→          }`}
   143→        >
   144→          <Icon className={`size-4 ${isCurrent || isCompleted ? color : 'text-muted-foreground/50'}`} />
   145→          {isCurrent && (
   146→            <span className="absolute -top-0.5 -right-0.5 flex items-center justify-center size-2.5 rounded-full bg-macos-blue animate-pulse" />
   147→          )}
   148→        </div>
   149→        <div className="text-center">
   150→          <p className={`text-[10px] font-semibold uppercase tracking-wider ${isCurrent ? color : isCompleted ? 'text-foreground' : 'text-muted-foreground/60'}`}>
   151→            {label}
   152→          </p>
   153→        </div>
   154→      </div>
   155→      {!isLast && (
   156→        <div className="flex items-center self-start mt-5 -ml-1 -mr-1">
   157→          <div className={`h-[2px] w-full min-w-[24px] rounded-full transition-colors ${
   158→            isCompleted ? 'bg-gg-golden/50' : 'bg-border'
   159→          }`} />
   160→          <ArrowRight className={`size-3 -ml-1 ${isCompleted ? 'text-gg-golden/50' : 'text-muted-foreground/30'}`} />
   161→        </div>
   162→      )}
   163→    </div>
   164→  )
   165→}

   167→// Order detail pipeline for a single order in the detail dialog
   168→function OrderDeliveryPipeline({ status }: { status: string }) {
   169→  const currentIdx = deliveryPipelineStages.indexOf(status)
   170→  const isCancelled = status === 'cancelled'
   171→
   172→  return (
   173→    <div className="py-3">
   174→      {isCancelled ? (
   175→        <div className="flex items-center gap-3 px-4 py-3 rounded-xl bg-macos-red/5 border border-macos-red/10">
   176→          <XCircle className="size-5 text-macos-red" />
   177→          <div>
   178→            <p className="text-[13px] font-semibold text-macos-red">Order Cancelled</p>
   179→            <p className="text-[11px] text-muted-foreground">This order has been cancelled</p>
   180→          </div>
   181→        </div>
   182→      ) : (
   183→        <div className="flex items-start gap-2">
   184→          {deliveryPipelineStages.map((stage, idx) => {
   185→            const config = statusConfig[stage]
   186→            const isLast = idx === deliveryPipelineStages.length - 1
   187→            const isCompleted = idx < currentIdx
   188→            const isCurrent = idx === currentIdx
   189→            return (
   190→              <DeliveryPipelineStep
   191→                key={stage}
   192→                label={config.label}
   193→                icon={config.icon}
   194→                isActive={isCompleted || isCurrent}
   195→                isCompleted={isCompleted}
   196→                isCurrent={isCurrent}
   197→                color={config.color}
   198→                bgColor={config.bgColor}
   199→                isLast={isLast}
   200→              />
   201→            )
   202→          })}
   203→        </div>
   204→      )}
   205→    </div>
   206→  )
   207→}
