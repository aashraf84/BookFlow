   128→// ── Cover colors for thumbnails ────────────────────────────────
   129→const coverColors = [
   130→  'bg-macos-blue', 'bg-macos-green', 'bg-macos-orange',
   131→  'bg-macos-purple', 'bg-macos-pink', 'bg-macos-teal',
   132→  'bg-macos-red', 'bg-macos-indigo',
   133→]
   134→
   135→const getCoverColor = (title: string) => {
   136→  let hash = 0
   137→  for (let i = 0; i < title.length; i++) {
   138→    hash = title.charCodeAt(i) + ((hash << 5) - hash)
   139→  }
   140→  return coverColors[Math.abs(hash) % coverColors.length]
   141→}

   143→// ── Component ────────────────────────────────────────────────────
   144→export default function BookCatalogModule() {
   145→  const [books, setBooks] = useState<Book[]>([])
   146→  const [totalPages, setTotalPages] = useState(1)
   147→  const [total, setTotal] = useState(0)
   148→  const [page, setPage] = useState(1)
   149→  const [loading, setLoading] = useState(true)
   150→  const [search, setSearch] = useState('')
   151→  const [filterCategory, setFilterCategory] = useState('')
   152→  const [filterAuthor, setFilterAuthor] = useState('')
   153→  const [filterPublisher, setFilterPublisher] = useState('')
   154→  const [dialogOpen, setDialogOpen] = useState(false)
   155→  const [editing, setEditing] = useState<Book | null>(null)
   156→  const [form, setForm] = useState<BookFormData>(emptyForm)
   157→  const [saving, setSaving] = useState(false)
   158→  const [deleting, setDeleting] = useState<string | null>(null)
   159→
   160→  // Dropdown data
   161→  const [authors, setAuthors] = useState<Author[]>([])
   162→  const [publishers, setPublishers] = useState<Publisher[]>([])
   163→  const [categories, setCategories] = useState<Category[]>([])
   164→
   165→  const limit = 10
   166→
   167→  // ── Fetch dropdown data ────────────────────────────────────
   168→  useEffect(() => {
   169→    Promise.all([
   170→      fetch('/api/authors?limit=200').then((r) => r.json()),
   171→      fetch('/api/publishers?limit=200').then((r) => r.json()),
   172→      fetch('/api/categories').then((r) => r.json()),
   173→    ]).then(([authorsJson, publishersJson, categoriesJson]) => {
   174→      setAuthors(authorsJson.data || [])
   175→      setPublishers(publishersJson.data || [])
   176→      setCategories(categoriesJson.data || [])
   177→    })
   178→  }, [])
   179→
   180→  // ── Fetch books ────────────────────────────────────────────
   181→  const fetchBooks = useCallback(async () => {
   182→    setLoading(true)
   183→    try {
   184→      const params = new URLSearchParams({
   185→        page: String(page),
   186→        limit: String(limit),
   187→      })
   188→      if (search) params.set('search', search)
   189→      if (filterCategory && filterCategory !== 'all') params.set('categoryId', filterCategory)
   190→      if (filterAuthor && filterAuthor !== 'all') params.set('authorId', filterAuthor)
   191→      if (filterPublisher && filterPublisher !== 'all') params.set('publisherId', filterPublisher)
   192→
   193→      const res = await fetch(`/api/books?${params}`)
   194→      if (!res.ok) throw new Error('Failed to fetch books')
   195→      const json = await res.json()
   196→      setBooks(json.data || [])
   197→      setTotalPages(json.pagination?.totalPages || 1)
   198→      setTotal(json.pagination?.total || 0)
   199→    } catch {
   200→      setBooks([])
   201→    } finally {
   202→      setLoading(false)
   203→    }
   204→  }, [page, search, filterCategory, filterAuthor, filterPublisher])
   205→
   206→  useEffect(() => {
   207→    fetchBooks()
   208→  }, [fetchBooks])
   209→
   210→  // Reset to page 1 when filters change
   211→  useEffect(() => {
   212→    setPage(1)
   213→  }, [search, filterCategory, filterAuthor, filterPublisher])
   214→
   215→  // ── Save book (create or update) ───────────────────────────
   216→  const handleSave = async () => {
   217→    if (!form.title.trim()) return
   218→    setSaving(true)
   219→    try {
   220→      const payload: Record<string, unknown> = {
   221→        title: form.title.trim(),
   222→        subtitle: form.subtitle || null,
   223→        isbn: form.isbn || null,
   224→        authorId: form.authorId || null,
   225→        publisherId: form.publisherId || null,
   226→        categoryId: form.categoryId || null,
   227→        edition: form.edition || null,
   228→        language: form.language || 'English',
   229→        pages: form.pages ? parseInt(form.pages) : null,
   230→        publishYear: form.publishYear ? parseInt(form.publishYear) : null,
   231→        description: form.description || null,
   232→        price: form.price ? parseFloat(form.price) : 0,
   233→        costPrice: form.costPrice ? parseFloat(form.costPrice) : 0,
   234→        discount: form.discount ? parseFloat(form.discount) : 0,
   235→        academicLevel: form.academicLevel || null,
   236→        subject: form.subject || null,
   237→        barcode: form.barcode || null,
   238→      }
   239→
   240→      let res: Response
   241→      if (editing) {
   242→        payload.id = editing.id
   243→        res = await fetch('/api/books', {
   244→          method: 'PUT',
   245→          headers: { 'Content-Type': 'application/json' },
   246→          body: JSON.stringify(payload),
   247→        })
   248→      } else {
   249→        res = await fetch('/api/books', {
   250→          method: 'POST',
   251→          headers: { 'Content-Type': 'application/json' },
   252→          body: JSON.stringify(payload),
   253→        })
   254→      }
   255→
   256→      if (!res.ok) {
   257→        const err = await res.json()
   258→        throw new Error(err.error || 'Failed to save book')
   259→      }
   260→
   261→      setDialogOpen(false)
   262→      setEditing(null)
   263→      setForm(emptyForm)
   264→      fetchBooks()
   265→    } catch (err) {
   266→      alert(err instanceof Error ? err.message : 'Failed to save book')
   267→    } finally {
   268→      setSaving(false)
   269→    }
   270→  }
   271→
   272→  // ── Delete book ────────────────────────────────────────────
   273→  const handleDelete = async (id: string) => {
   274→    if (!confirm('Are you sure you want to delete this book?')) return
   275→    setDeleting(id)
   276→    try {
   277→      const res = await fetch(`/api/books?id=${id}`, { method: 'DELETE' })
   278→      if (!res.ok) {
   279→        const err = await res.json()
   280→        throw new Error(err.error || 'Failed to delete')
   281→      }
   282→      fetchBooks()
   283→    } catch (err) {
   284→      alert(err instanceof Error ? err.message : 'Failed to delete book')
   285→    } finally {
   286→      setDeleting(null)
   287→    }
   288→  }
   289→
   290→  // ── Open dialog for edit ───────────────────────────────────
   291→  const openEdit = (book: Book) => {
   292→    setEditing(book)
   293→    setForm({
   294→      title: book.title,
   295→      subtitle: book.subtitle || '',
   296→      isbn: book.isbn || '',
   297→      authorId: book.authorId || '',
   298→      publisherId: book.publisherId || '',
   299→      categoryId: book.categoryId || '',
   300→      edition: book.edition || '',
   301→      language: book.language || 'English',
   302→      pages: book.pages?.toString() || '',
   303→      publishYear: book.publishYear?.toString() || '',
   304→      description: book.description || '',
   305→      price: book.price?.toString() || '',
   306→      costPrice: book.costPrice?.toString() || '',
   307→      discount: book.discount?.toString() || '',
   308→      academicLevel: book.academicLevel || '',
   309→      subject: book.subject || '',
   310→      barcode: book.barcode || '',
   311→    })
   312→    setDialogOpen(true)
   313→  }
   314→
   315→  // ── Open dialog for add ────────────────────────────────────
   316→  const openAdd = () => {
   317→    setEditing(null)
   318→    setForm(emptyForm)
   319→    setDialogOpen(true)
   320→  }
   321→
   322→  const formatCurrency = (value: number) =>
   323→    new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(value)
   324→
   325→  const hasActiveFilters = search || (filterCategory && filterCategory !== 'all') || (filterAuthor && filterAuthor !== 'all') || (filterPublisher && filterPublisher !== 'all')
   326→
   327→  return (
   328→    <div className="space-y-4">
   329→      {/* ── Header ──────────────────────────────────────────── */}
   330→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   331→        <div className="flex items-center gap-3">
   332→          <h2 className="text-xl font-bold tracking-tight">Book Catalog</h2>
   333→          <Badge className="rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 bg-macos-blue/10 text-macos-blue">
   334→            {total} books
   335→          </Badge>
   336→        </div>
   337→        <Button onClick={openAdd} className="bg-primary text-primary-foreground rounded-lg font-medium text-[13px] h-8">
   338→          <Plus className="size-4 mr-1.5" strokeWidth={1.5} /> Add Book
   339→        </Button>
   340→      </div>
   341→
   342→      {/* ── Search & Filters ────────────────────────────────── */}
   343→      <div className="rounded-2xl macos-shadow bg-card p-4">
   344→        {/* Spotlight-style search */}
   345→        <div className="relative mb-3">
   346→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" strokeWidth={1.5} />
   347→          <Input
   348→            placeholder="Search by title, ISBN, author..."
   349→            value={search}
   350→            onChange={(e) => setSearch(e.target.value)}
   351→            className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8 pl-8 focus-visible:ring-1 focus-visible:ring-primary/30"
   352→          />
   353→        </div>
   354→        {/* Filter pills */}
   355→        <div className="flex flex-wrap gap-2">
   356→          <Select value={filterCategory} onValueChange={setFilterCategory}>
   357→            <SelectTrigger className="rounded-lg bg-secondary/80 border-0 text-[13px] h-7 w-auto min-w-[130px] focus-visible:ring-1 focus-visible:ring-primary/30">
   358→              <Filter className="size-3 mr-1 text-muted-foreground" strokeWidth={1.5} />
   359→              <SelectValue placeholder="All Categories" />
   360→            </SelectTrigger>
   361→            <SelectContent>
   362→              <SelectItem value="all">All Categories</SelectItem>
   363→              {categories.map((cat) => (
   364→                <SelectItem key={cat.id} value={cat.id}>
   365→                  {cat.name}
   366→                </SelectItem>
   367→              ))}
   368→            </SelectContent>
   369→          </Select>
   370→          <Select value={filterAuthor} onValueChange={setFilterAuthor}>
   371→            <SelectTrigger className="rounded-lg bg-secondary/80 border-0 text-[13px] h-7 w-auto min-w-[130px] focus-visible:ring-1 focus-visible:ring-primary/30">
   372→              <Filter className="size-3 mr-1 text-muted-foreground" strokeWidth={1.5} />
   373→              <SelectValue placeholder="All Authors" />
   374→            </SelectTrigger>
   375→            <SelectContent>
   376→              <SelectItem value="all">All Authors</SelectItem>
   377→              {authors.map((a) => (
   378→                <SelectItem key={a.id} value={a.id}>
   379→                  {a.name}
   380→                </SelectItem>
   381→              ))}
   382→            </SelectContent>
   383→          </Select>
   384→          <Select value={filterPublisher} onValueChange={setFilterPublisher}>
   385→            <SelectTrigger className="rounded-lg bg-secondary/80 border-0 text-[13px] h-7 w-auto min-w-[130px] focus-visible:ring-1 focus-visible:ring-primary/30">
   386→              <Filter className="size-3 mr-1 text-muted-foreground" strokeWidth={1.5} />
   387→              <SelectValue placeholder="All Publishers" />
   388→            </SelectTrigger>
   389→            <SelectContent>
   390→              <SelectItem value="all">All Publishers</SelectItem>
   391→              {publishers.map((p) => (
   392→                <SelectItem key={p.id} value={p.id}>
   393→                  {p.name}
   394→                </SelectItem>
   395→              ))}
   396→            </SelectContent>
   397→          </Select>
   398→          {hasActiveFilters && (
   399→            <button
   400→              onClick={() => {
   401→                setSearch('')
   402→                setFilterCategory('')
   403→                setFilterAuthor('')
   404→                setFilterPublisher('')
   405→              }}
   406→              className="macos-pill flex items-center gap-1 bg-secondary/80 px-3 py-1 text-[11px] font-medium text-muted-foreground hover:bg-secondary"
   407→            >
   408→              <X className="size-3" strokeWidth={1.5} /> Clear
   409→            </button>
   410→          )}
   411→        </div>
   412→      </div>
   413→
   414→      {/* ── Books Table ─────────────────────────────────────── */}
   415→      <div className="rounded-2xl macos-shadow bg-card overflow-hidden">
   416→        {loading ? (
   417→          <div className="p-4 space-y-3">
   418→            {Array.from({ length: 5 }).map((_, i) => (
   419→              <Skeleton key={i} className="h-12 w-full" />
   420→            ))}
   421→          </div>
   422→        ) : books.length === 0 ? (
   423→          <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   424→            <BookOpen className="size-10 mb-3 opacity-40" strokeWidth={1.5} />
   425→            <p className="text-[13px] font-semibold">No books found</p>
   426→            <p className="text-[11px] mt-1">Try adjusting your search or filters</p>
   427→          </div>
   428→        ) : (
   429→          <div className="overflow-x-auto">
   430→            <Table>
   431→              <TableHeader>
   432→                <TableRow className="border-b">
   433→                  <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground min-w-[220px]">Title</TableHead>
   434→                  <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Author</TableHead>
   435→                  <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">ISBN</TableHead>
   436→                  <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Category</TableHead>
   437→                  <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Price</TableHead>
   438→                  <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Stock</TableHead>
   439→                  <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Actions</TableHead>
   440→                </TableRow>
   441→              </TableHeader>
   442→              <TableBody>
   443→                {books.map((book) => {
   444→                  const colorClass = getCoverColor(book.title)
   445→                  const initial = book.title.charAt(0).toUpperCase()
   446→                  return (
   447→                    <TableRow key={book.id} className="hover:bg-secondary/50">
   448→                      <TableCell>
   449→                        <div className="flex items-center gap-3">
   450→                          <div className={`size-8 rounded-lg ${colorClass} flex items-center justify-center shrink-0`}>
   451→                            <span className="text-white text-[11px] font-bold">{initial}</span>
   452→                          </div>
   453→                          <div className="min-w-0">
   454→                            <p className="text-[13px] font-medium truncate max-w-[200px]">{book.title}</p>
   455→                            {book.subtitle && (
   456→                              <p className="text-[11px] text-muted-foreground truncate max-w-[200px]">
   457→                                {book.subtitle}
   458→                              </p>
   459→                            )}
   460→                          </div>
   461→                        </div>
   462→                      </TableCell>
   463→                      <TableCell className="text-[13px] text-muted-foreground">
   464→                        {book.author?.name || '—'}
   465→                      </TableCell>
   466→                      <TableCell className="text-[13px] text-muted-foreground font-mono">
   467→                        {book.isbn || '—'}
   468→                      </TableCell>
   469→                      <TableCell>
   470→                        {book.category ? (
   471→                          <Badge className="rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 bg-macos-purple/10 text-macos-purple">
   472→                            {book.category.name}
   473→                          </Badge>
   474→                        ) : (
   475→                          <span className="text-[13px] text-muted-foreground">—</span>
   476→                        )}
   477→                      </TableCell>
   478→                      <TableCell className="text-[13px] text-right font-semibold tracking-tight">
   479→                        {formatCurrency(book.price)}
   480→                      </TableCell>
   481→                      <TableCell>
   482→                        <div className="flex items-center gap-1.5">
   483→                          <div className={`size-2 rounded-full ${
   484→                            book.isActive ? 'bg-macos-green' : 'bg-macos-red'
   485→                          }`} />
   486→                          <span className="text-[11px] text-muted-foreground">
   487→                            {book.isActive ? 'In Stock' : 'Out'}
   488→                          </span>
   489→                        </div>
   490→                      </TableCell>
   491→                      <TableCell className="text-right">
   492→                        <div className="flex justify-end gap-1">
   493→                          <Button
   494→                            variant="ghost"
   495→                            size="icon"
   496→                            className="size-7 rounded-lg"
   497→                            onClick={() => openEdit(book)}
   498→                          >
   499→                            <Pencil className="size-3.5" strokeWidth={1.5} />
   500→                          </Button>
   501→                          <Button
   502→                            variant="ghost"
   503→                            size="icon"
   504→                            className="size-7 rounded-lg text-macos-red hover:text-macos-red"
   505→                            onClick={() => handleDelete(book.id)}
   506→                            disabled={deleting === book.id}
   507→                          >
   508→                            {deleting === book.id ? (
   509→                              <RefreshCw className="size-3.5 animate-spin" strokeWidth={1.5} />
   510→                            ) : (
   511→                              <Trash2 className="size-3.5" strokeWidth={1.5} />
   512→                            )}
   513→                          </Button>
   514→                        </div>
   515→                      </TableCell>
   516→                    </TableRow>
   517→                  )
   518→                })}
   519→              </TableBody>
   520→            </Table>
   521→          </div>
   522→        )}
   523→      </div>
   524→
   525→      {/* ── Pagination ──────────────────────────────────────── */}
   526→      {totalPages > 1 && (
   527→        <div className="flex items-center justify-between">
   528→          <p className="text-[11px] text-muted-foreground">
   529→            Page {page} of {totalPages} · {total} results
   530→          </p>
   531→          <div className="flex gap-1">
   532→            <Button
   533→              variant="secondary"
   534→              size="sm"
   535→              className="rounded-lg h-7 text-[13px]"
   536→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   537→              disabled={page <= 1}
   538→            >
   539→              <ChevronLeft className="size-4" strokeWidth={1.5} />
   540→            </Button>
   541→            <Button
   542→              variant="secondary"
   543→              size="sm"
   544→              className="rounded-lg h-7 text-[13px]"
   545→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   546→              disabled={page >= totalPages}
   547→            >
   548→              <ChevronRight className="size-4" strokeWidth={1.5} />
   549→            </Button>
   550→          </div>
   551→        </div>
   552→      )}
   553→
   554→      {/* ── Add/Edit Dialog ─────────────────────────────────── */}
