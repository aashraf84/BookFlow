   246→function formatCurrency(amount: number, currency = 'USD'): string {
   247→  return new Intl.NumberFormat('en-US', { style: 'currency', currency, minimumFractionDigits: 2 }).format(amount)
   248→}

   250→function formatDate(dateStr: string | null | undefined): string {
   251→  if (!dateStr) return '—'
   252→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
   253→}

   298→function LineItemsEditor({
   299→  items,
   300→  onChange,
   301→}: {
   302→  items: InvoiceItem[] | QuotationItem[]
   303→  onChange: (items: InvoiceItem[] | QuotationItem[]) => void
   304→}) {
   305→  const addItem = () => {
   306→    onChange([...items, { description: '', quantity: 1, unitPrice: 0, discount: 0, total: 0 }])
   307→  }
   308→
   309→  const removeItem = (index: number) => {
   310→    const updated = items.filter((_, i) => i !== index)
   311→    onChange(updated)
   312→  }
   313→
   314→  const updateItem = (index: number, field: string, value: number | string) => {
   315→    const updated = [...items]
   316→    const item = { ...updated[index], [field]: value }
   317→    // Auto-calculate total
   318→    if (field === 'quantity' || field === 'unitPrice' || field === 'discount') {
   319→      const qty = field === 'quantity' ? Number(value) : item.quantity
   320→      const price = field === 'unitPrice' ? Number(value) : item.unitPrice
   321→      const disc = field === 'discount' ? Number(value) : item.discount
   322→      item.total = Math.max(0, qty * price - disc)
   323→    }
   324→    updated[index] = item
   325→    onChange(updated)
   326→  }
   327→
   328→  return (
   329→    <div className="space-y-3">
   330→      <div className="flex items-center justify-between">
   331→        <Label className="text-[12px] font-semibold text-foreground/80">Line Items</Label>
   332→        <MacosButton variant="ghost" icon={Plus} onClick={addItem}>Add Item</MacosButton>
   333→      </div>
   334→      {items.length === 0 && (
   335→        <div className="text-center py-6 text-[12px] text-muted-foreground/50 border border-dashed border-border/50 rounded-lg">
   336→          No items added. Click &quot;Add Item&quot; to add line items.
   337→        </div>
   338→      )}
   339→      {items.map((item, i) => (
   340→        <div key={i} className="grid grid-cols-12 gap-2 items-end p-2 rounded-lg bg-secondary/30 border border-border/30">
   341→          <div className="col-span-4">
   342→            <Label className="text-[10px] text-muted-foreground/60">Description</Label>
   343→            <Input
   344→              value={item.description}
   345→              onChange={(e) => updateItem(i, 'description', e.target.value)}
   346→              placeholder="Item description"
   347→              className="h-7 text-[12px]"
   348→            />
   349→          </div>
   350→          <div className="col-span-2">
   351→            <Label className="text-[10px] text-muted-foreground/60">Qty</Label>
   352→            <Input
   353→              type="number"
   354→              value={item.quantity}
   355→              onChange={(e) => updateItem(i, 'quantity', Number(e.target.value))}
   356→              min={1}
   357→              className="h-7 text-[12px]"
   358→            />
   359→          </div>
   360→          <div className="col-span-2">
   361→            <Label className="text-[10px] text-muted-foreground/60">Unit Price</Label>
   362→            <Input
   363→              type="number"
   364→              value={item.unitPrice}
   365→              onChange={(e) => updateItem(i, 'unitPrice', Number(e.target.value))}
   366→              min={0}
   367→              step={0.01}
   368→              className="h-7 text-[12px]"
   369→            />
   370→          </div>
   371→          <div className="col-span-2">
   372→            <Label className="text-[10px] text-muted-foreground/60">Discount</Label>
   373→            <Input
   374→              type="number"
   375→              value={item.discount}
   376→              onChange={(e) => updateItem(i, 'discount', Number(e.target.value))}
   377→              min={0}
   378→              step={0.01}
   379→              className="h-7 text-[12px]"
   380→            />
   381→          </div>
   382→          <div className="col-span-1">
   383→            <Label className="text-[10px] text-muted-foreground/60">Total</Label>
   384→            <div className="h-7 px-2 flex items-center text-[12px] font-semibold text-foreground/80 rounded-md bg-secondary/50">
   385→              {formatCurrency(item.total)}
   386→            </div>
   387→          </div>
   388→          <div className="col-span-1 flex items-center justify-center">
   389→            <button onClick={() => removeItem(i)} className="p-1 rounded hover:bg-macos-red/10 text-muted-foreground/50 hover:text-macos-red transition-colors">
   390→              <X className="size-3.5" />
   391→            </button>
   392→          </div>
   393→        </div>
   394→      ))}
   395→    </div>
   396→  )
   397→}
