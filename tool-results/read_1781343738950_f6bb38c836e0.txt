    91→    91→function getDeptBadge(dept: string) {
    92→    92→  const colors = departmentColors[dept]
    93→    93→  if (colors) return `${colors.bg} ${colors.text}`
    94→    94→  return 'bg-secondary text-secondary-foreground'
    95→    95→}
    96→    96→
    97→    97→function getDeptInitialsColor(dept: string) {
    98→    98→  const colors = departmentColors[dept]
    99→    99→  if (colors) return colors
   100→   100→  return { bg: 'bg-secondary', text: 'text-muted-foreground', icon: 'bg-secondary' }
   101→   101→}

   427→   427→function AddEmployeeDialog({
