   474→   474→  const [activeSection, setActiveSection] = React.useState('general')

   476→   476→  const [saved, setSaved] = React.useState(false)
   477→   477→  const [error, setError] = React.useState<string | null>(null)
   478→   478→  const [hasChanges, setHasChanges] = React.useState(false)
   479→   479→
   480→   480→  // Settings state
   481→   481→  const [general, setGeneral] = React.useState<GeneralSettings>(defaultGeneral)
   482→   482→  const [pos, setPos] = React.useState<PosSettings>(defaultPos)
   483→   483→  const [inventory, setInventory] = React.useState<InventorySettings>(defaultInventory)
   484→   484→  const [tax, setTax] = React.useState<TaxSettings>(defaultTax)
   485→   485→  const [sales, setSales] = React.useState<SalesSettings>(defaultSales)
   486→   486→  const [email, setEmail] = React.useState<EmailSettings>(defaultEmail)

   488→   488→  const [security, setSecurity] = React.useState<SecuritySettings>(defaultSecurity)
   489→   489→  const [printing, setPrinting] = React.useState<PrintingSettings>(defaultPrinting)
   490→   490→  const [appearance, setAppearance] = React.useState<AppearanceSettings>(defaultAppearance)
   491→   491→  const [backup, setBackup] = React.useState<BackupSettings>(defaultBackup)
   492→   492→  const [integrations, setIntegrations] = React.useState<IntegrationSettings>(defaultIntegration)
   493→   493→
   494→   494→  // Load settings from API
   495→   495→  React.useEffect(() => {
   496→   496→    async function loadSettings() {
   497→   497→      try {
   498→   498→        const res = await fetch('/api/settings')
   499→   499→        if (res.ok) {
   500→   500→          const data = await res.json()
   501→   501→          if (data.success && data.settings) {
   502→   502→            const s = data.settings
   503→   503→            if (s.general) setGeneral({ ...defaultGeneral, ...s.general })
   504→   504→            if (s.pos) setPos({ ...defaultPos, ...s.pos })
   505→   505→            if (s.inventory) setInventory({ ...defaultInventory, ...s.inventory })
   506→   506→            if (s.tax) setTax({ ...defaultTax, ...s.tax })
   507→   507→            if (s.sales) setSales({ ...defaultSales, ...s.sales })
   508→   508→            if (s.email) setEmail({ ...defaultEmail, ...s.email })
   509→   509→            if (s.notifications) setNotifications({ ...defaultNotification, ...s.notifications })
   510→   510→            if (s.security) setSecurity({ ...defaultSecurity, ...s.security })
   511→   511→            if (s.printing) setPrinting({ ...defaultPrinting, ...s.printing })
   512→   512→            if (s.appearance) setAppearance({ ...defaultAppearance, ...s.appearance })
   513→   513→            if (s.backup) setBackup({ ...defaultBackup, ...s.backup })
   514→   514→            if (s.integrations) setIntegrations({ ...defaultIntegration, ...s.integrations })
   515→   515→          }
   516→   516→        }
   517→   517→      } catch {
   518→   518→        // Settings will use defaults
   519→   519→      }
   520→   520→    }
   521→   521→    loadSettings()
   522→   522→  }, [])
   523→   523→
   524→   524→  // Mark changes
   525→   525→  const markChanged = React.useCallback(() => {
   526→   526→    if (!hasChanges) setHasChanges(true)
   527→   527→    setSaved(false)
   528→   528→  }, [hasChanges])
   529→   529→
