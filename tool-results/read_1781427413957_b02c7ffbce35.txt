   141→function CustomersTab() {
   142→  const { authUser } = useAppStore()
   143→  const isViewOnly = authUser?.isViewOnly ?? false
   144→  const [customers, setCustomers] = React.useState<Customer[]>([])
   145→  const [loading, setLoading] = React.useState(true)
   146→  const [search, setSearch] = React.useState('')
   147→  const [typeFilter, setTypeFilter] = React.useState('all')
   148→  const [showAddDialog, setShowAddDialog] = React.useState(false)
   149→  const [selectedCustomer, setSelectedCustomer] = React.useState<Customer | null>(null)
   150→
   151→  const fetchCustomers = React.useCallback(async () => {
   152→    setLoading(true)
   153→    try {
   154→      const params = new URLSearchParams()
   155→      if (search) params.set('search', search)
   156→      if (typeFilter && typeFilter !== 'all') params.set('type', typeFilter)
   157→      params.set('limit', '50')
   158→      const res = await fetch(`/api/customers?${params}`)
   159→      const json = await res.json()
   160→      if (json.data) setCustomers(json.data)
   161→    } catch (err) {
   162→      console.error('Failed to fetch customers:', err)
   163→    } finally {
   164→      setLoading(false)
   165→    }
   166→  }, [search, typeFilter])
   167→
   168→  React.useEffect(() => {
   169→    fetchCustomers()
   170→  }, [fetchCustomers])
   171→
   172→  const handleDelete = async (id: string) => {
   173→    if (!confirm('Are you sure you want to deactivate this customer?')) return
   174→    try {
   175→      await fetch(`/api/customers`, {
   176→        method: 'PUT',
   177→        headers: { 'Content-Type': 'application/json' },
   178→        body: JSON.stringify({ id, isActive: false }),
   179→      })
   180→      fetchCustomers()
   181→    } catch (err) {
   182→      console.error('Failed to deactivate customer:', err)
   183→    }
   184→  }
   185→
   186→  const getTypeBadge = (type: string) => {
   187→    const variants: Record<string, 'success' | 'info' | 'warning' | 'purple' | 'danger'> = {
   188→      retail: 'success',
   189→      wholesale: 'info',
   190→      school: 'warning',
   191→      dealer: 'purple',
   192→      distributor: 'danger',
   193→    }
   194→    return variants[type] || 'default'
   195→  }
   196→
   197→  const getLoyaltyTier = (points: number) => {
   198→    if (points >= 5000) return { name: 'Platinum', variant: 'purple' as const, icon: Crown }
   199→    if (points >= 2000) return { name: 'Gold', variant: 'warning' as const, icon: Award }
   200→    if (points >= 500) return { name: 'Silver', variant: 'default' as const, icon: Star }
   201→    return { name: 'Bronze', variant: 'warning' as const, icon: Gift }
   202→  }
   203→
   204→  const totalCustomers = customers.length
   205→  const totalRevenue = customers.reduce((sum, c) => sum + c.totalSpent, 0)
   206→  const totalOutstanding = customers.reduce((sum, c) => sum + c.outstanding, 0)
   207→  const totalPoints = customers.reduce((sum, c) => sum + c.loyaltyPoints, 0)
   208→
   209→  return (
   210→    <div className="space-y-5">
   211→      {/* Stats Row */}
   212→      <MacosStatGrid>
   213→        <MacosStatCard label="Total Customers" value={totalCustomers} icon={Users} iconColor="text-macos-blue" iconBg="bg-macos-blue/10" />
   214→        <MacosStatCard label="Total Revenue" value={`$${totalRevenue.toLocaleString()}`} icon={DollarSign} iconColor="text-macos-green" iconBg="bg-macos-green/10" />
   215→        <MacosStatCard label="Outstanding" value={`$${totalOutstanding.toLocaleString()}`} icon={TrendingUp} iconColor="text-macos-orange" iconBg="bg-macos-orange/10" />
   216→        <MacosStatCard label="Loyalty Points" value={totalPoints.toLocaleString()} icon={Sparkles} iconColor="text-gg-bridge" iconBg="bg-gg-bridge/10" />
   217→      </MacosStatGrid>
   218→
   219→      {/* Search & Filter Bar */}
   220→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   221→        <div className="flex gap-2 w-full sm:w-auto">
   222→          <MacosSearch value={search} onChange={setSearch} placeholder="Search customers..." className="w-full sm:w-64" />
   223→          <Select value={typeFilter} onValueChange={setTypeFilter}>
   224→            <SelectTrigger className="w-32 rounded-lg bg-secondary/80 border-0 text-[13px] h-8">
   225→              <SelectValue />
   226→            </SelectTrigger>
   227→            <SelectContent>
   228→              <SelectItem value="all">All Types</SelectItem>
   229→              <SelectItem value="retail">Retail</SelectItem>
   230→              <SelectItem value="wholesale">Wholesale</SelectItem>
   231→              <SelectItem value="school">School</SelectItem>
   232→              <SelectItem value="dealer">Dealer</SelectItem>
   233→              <SelectItem value="distributor">Distributor</SelectItem>
   234→            </SelectContent>
   235→          </Select>
   236→        </div>
   237→        {!isViewOnly && (
   238→          <MacosButton variant="primary" icon={Plus} onClick={() => setShowAddDialog(true)}>
   239→            Add Customer
   240→          </MacosButton>
   241→        )}
   242→      </div>
   243→
   244→      {/* Customer List */}
   245→      {loading ? (
   246→        <div className="macos-card macos-shadow-card p-6">
   247→          <div className="space-y-3">
   248→            {[1, 2, 3, 4, 5].map((i) => (
   249→              <div key={i} className="h-14 bg-muted/50 rounded-xl animate-pulse" />
   250→            ))}
   251→          </div>
   252→        </div>
   253→      ) : customers.length === 0 ? (
   254→        <MacosEmptyState
   255→          icon={Contact}
   256→          title="No customers found"
   257→          description="Add your first customer to get started"
   258→        />
   259→      ) : (
   260→        <div className="macos-card macos-shadow-card overflow-hidden">
   261→          <ScrollArea className="max-h-[500px] custom-scrollbar">
   262→            <Table>
   263→              <TableHeader>
   264→                <TableRow className="hover:bg-transparent border-b border-border/50">
   265→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Name</TableHead>
   266→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Contact</TableHead>
   267→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Type</TableHead>
   268→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Tier</TableHead>
   269→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 text-right">Points</TableHead>
   270→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 text-right">Total Spent</TableHead>
   271→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 text-right">Outstanding</TableHead>
   272→                  {!isViewOnly && <TableHead className="w-10"></TableHead>}
   273→                </TableRow>
   274→              </TableHeader>
   275→              <TableBody>
   276→                {customers.map((customer) => {
   277→                  const tier = getLoyaltyTier(customer.loyaltyPoints)
   278→                  const TierIcon = tier.icon
   279→                  return (
   280→                    <TableRow
   281→                      key={customer.id}
   282→                      className="macos-table-row cursor-pointer border-b border-border/30 last:border-0"
   283→                      onClick={() => setSelectedCustomer(customer)}
   284→                    >
   285→                      <TableCell className="font-medium text-[13px] tracking-tight">
   286→                        <div className="flex items-center gap-2.5">
   287→                          <div className="size-8 rounded-full bg-gg-bridge/10 flex items-center justify-center flex-shrink-0">
   288→                            <span className="text-[11px] font-semibold text-gg-bridge">{customer.name.charAt(0).toUpperCase()}</span>
   289→                          </div>
   290→                          <span>{customer.name}</span>
   291→                        </div>
   292→                      </TableCell>
   293→                      <TableCell>
   294→                        <div className="text-[11px] space-y-0.5">
   295→                          {customer.email && (
   296→                            <div className="flex items-center gap-1 text-muted-foreground">
   297→                              <Mail className="size-3" strokeWidth={1.8} /> {customer.email}
   298→                            </div>
   299→                          )}
   300→                          {customer.phone && (
   301→                            <div className="flex items-center gap-1 text-muted-foreground">
   302→                              <Phone className="size-3" strokeWidth={1.8} /> {customer.phone}
   303→                            </div>
   304→                          )}
   305→                        </div>
   306→                      </TableCell>
   307→                      <TableCell>
   308→                        <MacosBadge variant={getTypeBadge(customer.type)}>
   309→                          {customer.type}
   310→                        </MacosBadge>
   311→                      </TableCell>
   312→                      <TableCell>
   313→                        <div className="inline-flex items-center gap-1 macos-badge">
   314→                          <TierIcon className="size-3" strokeWidth={1.8} />
   315→                          {tier.name}
   316→                        </div>
   317→                      </TableCell>
   318→                      <TableCell className="text-right">
   319→                        <span className="text-gg-bridge font-medium text-[13px]">{customer.loyaltyPoints}</span>
   320→                      </TableCell>
   321→                      <TableCell className="text-right text-[13px]">
   322→                        ${customer.totalSpent.toLocaleString()}
   323→                      </TableCell>
   324→                      <TableCell className="text-right">
   325→                        <span className={`text-[13px] ${customer.outstanding > 0 ? 'text-macos-orange font-medium' : ''}`}>
   326→                          ${customer.outstanding.toLocaleString()}
   327→                        </span>
   328→                      </TableCell>
   329→                      {!isViewOnly && (
   330→                        <TableCell>
   331→                          <MacosButton
   332→                            variant="danger"
   333→                            className="h-6 w-6 p-0"
   334→                            onClick={(e) => {
   335→                              e.stopPropagation()
   336→                              handleDelete(customer.id)
   337→                            }}
   338→                          >
   339→                            <Trash2 className="size-3" strokeWidth={1.8} />
   340→                          </MacosButton>
   341→                        </TableCell>
   342→                      )}
   343→                    </TableRow>
   344→                  )
   345→                })}
   346→              </TableBody>
   347→            </Table>
   348→          </ScrollArea>
   349→        </div>
   350→      )}
   351→
   352→      <AddCustomerDialog open={showAddDialog} onOpenChange={setShowAddDialog} onSuccess={fetchCustomers} />
   353→      <CustomerDetailDialog
   354→        customer={selectedCustomer}
   355→        onOpenChange={(open) => !open && setSelectedCustomer(null)}
   356→      />
   357→    </div>
   358→  )
   359→}

   363→function LoyaltyTab() {
   364→  const [tiers, setTiers] = React.useState<LoyaltyTier[]>([])
   365→  const [loading, setLoading] = React.useState(true)
   366→
   367→  const fetchTiers = React.useCallback(async () => {
   368→    setLoading(true)
   369→    try {

   372→        const json = await res.json()
   373→        if (json.data) setTiers(json.data)
   374→      }
   375→    } catch (err) {
   376→      console.error('Failed to fetch loyalty tiers:', err)
   377→    } finally {
   378→      setLoading(false)
   379→    }

   382→  React.useEffect(() => {
   383→    fetchTiers()

   389→    Gold: Crown,
