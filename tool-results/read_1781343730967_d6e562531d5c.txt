   419→function AccountsTab() {
   420→  const [accounts, setAccounts] = React.useState<Account[]>([])
   421→  const [loading, setLoading] = React.useState(true)
   422→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   423→
   424→  const fetchAccounts = React.useCallback(async () => {
   425→    setLoading(true)
   426→    try {
   427→      const res = await fetch('/api/finance?type=accounts&limit=100')
   428→      const json = await res.json()
   429→      if (json.data) setAccounts(json.data)
   430→    } catch (err) {
   431→      console.error('Failed to fetch accounts:', err)
   432→    } finally {
   433→      setLoading(false)
   434→    }
   435→  }, [])
   436→
   437→  React.useEffect(() => {
   438→    fetchAccounts()
   439→  }, [fetchAccounts])
   440→
   441→  const getTypeBadge = (type: string) => {
   442→    switch (type) {
   443→      case 'asset':
   444→        return 'bg-macos-green/10 text-macos-green'
   445→      case 'liability':
   446→        return 'bg-macos-red/10 text-macos-red'
   447→      case 'equity':
   448→        return 'bg-macos-teal/10 text-macos-teal'
   449→      case 'revenue':
   450→        return 'bg-macos-orange/10 text-macos-orange'
   451→      case 'expense':
   452→        return 'bg-macos-purple/10 text-macos-purple'
   453→      default:
   454→        return 'bg-secondary text-secondary-foreground'
   455→    }
   456→  }
   457→
   458→  const groupedAccounts = accounts.reduce(
   459→    (groups, account) => {
   460→      if (!groups[account.type]) groups[account.type] = []
   461→      groups[account.type].push(account)
   462→      return groups
   463→    },
   464→    {} as Record<string, Account[]>
   465→  )
   466→
   467→  return (
   468→    <div className="space-y-4">
   469→      <div className="flex items-center justify-between">
   470→        <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">{accounts.length} accounts</p>
   471→        <Button onClick={() => setShowCreateDialog(true)} className="gap-1.5 bg-primary text-primary-foreground rounded-lg font-medium text-[13px]">
   472→          <Plus className="size-3.5" strokeWidth={2} /> Create Account
   473→        </Button>
   474→      </div>
   475→
   476→      {loading ? (
   477→        <Card className="rounded-2xl macos-shadow border-0">
   478→          <CardContent className="p-6">
   479→            <div className="space-y-3">
   480→              {[1, 2, 3, 4, 5].map((i) => (
   481→                <div key={i} className="h-12 bg-muted rounded-xl animate-pulse" />
   482→              ))}
   483→            </div>
   484→          </CardContent>
   485→        </Card>
   486→      ) : accounts.length === 0 ? (
   487→        <Card className="rounded-2xl macos-shadow border-0">
   488→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   489→            <Wallet className="size-10 text-muted-foreground mb-3" strokeWidth={1.5} />
   490→            <p className="text-[13px] text-muted-foreground">No accounts found</p>
   491→            <p className="text-[11px] text-muted-foreground">Create your chart of accounts</p>
   492→          </CardContent>
   493→        </Card>
   494→      ) : (
   495→        <Card className="rounded-2xl macos-shadow border-0">
   496→          <CardContent className="p-0">
   497→            <ScrollArea className="max-h-[500px]">
   498→              <Table>
   499→                <TableHeader>
   500→                  <TableRow className="hover:bg-transparent">
   501→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Code</TableHead>
   502→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Name</TableHead>
   503→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Type</TableHead>
   504→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Parent</TableHead>
   505→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Balance</TableHead>
   506→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Status</TableHead>
   507→                  </TableRow>
   508→                </TableHeader>
   509→                <TableBody>
   510→                  {Object.entries(groupedAccounts).map(([type, typeAccounts]) => (
   511→                    <React.Fragment key={type}>
   512→                      <TableRow className="bg-secondary/30 hover:bg-secondary/30">
   513→                        <TableCell colSpan={6} className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground py-2">
   514→                          {type} ({typeAccounts.length})
   515→                        </TableCell>

   985→  const totalCredit = lines.reduce((sum, l) => sum + (parseFloat(l.credit) || 0), 0)
   986→  const isBalanced = Math.abs(totalDebit - totalCredit) < 0.01 && totalDebit > 0
   987→

   989→    setLines([...lines, { key: crypto.randomUUID(), accountId: '', debit: '', credit: '', description: '' }])
   990→  }

   993→    if (lines.length > 2) {
   994→      setLines(lines.filter((l) => l.key !== key))
   995→    }
   996→  }
   997→
   998→  const updateLine = (key: string, updates: Partial<JournalLineForm>) => {
   999→    setLines(lines.map((l) => (l.key === key ? { ...l, ...updates } : l)))
  1000→  }
  1001→
  1002→  const handleSubmit = async (e: React.FormEvent) => {
  1003→    e.preventDefault()
  1004→    if (!form.description.trim() || !isBalanced) return
  1005→    setSaving(true)
  1006→    try {
  1007→      const res = await fetch('/api/finance?type=entries', {
