   129→   129→const coverColors = [
   130→   130→  'bg-macos-blue', 'bg-macos-green', 'bg-macos-orange',
   131→   131→  'bg-macos-purple', 'bg-macos-pink', 'bg-macos-teal',

   135→   135→const getCoverColor = (title: string) => {
   136→   136→  let hash = 0
   137→   137→  for (let i = 0; i < title.length; i++) {
   138→   138→    hash = title.charCodeAt(i) + ((hash << 5) - hash)
   139→   139→  }
   140→   140→  return coverColors[Math.abs(hash) % coverColors.length]
   141→   141→}
