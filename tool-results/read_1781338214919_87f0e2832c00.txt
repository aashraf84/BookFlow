   331→   331→function ContractsTab() {
   332→   332→  const [contracts, setContracts] = React.useState<SchoolContract[]>([])
   333→   333→  const [schools, setSchools] = React.useState<School[]>([])
   334→   334→  const [loading, setLoading] = React.useState(true)
   335→   335→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   336→   336→
   337→   337→  const fetchContracts = React.useCallback(async () => {
   338→   338→    setLoading(true)
   339→   339→    try {
   340→   340→      const res = await fetch('/api/schools/contracts')
   341→   341→      if (res.ok) {
   342→   342→        const json = await res.json()
   343→   343→        if (json.data) setContracts(json.data)
   344→   344→      } else {
   345→   345→        // Fallback: fetch schools with contracts
   346→   346→        const schoolRes = await fetch('/api/schools?limit=50')
   347→   347→        const schoolJson = await schoolRes.json()

   349→   349→          const allContracts: SchoolContract[] = []
   350→   350→          schoolJson.data.forEach((s: School & { contracts?: SchoolContract[] }) => {
   351→   351→            if (s.contracts) {
   352→   352→              s.contracts.forEach((c) => allContracts.push({ ...c, school: { id: s.id, name: s.name } }))
   353→   353→            }
   354→   354→          })
   355→   355→          setContracts(allContracts)
   356→   356→        }
   357→   357→      }
   358→   358→    } catch (err) {
   359→   359→      console.error('Failed to fetch contracts:', err)
   360→   360→    } finally {
   361→   361→      setLoading(false)
   362→   362→    }
   363→   363→  }, [])
   364→   364→
   365→   365→  const fetchSchools = React.useCallback(async () => {
   366→   366→    try {
   367→   367→      const res = await fetch('/api/schools?limit=50')
   368→   368→      const json = await res.json()
   369→   369→      if (json.data) setSchools(json.data)
   370→   370→    } catch (err) {
   371→   371→      console.error('Failed to fetch schools:', err)
   372→   372→    }
   373→   373→  }, [])

   376→   376→    fetchContracts()

   378→   378→  }, [fetchContracts, fetchSchools])

   383→   383→        return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400'

   385→   385→        return 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400'
   386→   386→      case 'cancelled':
   387→   387→        return 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400'
   388→   388→      default:
   389→   389→        return ''
   390→   390→    }
   391→   391→  }
   392→   392→
   393→   393→  return (
   394→   394→    <div className="space-y-4">
   395→   395→      <div className="flex items-center justify-between">
   396→   396→        <p className="text-sm text-muted-foreground">{contracts.length} contracts</p>
   397→   397→        <Button onClick={() => setShowCreateDialog(true)} className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">

   408→   408→              ))}
   409→   409→            </div>
   410→   410→          </CardContent>
   411→   411→        </Card>
   412→   412→      ) : contracts.length === 0 ? (
   413→   413→        <Card>
   414→   414→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   415→   415→            <FileText className="size-10 text-muted-foreground mb-3" />

   419→   419→        </Card>
   420→   420→      ) : (

   423→   423→            <ScrollArea className="max-h-[500px]">
