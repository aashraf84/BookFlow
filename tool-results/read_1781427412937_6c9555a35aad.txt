    81→// Helper: Network signal strength indicator
    82→function SignalBars({ strength }: { strength: number }) {
    83→  const bars = [1, 2, 3, 4]
    84→  return (
    85→    <div className="flex items-end gap-[2px]">
    86→      {bars.map((bar) => (
    87→        <div
    88→          key={bar}
    89→          className={`w-[3px] rounded-sm transition-colors ${
    90→            bar <= strength
    91→              ? 'bg-macos-green'
    92→              : 'bg-muted-foreground/20'
    93→          }`}
    94→          style={{ height: `${bar * 3 + 2}px` }}
    95→        />
    96→      ))}
    97→    </div>
    98→  )
    99→}
   100→
   101→// Get signal strength based on metrics
   102→function getSignalStrength(branch: Branch) {
   103→  const employees = branch._count?.employees || 0
   104→  const sales = branch._count?.sales || 0
   105→  if (employees >= 5 && sales >= 10) return 4
   106→  if (employees >= 3 && sales >= 5) return 3
   107→  if (employees >= 1) return 2
   108→  return 1
   109→}
