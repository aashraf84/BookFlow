    94→    94→const getPublisherColor = (name: string) => {
    95→    95→  let hash = 0
    96→    96→  for (let i = 0; i < name.length; i++) {
    97→    97→    hash = name.charCodeAt(i) + ((hash << 5) - hash)
    98→    98→  }
    99→    99→  return publisherColors[Math.abs(hash) % publisherColors.length]
   100→   100→}
