   165→// ==================== SCHOOLS TAB ====================
   166→
   167→function SchoolsTab({ isViewOnly }: { isViewOnly: boolean }) {
   168→  const [schools, setSchools] = React.useState<School[]>([])
   169→  const [loading, setLoading] = React.useState(true)
   170→  const [search, setSearch] = React.useState('')
   171→  const [showAddDialog, setShowAddDialog] = React.useState(false)
   172→  const [selectedSchool, setSelectedSchool] = React.useState<School | null>(null)
   173→
   174→  const fetchSchools = React.useCallback(async () => {
   175→    setLoading(true)
   176→    try {
   177→      const res = await fetch(`/api/schools?search=${encodeURIComponent(search)}&limit=50`)
   178→      const json = await res.json()
   179→      if (json.data) setSchools(json.data)
   180→    } catch (err) {
   181→      console.error('Failed to fetch schools:', err)
   182→    } finally {
   183→      setLoading(false)
   184→    }
   185→  }, [search])
   186→
   187→  React.useEffect(() => {
   188→    fetchSchools()
   189→  }, [fetchSchools])
   190→
   191→  const handleDelete = async (id: string) => {
   192→    if (!confirm('Are you sure you want to deactivate this school?')) return
   193→    try {
   194→      await fetch(`/api/schools?id=${id}`, { method: 'DELETE' })
   195→      fetchSchools()
   196→    } catch (err) {
   197→      console.error('Failed to delete school:', err)
   198→    }
   199→  }
   200→
   201→  const totalContracts = schools.reduce((s, sc) => s + (sc._count?.contracts || 0), 0)
   202→  const totalPackages = schools.reduce((s, sc) => s + (sc._count?.packages || 0), 0)
   203→  const activeSchools = schools.filter((s) => s.isActive).length
   204→
   205→  return (
   206→    <div className="space-y-4">
   207→      {/* Stats */}
   208→      <MacosStatGrid>
   209→        <MacosStatCard label="Total Schools" value={schools.length} icon={GraduationCap} iconColor="text-macos-blue" iconBg="bg-macos-blue/10" />
   210→        <MacosStatCard label="Active" value={activeSchools} icon={SchoolIcon} iconColor="text-macos-green" iconBg="bg-macos-green/10" />
   211→        <MacosStatCard label="Contracts" value={totalContracts} icon={FileText} iconColor="text-macos-purple" iconBg="bg-macos-purple/10" />
   212→        <MacosStatCard label="Packages" value={totalPackages} icon={Package} iconColor="text-macos-teal" iconBg="bg-macos-teal/10" />
   213→      </MacosStatGrid>
   214→
   215→      {/* Search & Add */}
   216→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   217→        <MacosSearch value={search} onChange={setSearch} placeholder="Search schools..." className="w-full sm:w-64" />
   218→        {!isViewOnly && (
   219→          <MacosButton variant="primary" icon={Plus} onClick={() => setShowAddDialog(true)}>
   220→            Add School
   221→          </MacosButton>
   222→        )}
   223→      </div>
   224→
   225→      {loading ? (
   226→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   227→          {[1, 2, 3].map((i) => (
   228→            <div key={i} className="macos-card macos-shadow-card p-4 animate-pulse">
   229→              <div className="h-4 bg-muted rounded w-2/3 mb-3" />
   230→              <div className="h-3 bg-muted rounded w-1/2 mb-2" />
   231→              <div className="h-3 bg-muted rounded w-3/4" />
   232→            </div>
   233→          ))}
   234→        </div>
   235→      ) : schools.length === 0 ? (
   236→        <MacosEmptyState
   237→          icon={GraduationCap}
   238→          title="No schools found"
   239→          description="Add your first school to get started"
   240→        />
   241→      ) : (
   242→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   243→          {schools.map((school) => (
   244→            <div
   245→              key={school.id}
   246→              className="macos-card macos-shadow-card macos-card-hover cursor-pointer group"
   247→              onClick={() => setSelectedSchool(school)}
   248→            >
   249→              <div className="p-4">
   250→                <div className="flex items-start justify-between mb-3">
   251→                  <div className="flex items-center gap-2.5">
   252→                    <div className="size-9 rounded-xl bg-macos-blue/10 flex items-center justify-center">
   253→                      <GraduationCap className="size-4 text-macos-blue" strokeWidth={1.8} />
   254→                    </div>
   255→                    <div>
   256→                      <h3 className="font-semibold text-[13px] leading-tight tracking-tight">{school.name}</h3>
   257→                      {school.academicSession && (
   258→                        <MacosBadge variant="purple" className="mt-0.5">{school.academicSession}</MacosBadge>
   259→                      )}
   260→                    </div>
   261→                  </div>
   262→                  <MacosBadge variant={school.isActive ? 'success' : 'danger'}>
   263→                    {school.isActive ? 'Active' : 'Inactive'}
   264→                  </MacosBadge>
   265→                </div>
   266→                <div className="space-y-1.5 text-[11px] text-muted-foreground">
   267→                  {school.address && (
   268→                    <div className="flex items-center gap-1.5">
   269→                      <MapPin className="size-3 shrink-0" strokeWidth={1.8} />
   270→                      <span className="truncate">{school.address}</span>
   271→                    </div>
   272→                  )}
   273→                  {school.phone && (
   274→                    <div className="flex items-center gap-1.5">
   275→                      <Phone className="size-3 shrink-0" strokeWidth={1.8} />
   276→                      {school.phone}
   277→                    </div>
   278→                  )}
   279→                  {school.contactPerson && (
   280→                    <div className="flex items-center gap-1.5">
   281→                      <User className="size-3 shrink-0" strokeWidth={1.8} />
   282→                      {school.contactPerson}
   283→                    </div>
   284→                  )}
   285→                </div>
   286→                <Separator className="my-3 bg-border/50" />
   287→                <div className="flex items-center justify-between text-[11px]">
   288→                  <span className="text-muted-foreground">
   289→                    {school._count?.contracts || 0} contracts
   290→                  </span>
   291→                  <span className="text-muted-foreground">
   292→                    {school._count?.packages || 0} packages
   293→                  </span>
   294→                  {!isViewOnly && (
   295→                    <MacosButton
   296→                      variant="danger"
   297→                      className="opacity-0 group-hover:opacity-100 h-6 text-[10px]"
   298→                      onClick={(e) => {
   299→                        e.stopPropagation()
   300→                        handleDelete(school.id)
   301→                      }}
   302→                    >
   303→                      <Trash2 className="size-3" strokeWidth={1.8} />
   304→                    </MacosButton>
   305→                  )}
   306→                </div>
   307→              </div>
   308→            </div>
   309→          ))}
   310→        </div>
   311→      )}
   312→
   313→      <AddSchoolDialog open={showAddDialog} onOpenChange={setShowAddDialog} onSuccess={fetchSchools} />
   314→      <SchoolDetailDialog school={selectedSchool} onOpenChange={(open) => !open && setSelectedSchool(null)} onSuccess={fetchSchools} isViewOnly={isViewOnly} />
   315→    </div>
   316→  )
   317→}

   321→function ContractsTab({ isViewOnly }: { isViewOnly: boolean }) {
   322→  const [contracts, setContracts] = React.useState<SchoolContract[]>([])
   323→  const [schools, setSchools] = React.useState<School[]>([])
   324→  const [loading, setLoading] = React.useState(true)
   325→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   326→
   327→  const fetchContracts = React.useCallback(async () => {
   328→    setLoading(true)
   329→    try {
   330→      const res = await fetch('/api/schools/contracts')
   331→      if (res.ok) {
   332→        const json = await res.json()
   333→        if (json.data) setContracts(json.data)
   334→      } else {
   335→        const schoolRes = await fetch('/api/schools?limit=50')
   336→        const schoolJson = await schoolRes.json()
   337→        if (schoolJson.data) {
   338→          const allContracts: SchoolContract[] = []
   339→          schoolJson.data.forEach((s: School & { contracts?: SchoolContract[] }) => {
   340→            if (s.contracts) {
   341→              s.contracts.forEach((c) => allContracts.push({ ...c, school: { id: s.id, name: s.name } }))
   342→            }
   343→          })
   344→          setContracts(allContracts)
   345→        }
   346→      }
   347→    } catch (err) {
   348→      console.error('Failed to fetch contracts:', err)
   349→    } finally {
   350→      setLoading(false)
   351→    }
   352→  }, [])
   353→
   354→  const fetchSchools = React.useCallback(async () => {
   355→    try {
   356→      const res = await fetch('/api/schools?limit=50')
   357→      const json = await res.json()
   358→      if (json.data) setSchools(json.data)
   359→    } catch (err) {
   360→      console.error('Failed to fetch schools:', err)
   361→    }
   362→  }, [])
   363→
   364→  React.useEffect(() => {
   365→    fetchContracts()
   366→    fetchSchools()
   367→  }, [fetchContracts, fetchSchools])
   368→
   369→  const getStatusBadge = (status: string) => {
   370→    switch (status) {
   371→      case 'active': return 'success' as const
   372→      case 'expired': return 'warning' as const
   373→      case 'cancelled': return 'danger' as const
   374→      case 'pending': return 'warning' as const
   375→      case 'confirmed': return 'info' as const
   376→      default: return 'default' as const
   377→    }
   378→  }
   379→
   380→  const totalValue = contracts.reduce((s, c) => s + c.totalValue, 0)
   381→  const activeContracts = contracts.filter((c) => c.status === 'active').length
   382→
   383→  return (
   384→    <div className="space-y-4">
   385→      {/* Stats */}
   386→      <MacosStatGrid className="grid-cols-2 lg:grid-cols-3">
   387→        <MacosStatCard label="Total Contracts" value={contracts.length} icon={FileText} iconColor="text-macos-purple" iconBg="bg-macos-purple/10" />
   388→        <MacosStatCard label="Active" value={activeContracts} icon={Calendar} iconColor="text-macos-green" iconBg="bg-macos-green/10" />
   389→        <MacosStatCard label="Total Value" value={`$${totalValue.toLocaleString()}`} icon={DollarSign} iconColor="text-gg-golden" iconBg="bg-gg-golden/10" />
   390→      </MacosStatGrid>
   391→
   392→      <div className="flex items-center justify-between">
   393→        <p className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">{contracts.length} contracts</p>
   394→        {!isViewOnly && (
   395→          <MacosButton variant="primary" icon={Plus} onClick={() => setShowCreateDialog(true)}>
   396→            Create Contract
   397→          </MacosButton>
   398→        )}
   399→      </div>
   400→
   401→      {loading ? (
   402→        <div className="macos-card macos-shadow-card p-6">
   403→          <div className="space-y-3">
   404→            {[1, 2, 3].map((i) => (
   405→              <div key={i} className="h-12 bg-muted rounded-xl animate-pulse" />
   406→            ))}
   407→          </div>
   408→        </div>
   409→      ) : contracts.length === 0 ? (
   410→        <MacosEmptyState
   411→          icon={FileText}
   412→          title="No contracts found"
   413→          description="Create your first school contract"
   414→        />
   415→      ) : (
   416→        <div className="macos-card macos-shadow-card overflow-hidden">
   417→          <ScrollArea className="max-h-[500px]">
   418→            <Table>
   419→              <TableHeader>
   420→                <TableRow className="border-b border-border/50 hover:bg-transparent">
   421→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Title</TableHead>
   422→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">School</TableHead>
   423→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Session</TableHead>
   424→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Value</TableHead>
   425→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Status</TableHead>
   426→                  <TableHead className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60">Period</TableHead>
   427→                </TableRow>
   428→              </TableHeader>
   429→              <TableBody>
   430→                {contracts.map((contract) => (
   431→                  <TableRow key={contract.id} className="macos-table-row border-b border-border/30 last:border-0">
   432→                    <TableCell className="font-medium text-[13px] tracking-tight">{contract.title}</TableCell>
   433→                    <TableCell className="text-[13px]">{contract.school?.name || '—'}</TableCell>
   434→                    <TableCell className="text-[13px]">{contract.academicSession || '—'}</TableCell>
   435→                    <TableCell className="text-[13px] font-medium">${contract.totalValue.toLocaleString()}</TableCell>
   436→                    <TableCell>
   437→                      <MacosBadge variant={getStatusBadge(contract.status)}>
   438→                        {contract.status}
   439→                      </MacosBadge>
   440→                    </TableCell>
   441→                    <TableCell className="text-[11px] text-muted-foreground">
   442→                      {contract.startDate
   443→                        ? `${new Date(contract.startDate).toLocaleDateString()} - ${contract.endDate ? new Date(contract.endDate).toLocaleDateString() : '...'}`
   444→                        : '—'}
   445→                    </TableCell>
   446→                  </TableRow>
   447→                ))}
   448→              </TableBody>
   449→            </Table>
   450→          </ScrollArea>
   451→        </div>
   452→      )}
   453→
   454→      <CreateContractDialog
   455→        open={showCreateDialog}
   456→        onOpenChange={setShowCreateDialog}
   457→        schools={schools}
   458→        onSuccess={fetchContracts}
   459→      />
   460→    </div>
   461→  )
   462→}

   466→function PackagesTab({ isViewOnly }: { isViewOnly: boolean }) {

   471→  const [loading, setLoading] = React.useState(true)
   472→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   473→  const [selectedPackage, setSelectedPackage] = React.useState<AcademicPackage | null>(null)
   474→
   475→  const fetchPackages = React.useCallback(async () => {
   476→    setLoading(true)
   477→    try {
   478→      const res = await fetch('/api/schools/packages')
   479→      if (res.ok) {
   480→        const json = await res.json()
   481→        if (json.data) setPackages(json.data)
   482→      } else {
   483→        const schoolRes = await fetch('/api/schools?limit=50')
   484→        const schoolJson = await schoolRes.json()

   486→          const allPackages: AcademicPackage[] = []
   487→          schoolJson.data.forEach((s: School & { packages?: AcademicPackage[] }) => {
   488→            if (s.packages) {
   489→              s.packages.forEach((p) => allPackages.push({ ...p, school: { id: s.id, name: s.name } }))
   490→            }
   491→          })
   492→          setPackages(allPackages)
   493→        }
   494→      }
   495→    } catch (err) {
   496→      console.error('Failed to fetch packages:', err)
   497→    } finally {
   498→      setLoading(false)
   499→    }
   500→  }, [])
   501→
   502→  const fetchReferenceData = React.useCallback(async () => {
   503→    try {
   504→      const [schoolsRes, booksRes, stationeryRes] = await Promise.all([
   505→        fetch('/api/schools?limit=50'),
   506→        fetch('/api/books?limit=100'),
   507→        fetch('/api/stationery?limit=100'),
   508→      ])
   509→      const [schoolsJson, booksJson, stationeryJson] = await Promise.all([
   510→        schoolsRes.json(),
   511→        booksRes.json(),
   512→        stationeryRes.json(),
   513→      ])
   514→      if (schoolsJson.data) setSchools(schoolsJson.data)
   515→      if (booksJson.data) setBooks(booksJson.data)
   516→      if (stationeryJson.data) setStationery(stationeryJson.data)
   517→    } catch (err) {
   518→      console.error('Failed to fetch reference data:', err)
   519→    }
   520→  }, [])
   521→
   522→  React.useEffect(() => {
   523→    fetchPackages()
   524→    fetchReferenceData()
   525→  }, [fetchPackages, fetchReferenceData])
   526→
   527→  const totalPackageValue = packages.reduce((s, p) => s + p.totalPrice, 0)
   528→
   529→  return (
   530→    <div className="space-y-4">
   531→      {/* Stats */}
   532→      <MacosStatGrid className="grid-cols-2 lg:grid-cols-3">
   533→        <MacosStatCard label="Packages" value={packages.length} icon={Package} iconColor="text-macos-teal" iconBg="bg-macos-teal/10" />
   534→        <MacosStatCard label="Active" value={packages.filter((p) => p.isActive).length} icon={BookOpen} iconColor="text-macos-green" iconBg="bg-macos-green/10" />
   535→        <MacosStatCard label="Total Value" value={`$${totalPackageValue.toLocaleString()}`} icon={DollarSign} iconColor="text-gg-golden" iconBg="bg-gg-golden/10" />
   536→      </MacosStatGrid>
   537→
   538→      <div className="flex items-center justify-between">
   539→        <p className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">{packages.length} academic packages</p>
   540→        {!isViewOnly && (
   541→          <MacosButton variant="primary" icon={Plus} onClick={() => setShowCreateDialog(true)}>
   542→            Create Package
   543→          </MacosButton>
   544→        )}
   545→      </div>
   546→
   547→      {loading ? (

  1005→interface PackageItemForm {
  1006→  key: string
  1007→  itemType: 'book' | 'stationery'
  1008→  itemId: string
  1009→  quantity: number
  1010→  unitPrice: number
  1011→  name: string
  1012→}
  1013→
  1014→function CreatePackageDialog({
  1015→  open,
  1016→  onOpenChange,
  1017→  schools,
  1018→  books,
  1019→  stationery,
  1020→  onSuccess,
  1021→}: {
  1022→  open: boolean
  1023→  onOpenChange: (open: boolean) => void
  1024→  schools: School[]
  1025→  books: Book[]
  1026→  stationery: StationeryProduct[]
  1027→  onSuccess: () => void
  1028→}) {
  1029→  const [saving, setSaving] = React.useState(false)
  1030→  const [form, setForm] = React.useState({
  1031→    schoolId: '',
  1032→    name: '',
  1033→    grade: '',
  1034→    academicLevel: '',
  1035→    description: '',
  1036→    discount: '',
  1037→  })
  1038→  const [items, setItems] = React.useState<PackageItemForm[]>([
  1039→    { key: crypto.randomUUID(), itemType: 'book', itemId: '', quantity: 1, unitPrice: 0, name: '' },
  1040→  ])
  1041→
  1042→  const totalPrice = items.reduce((sum, item) => sum + item.quantity * item.unitPrice, 0)
  1043→
  1044→  const addItem = () => {
  1045→    setItems([
  1046→      ...items,
  1047→      { key: crypto.randomUUID(), itemType: 'book', itemId: '', quantity: 1, unitPrice: 0, name: '' },
  1048→    ])
  1049→  }
  1050→
  1051→  const removeItem = (key: string) => {
  1052→    if (items.length > 1) {
  1053→      setItems(items.filter((i) => i.key !== key))
  1054→    }
  1055→  }
  1056→
  1057→  const updateItem = (key: string, updates: Partial<PackageItemForm>) => {
  1058→    setItems(
  1059→      items.map((item) => {
  1060→        if (item.key !== key) return item
  1061→        const updated = { ...item, ...updates }
  1062→        if (updates.itemId) {
  1063→          if (updated.itemType === 'book') {
  1064→            const book = books.find((b) => b.id === updates.itemId)
  1065→            if (book) {
  1066→              updated.unitPrice = book.price
  1067→              updated.name = book.title
  1068→            }
  1069→          } else {
  1070→            const product = stationery.find((s) => s.id === updates.itemId)
  1071→            if (product) {
  1072→              updated.unitPrice = product.price
  1073→              updated.name = product.name
  1074→            }
  1075→          }
  1076→        }
  1077→        return updated
  1078→      })
  1079→    )
  1080→  }
  1081→
