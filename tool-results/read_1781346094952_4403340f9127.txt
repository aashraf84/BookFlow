    94→    94→function getDeptBadge(dept: string) {
    95→    95→  const colors = departmentColors[dept]
    96→    96→  if (colors) return `${colors.bg} ${colors.text}`
    97→    97→  return 'bg-secondary text-secondary-foreground'
    98→    98→}
    99→    99→
   100→   100→function getDeptInitialsColor(dept: string) {
   101→   101→  const colors = departmentColors[dept]
   102→   102→  if (colors) return colors
   103→   103→  return { bg: 'bg-secondary', text: 'text-muted-foreground', icon: 'bg-secondary' }
   104→   104→}
