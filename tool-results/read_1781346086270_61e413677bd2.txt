   101→const statusWorkflow = ['submitted', 'reviewed', 'approved', 'published']
   102→const rejectionStates = ['rejected']
   103→
   104→function getNextStatus(current: string): string[] {
   105→  if (current === 'rejected') return []
   106→  const idx = statusWorkflow.indexOf(current)
   107→  if (idx === -1) return []
   108→  if (idx === statusWorkflow.length - 1) return []
   109→  return [statusWorkflow[idx + 1]]
   110→}

   112→// Pipeline step component for manuscript workflow visualization
   113→function PipelineStep({
   114→  label,
   115→  icon: Icon,
   116→  isActive,
   117→  isCompleted,
   118→  count,
   119→  color,
   120→  bgColor,
   121→  isLast,
   122→}: {
   123→  label: string
   124→  icon: React.ElementType
   125→  isActive: boolean
   126→  isCompleted: boolean
   127→  count: number
   128→  color: string
   129→  bgColor: string
   130→  isLast: boolean
   131→}) {
   132→  return (
   133→    <div className="flex items-center gap-0 flex-1 min-w-0">
   134→      <div className="flex flex-col items-center gap-1.5 min-w-0 flex-1">
   135→        <div
   136→          className={`relative flex items-center justify-center size-10 rounded-xl transition-all duration-300 ${
   137→            isActive
   138→              ? `${bgColor} ring-2 ring-offset-1 ring-offset-background ${color.replace('text-', 'ring-')}`
   139→              : isCompleted
   140→              ? bgColor
   141→              : 'bg-secondary/60'
   142→          }`}
   143→        >
   144→          <Icon className={`size-4 ${isActive || isCompleted ? color : 'text-muted-foreground/50'}`} />
   145→          {count > 0 && (
   146→            <span className={`absolute -top-1.5 -right-1.5 flex items-center justify-center min-w-[18px] h-[18px] rounded-full text-[9px] font-bold px-1 ${
   147→              isActive ? 'bg-macos-blue text-white' : isCompleted ? bgColor + ' ' + color : 'bg-secondary text-muted-foreground'
   148→            }`}>
   149→              {count}
   150→            </span>
   151→          )}
   152→        </div>
   153→        <div className="text-center">
   154→          <p className={`text-[10px] font-semibold uppercase tracking-wider ${isActive ? color : isCompleted ? 'text-foreground' : 'text-muted-foreground/60'}`}>
   155→            {label}
   156→          </p>
   157→        </div>
   158→      </div>
   159→      {!isLast && (
   160→        <div className="flex items-center pt-[-16px] self-start mt-5 -ml-1 -mr-1">
   161→          <div className={`h-[2px] w-full min-w-[24px] rounded-full transition-colors ${
   162→            isCompleted ? 'bg-macos-green/40' : 'bg-border'
   163→          }`} />
   164→          <ArrowRight className={`size-3 -ml-1 ${isCompleted ? 'text-macos-green/40' : 'text-muted-foreground/30'}`} />
   165→        </div>
   166→      )}
   167→    </div>
   168→  )
   169→}

   171→export function PublishingModule() {

   173→  const authUser = useAppStore((s) => s.authUser)
   174→  const isViewOnly = authUser?.isViewOnly ?? false
   175→
   176→  const [manuscripts, setManuscripts] = React.useState<Manuscript[]>([])
   177→  const [printOrders, setPrintOrders] = React.useState<PrintOrder[]>([])
   178→  const [books, setBooks] = React.useState<BookOption[]>([])

   190→  const fetchManuscripts = React.useCallback(async () => {

   198→    }
   199→  }, [toast])
   200→

   202→    try {
   203→      const res = await fetch('/api/publishing?type=print-orders')

   205→      const json = await res.json()
   206→      setPrintOrders(json.data || [])
   207→    } catch {
   208→      toast({ title: 'Error', description: 'Failed to fetch print orders', variant: 'destructive' })
   209→    }
   210→  }, [toast])

   217→      setBooks(json.data || [])

   227→      setLoading(false)
   228→    }
   229→    loadAll()
   230→  }, [fetchManuscripts, fetchPrintOrders, fetchBooks])
   231→

   233→    if (!msForm.title) {
   234→      toast({ title: 'Validation Error', description: 'Title is required', variant: 'destructive' })
   235→      return
   236→    }
   237→    try {
   238→      setSubmitting(true)
   239→      const res = await fetch('/api/publishing?type=manuscripts', {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify(msForm),

   253→    }

   258→      toast({ title: 'Validation Error', description: 'Book and quantity are required', variant: 'destructive' })

   264→        method: 'POST',
   265→        headers: { 'Content-Type': 'application/json' },

   267→          bookId: poForm.bookId,
   268→          quantity: parseInt(poForm.quantity),
   269→          costPerUnit: parseFloat(poForm.costPerUnit) || 0,

   276→      fetchPrintOrders()

   278→      toast({ title: 'Error', description: 'Failed to create print order', variant: 'destructive' })
   279→    } finally {
   280→      setSubmitting(false)
   281→    }
   282→  }
   283→

   286→      const res = await fetch('/api/publishing?type=manuscripts', {
   287→        method: 'PUT',
   288→        headers: { 'Content-Type': 'application/json' },

   295→      toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })

   298→
   299→  const handlePrintOrderStatus = async (order: PrintOrder, newStatus: string) => {

   302→        method: 'PUT',
   303→        headers: { 'Content-Type': 'application/json' },
   304→        body: JSON.stringify({ id: order.id, status: newStatus }),
   305→      })

   308→      fetchPrintOrders()

   310→      toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })
   311→    }
   312→  }
   313→
   314→  const msByStatus = manuscripts.reduce<Record<string, number>>((acc, m) => {
   315→    acc[m.status] = (acc[m.status] || 0) + 1
   316→    return acc

   318→
   319→  const totalPrintCost = printOrders.reduce((sum, o) => sum + o.totalCost, 0)
   320→

   325→      (m.author && m.author.toLowerCase().includes(searchQuery.toLowerCase()))

   328→  })
   329→
   330→  const tabs = [
   331→    { id: 'manuscripts', label: 'Manuscripts' },
   332→    { id: 'print-orders', label: 'Print Orders' },
   333→  ]
   334→
   335→  return (
   336→    <div className="space-y-6">
   337→      {/* Page Title */}
   338→      <div>
   339→        <h1 className="macos-page-title">Publishing</h1>
   340→        <p className="macos-page-subtitle">Manage manuscripts, reviews, and print production</p>
   341→      </div>
   342→
   343→      {/* Summary Metrics */}
   344→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-5">
   345→        <div className="macos-card macos-shadow-card p-4">
   346→          <div className="flex items-center justify-between">
   347→            <div>
   348→              <p className="text-[10px] uppercase font-semibold tracking-wider text-muted-foreground">Manuscripts</p>
   349→              <p className="text-2xl font-bold tracking-tight mt-0.5">{manuscripts.length}</p>
   350→            </div>
   351→            <div className="rounded-xl bg-macos-blue/10 p-2.5">
   352→              <FileText className="size-5 text-macos-blue" />
   353→            </div>
   354→          </div>
   355→        </div>
   356→        <div className="macos-card macos-shadow-card p-4">
   357→          <div className="flex items-center justify-between">
   358→            <div>
   359→              <p className="text-[10px] uppercase font-semibold tracking-wider text-muted-foreground">Submitted</p>
   360→              <p className="text-2xl font-bold tracking-tight mt-0.5">{msByStatus['submitted'] || 0}</p>
   361→            </div>
   362→            <div className="rounded-xl bg-macos-orange/10 p-2.5">
   363→              <Clock className="size-5 text-macos-orange" />
   364→            </div>
   365→          </div>

   370→              <p className="text-[10px] uppercase font-semibold tracking-wider text-muted-foreground">Published</p>
   371→              <p className="text-2xl font-bold tracking-tight mt-0.5">{msByStatus['published'] || 0}</p>
   372→            </div>
   373→            <div className="rounded-xl bg-macos-teal/10 p-2.5">
   374→              <Globe className="size-5 text-macos-teal" />
   375→            </div>
   376→          </div>
   377→        </div>
   378→        <div className="macos-card macos-shadow-card p-4">
   379→          <div className="flex items-center justify-between">
   380→            <div>
   381→              <p className="text-[10px] uppercase font-semibold tracking-wider text-muted-foreground">Print Orders</p>
   382→              <p className="text-2xl font-bold tracking-tight mt-0.5">{printOrders.length}</p>

   384→            <div className="rounded-xl bg-macos-purple/10 p-2.5">
   385→              <Printer className="size-5 text-macos-purple" />
   386→            </div>
   387→          </div>
   388→        </div>

   422→                label={config.label}

   442→          </div>
   443→        )}
   444→      </div>
   445→
   446→      {/* Header + Filter Bar */}
   447→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   448→        <div className="macos-filter-bar">
   449→          {tabs.map((tab) => (
   450→            <button
   451→              key={tab.id}
   452→              onClick={() => setActiveTab(tab.id)}
   453→              className={`macos-filter-item ${activeTab === tab.id ? 'macos-filter-item-active' : ''}`}
   454→            >
   455→              {tab.label}
   456→            </button>
   457→          ))}

   459→        <div className="flex items-center gap-2">

   465→                placeholder="Search manuscripts..."
   466→                value={searchQuery}
   467→                onChange={(e) => setSearchQuery(e.target.value)}
   468→                className="border-0 bg-transparent h-6 text-[13px] p-0 shadow-none focus-visible:ring-0 placeholder:text-muted-foreground/60"
   469→              />
   470→            </div>
   471→          )}
   472→          {/* Status Filter */}
   473→          {activeTab === 'manuscripts' && (
   474→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   475→              <SelectTrigger className="w-[140px] h-8 text-[13px] rounded-lg bg-secondary/80 border-0">
   476→                <SelectValue placeholder="Filter status" />
   477→              </SelectTrigger>
   478→              <SelectContent>
   479→                <SelectItem value="all">All Status</SelectItem>
   480→                <SelectItem value="submitted">Submitted</SelectItem>
   481→                <SelectItem value="reviewed">Reviewed</SelectItem>
   482→                <SelectItem value="approved">Approved</SelectItem>
   483→                <SelectItem value="published">Published</SelectItem>
   484→                <SelectItem value="rejected">Rejected</SelectItem>
   485→              </SelectContent>
   486→            </Select>
   487→          )}
   488→          {/* CRUD Buttons - hidden for view-only */}
   489→          {!isViewOnly && (
   490→            <>
   491→              <Button
   492→                onClick={() => setManuscriptDialogOpen(true)}
   493→                variant="outline"
   494→                className="rounded-lg font-medium text-[13px] h-8"
   495→              >
   496→                <Plus className="size-3.5 mr-1.5" strokeWidth={2} />
   497→                Submit Manuscript
   498→              </Button>
   499→              <Button
   500→                onClick={() => setPrintOrderDialogOpen(true)}
   501→                className="bg-primary text-primary-foreground rounded-lg font-medium text-[13px] h-8"
   502→              >
   503→                <Plus className="size-3.5 mr-1.5" strokeWidth={2} />
   504→                Print Order
   505→              </Button>
   506→            </>
   507→          )}
   508→        </div>
   509→      </div>
   510→
   511→      {/* Manuscripts Tab */}
   512→      {activeTab === 'manuscripts' && (
   513→        <div>
   514→          {loading ? (
   515→            <div className="macos-card macos-shadow-card p-8 text-center text-muted-foreground animate-pulse text-[13px]">
   516→              Loading manuscripts...
   517→            </div>
   518→          ) : filteredManuscripts.length === 0 ? (
   519→            <div className="macos-card macos-shadow-card macos-empty">
   520→              <FileText className="size-12 text-muted-foreground/40 mb-4" />
   521→              <h3 className="text-[13px] font-semibold tracking-tight mb-1">No manuscripts found</h3>
   522→              <p className="text-[13px] text-muted-foreground">
   523→                {searchQuery || statusFilter !== 'all' ? 'Try adjusting your filters' : 'Submit your first manuscript to get started'}
   524→              </p>
   525→            </div>
   526→          ) : (
