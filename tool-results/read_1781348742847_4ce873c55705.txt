    23→// ─── macOS Clock ──────────────────────────────────────────────
    24→function MacClock() {
    25→  const [time, setTime] = React.useState(new Date())
    26→
    27→  React.useEffect(() => {
    28→    const timer = setInterval(() => setTime(new Date()), 1000)
    29→    return () => clearInterval(timer)
    30→  }, [])
    31→
    32→  const hours = time.getHours()
    33→  const minutes = time.getMinutes().toString().padStart(2, '0')
    34→  const ampm = hours >= 12 ? 'PM' : 'AM'
    35→  const displayHours = hours % 12 || 12
    36→
    37→  const days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    38→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December']
    39→
    40→  return (
    41→    <div className="text-center mb-8 select-none">
    42→      <div className="text-[86px] font-extralight tracking-tight leading-none text-white/95 drop-shadow-[0_2px_8px_rgba(0,0,0,0.3)]">
    43→        {displayHours}:{minutes}
    44→        <span className="text-[32px] font-light ml-1 text-white/80">{ampm}</span>
    45→      </div>
    46→      <div className="text-[17px] font-medium text-white/80 mt-2 drop-shadow-[0_1px_3px_rgba(0,0,0,0.3)]">
    47→        {days[time.getDay()]}, {months[time.getMonth()]} {time.getDate()}
    48→      </div>
    49→    </div>
    50→  )
    51→}

   158→// ─── Login Gate Component ─────────────────────────────────────
   159→export function LoginGate() {
   160→  const { setAuth, setDeviceActivated } = useAppStore()
   161→  const { resolvedTheme, setTheme } = useTheme()
   162→  const [mounted, setMounted] = React.useState(false)
   163→  const [stage, setStage] = React.useState<LoginStage>('initial')
   164→  const [serialKey, setSerialKey] = React.useState('')
   165→  const [email, setEmail] = React.useState('')
   166→  const [password, setPassword] = React.useState('')
   167→  const [showPassword, setShowPassword] = React.useState(false)
   168→  const [error, setError] = React.useState('')
   169→  const [isActivating, setIsActivating] = React.useState(false)
   170→  const [isLoggingIn, setIsLoggingIn] = React.useState(false)
   171→
   172→  React.useEffect(() => setMounted(true), [])
   173→
   174→  // Check if device is already activated
   175→  React.useEffect(() => {
   176→    const activated = localStorage.getItem('bf_device_activated')
   177→    if (activated === 'true') {
   178→      setDeviceActivated(true)
   179→      setStage('login')
   180→    }
   181→  }, [setDeviceActivated])
   182→
   183→  // Check if already authenticated
   184→  React.useEffect(() => {
   185→    const token = localStorage.getItem('bf_auth_token')
   186→    const userStr = localStorage.getItem('bf_auth_user')
   187→    if (token && userStr) {
   188→      try {
   189→        const user = JSON.parse(userStr)
   190→        // Verify session
   191→        fetch('/api/auth/verify', {
   192→          headers: { Authorization: `Bearer ${token}` },
   193→        })
   194→          .then(res => res.json())
   195→          .then(data => {
   196→            if (data.success) {
   197→              setAuth(data.user, token)
   198→            } else {
   199→              localStorage.removeItem('bf_auth_token')
   200→              localStorage.removeItem('bf_auth_user')
   201→            }
   202→          })
   203→          .catch(() => {
   204→            localStorage.removeItem('bf_auth_token')
   205→            localStorage.removeItem('bf_auth_user')
   206→          })
   207→      } catch {
   208→        localStorage.removeItem('bf_auth_token')
   209→        localStorage.removeItem('bf_auth_user')
   210→      }
   211→    }
   212→  }, [setAuth])
   213→
   214→  const handleActivate = async () => {
   215→    if (!serialKey.trim()) {
   216→      setError('Please enter a serial key')
   217→      return
   218→    }
   219→    setIsActivating(true)
   220→    setError('')
   221→    try {
   222→      const res = await fetch('/api/auth/activate', {
   223→        method: 'POST',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({ key: serialKey.trim() }),
   226→      })
   227→      const data = await res.json()
   228→      if (data.success) {
   229→        localStorage.setItem('bf_device_activated', 'true')
   230→        setDeviceActivated(true)
   231→        setStage('login')
   232→      } else {
   233→        setError(data.error || 'Invalid serial key')
   234→      }
   235→    } catch {
   236→      setError('Failed to activate. Please try again.')
   237→    } finally {
   238→      setIsActivating(false)
   239→    }
   240→  }
   241→
   242→  const handleLogin = async () => {
   243→    if (!email.trim() || !password.trim()) {
   244→      setError('Please enter email and password')
   245→      return
   246→    }
   247→    setIsLoggingIn(true)
   248→    setError('')
   249→    try {
   250→      const res = await fetch('/api/auth/login', {
   251→        method: 'POST',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify({ email: email.trim(), password }),
   254→      })
   255→      const data = await res.json()
   256→      if (data.success) {
   257→        localStorage.setItem('bf_auth_token', data.token)
   258→        localStorage.setItem('bf_auth_user', JSON.stringify(data.user))
   259→        setAuth(data.user, data.token)
   260→      } else {
   261→        setError(data.error || 'Invalid credentials')
   262→      }
   263→    } catch {
   264→      setError('Login failed. Please try again.')
   265→    } finally {
   266→      setIsLoggingIn(false)
   267→    }
   268→  }
   269→
   270→  const handleDemoLogin = async () => {
   271→    setIsLoggingIn(true)
   272→    setError('')
   273→    try {
   274→      const res = await fetch('/api/auth/demo', {
   275→        method: 'POST',
   276→        headers: { 'Content-Type': 'application/json' },
   277→      })
   278→      const data = await res.json()
   279→      if (data.success) {
   280→        localStorage.setItem('bf_auth_token', data.token)
   281→        localStorage.setItem('bf_auth_user', JSON.stringify(data.user))
   282→        localStorage.setItem('bf_device_activated', 'true')
   283→        setDeviceActivated(true)
   284→        setAuth(data.user, data.token)
   285→      } else {
   286→        setError(data.error || 'Demo login failed')
   287→      }
   288→    } catch {
   289→      setError('Demo login failed. Please try again.')
   290→    } finally {
   291→      setIsLoggingIn(false)
   292→    }
   293→  }
   294→
   295→  const handleKeyDown = (e: React.KeyboardEvent, action: () => void) => {
   296→    if (e.key === 'Enter') action()
   297→  }
   298→
   299→  const isDark = mounted && resolvedTheme === 'dark'
   300→
   301→  return (
   302→    <div className="fixed inset-0 flex flex-col items-center justify-center overflow-hidden select-none">
   303→      <GoldenGateBackground />
   304→
   305→      {/* macOS Menu Bar */}
   306→      <div className="absolute top-0 left-0 right-0 z-50 flex items-center justify-between h-[30px] px-4 glass-thin bg-black/20">
   307→        <div className="flex items-center gap-4">
   308→          <BookOpen className="size-3.5 text-white/70" strokeWidth={2} />
   309→          <span className="text-[13px] font-medium text-white/70">BookFlow OS</span>
   310→          <span className="text-[13px] text-white/50">File</span>

   312→          <span className="text-[13px] text-white/50">View</span>
   313→          <span className="text-[13px] text-white/50">Window</span>
   314→          <span className="text-[13px] text-white/50">Help</span>
   315→        </div>
   316→        <div className="flex items-center gap-3">
   317→          {mounted && (
   318→            <button
   319→              onClick={() => setTheme(isDark ? 'light' : 'dark')}
   320→              className="text-[12px] text-white/50 hover:text-white/80 transition-colors"
   321→            >
   322→              {isDark ? '☀️' : '🌙'}
   323→            </button>
   324→          )}
   325→          <span className="flex items-center gap-1 text-[12px] text-white/50">

   331→      {/* Main Content Area */}

   336→        {/* User Avatar */}
