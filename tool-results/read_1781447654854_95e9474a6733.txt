   885→export function ReportsModule() {
   886→  const { authUser } = useAppStore()
   887→  const isViewOnly = authUser?.isViewOnly ?? false
   888→  const [activeCategory, setActiveCategory] = React.useState<ReportCategory>('sales')
   889→  const [dateRange, setDateRange] = React.useState('12m')
   890→  const [reportData, setReportData] = React.useState<ReportData>(emptyData)
   891→  const [loading, setLoading] = React.useState(true)
   892→  const [error, setError] = React.useState<string | null>(null)
   893→
   894→  // Fetch report data from API
   895→  React.useEffect(() => {
   896→    let cancelled = false
   897→
   898→    async function fetchReportData() {
   899→      setLoading(true)
   900→      setError(null)
   901→      try {
   902→        const params = new URLSearchParams()
   903→        if (authUser?.tenantId) {
   904→          params.set('tenantId', authUser.tenantId)
   905→        }
   906→        params.set('range', dateRange)
   907→
   908→        const response = await fetch(`/api/reports?${params.toString()}`)
   909→        const result = await response.json()
   910→
   911→        if (!cancelled) {
   912→          if (result.success && result.data) {
   913→            setReportData(result.data)
   914→          } else {
   915→            setError(result.error || 'Failed to load report data')
   916→          }
   917→        }
   918→      } catch (err) {
   919→        if (!cancelled) {
   920→          setError('Network error: Failed to fetch report data')
   921→          console.error('Reports fetch error:', err)
   922→        }
   923→      } finally {
   924→        if (!cancelled) {
   925→          setLoading(false)
   926→        }
   927→      }
   928→    }
   929→
   930→    fetchReportData()
   931→
   932→    return () => {
   933→      cancelled = true
   934→    }
   935→  }, [authUser?.tenantId, dateRange])
   936→
   937→  const renderReport = () => {
   938→    if (loading) return <ReportSkeleton />
   939→
   940→    if (error) {
   941→      return (
   942→        <div className="flex flex-col items-center justify-center py-16 gap-3">
   943→          <div className="size-12 rounded-full bg-destructive/10 flex items-center justify-center">
   944→            <BarChart3 className="size-6 text-destructive" />
   945→          </div>
   946→          <p className="text-sm text-muted-foreground">{error}</p>
   947→          <button
   948→            onClick={() => setDateRange(dateRange)}
   949→            className="text-sm text-primary hover:underline"
   950→          >
   951→            Retry
   952→          </button>
   953→        </div>
   954→      )
   955→    }
   956→
   957→    switch (activeCategory) {
   958→      case 'sales': return <SalesReport data={reportData} />
   959→      case 'inventory': return <InventoryReport data={reportData} />
   960→      case 'publisher': return <PublisherReport data={reportData} />
   961→      case 'author': return <AuthorReport data={reportData} />
   962→      case 'wholesale': return <WholesaleReport data={reportData} />
   963→      case 'school': return <SchoolContractReport data={reportData} />
   964→      case 'financial': return <FinancialReport data={reportData} />
   965→      default: return <SalesReport data={reportData} />
   966→    }
   967→  }
   968→
   969→  return (
   970→    <div className="space-y-5">
   971→      {/* macOS Page Header */}
   972→      <MacosPageHeader
   973→        title="Reports & Analytics"
   974→        subtitle="Comprehensive business intelligence and reporting"
   975→        icon={BarChart3}
   976→        iconColor="text-macos-purple"
   977→        iconBg="bg-macos-purple/10"
   978→      >
   979→        {isViewOnly && (
   980→          <MacosBadge variant="warning">
   981→            <Eye className="size-3 mr-1" />
   982→            VIEW ONLY
   983→          </MacosBadge>
   984→        )}
   985→        <MacosFilterBar
   986→          items={dateRanges.map((dr) => ({ id: dr.value, label: dr.label }))}
   987→          activeId={dateRange}
   988→          onChange={setDateRange}
   989→        />
   990→        <MacosButton
   991→          variant="secondary"
   992→          icon={Download}
   993→          disabled={isViewOnly}
   994→        >
   995→          Export
   996→        </MacosButton>
   997→      </MacosPageHeader>
   998→
   999→      {/* Category Filter Bar (macOS Activity Monitor style tabs) */}
  1000→      <div className="macos-filter-bar">
  1001→        {reportCategories.map((cat) => {
  1002→          const Icon = cat.icon
  1003→          const isActive = activeCategory === cat.id
  1004→          return (
  1005→            <button
  1006→              key={cat.id}
  1007→              onClick={() => setActiveCategory(cat.id)}
  1008→              className={`flex items-center gap-1.5 ${
  1009→                isActive ? 'macos-filter-item macos-filter-item-active' : 'macos-filter-item'
  1010→              }`}
  1011→            >
  1012→              <Icon className="size-3.5" />
  1013→              <span className="text-[12px]">{cat.label}</span>
  1014→              {loading && isActive && (
  1015→                <Loader2 className="size-3 animate-spin text-muted-foreground" />
  1016→              )}
  1017→            </button>
  1018→          )
  1019→        })}
  1020→      </div>
  1021→
  1022→      {/* Report Content */}
  1023→      {renderReport()}
  1024→    </div>
  1025→  )
  1026→}
