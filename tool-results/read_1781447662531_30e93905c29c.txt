   893→   893→
   894→   894→  // Fetch report data from API
   895→   895→  React.useEffect(() => {
   896→   896→    let cancelled = false
   897→   897→
   898→   898→    async function fetchReportData() {
   899→   899→      setLoading(true)
   900→   900→      setError(null)
   901→   901→      try {
   902→   902→        const params = new URLSearchParams()
   903→   903→        if (authUser?.tenantId) {
   904→   904→          params.set('tenantId', authUser.tenantId)
   905→   905→        }
   906→   906→        params.set('range', dateRange)
   907→   907→
   908→   908→        const response = await fetch(`/api/reports?${params.toString()}`)
   909→   909→        const result = await response.json()
   910→   910→
   911→   911→        if (!cancelled) {
   912→   912→          if (result.success && result.data) {
   913→   913→            setReportData(result.data)
   914→   914→          } else {
   915→   915→            setError(result.error || 'Failed to load report data')
   916→   916→          }
   917→   917→        }
   918→   918→      } catch (err) {
   919→   919→        if (!cancelled) {
   920→   920→          setError('Network error: Failed to fetch report data')
   921→   921→          console.error('Reports fetch error:', err)
   922→   922→        }
   923→   923→      } finally {
   924→   924→        if (!cancelled) {
   925→   925→          setLoading(false)
   926→   926→        }
   927→   927→      }
   928→   928→    }
   929→   929→
   930→   930→    fetchReportData()
   931→   931→
   932→   932→    return () => {
   933→   933→      cancelled = true
   934→   934→    }
