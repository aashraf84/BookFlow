    99→const getPublisherColor = (name: string) => {
   100→  let hash = 0
   101→  for (let i = 0; i < name.length; i++) {
   102→    hash = name.charCodeAt(i) + ((hash << 5) - hash)
   103→  }
   104→  return publisherColors[Math.abs(hash) % publisherColors.length]
   105→}
