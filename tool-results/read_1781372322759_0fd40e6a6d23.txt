   240→   240→          unitPrice: product.price,
   241→   241→          discount: product.discount,

   260→   260→      )
   261→   261→    )
   262→   262→  }
   263→   263→

   318→   318→

   338→   338→      // Clear cart

   345→   345→      setCompleting(false)
   346→   346→    }
