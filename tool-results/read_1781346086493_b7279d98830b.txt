   992→function CreateCampaignDialog({
   993→  open,
   994→  onOpenChange,
   995→  onSuccess,
   996→}: {
   997→  open: boolean
   998→  onOpenChange: (open: boolean) => void
   999→  onSuccess: () => void
  1000→}) {
  1001→  const [saving, setSaving] = React.useState(false)
  1002→  const [form, setForm] = React.useState({
  1003→    name: '',
  1004→    type: 'email',
  1005→    target: 'all',
  1006→    message: '',
  1007→    scheduledAt: '',
  1008→  })
  1009→
  1010→  const handleSubmit = async (e: React.FormEvent) => {
  1011→    e.preventDefault()
  1012→    if (!form.name.trim() || !form.message.trim()) return
  1013→    setSaving(true)
  1014→    try {
  1015→      const res = await fetch('/api/crm/campaigns', {
  1016→        method: 'POST',
  1017→        headers: { 'Content-Type': 'application/json' },
  1018→        body: JSON.stringify({
  1019→          ...form,
  1020→          scheduledAt: form.scheduledAt || null,
  1021→        }),
  1022→      })
  1023→      if (res.ok) {
  1024→        setForm({ name: '', type: 'email', target: 'all', message: '', scheduledAt: '' })
  1025→        onOpenChange(false)
  1026→        onSuccess()
  1027→      } else {
  1028→        const json = await res.json()
  1029→        alert(json.error || 'Failed to create campaign')
  1030→      }
  1031→    } catch (err) {
  1032→      console.error('Failed to create campaign:', err)
  1033→      alert('Failed to create campaign')
  1034→    } finally {
  1035→      setSaving(false)
  1036→    }
  1037→  }
  1038→
  1039→  return (
  1040→    <Dialog open={open} onOpenChange={onOpenChange}>
  1041→      <DialogContent className="sm:max-w-md rounded-2xl macos-shadow-float">

  1050→          </div>
  1051→          <div className="grid grid-cols-2 gap-3">
  1052→            <div className="space-y-2">
  1053→              <Label className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Campaign Type</Label>

  1055→                <SelectTrigger className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8">
  1056→                  <SelectValue />
  1057→                </SelectTrigger>
  1058→                <SelectContent>
