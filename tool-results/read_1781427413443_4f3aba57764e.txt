    52→// ─── Types ────────────────────────────────────────────────────────
    53→type ProductType = 'book' | 'stationery'
    54→
    55→interface POSProduct {
    56→  id: string
    57→  type: ProductType
    58→  name: string
    59→  sku: string | null
    60→  isbn: string | null
    61→  barcode: string | null
    62→  price: number
    63→  costPrice: number
    64→  discount: number
    65→  author?: string | null
    66→  brand?: string | null
    67→  category?: string | null
    68→}
    69→
    70→interface CartItem {
    71→  product: POSProduct
    72→  quantity: number
    73→  unitPrice: number
    74→  discount: number
    75→  totalPrice: number
    76→}
    77→
    78→interface Customer {
    79→  id: string
    80→  name: string
    81→  email?: string | null
    82→  phone?: string | null
    83→  loyaltyPoints: number
    84→}
    85→
    86→interface CompletedSale {
    87→  invoiceNumber: string
    88→  total: number
    89→  subtotal: number
    90→  discount: number
    91→  tax: number
    92→  paymentMethod: string
    93→  customerName: string | null
    94→  items: Array<{ name: string; quantity: number; unitPrice: number; totalPrice: number }>
    95→}

    97→// ─── Component ────────────────────────────────────────────────────
    98→export function RetailPOSModule() {
    99→  const { authUser } = useAppStore()
   100→  const isViewOnly = authUser?.isViewOnly ?? false
   101→
   102→  // Products
   103→  const [products, setProducts] = React.useState<POSProduct[]>([])
   104→  const [searchTerm, setSearchTerm] = React.useState('')
   105→  const [categoryFilter, setCategoryFilter] = React.useState<'all' | 'books' | 'stationery' | 'school'>('all')
   106→  const [searchLoading, setSearchLoading] = React.useState(false)
   107→
   108→  // Cart
   109→  const [cart, setCart] = React.useState<CartItem[]>([])
   110→  const [cartDiscount, setCartDiscount] = React.useState(0)
   111→  const [discountType, setDiscountType] = React.useState<'percentage' | 'fixed'>('percentage')
   112→  const [taxRate, setTaxRate] = React.useState(0)
   113→
   114→  // Customer
   115→  const [customerSearch, setCustomerSearch] = React.useState('')
   116→  const [customers, setCustomers] = React.useState<Customer[]>([])
   117→  const [selectedCustomer, setSelectedCustomer] = React.useState<Customer | null>(null)
   118→  const [customerLookupOpen, setCustomerLookupOpen] = React.useState(false)
   119→
   120→  // Payment
   121→  const [paymentMethod, setPaymentMethod] = React.useState('cash')
   122→
   123→  // Sale
   124→  const [completing, setCompleting] = React.useState(false)
   125→  const [completedSale, setCompletedSale] = React.useState<CompletedSale | null>(null)
   126→  const [error, setError] = React.useState<string | null>(null)
   127→
   128→  // Search products
   129→  React.useEffect(() => {
   130→    const searchProducts = async () => {
   131→      setSearchLoading(true)
   132→      try {
   133→        const params = new URLSearchParams()
   134→        params.set('limit', '30')
   135→        if (searchTerm) params.set('search', searchTerm)
   136→
   137→        const [booksRes, stationeryRes] = await Promise.all([
   138→          fetch(`/api/books?${params.toString()}`),
   139→          fetch(`/api/stationery?${params.toString()}`),
   140→        ])
   141→
   142→        const bookData = booksRes.ok ? await booksRes.json() : { data: [] }
   143→        const stationeryData = stationeryRes.ok ? await stationeryRes.json() : { data: [] }
   144→
   145→        const bookProducts: POSProduct[] = (bookData.data || []).map((b: Record<string, unknown>) => ({
   146→          id: b.id as string,
   147→          type: 'book' as ProductType,
   148→          name: b.title as string,
   149→          sku: null,
   150→          isbn: (b.isbn as string) || null,
   151→          barcode: (b.barcode as string) || null,
   152→          price: (b.price as number) || 0,
   153→          costPrice: (b.costPrice as number) || 0,
   154→          discount: (b.discount as number) || 0,
   155→          author: (b.author as { name: string } | null)?.name || null,
   156→          category: (b.category as { name: string } | null)?.name || null,
   157→        }))
   158→
   159→        const stationeryProducts: POSProduct[] = (stationeryData.data || []).map((s: Record<string, unknown>) => ({
   160→          id: s.id as string,
   161→          type: 'stationery' as ProductType,
   162→          name: s.name as string,
   163→          sku: (s.sku as string) || null,
   164→          isbn: null,
   165→          barcode: (s.barcode as string) || null,
   166→          price: (s.price as number) || 0,
   167→          costPrice: (s.costPrice as number) || 0,
   168→          discount: (s.discount as number) || 0,
   169→          brand: (s.brand as string) || null,
   170→          category: (s.category as { name: string } | null)?.name || null,
   171→        }))
   172→
   173→        let all = [...bookProducts, ...stationeryProducts]
   174→
   175→        // Apply category filter
   176→        if (categoryFilter === 'books') {
   177→          all = all.filter((p) => p.type === 'book')
   178→        } else if (categoryFilter === 'stationery') {
   179→          all = all.filter((p) => p.type === 'stationery')
   180→        } else if (categoryFilter === 'school') {
   181→          all = all.filter((p) => p.category?.toLowerCase().includes('school') || p.category?.toLowerCase().includes('educational'))
   182→        }
   183→
   184→        setProducts(all)
   185→      } catch {
   186→        // silent
   187→      } finally {
   188→        setSearchLoading(false)
   189→      }
   190→    }
   191→
   192→    const debounce = setTimeout(searchProducts, 300)
   193→    return () => clearTimeout(debounce)
   194→  }, [searchTerm, categoryFilter])
   195→
   196→  // Search customers
   197→  React.useEffect(() => {
   198→    if (!customerSearch) {
   199→      setCustomers([])
   200→      return
   201→    }
   202→    const searchCustomers = async () => {
   203→      try {
   204→        const res = await fetch(`/api/customers?search=${encodeURIComponent(customerSearch)}&type=retail&limit=10`)
   205→        if (res.ok) {
   206→          const json = await res.json()
   207→          setCustomers(json.data || [])
   208→        }
   209→      } catch {
   210→        // silent
   211→      }
   212→    }
   213→    const debounce = setTimeout(searchCustomers, 300)
   214→    return () => clearTimeout(debounce)
   215→  }, [customerSearch])
   216→
   217→  // Add to cart
   218→  const addToCart = (product: POSProduct) => {
   219→    if (isViewOnly) return
   220→    setCart((prev) => {
   221→      const existing = prev.find((item) => item.product.id === product.id && item.product.type === product.type)
   222→      if (existing) {
   223→        return prev.map((item) =>
   224→          item.product.id === product.id && item.product.type === product.type
   225→            ? {
   226→                ...item,
   227→                quantity: item.quantity + 1,
   228→                totalPrice: (item.quantity + 1) * item.unitPrice * (1 - item.discount / 100),
   229→              }
   230→            : item
   231→        )
   232→      }
   233→      const unitPrice = product.price * (1 - product.discount / 100)
   234→      return [
   235→        ...prev,
   236→        {
   237→          product,
   238→          quantity: 1,
   239→          unitPrice: product.price,
   240→          discount: product.discount,
   241→          totalPrice: unitPrice,
   242→        },
   243→      ]
   244→    })
   245→  }
   246→
   247→  // Update cart quantity
   248→  const updateQuantity = (index: number, newQty: number) => {
   249→    if (isViewOnly) return
   250→    if (newQty < 1) {
   251→      removeFromCart(index)
   252→      return
   253→    }
   254→    setCart((prev) =>
   255→      prev.map((item, i) =>
   256→        i === index
   257→          ? { ...item, quantity: newQty, totalPrice: newQty * item.unitPrice * (1 - item.discount / 100) }
   258→          : item
   259→      )
   260→    )
   261→  }
   262→
   263→  // Remove from cart
   264→  const removeFromCart = (index: number) => {
   265→    if (isViewOnly) return
   266→    setCart((prev) => prev.filter((_, i) => i !== index))
   267→  }
   268→
   269→  // Clear cart
   270→  const clearCart = () => {
   271→    if (isViewOnly) return
   272→    setCart([])
   273→    setCartDiscount(0)
   274→    setSelectedCustomer(null)
   275→    setCompletedSale(null)
   276→  }
   277→
   278→  // Compute totals
   279→  const subtotal = cart.reduce((s, item) => s + item.totalPrice, 0)
   280→  const discountAmount = discountType === 'percentage' ? subtotal * (cartDiscount / 100) : cartDiscount
   281→  const afterDiscount = Math.max(0, subtotal - discountAmount)
   282→  const taxAmount = afterDiscount * (taxRate / 100)
   283→  const total = afterDiscount + taxAmount
   284→
   285→  // Complete sale
   286→  const handleCompleteSale = async () => {
   287→    if (isViewOnly || cart.length === 0) return
   288→    setCompleting(true)
   289→    setError(null)
   290→    try {
   291→      const saleItems = cart.map((item) => ({
   292→        bookId: item.product.type === 'book' ? item.product.id : undefined,
   293→        stationeryId: item.product.type === 'stationery' ? item.product.id : undefined,
   294→        quantity: item.quantity,
   295→        unitPrice: item.unitPrice,
   296→        discount: item.discount,
   297→      }))
   298→
   299→      const res = await fetch('/api/sales', {
   300→        method: 'POST',
   301→        headers: { 'Content-Type': 'application/json' },
   302→        body: JSON.stringify({
   303→          customerId: selectedCustomer?.id || null,
   304→          discount: discountAmount,
   305→          tax: taxRate,
   306→          paymentMethod,
   307→          paymentStatus: 'paid',
   308→          saleType: 'retail',
   309→          items: saleItems,
   310→        }),
   311→      })
   312→
   313→      if (!res.ok) {
   314→        const errData = await res.json()
   315→        throw new Error(errData.error || 'Failed to complete sale')
   316→      }
   317→
   318→      const json = await res.json()
   319→      const sale = json.data
   320→
   321→      setCompletedSale({
   322→        invoiceNumber: sale.invoiceNumber,
   323→        total: sale.total,
   324→        subtotal: sale.subtotal,
   325→        discount: sale.discount,
   326→        tax: sale.tax,
   327→        paymentMethod: sale.paymentMethod,
   328→        customerName: selectedCustomer?.name || null,
   329→        items: cart.map((item) => ({
   330→          name: item.product.name,
   331→          quantity: item.quantity,
   332→          unitPrice: item.unitPrice,
   333→          totalPrice: item.totalPrice,
   334→        })),
   335→      })
   336→
   337→      // Clear cart
   338→      setCart([])
   339→      setCartDiscount(0)
   340→      setSelectedCustomer(null)
   341→    } catch (err) {
   342→      setError(err instanceof Error ? err.message : 'Failed to complete sale')
   343→    } finally {
   344→      setCompleting(false)
   345→    }
   346→  }
   347→
   348→  // Payment method icons
   349→  const paymentMethods = [
   350→    { value: 'cash', label: 'Cash', icon: Banknote, color: 'macos-green' as const },
   351→    { value: 'card', label: 'Card', icon: CreditCard, color: 'macos-blue' as const },
   352→    { value: 'bank', label: 'Bank', icon: Building, color: 'macos-purple' as const },
   353→    { value: 'wallet', label: 'Wallet', icon: Smartphone, color: 'macos-orange' as const },
   354→  ]
   355→
   356→  // Category filters for MacosFilterBar
   357→  const categoryFilterItems = [
   358→    { id: 'all', label: 'All' },
   359→    { id: 'books', label: 'Books' },
   360→    { id: 'stationery', label: 'Stationery' },
   361→    { id: 'school', label: 'School' },
   362→  ]
   363→
   364→  return (
   365→    <div className="flex flex-col h-[calc(100vh-52px-32px)] overflow-hidden">
   366→      {/* ─── Page Header ─── */}
   367→      <MacosPageHeader
   368→        title="Point of Sale"
   369→        subtitle="Process retail transactions — select products, add to cart, and complete sales"
   370→        icon={ShoppingCart}
   371→        iconColor="text-gg-bridge"
   372→        iconBg="bg-gg-bridge/10"
   373→      >
   374→        {isViewOnly && (
   375→          <MacosBadge variant="warning" className="flex items-center gap-1.5">
   376→            <Eye className="size-3" />
   377→            View Only
   378→          </MacosBadge>
   379→        )}
   380→      </MacosPageHeader>
   381→
   382→      {/* ─── Error Alert ─── */}
   383→      {error && (
   384→        <div className="macos-card macos-shadow-card flex items-center gap-2.5 p-3 mb-4 border-macos-red/20 bg-macos-red/5 flex-shrink-0">
   385→          <div className="size-6 rounded-md bg-macos-red/10 flex items-center justify-center flex-shrink-0">
   386→            <AlertCircle className="size-3.5 text-macos-red" />
   387→          </div>
   388→          <span className="text-[13px] font-medium text-macos-red flex-1">{error}</span>
   389→          <button
   390→            className="size-6 rounded-md hover:bg-macos-red/10 flex items-center justify-center transition-colors"
   391→            onClick={() => setError(null)}
   392→          >
   393→            <X className="size-3.5 text-macos-red" />
   394→          </button>
   395→        </div>
   396→      )}
   397→
   398→      {/* ─── Receipt Dialog ─── */}
   399→      <MacosDialog
   400→        open={!!completedSale}
   401→        onOpenChange={(open) => !open && setCompletedSale(null)}
   402→        title="Sale Completed"
   403→        description="Transaction processed successfully"
   404→        className="glass-card"
   405→        footer={
   406→          <Button
   407→            onClick={() => setCompletedSale(null)}
   408→            className="w-full bg-macos-green text-white rounded-lg font-semibold text-[13px] h-10 hover:bg-macos-green/90"
   409→          >
   410→            <Check className="size-4 mr-1.5" />
   411→            Done
   412→          </Button>
   413→        }
   414→      >
   415→        {completedSale && (
   416→          <div className="space-y-4">
   417→            <div className="text-center space-y-1.5 py-3">
   418→              <div className="flex items-center justify-center gap-1.5">
   419→                <Sparkles className="size-3.5 text-gg-golden" />
   420→                <p className="text-[15px] font-bold tracking-tight">BookFlow OS</p>
   421→                <Sparkles className="size-3.5 text-gg-golden" />
   422→              </div>
   423→              <p className="text-[11px] text-muted-foreground font-medium">Invoice {completedSale.invoiceNumber}</p>
   424→              <p className="text-[10px] text-muted-foreground">{new Date().toLocaleString()}</p>
   425→            </div>
   426→
   427→            <div className="rounded-xl bg-secondary/50 p-3 space-y-2">
   428→              {completedSale.items.map((item, i) => (
   429→                <div key={i} className="flex justify-between text-[13px] tracking-tight">
   430→                  <span className="flex-1 truncate mr-3 font-medium">{item.name} <span className="text-muted-foreground">×{item.quantity}</span></span>
   431→                  <span className="font-semibold tabular-nums">${item.totalPrice.toFixed(2)}</span>
   432→                </div>
   433→              ))}
   434→            </div>
   435→
   436→            <Separator className="opacity-50" />
   437→
   438→            <div className="space-y-2 text-[13px] tracking-tight">
   439→              <div className="flex justify-between">
   440→                <span className="text-muted-foreground">Subtotal</span>
   441→                <span className="tabular-nums">${completedSale.subtotal.toFixed(2)}</span>
   442→              </div>
   443→              {completedSale.discount > 0 && (
   444→                <div className="flex justify-between text-macos-green">
   445→                  <span>Discount</span>
   446→                  <span className="tabular-nums">-${completedSale.discount.toFixed(2)}</span>
   447→                </div>
   448→              )}
   449→              {completedSale.tax > 0 && (
   450→                <div className="flex justify-between">
   451→                  <span className="text-muted-foreground">Tax</span>
   452→                  <span className="tabular-nums">${completedSale.tax.toFixed(2)}</span>
   453→                </div>
   454→              )}
   455→              <Separator className="opacity-50" />
   456→              <div className="flex justify-between font-bold text-[15px] tracking-tight">
   457→                <span>Total</span>
   458→                <span className="tabular-nums">${completedSale.total.toFixed(2)}</span>
   459→              </div>
   460→            </div>
   461→
   462→            <div className="flex justify-between text-[10px] text-muted-foreground uppercase tracking-wider font-semibold">
   463→              <span>Paid: {completedSale.paymentMethod}</span>
   464→              {completedSale.customerName && <span>Customer: {completedSale.customerName}</span>}
   465→            </div>
   466→          </div>
   467→        )}
   468→      </MacosDialog>
   469→
   470→      {/* ─── Main POS Split Layout ─── */}
   471→      <div className="flex-1 grid grid-cols-1 lg:grid-cols-5 gap-4 min-h-0">
   472→        {/* ─── Left: Products ─── */}
   473→        <div className="lg:col-span-3 flex flex-col min-h-0 gap-4">
   474→          {/* Search + Category Filter Bar */}
   475→          <div className="macos-card macos-shadow-card p-4 flex-shrink-0">
   476→            <MacosSearch
   477→              value={searchTerm}
   478→              onChange={setSearchTerm}
   479→              placeholder="Search by name, ISBN, or barcode…"
   480→              className="h-9"
   481→            />
   482→
   483→            <div className="mt-3">
   484→              <MacosFilterBar
   485→                items={categoryFilterItems}
   486→                activeId={categoryFilter}
   487→                onChange={(id) => setCategoryFilter(id as 'all' | 'books' | 'stationery' | 'school')}
   488→              />
   489→            </div>
   490→          </div>
   491→
   492→          {/* Product Grid */}
   493→          <div className="flex-1 min-h-0 overflow-y-auto custom-scrollbar pr-1">
   494→            {searchLoading ? (
   495→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   496→                {Array.from({ length: 6 }).map((_, i) => (
   497→                  <div key={i} className="macos-card macos-shadow-card p-4 animate-pulse">
   498→                    <div className="h-3 bg-secondary rounded-md w-3/4 mb-3" />
   499→                    <div className="h-2.5 bg-secondary rounded-md w-1/2 mb-3" />
   500→                    <div className="h-4 bg-secondary rounded-md w-1/3" />
   501→                  </div>
   502→                ))}
   503→              </div>
   504→            ) : products.length === 0 ? (
   505→              <MacosEmptyState
   506→                icon={Package}
   507→                title="No products found"
   508→                description="Try a different search or category"
   509→                className="macos-card macos-shadow-card"
   510→              />
   511→            ) : (
   512→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   513→                {products.map((product) => {
   514→                  const inCart = cart.find((c) => c.product.id === product.id && c.product.type === product.type)
   515→                  return (
   516→                    <button
   517→                      key={`${product.type}-${product.id}`}
   518→                      className={`macos-card macos-shadow-card macos-card-hover p-4 text-left w-full cursor-pointer group ${inCart ? 'ring-1 ring-macos-green/30 bg-macos-green/[0.02]' : 'bg-card'} ${isViewOnly ? 'cursor-default' : ''}`}
   519→                      onClick={() => addToCart(product)}
   520→                      disabled={isViewOnly}
   521→                    >
   522→                      <div className="space-y-2">
   523→                        <div className="flex items-start justify-between gap-1.5">
   524→                          <p className="text-[13px] font-semibold leading-snug line-clamp-2 tracking-tight group-hover:text-macos-blue transition-colors">{product.name}</p>
   525→                          <MacosBadge variant={product.type === 'book' ? 'danger' : 'warning'}>
   526→                            {product.type === 'book' ? 'Book' : 'Item'}
   527→                          </MacosBadge>
   528→                        </div>
   529→                        {product.author && (
   530→                          <p className="text-[11px] text-muted-foreground truncate tracking-tight">by {product.author}</p>
   531→                        )}
   532→                        {product.brand && (
   533→                          <p className="text-[11px] text-muted-foreground truncate tracking-tight">{product.brand}</p>
   534→                        )}
   535→                        {product.category && (
   536→                          <MacosBadge>{product.category}</MacosBadge>
   537→                        )}
   538→                        <div className="flex items-center justify-between pt-1">
   539→                          <div>
   540→                            {product.discount > 0 ? (
   541→                              <div className="flex items-center gap-1.5">
   542→                                <span className="text-[11px] text-muted-foreground line-through tracking-tight">${product.price.toFixed(2)}</span>
   543→                                <span className="text-[14px] font-bold text-macos-green tracking-tight">${(product.price * (1 - product.discount / 100)).toFixed(2)}</span>
   544→                              </div>
   545→                            ) : (
   546→                              <span className="text-[14px] font-bold tracking-tight">${product.price.toFixed(2)}</span>
   547→                            )}
   548→                          </div>
   549→                          {inCart ? (
   550→                            <MacosBadge variant="success">×{inCart.quantity}</MacosBadge>
   551→                          ) : !isViewOnly ? (
   552→                            <div className="size-6 rounded-lg bg-macos-blue/10 flex items-center justify-center group-hover:bg-macos-blue/20 transition-colors">

   775→                      >

   785→              {/* Complete Sale */}
