   103→   103→export function InventoryModule() {
   104→   104→  const [activeTab, setActiveTab] = React.useState('overview')
   105→   105→  const [items, setItems] = React.useState<InventoryItem[]>([])
   106→   106→  const [warehouses, setWarehouses] = React.useState<Warehouse[]>([])
   107→   107→  const [transfers, setTransfers] = React.useState<StockTransfer[]>([])
   108→   108→  const [branches, setBranches] = React.useState<Branch[]>([])
   109→   109→  const [loading, setLoading] = React.useState(true)
   110→   110→  const [error, setError] = React.useState<string | null>(null)
   111→   111→
   112→   112→  // Filters
   113→   113→  const [branchFilter, setBranchFilter] = React.useState('all')
   114→   114→  const [warehouseFilter, setWarehouseFilter] = React.useState('all')
   115→   115→  const [lowStockOnly, setLowStockOnly] = React.useState(false)
   116→   116→
   117→   117→  // Dialogs
   118→   118→  const [adjustDialogOpen, setAdjustDialogOpen] = React.useState(false)
   119→   119→  const [transferDialogOpen, setTransferDialogOpen] = React.useState(false)
   120→   120→  const [adjustingItem, setAdjustingItem] = React.useState<InventoryItem | null>(null)
   121→   121→  const [saving, setSaving] = React.useState(false)
   122→   122→
   123→   123→  // Adjust form
   124→   124→  const [adjustQty, setAdjustQty] = React.useState('')
   125→   125→  const [adjustOperation, setAdjustOperation] = React.useState('set')
   126→   126→
   127→   127→  // Transfer form
   128→   128→  const [transferFrom, setTransferFrom] = React.useState('')
   129→   129→  const [transferTo, setTransferTo] = React.useState('')
   130→   130→  const [transferNotes, setTransferNotes] = React.useState('')
   131→   131→
   132→   132→  // Fetch inventory items
   133→   133→  const fetchItems = React.useCallback(async () => {
   134→   134→    setLoading(true)
   135→   135→    setError(null)
   136→   136→    try {
   137→   137→      const params = new URLSearchParams()
   138→   138→      params.set('limit', '100')
   139→   139→      if (branchFilter && branchFilter !== 'all') params.set('branchId', branchFilter)
   140→   140→      if (warehouseFilter && warehouseFilter !== 'all') params.set('warehouseId', warehouseFilter)
   141→   141→      if (lowStockOnly) params.set('lowStock', 'true')
   142→   142→
   143→   143→      const res = await fetch(`/api/inventory?${params.toString()}`)
   144→   144→      if (!res.ok) throw new Error('Failed to fetch inventory')
   145→   145→      const json = await res.json()
   146→   146→      setItems(json.data || [])
   147→   147→    } catch (err) {
   148→   148→      setError(err instanceof Error ? err.message : 'Unknown error')
   149→   149→    } finally {
   150→   150→      setLoading(false)
   151→   151→    }
   152→   152→  }, [branchFilter, warehouseFilter, lowStockOnly])
   153→   153→
   154→   154→  // Fetch warehouses
   155→   155→  const fetchWarehouses = React.useCallback(async () => {
   156→   156→    try {
   157→   157→      const res = await fetch('/api/warehouses')
   158→   158→      if (!res.ok) throw new Error('Failed to fetch warehouses')
   159→   159→      const json = await res.json()
   160→   160→      setWarehouses(json.data || [])
   161→   161→    } catch {
   162→   162→      // silent
   163→   163→    }
   164→   164→  }, [])
   165→   165→
   166→   166→  // Fetch branches
   167→   167→  const fetchBranches = React.useCallback(async () => {
   168→   168→    try {
   169→   169→      const res = await fetch('/api/branches')
   170→   170→      if (!res.ok) throw new Error('Failed to fetch branches')
   171→   171→      const json = await res.json()
   172→   172→      setBranches(json.data || [])
   173→   173→    } catch {
   174→   174→      // silent
   175→   175→    }
   176→   176→  }, [])
   177→   177→
   178→   178→  // Fetch transfers
   179→   179→  const fetchTransfers = React.useCallback(async () => {
   180→   180→    try {
   181→   181→      const res = await fetch('/api/inventory?transfers=true&limit=50')
   182→   182→      if (!res.ok) throw new Error('Failed to fetch transfers')
   183→   183→      const json = await res.json()
   184→   184→      setTransfers(json.transfers || [])
   185→   185→    } catch {
   186→   186→      // silent
   187→   187→    }
   188→   188→  }, [])
   189→   189→
   190→   190→  React.useEffect(() => {
   191→   191→    fetchItems()
   192→   192→  }, [fetchItems])
   193→   193→
   194→   194→  React.useEffect(() => {
   195→   195→    fetchWarehouses()
   196→   196→    fetchBranches()
   197→   197→    fetchTransfers()
   198→   198→  }, [fetchWarehouses, fetchBranches, fetchTransfers])
   199→   199→
   200→   200→  // Product name helper
   201→   201→  const getProductName = (item: InventoryItem): string => {
   202→   202→    if (item.book) return item.book.title
   203→   203→    if (item.stationery) return item.stationery.name
   204→   204→    return 'Unknown Product'
   205→   205→  }
   206→   206→
   207→   207→  // Stock status
   208→   208→  const getStockBadge = (item: InventoryItem) => {
   209→   209→    if (item.quantity === 0) {
   210→   210→      return <Badge className="bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400 border-red-200 dark:border-red-800 text-xs"><span className="mr-1 inline-block size-1.5 rounded-full bg-red-500" />Out of Stock</Badge>
   211→   211→    }
   212→   212→    if (item.isLowStock || item.quantity <= item.reorderLevel) {
   213→   213→      return <Badge variant="secondary" className="bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400 border-amber-200 dark:border-amber-800 text-xs"><span className="mr-1 inline-block size-1.5 rounded-full bg-amber-500" />Low Stock</Badge>
   214→   214→    }
   215→   215→    return <Badge className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800 text-xs"><span className="mr-1 inline-block size-1.5 rounded-full bg-emerald-500" />In Stock</Badge>
   216→   216→  }
   217→   217→
   218→   218→  // Transfer status badge
   219→   219→  const getTransferStatusBadge = (status: string) => {
   220→   220→    const map: Record<string, { className: string; label: string }> = {
   221→   221→      pending: { className: 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400 border-amber-200 dark:border-amber-800', label: 'Pending' },
   222→   222→      'in-transit': { className: 'bg-sky-100 text-sky-700 dark:bg-sky-950/40 dark:text-sky-400 border-sky-200 dark:border-sky-800', label: 'In Transit' },
   223→   223→      completed: { className: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800', label: 'Completed' },
   224→   224→      cancelled: { className: 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400 border-red-200 dark:border-red-800', label: 'Cancelled' },
   225→   225→    }
   226→   226→    const s = map[status] || { className: '', label: status }
   227→   227→    return <Badge className={`text-xs ${s.className}`}>{s.label}</Badge>
   228→   228→  }
   229→   229→
   230→   230→  // Open adjust dialog
   231→   231→  const handleAdjust = (item: InventoryItem) => {
   232→   232→    setAdjustingItem(item)
   233→   233→    setAdjustQty('')
   234→   234→    setAdjustOperation('set')
   235→   235→    setAdjustDialogOpen(true)
   236→   236→  }
   237→   237→
   238→   238→  // Save stock adjustment
   239→   239→  const handleAdjustSave = async () => {
   240→   240→    if (!adjustingItem || !adjustQty) return
   241→   241→    setSaving(true)
   242→   242→    try {
   243→   243→      const res = await fetch('/api/inventory', {
   244→   244→        method: 'PUT',
   245→   245→        headers: { 'Content-Type': 'application/json' },
   246→   246→        body: JSON.stringify({
   247→   247→          id: adjustingItem.id,
   248→   248→          quantity: parseInt(adjustQty),
   249→   249→          operation: adjustOperation,
   250→   250→        }),
   251→   251→      })
   252→   252→      if (!res.ok) throw new Error('Failed to adjust stock')
   253→   253→      setAdjustDialogOpen(false)
   254→   254→      fetchItems()
   255→   255→      fetchWarehouses()
   256→   256→    } catch (err) {
   257→   257→      setError(err instanceof Error ? err.message : 'Failed to adjust stock')
   258→   258→    } finally {
   259→   259→      setSaving(false)
   260→   260→    }
   261→   261→  }
   262→   262→
   263→   263→  // Create transfer
   264→   264→  const handleTransferCreate = async () => {
   265→   265→    if (!transferFrom || !transferTo || transferFrom === transferTo) return
   266→   266→    setSaving(true)
   267→   267→    try {
   268→   268→      const res = await fetch('/api/inventory', {
   269→   269→        method: 'POST',
   270→   270→        headers: { 'Content-Type': 'application/json' },
   271→   271→        body: JSON.stringify({
   272→   272→          type: 'transfer',
   273→   273→          fromBranchId: transferFrom,
   274→   274→          toBranchId: transferTo,
   275→   275→          notes: transferNotes,
   276→   276→        }),
   277→   277→      })
   278→   278→      if (!res.ok) throw new Error('Failed to create transfer')
   279→   279→      setTransferDialogOpen(false)
   280→   280→      setTransferFrom('')
   281→   281→      setTransferTo('')
   282→   282→      setTransferNotes('')
   283→   283→      fetchTransfers()
   284→   284→    } catch (err) {
   285→   285→      setError(err instanceof Error ? err.message : 'Failed to create transfer')
   286→   286→    } finally {
   287→   287→      setSaving(false)
   288→   288→    }
   289→   289→  }
   290→   290→
   291→   291→  // Compute stats
   292→   292→  const totalItems = items.length
   293→   293→  const lowStockItems = items.filter((i) => i.isLowStock || i.quantity <= i.reorderLevel)
   294→   294→  const outOfStockItems = items.filter((i) => i.quantity === 0)
   295→   295→  const totalValue = items.reduce((s, i) => s + i.quantity * i.costPrice, 0)
   296→   296→
   297→   297→  return (
   298→   298→    <div className="space-y-6">
   299→   299→      {/* Header */}
   300→   300→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   301→   301→        <div>
   302→   302→          <h2 className="text-2xl font-bold tracking-tight">Inventory Management</h2>
   303→   303→          <p className="text-muted-foreground text-sm">Real-time stock tracking across warehouses and branches</p>
   304→   304→        </div>
   305→   305→        <div className="flex gap-2">
   306→   306→          <Button variant="outline" onClick={() => setTransferDialogOpen(true)}>
   307→   307→            <ArrowRightLeft className="size-4 mr-1.5" />
   308→   308→            New Transfer
   309→   309→          </Button>
   310→   310→        </div>
   311→   311→      </div>
   312→   312→
   313→   313→      {/* Stats Cards */}
   314→   314→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   315→   315→        <Card>
   316→   316→          <CardContent className="p-4">
   317→   317→            <div className="flex items-center justify-between">
   318→   318→              <div>
   319→   319→                <p className="text-sm text-muted-foreground">Total Items</p>
   320→   320→                <p className="text-2xl font-bold">{totalItems}</p>
   321→   321→              </div>
   322→   322→              <div className="rounded-lg bg-emerald-100 dark:bg-emerald-950/30 p-2">
   323→   323→                <Package className="size-5 text-emerald-600" />
   324→   324→              </div>
   325→   325→            </div>
   326→   326→          </CardContent>
   327→   327→        </Card>
   328→   328→        <Card>
   329→   329→          <CardContent className="p-4">
   330→   330→            <div className="flex items-center justify-between">
   331→   331→              <div>
   332→   332→                <p className="text-sm text-muted-foreground">Low Stock</p>
   333→   333→                <p className="text-2xl font-bold text-amber-600">{lowStockItems.length}</p>
   334→   334→              </div>
   335→   335→              <div className="rounded-lg bg-amber-100 dark:bg-amber-950/30 p-2">
   336→   336→                <TrendingDown className="size-5 text-amber-600" />
   337→   337→              </div>
   338→   338→            </div>
   339→   339→          </CardContent>
   340→   340→        </Card>
   341→   341→        <Card>
   342→   342→          <CardContent className="p-4">
   343→   343→            <div className="flex items-center justify-between">
   344→   344→              <div>
   345→   345→                <p className="text-sm text-muted-foreground">Out of Stock</p>
   346→   346→                <p className="text-2xl font-bold text-red-600">{outOfStockItems.length}</p>
   347→   347→              </div>
   348→   348→              <div className="rounded-lg bg-red-100 dark:bg-red-950/30 p-2">
   349→   349→                <AlertTriangle className="size-5 text-red-600" />
   350→   350→              </div>
   351→   351→            </div>
   352→   352→          </CardContent>
   353→   353→        </Card>
   354→   354→        <Card>
   355→   355→          <CardContent className="p-4">
   356→   356→            <div className="flex items-center justify-between">
   357→   357→              <div>
   358→   358→                <p className="text-sm text-muted-foreground">Total Value</p>
   359→   359→                <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   360→   360→              </div>
   361→   361→              <div className="rounded-lg bg-teal-100 dark:bg-teal-950/30 p-2">
   362→   362→                <WarehouseIcon className="size-5 text-teal-600" />
   363→   363→              </div>
   364→   364→            </div>
   365→   365→          </CardContent>
   366→   366→        </Card>
   367→   367→      </div>
   368→   368→
   369→   369→      {/* Error Alert */}
   370→   370→      {error && (
   371→   371→        <div className="flex items-center gap-2 rounded-lg border border-red-200 bg-red-50 dark:border-red-900 dark:bg-red-950/30 p-3 text-sm text-red-700 dark:text-red-400">
   372→   372→          <AlertCircle className="size-4 shrink-0" />
   373→   373→          {error}
   374→   374→          <Button variant="ghost" size="sm" className="ml-auto" onClick={() => setError(null)}>
   375→   375→            Dismiss
   376→   376→          </Button>
   377→   377→        </div>
   378→   378→      )}
   379→   379→
   380→   380→      {/* Tabs */}
   381→   381→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   382→   382→        <TabsList className="grid w-full grid-cols-4">
   383→   383→          <TabsTrigger value="overview" className="text-xs sm:text-sm">Stock Overview</TabsTrigger>
   384→   384→          <TabsTrigger value="low-stock" className="text-xs sm:text-sm">Low Stock</TabsTrigger>
   385→   385→          <TabsTrigger value="warehouses" className="text-xs sm:text-sm">Warehouses</TabsTrigger>
   386→   386→          <TabsTrigger value="transfers" className="text-xs sm:text-sm">Transfers</TabsTrigger>
   387→   387→        </TabsList>
   388→   388→
   389→   389→        {/* Stock Overview Tab */}
   390→   390→        <TabsContent value="overview" className="space-y-4">
   391→   391→          {/* Filters */}
   392→   392→          <Card>
   393→   393→            <CardContent className="p-4">
   394→   394→              <div className="flex flex-col gap-3 sm:flex-row">
   395→   395→                <div className="flex items-center gap-2 flex-1">
   396→   396→                  <Building2 className="size-4 text-muted-foreground shrink-0" />
   397→   397→                  <Select value={branchFilter} onValueChange={setBranchFilter}>
   398→   398→                    <SelectTrigger className="w-full">
   399→   399→                      <SelectValue placeholder="All Branches" />
   400→   400→                    </SelectTrigger>
   401→   401→                    <SelectContent>
   402→   402→                      <SelectItem value="all">All Branches</SelectItem>
   403→   403→                      {branches.map((b) => (
   404→   404→                        <SelectItem key={b.id} value={b.id}>{b.name}</SelectItem>
   405→   405→                      ))}
   406→   406→                    </SelectContent>
   407→   407→                  </Select>
   408→   408→                </div>
   409→   409→                <div className="flex items-center gap-2 flex-1">
   410→   410→                  <WarehouseIcon className="size-4 text-muted-foreground shrink-0" />
   411→   411→                  <Select value={warehouseFilter} onValueChange={setWarehouseFilter}>
   412→   412→                    <SelectTrigger className="w-full">
   413→   413→                      <SelectValue placeholder="All Warehouses" />
   414→   414→                    </SelectTrigger>
   415→   415→                    <SelectContent>
   416→   416→                      <SelectItem value="all">All Warehouses</SelectItem>
   417→   417→                      {warehouses.map((w) => (
   418→   418→                        <SelectItem key={w.id} value={w.id}>{w.name}</SelectItem>
   419→   419→                      ))}
   420→   420→                    </SelectContent>
   421→   421→                  </Select>
   422→   422→                </div>
   423→   423→                <Button
   424→   424→                  variant={lowStockOnly ? 'default' : 'outline'}
   425→   425→                  size="sm"
   426→   426→                  className={lowStockOnly ? 'bg-amber-600 hover:bg-amber-700' : ''}
   427→   427→                  onClick={() => setLowStockOnly(!lowStockOnly)}
   428→   428→                >
   429→   429→                  <AlertTriangle className="size-4 mr-1.5" />
   430→   430→                  Low Stock Only
   431→   431→                </Button>
   432→   432→              </div>
   433→   433→            </CardContent>
   434→   434→          </Card>
   435→   435→
   436→   436→          {/* Inventory Table */}
   437→   437→          <Card>
   438→   438→            <CardContent className="p-0">
   439→   439→              {loading ? (
   440→   440→                <div className="p-4 space-y-3">
   441→   441→                  {Array.from({ length: 5 }).map((_, i) => (
   442→   442→                    <Skeleton key={i} className="h-12 w-full" />
   443→   443→                  ))}
   444→   444→                </div>
   445→   445→              ) : items.length === 0 ? (
   446→   446→                <div className="flex flex-col items-center justify-center py-12 text-center">
   447→   447→                  <Package className="size-12 text-muted-foreground/40 mb-3" />
   448→   448→                  <p className="text-muted-foreground">No inventory items found</p>
   449→   449→                </div>
   450→   450→              ) : (
   451→   451→                <div className="overflow-x-auto">
   452→   452→                  <Table>
