   176→function SchoolsTab() {
   177→  const [schools, setSchools] = React.useState<School[]>([])
   178→  const [loading, setLoading] = React.useState(true)
   179→  const [search, setSearch] = React.useState('')
   180→  const [showAddDialog, setShowAddDialog] = React.useState(false)
   181→  const [selectedSchool, setSelectedSchool] = React.useState<School | null>(null)
   182→
   183→  const fetchSchools = React.useCallback(async () => {
   184→    setLoading(true)
   185→    try {
   186→      const res = await fetch(`/api/schools?search=${encodeURIComponent(search)}&limit=50`)
   187→      const json = await res.json()
   188→      if (json.data) setSchools(json.data)
   189→    } catch (err) {
   190→      console.error('Failed to fetch schools:', err)
   191→    } finally {
   192→      setLoading(false)
   193→    }
   194→  }, [search])
   195→
   196→  React.useEffect(() => {
   197→    fetchSchools()
   198→  }, [fetchSchools])
   199→
   200→  const handleDelete = async (id: string) => {
   201→    if (!confirm('Are you sure you want to deactivate this school?')) return
   202→    try {
   203→      await fetch(`/api/schools?id=${id}`, { method: 'DELETE' })
   204→      fetchSchools()
   205→    } catch (err) {
   206→      console.error('Failed to delete school:', err)
   207→    }
   208→  }
   209→
   210→  return (
   211→    <div className="space-y-4">
   212→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   213→        <div className="relative flex-1 w-full sm:max-w-sm">
   214→          <Search className="absolute left-2.5 top-1.5 size-3.5 text-muted-foreground" />
   215→          <Input
   216→            placeholder="Search schools..."
   217→            value={search}
   218→            onChange={(e) => setSearch(e.target.value)}
   219→            className="pl-8 rounded-lg bg-secondary/80 border-0 text-[13px] h-8"
   220→          />
   221→        </div>
   222→        <Button onClick={() => setShowAddDialog(true)} className="gap-1.5 bg-primary text-primary-foreground rounded-lg font-medium text-[13px]">
   223→          <Plus className="size-3.5" strokeWidth={2} /> Add School
   224→        </Button>
   225→      </div>
   226→
   227→      {loading ? (
   228→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   229→          {[1, 2, 3].map((i) => (
   230→            <Card key={i} className="rounded-2xl macos-shadow border-0 animate-pulse">
   231→              <CardContent className="p-4">
   232→                <div className="h-4 bg-muted rounded w-2/3 mb-3" />
   233→                <div className="h-3 bg-muted rounded w-1/2 mb-2" />
   234→                <div className="h-3 bg-muted rounded w-3/4" />
   235→              </CardContent>
   236→            </Card>
   237→          ))}
   238→        </div>
   239→      ) : schools.length === 0 ? (
   240→        <Card className="rounded-2xl macos-shadow border-0">
   241→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   242→            <GraduationCap className="size-10 text-muted-foreground mb-3" strokeWidth={1.5} />
   243→            <p className="text-[13px] text-muted-foreground">No schools found</p>
   244→            <p className="text-[11px] text-muted-foreground">Add your first school to get started</p>
   245→          </CardContent>
   246→        </Card>
   247→      ) : (
   248→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   249→          {schools.map((school) => (
   250→            <Card
   251→              key={school.id}
   252→              className="rounded-2xl macos-shadow border-0 macos-card-hover cursor-pointer group"
   253→              onClick={() => setSelectedSchool(school)}
   254→            >
   255→              <CardContent className="p-4">
   256→                <div className="flex items-start justify-between mb-3">
   257→                  <div className="flex items-center gap-2.5">
   258→                    <div className="size-8 rounded-xl bg-macos-blue/10 flex items-center justify-center">
   259→                      <GraduationCap className="size-4 text-macos-blue" strokeWidth={1.5} />
   260→                    </div>
   261→                    <div>
   262→                      <h3 className="font-semibold text-[13px] leading-tight tracking-tight">{school.name}</h3>
   263→                      {school.academicSession && (
   264→                        <Badge className="rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 bg-macos-purple/10 text-macos-purple">
   265→                          {school.academicSession}
   266→                        </Badge>
   267→                      )}
   268→                    </div>
   269→                  </div>
   270→                  <Badge
   271→                    className={`rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 ${
   272→                      school.isActive
   273→                        ? 'bg-macos-green/10 text-macos-green'
   274→                        : 'bg-macos-red/10 text-macos-red'
   275→                    }`}
   276→                  >
   277→                    {school.isActive ? 'Active' : 'Inactive'}
   278→                  </Badge>
   279→                </div>
   280→                <div className="space-y-1.5 text-[11px] text-muted-foreground">
   281→                  {school.address && (
   282→                    <div className="flex items-center gap-1.5">
   283→                      <MapPin className="size-3 shrink-0" strokeWidth={1.5} />
   284→                      <span className="truncate">{school.address}</span>

   286→                  )}
   287→                  {school.phone && (
   288→                    <div className="flex items-center gap-1.5">
   289→                      <Phone className="size-3 shrink-0" strokeWidth={1.5} />
   290→                      {school.phone}
   291→                    </div>
   292→                  )}
   293→                  {school.contactPerson && (
   294→                    <div className="flex items-center gap-1.5">
   295→                      <User className="size-3 shrink-0" strokeWidth={1.5} />
   296→                      {school.contactPerson}

   298→                  )}
   299→                </div>
   300→                <Separator className="my-3 bg-border/50" />
   301→                <div className="flex items-center justify-between text-[11px]">
   302→                  <span className="text-muted-foreground">
   303→                    {school._count?.contracts || 0} contracts
   304→                  </span>
   305→                  <span className="text-muted-foreground">
   306→                    {school._count?.packages || 0} packages
   307→                  </span>
   308→                  <Button
   309→                    variant="ghost"
   310→                    size="sm"
   311→                    className="h-6 px-2 text-[11px] opacity-0 group-hover:opacity-100 text-macos-red hover:text-macos-red hover:bg-macos-red/10 rounded-lg"
   312→                    onClick={(e) => {
   313→                      e.stopPropagation()
   314→                      handleDelete(school.id)
   315→                    }}
   316→                  >
   317→                    <Trash2 className="size-3" strokeWidth={1.5} />
   318→                  </Button>
   319→                </div>
   320→              </CardContent>
   321→            </Card>
   322→          ))}
   323→        </div>
   324→      )}
   325→
   326→      <AddSchoolDialog open={showAddDialog} onOpenChange={setShowAddDialog} onSuccess={fetchSchools} />
   327→      <SchoolDetailDialog school={selectedSchool} onOpenChange={(open) => !open && setSelectedSchool(null)} onSuccess={fetchSchools} />
   328→    </div>
   329→  )
   330→}

   334→function ContractsTab() {
   335→  const [contracts, setContracts] = React.useState<SchoolContract[]>([])
   336→  const [schools, setSchools] = React.useState<School[]>([])
   337→  const [loading, setLoading] = React.useState(true)
   338→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   339→
   340→  const fetchContracts = React.useCallback(async () => {
   341→    setLoading(true)
   342→    try {
   343→      const res = await fetch('/api/schools/contracts')
   344→      if (res.ok) {
   345→        const json = await res.json()
   346→        if (json.data) setContracts(json.data)
   347→      } else {
   348→        const schoolRes = await fetch('/api/schools?limit=50')
   349→        const schoolJson = await schoolRes.json()
   350→        if (schoolJson.data) {
   351→          const allContracts: SchoolContract[] = []
   352→          schoolJson.data.forEach((s: School & { contracts?: SchoolContract[] }) => {
   353→            if (s.contracts) {
   354→              s.contracts.forEach((c) => allContracts.push({ ...c, school: { id: s.id, name: s.name } }))
   355→            }
   356→          })
   357→          setContracts(allContracts)
   358→        }
   359→      }
   360→    } catch (err) {
   361→      console.error('Failed to fetch contracts:', err)
   362→    } finally {
   363→      setLoading(false)
   364→    }
   365→  }, [])
   366→
   367→  const fetchSchools = React.useCallback(async () => {
   368→    try {
   369→      const res = await fetch('/api/schools?limit=50')
   370→      const json = await res.json()
   371→      if (json.data) setSchools(json.data)
   372→    } catch (err) {
   373→      console.error('Failed to fetch schools:', err)
   374→    }
   375→  }, [])
   376→
   377→  React.useEffect(() => {
   378→    fetchContracts()
   379→    fetchSchools()
   380→  }, [fetchContracts, fetchSchools])
   381→
   382→  const getStatusBadge = (status: string) => {
   383→    switch (status) {
   384→      case 'active':
   385→        return 'bg-macos-green/10 text-macos-green'
   386→      case 'expired':
   387→        return 'bg-macos-orange/10 text-macos-orange'
   388→      case 'cancelled':
   389→        return 'bg-macos-red/10 text-macos-red'
   390→      default:
   391→        return 'bg-secondary text-secondary-foreground'
   392→    }
   393→  }
   394→
   395→  return (
   396→    <div className="space-y-4">
   397→      <div className="flex items-center justify-between">
   398→        <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">{contracts.length} contracts</p>
   399→        <Button onClick={() => setShowCreateDialog(true)} className="gap-1.5 bg-primary text-primary-foreground rounded-lg font-medium text-[13px]">
   400→          <Plus className="size-3.5" strokeWidth={2} /> Create Contract
   401→        </Button>
   402→      </div>
   403→
   404→      {loading ? (
   405→        <Card className="rounded-2xl macos-shadow border-0">
   406→          <CardContent className="p-6">
   407→            <div className="space-y-3">
   408→              {[1, 2, 3].map((i) => (
   409→                <div key={i} className="h-12 bg-muted rounded-xl animate-pulse" />
   410→              ))}
   411→            </div>
   412→          </CardContent>
   413→        </Card>
   414→      ) : contracts.length === 0 ? (
   415→        <Card className="rounded-2xl macos-shadow border-0">
   416→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   417→            <FileText className="size-10 text-muted-foreground mb-3" strokeWidth={1.5} />
   418→            <p className="text-[13px] text-muted-foreground">No contracts found</p>
   419→            <p className="text-[11px] text-muted-foreground">Create your first school contract</p>
   420→          </CardContent>
   421→        </Card>
   422→      ) : (
   423→        <Card className="rounded-2xl macos-shadow border-0">
   424→          <CardContent className="p-0">
   425→            <ScrollArea className="max-h-[500px]">
   426→              <Table>
   427→                <TableHeader>
   428→                  <TableRow className="hover:bg-transparent">
   429→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Title</TableHead>
   430→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">School</TableHead>
   431→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Session</TableHead>
   432→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Value</TableHead>
   433→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Status</TableHead>
   434→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Period</TableHead>
   435→                  </TableRow>
   436→                </TableHeader>
   437→                <TableBody>
   438→                  {contracts.map((contract) => (
   439→                    <TableRow key={contract.id} className="hover:bg-secondary/50">

   448→                      </TableCell>

   453→                      </TableCell>

   455→                  ))}

   457→              </Table>
   458→            </ScrollArea>
   459→          </CardContent>
   460→        </Card>
   461→      )}
   462→
   463→      <CreateContractDialog
   464→        open={showCreateDialog}
   465→        onOpenChange={setShowCreateDialog}
   466→        schools={schools}
   467→        onSuccess={fetchContracts}
   468→      />
   469→    </div>
   470→  )
   471→}

   475→function PackagesTab() {

   478→  const [books, setBooks] = React.useState<Book[]>([])

   482→  const [selectedPackage, setSelectedPackage] = React.useState<AcademicPackage | null>(null)
   483→

   485→    setLoading(true)
   486→    try {
   487→      const res = await fetch('/api/schools/packages')
   488→      if (res.ok) {
   489→        const json = await res.json()
   490→        if (json.data) setPackages(json.data)
   491→      } else {
   492→        const schoolRes = await fetch('/api/schools?limit=50')
   493→        const schoolJson = await schoolRes.json()

   496→          schoolJson.data.forEach((s: School & { packages?: AcademicPackage[] }) => {

   499→            }
   500→          })
   501→          setPackages(allPackages)
   502→        }
   503→      }
   504→    } catch (err) {
   505→      console.error('Failed to fetch packages:', err)
   506→    } finally {
   507→      setLoading(false)
   508→    }
   509→  }, [])
   510→
   511→  const fetchReferenceData = React.useCallback(async () => {
   512→    try {
   513→      const [schoolsRes, booksRes, stationeryRes] = await Promise.all([
   514→        fetch('/api/schools?limit=50'),
   515→        fetch('/api/books?limit=100'),
   516→        fetch('/api/stationery?limit=100'),
   517→      ])
   518→      const [schoolsJson, booksJson, stationeryJson] = await Promise.all([
   519→        schoolsRes.json(),
   520→        booksRes.json(),
   521→        stationeryRes.json(),
   522→      ])
   523→      if (schoolsJson.data) setSchools(schoolsJson.data)
   524→      if (booksJson.data) setBooks(booksJson.data)
   525→      if (stationeryJson.data) setStationery(stationeryJson.data)
   526→    } catch (err) {
   527→      console.error('Failed to fetch reference data:', err)
   528→    }
   529→  }, [])
   530→
   531→  React.useEffect(() => {
   532→    fetchPackages()
   533→    fetchReferenceData()
   534→  }, [fetchPackages, fetchReferenceData])
   535→
   536→  return (
   537→    <div className="space-y-4">
   538→      <div className="flex items-center justify-between">
   539→        <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">{packages.length} academic packages</p>
   540→        <Button onClick={() => setShowCreateDialog(true)} className="gap-1.5 bg-primary text-primary-foreground rounded-lg font-medium text-[13px]">
   541→          <Plus className="size-3.5" strokeWidth={2} /> Create Package
   542→        </Button>
   543→      </div>
   544→
   545→      {loading ? (
   546→        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   547→          {[1, 2, 3, 4].map((i) => (
   548→            <Card key={i} className="rounded-2xl macos-shadow border-0 animate-pulse">
   549→              <CardContent className="p-4">
   550→                <div className="h-4 bg-muted rounded w-2/3 mb-3" />
   551→                <div className="h-3 bg-muted rounded w-1/2 mb-2" />

  1060→  const totalPrice = items.reduce((sum, item) => sum + item.quantity * item.unitPrice, 0)
  1061→
  1062→  const addItem = () => {
  1063→    setItems([
  1064→      ...items,
  1065→      { key: crypto.randomUUID(), itemType: 'book', itemId: '', quantity: 1, unitPrice: 0, name: '' },
  1066→    ])
  1067→  }
  1068→
  1069→  const removeItem = (key: string) => {
  1070→    if (items.length > 1) {
  1071→      setItems(items.filter((i) => i.key !== key))
  1072→    }
  1073→  }
  1074→
  1075→  const updateItem = (key: string, updates: Partial<PackageItemForm>) => {
  1076→    setItems(
  1077→      items.map((item) => {
  1078→        if (item.key !== key) return item
  1079→        const updated = { ...item, ...updates }
  1080→        if (updates.itemId) {
  1081→          if (updated.itemType === 'book') {
  1082→            const book = books.find((b) => b.id === updates.itemId)
  1083→            if (book) {
  1084→              updated.unitPrice = book.price
  1085→              updated.name = book.title
  1086→            }
  1087→          } else {
  1088→            const product = stationery.find((s) => s.id === updates.itemId)
  1089→            if (product) {
  1090→              updated.unitPrice = product.price
  1091→              updated.name = product.name
  1092→            }
  1093→          }
  1094→        }
  1095→        return updated
  1096→      })
  1097→    )
  1098→  }
  1099→
  1100→  const handleSubmit = async (e: React.FormEvent) => {
  1101→    e.preventDefault()
  1102→    if (!form.name.trim()) return
  1103→    setSaving(true)
  1104→    try {
  1105→      const res = await fetch('/api/schools/packages', {
  1106→        method: 'POST',
  1107→        headers: { 'Content-Type': 'application/json' },
  1108→        body: JSON.stringify({
  1109→          schoolId: form.schoolId || null,
  1110→          name: form.name,
  1111→          grade: form.grade,
  1112→          academicLevel: form.academicLevel,
  1113→          description: form.description,
  1114→          totalPrice,
  1115→          discount: parseFloat(form.discount) || 0,
  1116→          items: items
  1117→            .filter((i) => i.itemId)
  1118→            .map((i) => ({
  1119→              bookId: i.itemType === 'book' ? i.itemId : null,
  1120→              stationeryId: i.itemType === 'stationery' ? i.itemId : null,
  1121→              quantity: i.quantity,
  1122→              unitPrice: i.unitPrice,
  1123→            })),
  1124→        }),
  1125→      })
  1126→      if (res.ok) {
  1127→        setForm({ schoolId: '', name: '', grade: '', academicLevel: '', description: '', discount: '' })
  1128→        setItems([{ key: crypto.randomUUID(), itemType: 'book', itemId: '', quantity: 1, unitPrice: 0, name: '' }])
  1129→        onOpenChange(false)
  1130→        onSuccess()
  1131→      } else {
  1132→        const json = await res.json()
  1133→        alert(json.error || 'Failed to create package')
  1134→      }
  1135→    } catch (err) {
  1136→      console.error('Failed to create package:', err)
  1137→      alert('Failed to create package')
  1138→    } finally {
  1139→      setSaving(false)
  1140→    }
  1141→  }
  1142→
  1143→  return (
  1144→    <Dialog open={open} onOpenChange={onOpenChange}>
  1145→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto rounded-2xl">
  1146→        <DialogHeader>
  1147→          <DialogTitle className="text-[13px] font-semibold tracking-tight">Create Academic Package</DialogTitle>
  1148→          <DialogDescription className="text-[11px]">Build a grade-wise package with books and stationery</DialogDescription>
  1149→        </DialogHeader>
  1150→        <form onSubmit={handleSubmit} className="space-y-4">
  1151→          <div className="grid grid-cols-2 gap-3">
  1152→            <div className="space-y-2">
  1153→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">School</Label>
  1154→              <Select value={form.schoolId} onValueChange={(v) => setForm({ ...form, schoolId: v })}>
  1155→                <SelectTrigger className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8">
  1156→                  <SelectValue placeholder="Select school" />
  1157→                </SelectTrigger>
  1158→                <SelectContent>
  1159→                  {schools.map((s) => (
  1160→                    <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
  1161→                  ))}
  1162→                </SelectContent>
  1163→              </Select>
  1164→            </div>
  1165→            <div className="space-y-2">
  1166→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Package Name *</Label>
  1167→              <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="Grade 1 Package" className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" required />
  1168→            </div>
  1169→          </div>
  1170→          <div className="grid grid-cols-3 gap-3">
  1171→            <div className="space-y-2">
  1172→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Grade</Label>
  1173→              <Input value={form.grade} onChange={(e) => setForm({ ...form, grade: e.target.value })} placeholder="1" className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" />
  1174→            </div>
  1175→            <div className="space-y-2">
  1176→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Academic Level</Label>
  1177→              <Input value={form.academicLevel} onChange={(e) => setForm({ ...form, academicLevel: e.target.value })} placeholder="Primary" className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" />
  1178→            </div>
  1179→            <div className="space-y-2">
  1180→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Discount (%)</Label>
  1181→              <Input type="number" min="0" max="100" value={form.discount} onChange={(e) => setForm({ ...form, discount: e.target.value })} placeholder="10" className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" />
  1182→            </div>
  1183→          </div>
  1184→          <div className="space-y-2">
  1185→            <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Description</Label>
  1186→            <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} placeholder="Package description..." rows={2} className="rounded-lg bg-secondary/80 border-0 text-[13px]" />
  1187→          </div>
  1188→
  1189→          <Separator className="bg-border/50" />
  1190→
  1191→          <div className="space-y-3">
  1192→            <div className="flex items-center justify-between">
  1193→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Package Items</Label>
  1194→              <Button type="button" variant="ghost" size="sm" onClick={addItem} className="gap-1 text-[11px] text-macos-blue hover:text-macos-blue rounded-lg">
  1195→                <Plus className="size-3" strokeWidth={2} /> Add Item
  1196→              </Button>
  1197→            </div>
  1198→            {items.map((item) => (
  1199→              <div key={item.key} className="flex items-end gap-2 p-3 rounded-xl bg-secondary/50">
  1200→                <div className="flex-1 space-y-1">
  1201→                  <Label className="text-[10px] font-semibold tracking-wide text-muted-foreground">Type</Label>
  1202→                  <Select
  1203→                    value={item.itemType}
  1204→                    onValueChange={(v: 'book' | 'stationery') =>
  1205→                      updateItem(item.key, { itemType: v, itemId: '', unitPrice: 0, name: '' })
  1206→                    }
  1207→                  >
  1208→                    <SelectTrigger className="h-7 text-[11px] rounded-lg border-0 bg-background">
  1209→                      <SelectValue />
  1210→                    </SelectTrigger>
  1211→                    <SelectContent>
  1212→                      <SelectItem value="book"><span className="flex items-center gap-1"><BookOpen className="size-3" /> Book</span></SelectItem>
  1213→                      <SelectItem value="stationery"><span className="flex items-center gap-1"><PencilRuler className="size-3" /> Stationery</span></SelectItem>
  1214→                    </SelectContent>
  1215→                  </Select>
  1216→                </div>
  1217→                <div className="flex-[2] space-y-1">
  1218→                  <Label className="text-[10px] font-semibold tracking-wide text-muted-foreground">Item</Label>
  1219→                  <Select value={item.itemId} onValueChange={(v) => updateItem(item.key, { itemId: v })}>
  1220→                    <SelectTrigger className="h-7 text-[11px] rounded-lg border-0 bg-background">
  1221→                      <SelectValue placeholder={`Select ${item.itemType}...`} />
  1222→                    </SelectTrigger>
  1223→                    <SelectContent>
  1224→                      {(item.itemType === 'book' ? books : stationery).map((b) => (
  1225→                        <SelectItem key={b.id} value={b.id}>
  1226→                          {b.name || b.title} — ${b.price}
  1227→                        </SelectItem>
  1228→                      ))}
  1229→                    </SelectContent>
  1230→                  </Select>
  1231→                </div>
  1232→                <div className="w-16 space-y-1">
  1233→                  <Label className="text-[10px] font-semibold tracking-wide text-muted-foreground">Qty</Label>
  1234→                  <Input type="number" min="1" value={item.quantity} onChange={(e) => updateItem(item.key, { quantity: parseInt(e.target.value) || 1 })} className="h-7 text-[11px] rounded-lg border-0 bg-background" />
  1235→                </div>
  1236→                <div className="w-20 space-y-1">
  1237→                  <Label className="text-[10px] font-semibold tracking-wide text-muted-foreground">Price</Label>
  1238→                  <Input type="number" step="0.01" min="0" value={item.unitPrice} onChange={(e) => updateItem(item.key, { unitPrice: parseFloat(e.target.value) || 0 })} className="h-7 text-[11px] rounded-lg border-0 bg-background" />
  1239→                </div>
  1240→                <Button
  1241→                  type="button"
  1242→                  variant="ghost"
  1243→                  size="sm"
  1244→                  className="h-7 w-7 p-0 text-macos-red hover:text-macos-red hover:bg-macos-red/10 rounded-lg"
  1245→                  onClick={() => removeItem(item.key)}
  1246→                  disabled={items.length <= 1}
  1247→                >
  1248→                  <X className="size-3" />
  1249→                </Button>
  1250→              </div>
  1251→            ))}
  1252→          </div>
  1253→
  1254→          <div className="flex items-center justify-between p-3 rounded-xl bg-macos-green/10">
  1255→            <span className="text-[13px] font-medium">Total Package Price</span>
  1256→            <span className="text-[13px] font-bold tracking-tight text-macos-green">${totalPrice.toLocaleString()}</span>
  1257→          </div>
  1258→
  1259→          <DialogFooter>
  1260→            <Button type="button" variant="ghost" onClick={() => onOpenChange(false)} className="rounded-lg text-[13px]">Cancel</Button>
  1261→            <Button type="submit" disabled={saving || !form.name.trim()} className="bg-primary text-primary-foreground rounded-lg font-medium text-[13px]">
  1262→              {saving ? 'Creating...' : 'Create Package'}
  1263→            </Button>
  1264→          </DialogFooter>
  1265→        </form>
  1266→      </DialogContent>
  1267→    </Dialog>
  1268→  )
  1269→}
