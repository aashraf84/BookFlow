   104→   104→function getNextStatus(current: string): string[] {
   105→   105→  if (current === 'rejected') return []
   106→   106→  const idx = statusWorkflow.indexOf(current)
   107→   107→  if (idx === -1) return []
   108→   108→  if (idx === statusWorkflow.length - 1) return []
   109→   109→  return [statusWorkflow[idx + 1]]
   110→   110→}

   113→   113→function PipelineStep({
   114→   114→  label,
   115→   115→  icon: Icon,
   116→   116→  isActive,
   117→   117→  isCompleted,
   118→   118→  count,
   119→   119→  color,
   120→   120→  bgColor,
   121→   121→  isLast,
   122→   122→}: {
   123→   123→  label: string
   124→   124→  icon: React.ElementType
   125→   125→  isActive: boolean
   126→   126→  isCompleted: boolean
   127→   127→  count: number
   128→   128→  color: string
   129→   129→  bgColor: string
   130→   130→  isLast: boolean
   131→   131→}) {
   132→   132→  return (
   133→   133→    <div className="flex items-center gap-0 flex-1 min-w-0">

   135→   135→        <div
   136→   136→          className={`relative flex items-center justify-center size-10 rounded-xl transition-all duration-300 ${
   137→   137→            isActive
   138→   138→              ? `${bgColor} ring-2 ring-offset-1 ring-offset-background ${color.replace('text-', 'ring-')}`
   139→   139→              : isCompleted
   140→   140→              ? bgColor
   141→   141→              : 'bg-secondary/60'
   142→   142→          }`}
   143→   143→        >
   144→   144→          <Icon className={`size-4 ${isActive || isCompleted ? color : 'text-muted-foreground/50'}`} />
   145→   145→          {count > 0 && (
   146→   146→            <span className={`absolute -top-1.5 -right-1.5 flex items-center justify-center min-w-[18px] h-[18px] rounded-full text-[9px] font-bold px-1 ${
   147→   147→              isActive ? 'bg-macos-blue text-white' : isCompleted ? bgColor + ' ' + color : 'bg-secondary text-muted-foreground'

   150→   150→            </span>
   151→   151→          )}
   152→   152→        </div>
   153→   153→        <div className="text-center">
   154→   154→          <p className={`text-[10px] font-semibold uppercase tracking-wider ${isActive ? color : isCompleted ? 'text-foreground' : 'text-muted-foreground/60'}`}>

   157→   157→        </div>
   158→   158→      </div>

   160→   160→        <div className="flex items-center pt-[-16px] self-start mt-5 -ml-1 -mr-1">
   161→   161→          <div className={`h-[2px] w-full min-w-[24px] rounded-full transition-colors ${
   162→   162→            isCompleted ? 'bg-macos-green/40' : 'bg-border'
   163→   163→          }`} />
   164→   164→          <ArrowRight className={`size-3 -ml-1 ${isCompleted ? 'text-macos-green/40' : 'text-muted-foreground/30'}`} />
   165→   165→        </div>
   166→   166→      )}
   167→   167→    </div>
   168→   168→  )
   169→   169→}
