    88→  ordered: 'bg-macos-orange/10 text-macos-orange',
    89→  printing: 'bg-macos-blue/10 text-macos-blue',
    90→  completed: 'bg-macos-green/10 text-macos-green',
    91→}
    92→
    93→const statusWorkflow = ['submitted', 'reviewed', 'approved', 'published']
    94→const rejectionStates = ['rejected']
    95→
    96→function getNextStatus(current: string): string[] {
    97→  if (current === 'rejected') return []
    98→  const idx = statusWorkflow.indexOf(current)
    99→  if (idx === -1) return []
   100→  if (idx === statusWorkflow.length - 1) return []
   101→  return [statusWorkflow[idx + 1]]
   102→}

   104→export function PublishingModule() {
   105→  const { toast } = useToast()
   106→  const [manuscripts, setManuscripts] = React.useState<Manuscript[]>([])

   108→  const [books, setBooks] = React.useState<BookOption[]>([])
   109→  const [loading, setLoading] = React.useState(true)
   110→  const [manuscriptDialogOpen, setManuscriptDialogOpen] = React.useState(false)

   113→  const [activeTab, setActiveTab] = React.useState('manuscripts')
   114→

   123→      setManuscripts(json.data || [])

   125→      toast({ title: 'Error', description: 'Failed to fetch manuscripts', variant: 'destructive' })
   126→    }
   127→  }, [toast])
   128→
   129→  const fetchPrintOrders = React.useCallback(async () => {
   130→    try {
   131→      const res = await fetch('/api/publishing?type=print-orders')

   133→      const json = await res.json()
   134→      setPrintOrders(json.data || [])
   135→    } catch {
   136→      toast({ title: 'Error', description: 'Failed to fetch print orders', variant: 'destructive' })
   137→    }

   139→

   141→    try {
   142→      const res = await fetch('/api/books')

   145→      setBooks(json.data || [])
   146→    } catch {
   147→      // Silently fail
   148→    }
   149→  }, [])
   150→
   151→  React.useEffect(() => {
   152→    const loadAll = async () => {
   153→      setLoading(true)
   154→      await Promise.all([fetchManuscripts(), fetchPrintOrders(), fetchBooks()])
   155→      setLoading(false)
   156→    }
   157→    loadAll()
   158→  }, [fetchManuscripts, fetchPrintOrders, fetchBooks])
   159→
   160→  const handleSubmitManuscript = async () => {

   162→      toast({ title: 'Validation Error', description: 'Title is required', variant: 'destructive' })
   163→      return
   164→    }

   167→      const res = await fetch('/api/publishing?type=manuscripts', {

   175→      setMsForm({ title: '', author: '', notes: '' })

   178→      toast({ title: 'Error', description: 'Failed to submit manuscript', variant: 'destructive' })
   179→    } finally {
   180→      setSubmitting(false)
   181→    }
   182→  }
   183→

   185→    if (!poForm.bookId || !poForm.quantity) {
   186→      toast({ title: 'Validation Error', description: 'Book and quantity are required', variant: 'destructive' })
   187→      return
   188→    }
   189→    try {
   190→      setSubmitting(true)
   191→      const res = await fetch('/api/publishing?type=print-orders', {
   192→        method: 'POST',

   195→          bookId: poForm.bookId,
   196→          quantity: parseInt(poForm.quantity),

   201→      toast({ title: 'Success', description: 'Print order created successfully' })

   203→      setPoForm({ bookId: '', quantity: '', costPerUnit: '' })
   204→      fetchPrintOrders()
   205→    } catch {
   206→      toast({ title: 'Error', description: 'Failed to create print order', variant: 'destructive' })
   207→    } finally {
   208→      setSubmitting(false)
   209→    }
   210→  }
   211→
   212→  const handleManuscriptStatus = async (manuscript: Manuscript, newStatus: string) => {
   213→    try {
   214→      const res = await fetch('/api/publishing?type=manuscripts', {
   215→        method: 'PUT',
   216→        headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({ id: manuscript.id, status: newStatus }),
   218→      })

   220→      toast({ title: 'Success', description: `Manuscript status updated to ${newStatus}` })

   223→      toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })
   224→    }

   226→

   228→    try {
   229→      const res = await fetch('/api/publishing?type=print-orders', {
   230→        method: 'PUT',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({ id: order.id, status: newStatus }),

   236→      fetchPrintOrders()
   237→    } catch {
   238→      toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })
   239→    }
   240→  }
   241→

   248→

   252→  ]

   256→      {/* Summary Metrics */}

   258→        <div className="rounded-2xl macos-shadow p-4 bg-card">
   259→          <div className="flex items-center justify-between">
   260→            <div>
   261→              <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Manuscripts</p>

   263→            </div>
   264→            <div className="rounded-lg bg-macos-blue/10 p-2.5">

   266→            </div>
   267→          </div>
   268→        </div>

   278→          </div>
   279→        </div>
   280→        <div className="rounded-2xl macos-shadow p-4 bg-card">
   281→          <div className="flex items-center justify-between">
   282→            <div>
   283→              <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Published</p>
   284→              <p className="text-2xl font-bold tracking-tight">{msByStatus['published'] || 0}</p>
   285→            </div>
   286→            <div className="rounded-lg bg-macos-purple/10 p-2.5">
   287→              <BookOpen className="size-5 text-macos-purple" />
   288→            </div>
   289→          </div>
   290→        </div>

   292→          <div className="flex items-center justify-between">
   293→            <div>

   295→              <p className="text-2xl font-bold tracking-tight">{printOrders.length}</p>
   296→            </div>
   297→            <div className="rounded-lg bg-macos-teal/10 p-2.5">
   298→              <Printer className="size-5 text-macos-teal" />
   299→            </div>
   300→          </div>
   301→        </div>
   302→        <div className="rounded-2xl macos-shadow p-4 bg-card">
   303→          <div className="flex items-center justify-between">
   304→            <div>
   305→              <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Print Cost</p>
   306→              <p className="text-2xl font-bold tracking-tight">${totalPrintCost.toLocaleString()}</p>
   307→            </div>
   308→            <div className="rounded-lg bg-macos-green/10 p-2.5">
   309→              <DollarSign className="size-5 text-macos-green" />
   310→            </div>
   311→          </div>
   312→        </div>
   313→      </div>
   314→
   315→      {/* Header + Segmented Tabs */}
   316→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   317→        <div>
   318→          <h2 className="text-xl font-bold tracking-tight">Publishing</h2>
   319→          <p className="text-[13px] text-muted-foreground mt-0.5">Manage manuscripts, reviews, and print production</p>
   320→        </div>
   321→        <div className="flex items-center gap-3">
   322→          <div className="macos-segmented inline-flex p-0.5">
   323→            {tabs.map((tab) => (
   324→              <button
   325→                key={tab.id}
   326→                onClick={() => setActiveTab(tab.id)}
   327→                className={`px-3 py-1 text-[12px] font-medium rounded-md transition-colors ${
   328→                  activeTab === tab.id ? 'macos-segmented-active' : 'text-muted-foreground hover:text-foreground'
   329→                }`}
   330→              >
   331→                {tab.label}
   332→              </button>
   333→            ))}
   334→          </div>
   335→          <Button
   336→            onClick={() => setManuscriptDialogOpen(true)}
   337→            className="bg-secondary text-secondary-foreground rounded-lg font-medium text-[13px]"
   338→          >
   339→            <Plus className="size-3.5 mr-1.5" strokeWidth={2} />
   340→            Submit Manuscript
   341→          </Button>
   342→          <Button
   343→            onClick={() => setPrintOrderDialogOpen(true)}
   344→            className="bg-primary text-primary-foreground rounded-lg font-medium text-[13px]"
   345→          >
   346→            <Plus className="size-3.5 mr-1.5" strokeWidth={2} />
   347→            Print Order
   348→          </Button>
   349→        </div>
   350→      </div>
   351→
   352→      {/* Manuscripts Tab */}
   353→      {activeTab === 'manuscripts' && (
   354→        <div>
   355→          {loading ? (
   356→            <div className="rounded-2xl macos-shadow p-8 bg-card text-center text-muted-foreground animate-pulse text-[13px]">
   357→              Loading manuscripts...
   358→            </div>
   359→          ) : manuscripts.length === 0 ? (
   360→            <div className="rounded-2xl macos-shadow p-5 bg-card text-center py-16">
   361→              <FileText className="size-12 text-muted-foreground mx-auto mb-4" />
   362→              <h3 className="text-xl font-semibold tracking-tight mb-1">No manuscripts yet</h3>
   363→              <p className="text-[13px] text-muted-foreground">Submit your first manuscript to get started</p>
   364→            </div>
   365→          ) : (
   366→            <div className="rounded-2xl macos-shadow bg-card overflow-hidden">
   367→              <Table>
   368→                <TableHeader>
   369→                  <TableRow className="hover:bg-transparent">
   370→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Title</TableHead>
   371→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Author</TableHead>
   372→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Status</TableHead>
   373→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Submitted</TableHead>
   374→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Actions</TableHead>
   375→                  </TableRow>
   376→                </TableHeader>
   377→                <TableBody>
   378→                  {manuscripts.map((ms) => {
   379→                    const nextStatuses = getNextStatus(ms.status)
   380→                    return (
   381→                      <TableRow key={ms.id} className="hover:bg-secondary/50">
   382→                        <TableCell className="text-[13px] font-medium max-w-[200px] truncate">{ms.title}</TableCell>
   383→                        <TableCell className="text-[13px] text-muted-foreground">{ms.author || '—'}</TableCell>
   384→                        <TableCell>
   385→                          <Badge className={`rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 ${manuscriptStatusColors[ms.status] || ''}`}>
   386→                            {ms.status.charAt(0).toUpperCase() + ms.status.slice(1)}
   387→                          </Badge>
   388→                        </TableCell>
   389→                        <TableCell className="text-[13px] text-muted-foreground">
   390→                          {new Date(ms.submittedAt).toLocaleDateString()}
   391→                        </TableCell>
   392→                        <TableCell>
   393→                          <div className="flex items-center gap-1">
   394→                            {nextStatuses.map((ns) => (
   395→                              <Button
   396→                                key={ns}
   397→                                size="sm"
   398→                                variant="outline"
   399→                                className="h-7 text-[11px] rounded-md font-medium px-2"
   400→                                onClick={() => handleManuscriptStatus(ms, ns)}
   401→                              >
   402→                                <ArrowRight className="size-3 mr-1" />
   403→                                {ns.charAt(0).toUpperCase() + ns.slice(1)}
   404→                              </Button>
   405→                            ))}
   406→                            {!rejectionStates.includes(ms.status) && ms.status !== 'published' && (
   407→                              <Button
   408→                                size="sm"
   409→                                variant="outline"
   410→                                className="h-7 text-[11px] rounded-md font-medium px-2 text-macos-red border-macos-red/20 hover:bg-macos-red/10"
   411→                                onClick={() => handleManuscriptStatus(ms, 'rejected')}
   412→                              >
   413→                                <XCircle className="size-3 mr-1" />
   414→                                Reject
   415→                              </Button>
   416→                            )}
   417→                            {ms.status === 'published' && (
   418→                              <CheckCircle2 className="size-4 text-macos-green" />
   419→                            )}
   420→                            {ms.status === 'rejected' && (
   421→                              <XCircle className="size-4 text-macos-red" />
   422→                            )}
   423→                          </div>
   424→                        </TableCell>
   425→                      </TableRow>
   426→                    )
   427→                  })}
   428→                </TableBody>
   429→              </Table>
   430→            </div>
   431→          )}
   432→        </div>
   433→      )}
   434→
   435→      {/* Print Orders Tab */}
   436→      {activeTab === 'print-orders' && (
   437→        <div>
   438→          {loading ? (
   439→            <div className="rounded-2xl macos-shadow p-8 bg-card text-center text-muted-foreground animate-pulse text-[13px]">
   440→              Loading print orders...
   441→            </div>
   442→          ) : printOrders.length === 0 ? (
   443→            <div className="rounded-2xl macos-shadow p-5 bg-card text-center py-16">
   444→              <Printer className="size-12 text-muted-foreground mx-auto mb-4" />
   445→              <h3 className="text-xl font-semibold tracking-tight mb-1">No print orders yet</h3>
   446→              <p className="text-[13px] text-muted-foreground">Create a print order for a published book</p>
   447→            </div>
   448→          ) : (
   449→            <div className="rounded-2xl macos-shadow bg-card overflow-hidden">
   450→              <Table>
   451→                <TableHeader>
   452→                  <TableRow className="hover:bg-transparent">
   453→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Book</TableHead>
   454→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Quantity</TableHead>
   455→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Cost/Unit</TableHead>
   456→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Total Cost</TableHead>
   457→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Status</TableHead>
   458→                    <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Actions</TableHead>
   459→                  </TableRow>
   460→                </TableHeader>
   461→                <TableBody>
   462→                  {printOrders.map((po) => (
   463→                    <TableRow key={po.id} className="hover:bg-secondary/50">
   464→                      <TableCell className="text-[13px] font-medium">
   465→                        {po.book?.title || 'Unknown Book'}
   466→                      </TableCell>
   467→                      <TableCell className="text-[13px]">{po.quantity.toLocaleString()}</TableCell>
   468→                      <TableCell className="text-[13px]">${po.costPerUnit.toFixed(2)}</TableCell>
   469→                      <TableCell className="text-[13px] font-medium">${po.totalCost.toLocaleString()}</TableCell>
   470→                      <TableCell>
   471→                        <Badge className={`rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 ${printStatusColors[po.status] || ''}`}>
   472→                          {po.status.charAt(0).toUpperCase() + po.status.slice(1)}
   473→                        </Badge>
   474→                      </TableCell>
   475→                      <TableCell>
   476→                        <div className="flex items-center gap-1">
   477→                          {po.status === 'ordered' && (
   478→                            <Button
   479→                              size="sm"
   480→                              variant="outline"
   481→                              className="h-7 text-[11px] rounded-md font-medium px-2"
   482→                              onClick={() => handlePrintOrderStatus(po, 'printing')}
   483→                            >
   484→                              <ArrowRight className="size-3 mr-1" />
   485→                              Start Printing
   486→                            </Button>
   487→                          )}
   488→                          {po.status === 'printing' && (
   489→                            <Button
   490→                              size="sm"
   491→                              variant="outline"
   492→                              className="h-7 text-[11px] rounded-md font-medium px-2 text-macos-green border-macos-green/20 hover:bg-macos-green/10"
   493→                              onClick={() => handlePrintOrderStatus(po, 'completed')}
   494→                            >
   495→                              <CheckCircle2 className="size-3 mr-1" />
   496→                              Complete
   497→                            </Button>
   498→                          )}
   499→                          {po.status === 'completed' && (
   500→                            <Badge className="rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 bg-macos-green/10 text-macos-green">
   501→                              <CheckCircle2 className="size-3 mr-1" />
   502→                              Done
   503→                            </Badge>
   504→                          )}
   505→                        </div>
   506→                      </TableCell>
   507→                    </TableRow>
   508→                  ))}
   509→                </TableBody>
   510→              </Table>
   511→            </div>
   512→          )}
   513→        </div>
