    51→// ─── Types ────────────────────────────────────────────────────────
    52→type ProductType = 'book' | 'stationery'
    53→
    54→interface POSProduct {
    55→  id: string
    56→  type: ProductType
    57→  name: string
    58→  sku: string | null
    59→  isbn: string | null
    60→  barcode: string | null
    61→  price: number
    62→  costPrice: number
    63→  discount: number
    64→  author?: string | null
    65→  brand?: string | null
    66→  category?: string | null
    67→}
    68→
    69→interface CartItem {
    70→  product: POSProduct
    71→  quantity: number
    72→  unitPrice: number
    73→  discount: number
    74→  totalPrice: number
    75→}
    76→
    77→interface Customer {
    78→  id: string
    79→  name: string
    80→  email?: string | null
    81→  phone?: string | null
    82→  loyaltyPoints: number
    83→}
    84→
    85→interface CompletedSale {
    86→  invoiceNumber: string
    87→  total: number
    88→  subtotal: number
    89→  discount: number
    90→  tax: number
    91→  paymentMethod: string
    92→  customerName: string | null
    93→  items: Array<{ name: string; quantity: number; unitPrice: number; totalPrice: number }>
    94→}

    96→// ─── Component ────────────────────────────────────────────────────
    97→export function RetailPOSModule() {
    98→  // Products
    99→  const [products, setProducts] = React.useState<POSProduct[]>([])
   100→  const [searchTerm, setSearchTerm] = React.useState('')
   101→  const [categoryFilter, setCategoryFilter] = React.useState<'all' | 'books' | 'stationery' | 'school'>('all')
   102→  const [searchLoading, setSearchLoading] = React.useState(false)
   103→
   104→  // Cart
   105→  const [cart, setCart] = React.useState<CartItem[]>([])
   106→  const [cartDiscount, setCartDiscount] = React.useState(0)
   107→  const [discountType, setDiscountType] = React.useState<'percentage' | 'fixed'>('percentage')
   108→  const [taxRate, setTaxRate] = React.useState(0)
   109→
   110→  // Customer
   111→  const [customerSearch, setCustomerSearch] = React.useState('')
   112→  const [customers, setCustomers] = React.useState<Customer[]>([])
   113→  const [selectedCustomer, setSelectedCustomer] = React.useState<Customer | null>(null)
   114→  const [customerLookupOpen, setCustomerLookupOpen] = React.useState(false)
   115→
   116→  // Payment
   117→  const [paymentMethod, setPaymentMethod] = React.useState('cash')
   118→
   119→  // Sale
   120→  const [completing, setCompleting] = React.useState(false)
   121→  const [completedSale, setCompletedSale] = React.useState<CompletedSale | null>(null)
   122→  const [error, setError] = React.useState<string | null>(null)
   123→
   124→  // Search products
   125→  React.useEffect(() => {
   126→    const searchProducts = async () => {
   127→      setSearchLoading(true)
   128→      try {
   129→        const params = new URLSearchParams()
   130→        params.set('limit', '30')
   131→        if (searchTerm) params.set('search', searchTerm)
   132→
   133→        const [booksRes, stationeryRes] = await Promise.all([
   134→          fetch(`/api/books?${params.toString()}`),
   135→          fetch(`/api/stationery?${params.toString()}`),
   136→        ])
   137→
   138→        const bookData = booksRes.ok ? await booksRes.json() : { data: [] }
   139→        const stationeryData = stationeryRes.ok ? await stationeryRes.json() : { data: [] }
   140→
   141→        const bookProducts: POSProduct[] = (bookData.data || []).map((b: Record<string, unknown>) => ({
   142→          id: b.id as string,
   143→          type: 'book' as ProductType,
   144→          name: b.title as string,
   145→          sku: null,
   146→          isbn: (b.isbn as string) || null,
   147→          barcode: (b.barcode as string) || null,
   148→          price: (b.price as number) || 0,
   149→          costPrice: (b.costPrice as number) || 0,
   150→          discount: (b.discount as number) || 0,
   151→          author: (b.author as { name: string } | null)?.name || null,
   152→          category: (b.category as { name: string } | null)?.name || null,
   153→        }))
   154→
   155→        const stationeryProducts: POSProduct[] = (stationeryData.data || []).map((s: Record<string, unknown>) => ({
   156→          id: s.id as string,
   157→          type: 'stationery' as ProductType,
   158→          name: s.name as string,
   159→          sku: (s.sku as string) || null,
   160→          isbn: null,
   161→          barcode: (s.barcode as string) || null,
   162→          price: (s.price as number) || 0,
   163→          costPrice: (s.costPrice as number) || 0,
   164→          discount: (s.discount as number) || 0,
   165→          brand: (s.brand as string) || null,
   166→          category: (s.category as { name: string } | null)?.name || null,
   167→        }))
   168→
   169→        let all = [...bookProducts, ...stationeryProducts]
   170→
   171→        // Apply category filter
   172→        if (categoryFilter === 'books') {
   173→          all = all.filter((p) => p.type === 'book')
   174→        } else if (categoryFilter === 'stationery') {
   175→          all = all.filter((p) => p.type === 'stationery')
   176→        } else if (categoryFilter === 'school') {
   177→          all = all.filter((p) => p.category?.toLowerCase().includes('school') || p.category?.toLowerCase().includes('educational'))
   178→        }
   179→
   180→        setProducts(all)
   181→      } catch {
   182→        // silent
   183→      } finally {
   184→        setSearchLoading(false)
   185→      }
   186→    }
   187→
   188→    const debounce = setTimeout(searchProducts, 300)
   189→    return () => clearTimeout(debounce)
   190→  }, [searchTerm, categoryFilter])
   191→
   192→  // Search customers
   193→  React.useEffect(() => {
   194→    if (!customerSearch) {
   195→      setCustomers([])
   196→      return
   197→    }
   198→    const searchCustomers = async () => {
   199→      try {
   200→        const res = await fetch(`/api/customers?search=${encodeURIComponent(customerSearch)}&type=retail&limit=10`)
   201→        if (res.ok) {
   202→          const json = await res.json()
   203→          setCustomers(json.data || [])
   204→        }
   205→      } catch {
   206→        // silent
   207→      }
   208→    }
   209→    const debounce = setTimeout(searchCustomers, 300)
   210→    return () => clearTimeout(debounce)
   211→  }, [customerSearch])
   212→
   213→  // Add to cart
   214→  const addToCart = (product: POSProduct) => {
   215→    setCart((prev) => {
   216→      const existing = prev.find((item) => item.product.id === product.id && item.product.type === product.type)
   217→      if (existing) {
   218→        return prev.map((item) =>
   219→          item.product.id === product.id && item.product.type === product.type
   220→            ? {
   221→                ...item,
   222→                quantity: item.quantity + 1,
   223→                totalPrice: (item.quantity + 1) * item.unitPrice * (1 - item.discount / 100),
   224→              }
   225→            : item
   226→        )
   227→      }
   228→      const unitPrice = product.price * (1 - product.discount / 100)
   229→      return [
   230→        ...prev,
   231→        {
   232→          product,
   233→          quantity: 1,
   234→          unitPrice: product.price,
   235→          discount: product.discount,
   236→          totalPrice: unitPrice,
   237→        },
   238→      ]
   239→    })
   240→  }
   241→
   242→  // Update cart quantity
   243→  const updateQuantity = (index: number, newQty: number) => {
   244→    if (newQty < 1) {
   245→      removeFromCart(index)
   246→      return
   247→    }
   248→    setCart((prev) =>
   249→      prev.map((item, i) =>
   250→        i === index
   251→          ? { ...item, quantity: newQty, totalPrice: newQty * item.unitPrice * (1 - item.discount / 100) }
   252→          : item
   253→      )
   254→    )
   255→  }
   256→
   257→  // Remove from cart
   258→  const removeFromCart = (index: number) => {
   259→    setCart((prev) => prev.filter((_, i) => i !== index))
   260→  }
   261→
   262→  // Clear cart
   263→  const clearCart = () => {
   264→    setCart([])
   265→    setCartDiscount(0)
   266→    setSelectedCustomer(null)
   267→    setCompletedSale(null)
   268→  }
   269→
   270→  // Compute totals
   271→  const subtotal = cart.reduce((s, item) => s + item.totalPrice, 0)
   272→  const discountAmount = discountType === 'percentage' ? subtotal * (cartDiscount / 100) : cartDiscount
   273→  const afterDiscount = Math.max(0, subtotal - discountAmount)
   274→  const taxAmount = afterDiscount * (taxRate / 100)
   275→  const total = afterDiscount + taxAmount
   276→
   277→  // Complete sale
   278→  const handleCompleteSale = async () => {
   279→    if (cart.length === 0) return
   280→    setCompleting(true)
   281→    setError(null)
   282→    try {
   283→      const saleItems = cart.map((item) => ({
   284→        bookId: item.product.type === 'book' ? item.product.id : undefined,
   285→        stationeryId: item.product.type === 'stationery' ? item.product.id : undefined,
   286→        quantity: item.quantity,
   287→        unitPrice: item.unitPrice,
   288→        discount: item.discount,
   289→      }))
   290→
   291→      const res = await fetch('/api/sales', {
   292→        method: 'POST',
   293→        headers: { 'Content-Type': 'application/json' },
   294→        body: JSON.stringify({
   295→          customerId: selectedCustomer?.id || null,
   296→          discount: discountAmount,
   297→          tax: taxRate,
   298→          paymentMethod,
   299→          paymentStatus: 'paid',
   300→          saleType: 'retail',
   301→          items: saleItems,
   302→        }),
   303→      })
   304→
   305→      if (!res.ok) {
   306→        const errData = await res.json()
   307→        throw new Error(errData.error || 'Failed to complete sale')
   308→      }
   309→
   310→      const json = await res.json()
   311→      const sale = json.data
   312→
   313→      setCompletedSale({
   314→        invoiceNumber: sale.invoiceNumber,
   315→        total: sale.total,
   316→        subtotal: sale.subtotal,
   317→        discount: sale.discount,
   318→        tax: sale.tax,
   319→        paymentMethod: sale.paymentMethod,
   320→        customerName: selectedCustomer?.name || null,
   321→        items: cart.map((item) => ({
   322→          name: item.product.name,
   323→          quantity: item.quantity,
   324→          unitPrice: item.unitPrice,
   325→          totalPrice: item.totalPrice,
   326→        })),
   327→      })
   328→
   329→      // Clear cart
   330→      setCart([])
   331→      setCartDiscount(0)
   332→      setSelectedCustomer(null)
   333→    } catch (err) {
   334→      setError(err instanceof Error ? err.message : 'Failed to complete sale')
   335→    } finally {
   336→      setCompleting(false)
   337→    }
   338→  }
   339→
   340→  // Payment method icons
   341→  const paymentMethods = [
   342→    { value: 'cash', label: 'Cash', icon: Banknote },
   343→    { value: 'card', label: 'Card', icon: CreditCard },
   344→    { value: 'bank', label: 'Bank Transfer', icon: Building },
   345→    { value: 'wallet', label: 'Digital Wallet', icon: Smartphone },
   346→  ]
   347→
   348→  // Category filters
   349→  const categoryFilters = [
   350→    { value: 'all' as const, label: 'All', icon: Package },
   351→    { value: 'books' as const, label: 'Books', icon: BookOpen },
   352→    { value: 'stationery' as const, label: 'Stationery', icon: PencilRuler },
   353→    { value: 'school' as const, label: 'School', icon: GraduationCap },
   354→  ]
   355→
   356→  return (
   357→    <div className="space-y-4">
   358→      {/* Header */}
   359→      <div>
   360→        <h2 className="text-2xl font-bold tracking-tight">Point of Sale</h2>
   361→        <p className="text-muted-foreground text-sm">Process retail sales quickly and efficiently</p>
   362→      </div>
   363→
   364→      {/* Error Alert */}
   365→      {error && (
   366→        <div className="flex items-center gap-2 rounded-lg border border-red-200 bg-red-50 dark:border-red-900 dark:bg-red-950/30 p-3 text-sm text-red-700 dark:text-red-400">
   367→          <AlertCircle className="size-4 shrink-0" />
   368→          {error}
   369→          <Button variant="ghost" size="sm" className="ml-auto" onClick={() => setError(null)}>
   370→            Dismiss
   371→          </Button>
   372→        </div>
   373→      )}
   374→
   375→      {/* Receipt Dialog */}
   376→      <Dialog open={!!completedSale} onOpenChange={(open) => !open && setCompletedSale(null)}>
   377→        <DialogContent className="sm:max-w-[450px]">
   378→          <DialogHeader>
   379→            <DialogTitle className="flex items-center gap-2">
   380→              <Receipt className="size-5 text-emerald-600" />
   381→              Sale Completed
   382→            </DialogTitle>
   383→            <DialogDescription>Your transaction has been processed successfully</DialogDescription>
   384→          </DialogHeader>
   385→          {completedSale && (
   386→            <div className="space-y-4 py-4">
   387→              <div className="text-center space-y-1">
   388→                <p className="text-lg font-bold">BookFlow OS</p>
   389→                <p className="text-xs text-muted-foreground">Invoice: {completedSale.invoiceNumber}</p>
   390→                <p className="text-xs text-muted-foreground">{new Date().toLocaleString()}</p>
   391→              </div>
   392→              <Separator />
   393→              <div className="space-y-2">
   394→                {completedSale.items.map((item, i) => (
   395→                  <div key={i} className="flex justify-between text-sm">
   396→                    <span className="flex-1 truncate mr-2">{item.name} x{item.quantity}</span>
   397→                    <span className="font-medium">${item.totalPrice.toFixed(2)}</span>
   398→                  </div>
   399→                ))}
   400→              </div>
   401→              <Separator />
   402→              <div className="space-y-1 text-sm">
   403→                <div className="flex justify-between">
   404→                  <span className="text-muted-foreground">Subtotal</span>
   405→                  <span>${completedSale.subtotal.toFixed(2)}</span>
   406→                </div>
   407→                {completedSale.discount > 0 && (
   408→                  <div className="flex justify-between text-emerald-600">
   409→                    <span>Discount</span>
   410→                    <span>-${completedSale.discount.toFixed(2)}</span>
   411→                  </div>
   412→                )}
   413→                {completedSale.tax > 0 && (
   414→                  <div className="flex justify-between">
   415→                    <span className="text-muted-foreground">Tax</span>
   416→                    <span>${completedSale.tax.toFixed(2)}</span>
   417→                  </div>
   418→                )}
   419→                <Separator />
   420→                <div className="flex justify-between font-bold text-lg">
   421→                  <span>Total</span>
   422→                  <span>${completedSale.total.toFixed(2)}</span>
   423→                </div>
   424→              </div>
   425→              <div className="flex justify-between text-sm text-muted-foreground">
   426→                <span>Payment: {completedSale.paymentMethod}</span>
   427→                {completedSale.customerName && <span>Customer: {completedSale.customerName}</span>}
   428→              </div>
   429→            </div>
   430→          )}
   431→          <DialogFooter>
   432→            <Button onClick={() => setCompletedSale(null)} className="bg-emerald-600 hover:bg-emerald-700">
   433→              <Check className="size-4 mr-1.5" />
   434→              Done
   435→            </Button>
   436→          </DialogFooter>
   437→        </DialogContent>
   438→      </Dialog>
   439→
   440→      {/* Main POS Layout */}
   441→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-4">
   442→        {/* Left: Products */}
   443→        <div className="lg:col-span-3 space-y-4">
   444→          {/* Search + Category Filter */}
   445→          <Card>
   446→            <CardContent className="p-4 space-y-3">
   447→              <div className="relative">
   448→                <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   449→                <Input
   450→                  placeholder="Search by name, ISBN, or barcode..."
   451→                  value={searchTerm}
   452→                  onChange={(e) => setSearchTerm(e.target.value)}
   453→                  className="pl-8"
   454→                />
   455→              </div>
   456→              <div className="flex gap-2 flex-wrap">
   457→                {categoryFilters.map((cat) => {
   458→                  const Icon = cat.icon
   459→                  const isActive = categoryFilter === cat.value
   460→                  return (
   461→                    <Button
   462→                      key={cat.value}
   463→                      variant={isActive ? 'default' : 'outline'}
   464→                      size="sm"
   465→                      className={isActive ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   466→                      onClick={() => setCategoryFilter(cat.value)}
   467→                    >
   468→                      <Icon className="size-3.5 mr-1.5" />
   469→                      {cat.label}
   470→                    </Button>
   471→                  )
   472→                })}
   473→              </div>
   474→            </CardContent>
   475→          </Card>
   476→
   477→          {/* Product Grid */}
   478→          {searchLoading ? (
   479→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   480→              {Array.from({ length: 6 }).map((_, i) => (
   481→                <Card key={i} className="animate-pulse">
   482→                  <CardContent className="p-3 space-y-2">
   483→                    <div className="h-4 bg-muted rounded w-3/4" />
   484→                    <div className="h-3 bg-muted rounded w-1/2" />
   485→                    <div className="h-5 bg-muted rounded w-1/3" />
   486→                  </CardContent>
   487→                </Card>
   488→              ))}
   489→            </div>
   490→          ) : products.length === 0 ? (
   491→            <Card>
   492→              <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   493→                <Package className="size-12 text-muted-foreground/40 mb-3" />
   494→                <p className="text-muted-foreground">No products found</p>
   495→                <p className="text-xs text-muted-foreground mt-1">Try a different search or category</p>
   496→              </CardContent>
   497→            </Card>
   498→          ) : (
   499→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   500→              {products.map((product) => {
   501→                const inCart = cart.find((c) => c.product.id === product.id && c.product.type === product.type)
   502→                return (
   503→                  <Card
   504→                    key={`${product.type}-${product.id}`}
   505→                    className={`cursor-pointer transition-all hover:shadow-md hover:border-emerald-300 dark:hover:border-emerald-700 ${inCart ? 'ring-2 ring-emerald-500/50 border-emerald-300 dark:border-emerald-700' : ''}`}
   506→                    onClick={() => addToCart(product)}
   507→                  >
   508→                    <CardContent className="p-3 space-y-1.5">
   509→                      <div className="flex items-start justify-between gap-1">
   510→                        <p className="font-medium text-sm leading-tight line-clamp-2">{product.name}</p>
   511→                        <Badge variant="outline" className="text-[10px] shrink-0">
   512→                          {product.type === 'book' ? 'Book' : 'Item'}
   513→                        </Badge>
   514→                      </div>
   515→                      {product.author && (
   516→                        <p className="text-xs text-muted-foreground truncate">by {product.author}</p>
   517→                      )}
   518→                      {product.brand && (
   519→                        <p className="text-xs text-muted-foreground truncate">{product.brand}</p>
   520→                      )}
   521→                      <div className="flex items-center justify-between">
   522→                        <div>
   523→                          {product.discount > 0 ? (
   524→                            <div className="flex items-center gap-1">
   525→                              <span className="text-xs text-muted-foreground line-through">${product.price.toFixed(2)}</span>
   526→                              <span className="text-sm font-bold text-emerald-600">${(product.price * (1 - product.discount / 100)).toFixed(2)}</span>
   527→                            </div>
   528→                          ) : (
   529→                            <span className="text-sm font-bold">${product.price.toFixed(2)}</span>
   530→                          )}
   531→                        </div>
   532→                        {inCart ? (
   533→                          <Badge className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800 text-xs">
   534→                            x{inCart.quantity}
   535→                          </Badge>
   536→                        ) : (
   537→                          <Plus className="size-4 text-emerald-600" />
   538→                        )}
   539→                      </div>
   540→                    </CardContent>
   541→                  </Card>
   542→                )
   543→              })}
   544→            </div>
   545→          )}
   546→        </div>
   547→
   548→        {/* Right: Cart */}
   549→        <div className="lg:col-span-2">
   550→          <Card className="sticky top-4">
   551→            <CardHeader className="pb-3">
   552→              <div className="flex items-center justify-between">
   553→                <CardTitle className="text-base flex items-center gap-2">
   554→                  <ShoppingCart className="size-4" />
   555→                  Cart ({cart.length})
   556→                </CardTitle>
   557→                {cart.length > 0 && (
   558→                  <Button variant="ghost" size="sm" className="text-red-600 hover:text-red-700" onClick={clearCart}>
   559→                    <Trash2 className="size-3.5 mr-1" />
   560→                    Clear
   561→                  </Button>
   562→                )}
   563→              </div>
   564→            </CardHeader>
   565→            <CardContent className="space-y-4">
   566→              {/* Customer Selection */}
   567→              <div className="space-y-2">
   568→                <div className="flex items-center justify-between">
   569→                  <Label className="text-xs text-muted-foreground">Customer</Label>
   570→                  <Button variant="ghost" size="sm" className="h-6 text-xs" onClick={() => setCustomerLookupOpen(true)}>
   571→                    <UserCircle className="size-3 mr-1" />
   572→                    {selectedCustomer ? 'Change' : 'Select'}
   573→                  </Button>
   574→                </div>
   575→                {selectedCustomer ? (
   576→                  <div className="flex items-center justify-between rounded-lg border p-2">
   577→                    <div>
   578→                      <p className="text-sm font-medium">{selectedCustomer.name}</p>
   579→                      <p className="text-xs text-muted-foreground">{selectedCustomer.loyaltyPoints} pts</p>
   580→                    </div>
   581→                    <Button variant="ghost" size="icon" className="size-6" onClick={() => setSelectedCustomer(null)}>
   582→                      <X className="size-3" />
   583→                    </Button>
   584→                  </div>
   585→                ) : (
   586→                  <p className="text-xs text-muted-foreground">Walk-in customer</p>
   587→                )}
   588→              </div>
   589→
   590→              <Separator />
   591→
   592→              {/* Cart Items */}
   593→              <ScrollArea className="max-h-64">
   594→                {cart.length === 0 ? (
   595→                  <div className="flex flex-col items-center justify-center py-8 text-center">
   596→                    <ShoppingCart className="size-8 text-muted-foreground/40 mb-2" />
   597→                    <p className="text-sm text-muted-foreground">Cart is empty</p>
   598→                  </div>
   599→                ) : (
   600→                  <div className="space-y-2 pr-1">
   601→                    {cart.map((item, index) => (
   602→                      <div key={`${item.product.type}-${item.product.id}`} className="flex items-start gap-2 rounded-lg border p-2">
   603→                        <div className="flex-1 min-w-0">
   604→                          <p className="text-sm font-medium truncate">{item.product.name}</p>
   605→                          <div className="flex items-center gap-2 mt-1">
   606→                            <span className="text-xs text-muted-foreground">${item.unitPrice.toFixed(2)}</span>
   607→                            {item.discount > 0 && (
   608→                              <Badge className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 text-[10px] px-1 py-0">
   609→                                -{item.discount}%
   610→                              </Badge>
   611→                            )}
   612→                          </div>
   613→                        </div>
   614→                        <div className="flex items-center gap-1">
   615→                          <Button
   616→                            variant="outline"
   617→                            size="icon"
   618→                            className="size-6"
   619→                            onClick={() => updateQuantity(index, item.quantity - 1)}
   620→                          >
   621→                            <Minus className="size-3" />
   622→                          </Button>
   623→                          <span className="w-8 text-center text-sm font-medium">{item.quantity}</span>
   624→                          <Button
   625→                            variant="outline"
   626→                            size="icon"
   627→                            className="size-6"
   628→                            onClick={() => updateQuantity(index, item.quantity + 1)}
   629→                          >
   630→                            <Plus className="size-3" />
   631→                          </Button>
   632→                        </div>
   633→                        <div className="text-right min-w-[60px]">
   634→                          <p className="text-sm font-medium">${item.totalPrice.toFixed(2)}</p>
   635→                          <Button
   636→                            variant="ghost"
   637→                            size="icon"
   638→                            className="size-5 text-red-500 hover:text-red-600 mt-0.5"
   639→                            onClick={() => removeFromCart(index)}
   640→                          >
   641→                            <X className="size-3" />
   642→                          </Button>
   643→                        </div>
   644→                      </div>
   645→                    ))}
   646→                  </div>
   647→                )}
   648→              </ScrollArea>
   649→
   650→              <Separator />
   651→
   652→              {/* Discount */}
   653→              <div className="space-y-2">
   654→                <Label className="text-xs text-muted-foreground flex items-center gap-1">
   655→                  <Tag className="size-3" />
   656→                  Cart Discount
   657→                </Label>
   658→                <div className="flex gap-2">
   659→                  <Select value={discountType} onValueChange={(v: 'percentage' | 'fixed') => setDiscountType(v)}>
   660→                    <SelectTrigger className="w-[110px]">
   661→                      <SelectValue />
   662→                    </SelectTrigger>
   663→                    <SelectContent>
   664→                      <SelectItem value="percentage">Percent %</SelectItem>
   665→                      <SelectItem value="fixed">Fixed $</SelectItem>
   666→                    </SelectContent>
   667→                  </Select>
   668→                  <Input
   669→                    type="number"
   670→                    min="0"
   671→                    step={discountType === 'percentage' ? '1' : '0.01'}
   672→                    value={cartDiscount || ''}
   673→                    onChange={(e) => setCartDiscount(parseFloat(e.target.value) || 0)}
   674→                    placeholder="0"
   675→                    className="w-24"
   676→                  />
   677→                </div>
   678→              </div>
   679→
   680→              {/* Tax */}
   681→              <div className="flex items-center justify-between">
   682→                <Label className="text-xs text-muted-foreground">Tax Rate (%)</Label>
   683→                <Input
   684→                  type="number"
   685→                  min="0"
   686→                  step="0.5"
   687→                  value={taxRate || ''}
   688→                  onChange={(e) => setTaxRate(parseFloat(e.target.value) || 0)}
   689→                  placeholder="0"
   690→                  className="w-20 text-right"
   691→                />
   692→              </div>
   693→
   694→              <Separator />
   695→
   696→              {/* Totals */}
   697→              <div className="space-y-1.5 text-sm">
   698→                <div className="flex justify-between">
   699→                  <span className="text-muted-foreground">Subtotal</span>
   700→                  <span>${subtotal.toFixed(2)}</span>
   701→                </div>
   702→                {discountAmount > 0 && (
   703→                  <div className="flex justify-between text-emerald-600">
   704→                    <span>Discount</span>
   705→                    <span>-${discountAmount.toFixed(2)}</span>
   706→                  </div>
   707→                )}
   708→                {taxAmount > 0 && (
   709→                  <div className="flex justify-between">
   710→                    <span className="text-muted-foreground">Tax ({taxRate}%)</span>
   711→                    <span>${taxAmount.toFixed(2)}</span>
   712→                  </div>
   713→                )}
   714→                <Separator />
   715→                <div className="flex justify-between font-bold text-lg">
   716→                  <span>Total</span>
   717→                  <span>${total.toFixed(2)}</span>
   718→                </div>
   719→              </div>
   720→
   721→              {/* Payment Method */}
   722→              <div className="space-y-2">
   723→                <Label className="text-xs text-muted-foreground">Payment Method</Label>
   724→                <RadioGroup value={paymentMethod} onValueChange={setPaymentMethod} className="grid grid-cols-2 gap-2">
   725→                  {paymentMethods.map((pm) => {
   726→                    const Icon = pm.icon
   727→                    return (
   728→                      <Label
   729→                        key={pm.value}
   730→                        className={`flex items-center gap-2 rounded-lg border p-2.5 cursor-pointer transition-colors ${paymentMethod === pm.value ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30' : 'hover:bg-muted/50'}`}
   731→                      >
   732→                        <RadioGroupItem value={pm.value} className="sr-only" />
   733→                        <Icon className="size-4" />
   734→                        <span className="text-sm">{pm.label}</span>
   735→                      </Label>
   736→                    )
   737→                  })}
   738→                </RadioGroup>
   739→              </div>
   740→
   741→              {/* Complete Sale */}
   742→              <Button
   743→                className="w-full bg-emerald-600 hover:bg-emerald-700 h-12 text-base font-semibold"
   744→                onClick={handleCompleteSale}
   745→                disabled={cart.length === 0 || completing}
   746→              >
   747→                {completing ? (
   748→                  <>
   749→                    <Loader2 className="size-5 mr-2 animate-spin" />
   750→                    Processing...
   751→                  </>
   752→                ) : (
   753→                  <>
   754→                    <Check className="size-5 mr-2" />
   755→                    Complete Sale — ${total.toFixed(2)}
   756→                  </>
   757→                )}
   758→              </Button>
   759→            </CardContent>
   760→          </Card>
   761→        </div>
   762→      </div>
   763→
   764→      {/* Customer Lookup Dialog */}
   765→      <Dialog open={customerLookupOpen} onOpenChange={setCustomerLookupOpen}>
   766→        <DialogContent className="sm:max-w-[425px]">
   767→          <DialogHeader>
   768→            <DialogTitle>Select Customer</DialogTitle>
   769→            <DialogDescription>Search for an existing customer</DialogDescription>
   770→          </DialogHeader>
   771→          <div className="space-y-4 py-4">
   772→            <div className="relative">
   773→              <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   774→              <Input
   775→                placeholder="Search by name, email, or phone..."
   776→                value={customerSearch}
   777→                onChange={(e) => setCustomerSearch(e.target.value)}
   778→                className="pl-8"
   779→              />
   780→            </div>
   781→            <div className="max-h-64 overflow-y-auto space-y-1">
   782→              {customers.length === 0 && customerSearch ? (
   783→                <p className="text-sm text-muted-foreground text-center py-4">No customers found</p>
   784→              ) : (
   785→                customers.map((c) => (
   786→                  <button
   787→                    key={c.id}
   788→                    className="w-full flex items-center justify-between rounded-lg border p-3 hover:bg-muted/50 transition-colors text-left"
   789→                    onClick={() => {
   790→                      setSelectedCustomer(c)
   791→                      setCustomerLookupOpen(false)
   792→                      setCustomerSearch('')
   793→                    }}
   794→                  >
   795→                    <div>
   796→                      <p className="text-sm font-medium">{c.name}</p>
   797→                      <p className="text-xs text-muted-foreground">{c.email || c.phone || 'No contact info'}</p>
   798→                    </div>
   799→                    <Badge variant="outline" className="text-xs">{c.loyaltyPoints} pts</Badge>
   800→                  </button>
   801→                ))
   802→              )}
   803→            </div>
   804→          </div>
   805→          <DialogFooter>
   806→            <Button
   807→              variant="outline"
   808→              onClick={() => {
   809→                setCustomerLookupOpen(false)
   810→                setCustomerSearch('')
   811→                setSelectedCustomer(null)
   812→              }}
   813→            >
   814→              Walk-in Customer
   815→            </Button>
   816→          </DialogFooter>
   817→        </DialogContent>
   818→      </Dialog>
   819→    </div>
   820→  )
