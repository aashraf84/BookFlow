    80→export function EmployeesModule() {
    81→  const [employees, setEmployees] = React.useState<Employee[]>([])
    82→  const [branches, setBranches] = React.useState<Branch[]>([])
    83→  const [loading, setLoading] = React.useState(true)
    84→  const [search, setSearch] = React.useState('')
    85→  const [departmentFilter, setDepartmentFilter] = React.useState('all')
    86→  const [showAddDialog, setShowAddDialog] = React.useState(false)
    87→  const [editEmployee, setEditEmployee] = React.useState<Employee | null>(null)
    88→  const [selectedEmployee, setSelectedEmployee] = React.useState<Employee | null>(null)
    89→
    90→  const fetchEmployees = React.useCallback(async () => {

    92→    try {

    94→      if (search) params.set('search', search)
    95→      if (departmentFilter && departmentFilter !== 'all') params.set('department', departmentFilter)
    96→      params.set('limit', '50')
    97→      const res = await fetch(`/api/employees?${params}`)
    98→      const json = await res.json()
    99→      if (json.data) setEmployees(json.data)
   100→    } catch (err) {
   101→      console.error('Failed to fetch employees:', err)
   102→    } finally {
   103→      setLoading(false)
   104→    }
   105→  }, [search, departmentFilter])
   106→
   107→  const fetchBranches = React.useCallback(async () => {
   108→    try {
   109→      const res = await fetch('/api/branches')
   110→      const json = await res.json()
   111→      if (json.data) setBranches(json.data)
   112→    } catch (err) {
   113→      console.error('Failed to fetch branches:', err)
   114→    }
   115→  }, [])
   116→
   117→  React.useEffect(() => {
   118→    fetchEmployees()
   119→  }, [fetchEmployees])
   120→
   121→  React.useEffect(() => {
   122→    fetchBranches()
   123→  }, [fetchBranches])
   124→
   125→  const handleDelete = async (id: string) => {
   126→    if (!confirm('Are you sure you want to deactivate this employee?')) return
   127→    try {
   128→      await fetch(`/api/employees?id=${id}`, { method: 'DELETE' })
   129→      fetchEmployees()
   130→    } catch (err) {
   131→      console.error('Failed to deactivate employee:', err)
   132→    }
   133→  }
   134→
   135→  // Department stats
   136→  const departments = employees.reduce(
   137→    (acc, emp) => {
   138→      const dept = emp.department || 'Unassigned'
   139→      if (!acc[dept]) acc[dept] = { total: 0, active: 0, inactive: 0 }
   140→      acc[dept].total++
   141→      if (emp.isActive) acc[dept].active++
   142→      else acc[dept].inactive++
   143→      return acc
   144→    },
   145→    {} as Record<string, { total: number; active: number; inactive: number }>
   146→  )
   147→
   148→  const totalEmployees = employees.length
   149→  const activeEmployees = employees.filter((e) => e.isActive).length
   150→  const inactiveEmployees = employees.filter((e) => !e.isActive).length
   151→  const totalPayroll = employees.filter((e) => e.isActive).reduce((sum, e) => sum + e.salary, 0)
   152→
   153→  const departmentList = [...new Set(employees.map((e) => e.department).filter(Boolean))] as string[]
   154→
   155→  return (
   156→    <div className="space-y-6">
   157→      <div className="flex items-center gap-3">
   158→        <div className="rounded-lg bg-emerald-100 p-2 dark:bg-emerald-950/40">
   159→          <UserCog className="size-5 text-emerald-600" />
   160→        </div>
   161→        <div>
   162→          <h2 className="text-xl font-bold tracking-tight">Employees</h2>
   163→          <p className="text-sm text-muted-foreground">
   164→            Manage your team, departments, and payroll
   165→          </p>
   166→        </div>
   167→      </div>
   168→
   169→      {/* Department Stats */}
   170→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   171→        <Card>
   172→          <CardContent className="p-3">
   173→            <div className="flex items-center gap-2">
   174→              <div className="rounded p-1.5 bg-emerald-100 dark:bg-emerald-950/40">
   175→                <Users className="size-4 text-emerald-600" />
   176→              </div>
   177→              <div>
   178→                <p className="text-xs text-muted-foreground">Total</p>
   179→                <p className="text-lg font-bold">{totalEmployees}</p>
   180→              </div>
   181→            </div>
   182→          </CardContent>
   183→        </Card>
   184→        <Card>
   185→          <CardContent className="p-3">
   186→            <div className="flex items-center gap-2">
   187→              <div className="rounded p-1.5 bg-teal-100 dark:bg-teal-950/40">
   188→                <UserCheck className="size-4 text-teal-600" />
   189→              </div>
   190→              <div>
   191→                <p className="text-xs text-muted-foreground">Active</p>
   192→                <p className="text-lg font-bold text-teal-600">{activeEmployees}</p>
   193→              </div>
   194→            </div>
   195→          </CardContent>
   196→        </Card>
   197→        <Card>
   198→          <CardContent className="p-3">
   199→            <div className="flex items-center gap-2">
   200→              <div className="rounded p-1.5 bg-amber-100 dark:bg-amber-950/40">
   201→                <UserX className="size-4 text-amber-600" />
   202→              </div>
   203→              <div>
   204→                <p className="text-xs text-muted-foreground">Inactive</p>
   205→                <p className="text-lg font-bold text-amber-600">{inactiveEmployees}</p>
   206→              </div>
   207→            </div>
   208→          </CardContent>
   209→        </Card>
   210→        <Card>
   211→          <CardContent className="p-3">
   212→            <div className="flex items-center gap-2">
   213→              <div className="rounded p-1.5 bg-emerald-100 dark:bg-emerald-950/40">
   214→                <DollarSign className="size-4 text-emerald-600" />
   215→              </div>
   216→              <div>
   217→                <p className="text-xs text-muted-foreground">Monthly Payroll</p>
   218→                <p className="text-lg font-bold text-emerald-600">${totalPayroll.toLocaleString()}</p>
   219→              </div>
   220→            </div>
   221→          </CardContent>
   222→        </Card>
   223→      </div>
   224→
   225→      {/* Department breakdown */}
   226→      {Object.keys(departments).length > 0 && (
   227→        <Card>
   228→          <CardHeader className="pb-3">
   229→            <CardTitle className="text-sm">Department Breakdown</CardTitle>
   230→          </CardHeader>
   231→          <CardContent>
   232→            <div className="flex flex-wrap gap-2">
   233→              {Object.entries(departments).map(([dept, stats]) => (
   234→                <div
   235→                  key={dept}
   236→                  className="flex items-center gap-2 rounded-lg border px-3 py-2 text-sm cursor-pointer hover:bg-muted/50 transition-colors"
   237→                  onClick={() => setDepartmentFilter(dept === 'Unassigned' ? 'all' : dept)}
   238→                >
   239→                  <span className="font-medium">{dept}</span>
   240→                  <Badge variant="secondary" className="text-[10px]">
   241→                    {stats.active} active
   242→                  </Badge>
   243→                </div>
   244→              ))}
   245→            </div>
   246→          </CardContent>
   247→        </Card>
   248→      )}
   249→
   250→      {/* Search & Filter */}
   251→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   252→        <div className="flex gap-2 w-full sm:w-auto">
   253→          <div className="relative flex-1 sm:flex-initial sm:w-64">
   254→            <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   255→            <Input
   256→              placeholder="Search employees..."
   257→              value={search}
   258→              onChange={(e) => setSearch(e.target.value)}
   259→              className="pl-8"
   260→            />
   261→          </div>
   262→          <Select value={departmentFilter} onValueChange={setDepartmentFilter}>
   263→            <SelectTrigger className="w-36">
   264→              <SelectValue />
   265→            </SelectTrigger>
   266→            <SelectContent>
   267→              <SelectItem value="all">All Departments</SelectItem>
   268→              {departmentList.map((dept) => (
   269→                <SelectItem key={dept} value={dept!}>
   270→                  {dept}
   271→                </SelectItem>
   272→              ))}
   273→            </SelectContent>
   274→          </Select>
   275→        </div>
   276→        <Button onClick={() => setShowAddDialog(true)} className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   277→          <Plus className="size-4" /> Add Employee
   278→        </Button>
   279→      </div>
   280→
   281→      {/* Employee Table */}
   282→      {loading ? (
   283→        <Card>
   284→          <CardContent className="p-6">
   285→            <div className="space-y-3">
   286→              {[1, 2, 3, 4, 5].map((i) => (
   287→                <div key={i} className="h-14 bg-muted rounded animate-pulse" />
   288→              ))}
   289→            </div>
   290→          </CardContent>
   291→        </Card>
   292→      ) : employees.length === 0 ? (
   293→        <Card>
   294→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   295→            <UserCog className="size-10 text-muted-foreground mb-3" />
   296→            <p className="text-muted-foreground">No employees found</p>
   297→            <p className="text-sm text-muted-foreground">Add your first team member</p>
   298→          </CardContent>
   299→        </Card>
   300→      ) : (
   301→        <Card>
   302→          <CardContent className="p-0">
   303→            <ScrollArea className="max-h-[500px]">
   304→              <Table>
   305→                <TableHeader>
   306→                  <TableRow>
   307→                    <TableHead>Employee</TableHead>
   308→                    <TableHead>Department</TableHead>
   309→                    <TableHead>Position</TableHead>
   310→                    <TableHead>Branch</TableHead>
   311→                    <TableHead className="text-right">Salary</TableHead>
   312→                    <TableHead>Status</TableHead>
   313→                    <TableHead className="w-20">Actions</TableHead>
   314→                  </TableRow>
   315→                </TableHeader>
   316→                <TableBody>
   317→                  {employees.map((employee) => (
   318→                    <TableRow
   319→                      key={employee.id}
   320→                      className="cursor-pointer"
   321→                      onClick={() => setSelectedEmployee(employee)}
   322→                    >
   323→                      <TableCell>
   324→                        <div className="flex items-center gap-3">
   325→                          <Avatar className="size-8">
   326→                            <AvatarFallback className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 text-xs font-semibold">
   327→                              {employee.name.split(' ').map((n) => n[0]).join('').slice(0, 2).toUpperCase()}
   328→                            </AvatarFallback>
   329→                          </Avatar>
   330→                          <div>
   331→                            <p className="font-medium text-sm">{employee.name}</p>
   332→                            <p className="text-xs text-muted-foreground">{employee.email}</p>
   333→                          </div>
   334→                        </div>
   335→                      </TableCell>
   336→                      <TableCell>
   337→                        <Badge variant="secondary" className="text-[10px]">
   338→                          {employee.department || 'Unassigned'}
   339→                        </Badge>
   340→                      </TableCell>
   341→                      <TableCell className="text-sm">{employee.position || '—'}</TableCell>
   342→                      <TableCell className="text-sm">
   343→                        {employee.branch?.name || '—'}
   344→                      </TableCell>
   345→                      <TableCell className="text-right font-medium text-sm">
   346→                        ${employee.salary.toLocaleString()}
   347→                      </TableCell>
   348→                      <TableCell>
   349→                        <Badge
   350→                          className={
   351→                            employee.isActive
   352→                              ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 text-[10px]'
   353→                              : 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400 text-[10px]'
   354→                          }
   355→                        >
   356→                          {employee.isActive ? 'Active' : 'Inactive'}
   357→                        </Badge>
   358→                      </TableCell>
   359→                      <TableCell>
   360→                        <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
   361→                          <Button
   362→                            variant="ghost"
   363→                            size="sm"
   364→                            className="h-7 w-7 p-0 text-teal-600 hover:text-teal-700"
   365→                            onClick={() => setEditEmployee(employee)}
   366→                          >
   367→                            <Pencil className="size-3.5" />
   368→                          </Button>
   369→                          <Button
   370→                            variant="ghost"
   371→                            size="sm"
   372→                            className="h-7 w-7 p-0 text-red-500 hover:text-red-700"
   373→                            onClick={() => handleDelete(employee.id)}
   374→                          >
   375→                            <Trash2 className="size-3.5" />
   376→                          </Button>
   377→                        </div>
   378→                      </TableCell>
   379→                    </TableRow>
   380→                  ))}
   381→                </TableBody>
   382→              </Table>
   383→            </ScrollArea>
   384→          </CardContent>
   385→        </Card>
   386→      )}
   387→
   388→      <AddEmployeeDialog
   389→        open={showAddDialog}
   390→        onOpenChange={setShowAddDialog}
   391→        branches={branches}
   392→        onSuccess={fetchEmployees}
   393→      />
   394→      <EditEmployeeDialog
   395→        employee={editEmployee}
   396→        onOpenChange={(open) => !open && setEditEmployee(null)}
   397→        branches={branches}
   398→        onSuccess={fetchEmployees}
   399→      />
   400→      <EmployeeDetailDialog
   401→        employee={selectedEmployee}
   402→        onOpenChange={(open) => !open && setSelectedEmployee(null)}
   403→      />
   404→    </div>
   405→  )
   406→}

   408→// ==================== ADD EMPLOYEE DIALOG ====================
   409→
   410→function AddEmployeeDialog({
   411→  open,
   412→  onOpenChange,
   413→  branches,
   414→  onSuccess,
   415→}: {
   416→  open: boolean
   417→  onOpenChange: (open: boolean) => void
   418→  branches: Branch[]
   419→  onSuccess: () => void
   420→}) {
   421→  const [saving, setSaving] = React.useState(false)
   422→  const [form, setForm] = React.useState({
   423→    name: '',
   424→    email: '',
   425→    phone: '',
   426→    department: '',
   427→    position: '',
   428→    salary: '',
   429→    hireDate: '',
   430→    branchId: '',
   431→  })
   432→
   433→  const handleSubmit = async (e: React.FormEvent) => {
   434→    e.preventDefault()
   435→    if (!form.name.trim() || !form.email.trim()) return
   436→    setSaving(true)
   437→    try {
   438→      const res = await fetch('/api/employees', {
   439→        method: 'POST',
   440→        headers: { 'Content-Type': 'application/json' },
   441→        body: JSON.stringify({
   442→          ...form,
   443→          salary: parseFloat(form.salary) || 0,
   444→          hireDate: form.hireDate || null,
   445→          branchId: form.branchId || null,
   446→        }),
   447→      })
   448→      if (res.ok) {
   449→        setForm({ name: '', email: '', phone: '', department: '', position: '', salary: '', hireDate: '', branchId: '' })
   450→        onOpenChange(false)
   451→        onSuccess()
   452→      } else {
   453→        const json = await res.json()
   454→        alert(json.error || 'Failed to create employee')
   455→      }
   456→    } catch (err) {
   457→      console.error('Failed to create employee:', err)
   458→      alert('Failed to create employee')
   459→    } finally {
   460→      setSaving(false)
   461→    }
   462→  }
   463→
   464→  return (
   465→    <Dialog open={open} onOpenChange={onOpenChange}>
   466→      <DialogContent className="sm:max-w-md">
   467→        <DialogHeader>
   468→          <DialogTitle>Add Employee</DialogTitle>
   469→          <DialogDescription>Register a new team member</DialogDescription>
   470→        </DialogHeader>
   471→        <form onSubmit={handleSubmit} className="space-y-4">
   472→          <div className="space-y-2">
   473→            <Label htmlFor="emp-name">Full Name *</Label>
   474→            <Input
   475→              id="emp-name"
   476→              value={form.name}
   477→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   478→              placeholder="John Smith"
   479→              required
   480→            />
   481→          </div>
   482→          <div className="grid grid-cols-2 gap-3">
   483→            <div className="space-y-2">
   484→              <Label htmlFor="emp-email">Email *</Label>
   485→              <Input
   486→                id="emp-email"
   487→                type="email"
   488→                value={form.email}
   489→                onChange={(e) => setForm({ ...form, email: e.target.value })}
   490→                placeholder="john@bookflow.com"
   491→                required
   492→              />
   493→            </div>
   494→            <div className="space-y-2">
   495→              <Label htmlFor="emp-phone">Phone</Label>
   496→              <Input
   497→                id="emp-phone"
   498→                value={form.phone}
   499→                onChange={(e) => setForm({ ...form, phone: e.target.value })}
   500→                placeholder="+1-555-0100"
   501→              />
   502→            </div>
   503→          </div>
   504→          <div className="grid grid-cols-2 gap-3">
   505→            <div className="space-y-2">
   506→              <Label>Department</Label>
   507→              <Select value={form.department} onValueChange={(v) => setForm({ ...form, department: v })}>
   508→                <SelectTrigger>
   509→                  <SelectValue placeholder="Select..." />
   510→                </SelectTrigger>
   511→                <SelectContent>
   512→                  <SelectItem value="Sales">Sales</SelectItem>
   513→                  <SelectItem value="Management">Management</SelectItem>
   514→                  <SelectItem value="Finance">Finance</SelectItem>
   515→                  <SelectItem value="Warehouse">Warehouse</SelectItem>
   516→                  <SelectItem value="IT">IT</SelectItem>
   517→                  <SelectItem value="Marketing">Marketing</SelectItem>
   518→                  <SelectItem value="HR">HR</SelectItem>
   519→                  <SelectItem value="Operations">Operations</SelectItem>
   520→                </SelectContent>
   521→              </Select>
   522→            </div>
   523→            <div className="space-y-2">
   524→              <Label htmlFor="emp-position">Position</Label>
   525→              <Input
   526→                id="emp-position"
   527→                value={form.position}
   528→                onChange={(e) => setForm({ ...form, position: e.target.value })}
   529→                placeholder="Sales Manager"
   530→              />
   531→            </div>
   532→          </div>
   533→          <div className="grid grid-cols-2 gap-3">
   534→            <div className="space-y-2">
   535→              <Label htmlFor="emp-salary">Monthly Salary ($)</Label>
   536→              <Input
   537→                id="emp-salary"
   538→                type="number"
   539→                step="0.01"
   540→                min="0"
   541→                value={form.salary}
   542→                onChange={(e) => setForm({ ...form, salary: e.target.value })}
   543→                placeholder="5000"
   544→              />
   545→            </div>
   546→            <div className="space-y-2">
   547→              <Label htmlFor="emp-hire">Hire Date</Label>
   548→              <Input
   549→                id="emp-hire"
   550→                type="date"
   551→                value={form.hireDate}
   552→                onChange={(e) => setForm({ ...form, hireDate: e.target.value })}
   553→              />
   554→            </div>
   555→          </div>
   556→          <div className="space-y-2">
   557→            <Label>Branch</Label>
   558→            <Select value={form.branchId} onValueChange={(v) => setForm({ ...form, branchId: v })}>
   559→              <SelectTrigger>
   560→                <SelectValue placeholder="Select branch..." />
   561→              </SelectTrigger>
   562→              <SelectContent>
   563→                {branches.map((b) => (
   564→                  <SelectItem key={b.id} value={b.id}>
   565→                    {b.name} ({b.code})
   566→                  </SelectItem>
   567→                ))}
   568→              </SelectContent>
   569→            </Select>
   570→          </div>
   571→          <DialogFooter>
   572→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   573→              Cancel
   574→            </Button>
   575→            <Button
   576→              type="submit"
   577→              disabled={saving || !form.name.trim() || !form.email.trim()}
   578→              className="bg-emerald-600 hover:bg-emerald-700"
   579→            >
   580→              {saving ? 'Saving...' : 'Add Employee'}
   581→            </Button>
   582→          </DialogFooter>
   583→        </form>
   584→      </DialogContent>
   585→    </Dialog>
   586→  )
   587→}

   591→function EditEmployeeDialog({
   592→  employee,
   593→  onOpenChange,
   594→  branches,
   595→  onSuccess,
   596→}: {
   597→  employee: Employee | null
   598→  onOpenChange: (open: boolean) => void
   599→  branches: Branch[]
   600→  onSuccess: () => void
   601→}) {
   602→  const [saving, setSaving] = React.useState(false)
   603→  const [form, setForm] = React.useState({
   604→    name: '',
   605→    email: '',
   606→    phone: '',
   607→    department: '',
   608→    position: '',
   609→    salary: '',
   610→    hireDate: '',
   611→    branchId: '',
   612→    isActive: true,
   613→  })
   614→
   615→  React.useEffect(() => {
   616→    if (employee) {
   617→      setForm({
   618→        name: employee.name || '',
   619→        email: employee.email || '',
   620→        phone: employee.phone || '',
   621→        department: employee.department || '',
   622→        position: employee.position || '',
   623→        salary: employee.salary?.toString() || '',
   624→        hireDate: employee.hireDate ? new Date(employee.hireDate).toISOString().split('T')[0] : '',
   625→        branchId: employee.branchId || '',
   626→        isActive: employee.isActive,
   627→      })
   628→    }
   629→  }, [employee])
   630→
   631→  const handleSubmit = async (e: React.FormEvent) => {
   632→    e.preventDefault()
   633→    if (!employee) return
   634→    setSaving(true)
   635→    try {
   636→      const res = await fetch('/api/employees', {
   637→        method: 'PUT',
   638→        headers: { 'Content-Type': 'application/json' },
   639→        body: JSON.stringify({
   640→          id: employee.id,
   641→          name: form.name,
   642→          email: form.email,
   643→          phone: form.phone,
   644→          department: form.department,
   645→          position: form.position,
   646→          salary: parseFloat(form.salary) || 0,
   647→          hireDate: form.hireDate || null,
   648→          branchId: form.branchId || null,
   649→          isActive: form.isActive,
   650→        }),
   651→      })
   652→      if (res.ok) {
   653→        onOpenChange(false)
   654→        onSuccess()
   655→      } else {
   656→        const json = await res.json()
   657→        alert(json.error || 'Failed to update employee')
   658→      }
   659→    } catch (err) {
   660→      console.error('Failed to update employee:', err)
   661→      alert('Failed to update employee')
   662→    } finally {
   663→      setSaving(false)
   664→    }
   665→  }
   666→
   667→  if (!employee) return null
   668→
   669→  return (
   670→    <Dialog open={!!employee} onOpenChange={onOpenChange}>
   671→      <DialogContent className="sm:max-w-md">
   672→        <DialogHeader>
   673→          <DialogTitle>Edit Employee</DialogTitle>
   674→          <DialogDescription>Update employee information</DialogDescription>
   675→        </DialogHeader>
   676→        <form onSubmit={handleSubmit} className="space-y-4">
   677→          <div className="space-y-2">
   678→            <Label>Full Name *</Label>
   679→            <Input
   680→              value={form.name}
   681→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   682→              required
   683→            />
   684→          </div>
   685→          <div className="grid grid-cols-2 gap-3">
   686→            <div className="space-y-2">
   687→              <Label>Email *</Label>
   688→              <Input
   689→                type="email"
   690→                value={form.email}
   691→                onChange={(e) => setForm({ ...form, email: e.target.value })}
   692→                required
   693→              />
   694→            </div>
   695→            <div className="space-y-2">
   696→              <Label>Phone</Label>
   697→              <Input
   698→                value={form.phone}
   699→                onChange={(e) => setForm({ ...form, phone: e.target.value })}
   700→              />
   701→            </div>
   702→          </div>
   703→          <div className="grid grid-cols-2 gap-3">
   704→            <div className="space-y-2">
   705→              <Label>Department</Label>
   706→              <Select value={form.department} onValueChange={(v) => setForm({ ...form, department: v })}>
   707→                <SelectTrigger>
   708→                  <SelectValue placeholder="Select..." />
   709→                </SelectTrigger>
   710→                <SelectContent>
   711→                  <SelectItem value="Sales">Sales</SelectItem>
   712→                  <SelectItem value="Management">Management</SelectItem>
   713→                  <SelectItem value="Finance">Finance</SelectItem>
   714→                  <SelectItem value="Warehouse">Warehouse</SelectItem>
   715→                  <SelectItem value="IT">IT</SelectItem>
   716→                  <SelectItem value="Marketing">Marketing</SelectItem>
   717→                  <SelectItem value="HR">HR</SelectItem>
   718→                  <SelectItem value="Operations">Operations</SelectItem>
   719→                </SelectContent>
   720→              </Select>
   721→            </div>
   722→            <div className="space-y-2">
   723→              <Label>Position</Label>
   724→              <Input
   725→                value={form.position}
   726→                onChange={(e) => setForm({ ...form, position: e.target.value })}
   727→              />
   728→            </div>
   729→          </div>
   730→          <div className="grid grid-cols-2 gap-3">
   731→            <div className="space-y-2">
   732→              <Label>Monthly Salary ($)</Label>
   733→              <Input
   734→                type="number"
   735→                step="0.01"
   736→                min="0"
   737→                value={form.salary}
   738→                onChange={(e) => setForm({ ...form, salary: e.target.value })}
   739→              />
   740→            </div>
   741→            <div className="space-y-2">
   742→              <Label>Hire Date</Label>
   743→              <Input
   744→                type="date"
   745→                value={form.hireDate}
   746→                onChange={(e) => setForm({ ...form, hireDate: e.target.value })}
   747→              />
   748→            </div>
   749→          </div>
   750→          <div className="grid grid-cols-2 gap-3">
   751→            <div className="space-y-2">
   752→              <Label>Branch</Label>
   753→              <Select value={form.branchId} onValueChange={(v) => setForm({ ...form, branchId: v })}>
   754→                <SelectTrigger>
   755→                  <SelectValue placeholder="Select branch..." />
   756→                </SelectTrigger>
   757→                <SelectContent>
   758→                  {branches.map((b) => (
   759→                    <SelectItem key={b.id} value={b.id}>
   760→                      {b.name} ({b.code})
   761→                    </SelectItem>
   762→                  ))}
   763→                </SelectContent>
   764→              </Select>
   765→            </div>
   766→            <div className="space-y-2">
   767→              <Label>Status</Label>
   768→              <Select value={form.isActive ? 'active' : 'inactive'} onValueChange={(v) => setForm({ ...form, isActive: v === 'active' })}>
   769→                <SelectTrigger>
   770→                  <SelectValue />
   771→                </SelectTrigger>
   772→                <SelectContent>
   773→                  <SelectItem value="active">Active</SelectItem>
   774→                  <SelectItem value="inactive">Inactive</SelectItem>
   775→                </SelectContent>
   776→              </Select>
   777→            </div>
   778→          </div>
   779→          <DialogFooter>
   780→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   781→              Cancel
   782→            </Button>
   783→            <Button type="submit" disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   784→              {saving ? 'Saving...' : 'Update Employee'}
   785→            </Button>
   786→          </DialogFooter>
   787→        </form>
   788→      </DialogContent>
   789→    </Dialog>
   790→  )
   791→}
