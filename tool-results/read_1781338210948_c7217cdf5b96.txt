   151→function CustomersTab() {
   152→  const [customers, setCustomers] = React.useState<Customer[]>([])
   153→  const [loading, setLoading] = React.useState(true)
   154→  const [search, setSearch] = React.useState('')
   155→  const [typeFilter, setTypeFilter] = React.useState('all')
   156→  const [showAddDialog, setShowAddDialog] = React.useState(false)
   157→  const [selectedCustomer, setSelectedCustomer] = React.useState<Customer | null>(null)
   158→
   159→  const fetchCustomers = React.useCallback(async () => {
   160→    setLoading(true)
   161→    try {
   162→      const params = new URLSearchParams()
   163→      if (search) params.set('search', search)
   164→      if (typeFilter && typeFilter !== 'all') params.set('type', typeFilter)
   165→      params.set('limit', '50')
   166→      const res = await fetch(`/api/customers?${params}`)
   167→      const json = await res.json()
   168→      if (json.data) setCustomers(json.data)
   169→    } catch (err) {
   170→      console.error('Failed to fetch customers:', err)
   171→    } finally {
   172→      setLoading(false)
   173→    }
   174→  }, [search, typeFilter])
   175→
   176→  React.useEffect(() => {
   177→    fetchCustomers()
   178→  }, [fetchCustomers])
   179→
   180→  const handleDelete = async (id: string) => {
   181→    if (!confirm('Are you sure you want to deactivate this customer?')) return
   182→    try {
   183→      await fetch(`/api/customers`, {
   184→        method: 'PUT',
   185→        headers: { 'Content-Type': 'application/json' },
   186→        body: JSON.stringify({ id, isActive: false }),
   187→      })
   188→      fetchCustomers()
   189→    } catch (err) {
   190→      console.error('Failed to deactivate customer:', err)
   191→    }
   192→  }
   193→
   194→  const getTypeBadge = (type: string) => {
   195→    const styles: Record<string, string> = {
   196→      retail: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400',
   197→      wholesale: 'bg-teal-100 text-teal-700 dark:bg-teal-950/40 dark:text-teal-400',
   198→      school: 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400',
   199→      dealer: 'bg-purple-100 text-purple-700 dark:bg-purple-950/40 dark:text-purple-400',
   200→      distributor: 'bg-rose-100 text-rose-700 dark:bg-rose-950/40 dark:text-rose-400',
   201→    }
   202→    return styles[type] || 'bg-gray-100 text-gray-700 dark:bg-gray-950/40 dark:text-gray-400'
   203→  }
   204→
   205→  const totalCustomers = customers.length
   206→  const totalRevenue = customers.reduce((sum, c) => sum + c.totalSpent, 0)
   207→  const totalOutstanding = customers.reduce((sum, c) => sum + c.outstanding, 0)
   208→  const totalPoints = customers.reduce((sum, c) => sum + c.loyaltyPoints, 0)
   209→
   210→  return (
   211→    <div className="space-y-4">
   212→      {/* Summary Cards */}
   213→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   214→        <Card>
   215→          <CardContent className="p-3">
   216→            <p className="text-xs text-muted-foreground">Total Customers</p>
   217→            <p className="text-lg font-bold">{totalCustomers}</p>
   218→          </CardContent>
   219→        </Card>
   220→        <Card>
   221→          <CardContent className="p-3">
   222→            <p className="text-xs text-muted-foreground">Total Revenue</p>
   223→            <p className="text-lg font-bold text-emerald-600">${totalRevenue.toLocaleString()}</p>
   224→          </CardContent>
   225→        </Card>
   226→        <Card>
   227→          <CardContent className="p-3">
   228→            <p className="text-xs text-muted-foreground">Outstanding</p>
   229→            <p className="text-lg font-bold text-amber-600">${totalOutstanding.toLocaleString()}</p>
   230→          </CardContent>
   231→        </Card>
   232→        <Card>
   233→          <CardContent className="p-3">
   234→            <p className="text-xs text-muted-foreground">Loyalty Points</p>
   235→            <p className="text-lg font-bold text-teal-600">{totalPoints.toLocaleString()}</p>
   236→          </CardContent>
   237→        </Card>
   238→      </div>
   239→
   240→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   241→        <div className="flex gap-2 w-full sm:w-auto">
   242→          <div className="relative flex-1 sm:flex-initial sm:w-64">
   243→            <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   244→            <Input
   245→              placeholder="Search customers..."
   246→              value={search}
   247→              onChange={(e) => setSearch(e.target.value)}
   248→              className="pl-8"
   249→            />
   250→          </div>
   251→          <Select value={typeFilter} onValueChange={setTypeFilter}>
   252→            <SelectTrigger className="w-32">
   253→              <SelectValue />
   254→            </SelectTrigger>
   255→            <SelectContent>
   256→              <SelectItem value="all">All Types</SelectItem>
   257→              <SelectItem value="retail">Retail</SelectItem>
   258→              <SelectItem value="wholesale">Wholesale</SelectItem>
   259→              <SelectItem value="school">School</SelectItem>
   260→              <SelectItem value="dealer">Dealer</SelectItem>
   261→              <SelectItem value="distributor">Distributor</SelectItem>
   262→            </SelectContent>
   263→          </Select>
   264→        </div>
   265→        <Button onClick={() => setShowAddDialog(true)} className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   266→          <Plus className="size-4" /> Add Customer
   267→        </Button>
   268→      </div>
   269→
   270→      {loading ? (
   271→        <Card>
   272→          <CardContent className="p-6">
   273→            <div className="space-y-3">
   274→              {[1, 2, 3, 4, 5].map((i) => (
   275→                <div key={i} className="h-12 bg-muted rounded animate-pulse" />
   276→              ))}
   277→            </div>
   278→          </CardContent>
   279→        </Card>
   280→      ) : customers.length === 0 ? (
   281→        <Card>
   282→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   283→            <Users className="size-10 text-muted-foreground mb-3" />
   284→            <p className="text-muted-foreground">No customers found</p>
   285→            <p className="text-sm text-muted-foreground">Add your first customer to get started</p>
   286→          </CardContent>
   287→        </Card>
   288→      ) : (
   289→        <Card>
   290→          <CardContent className="p-0">
   291→            <ScrollArea className="max-h-[500px]">
   292→              <Table>
   293→                <TableHeader>
   294→                  <TableRow>
   295→                    <TableHead>Name</TableHead>
   296→                    <TableHead>Contact</TableHead>
   297→                    <TableHead>Type</TableHead>
   298→                    <TableHead className="text-right">Points</TableHead>
   299→                    <TableHead className="text-right">Total Spent</TableHead>
   300→                    <TableHead className="text-right">Outstanding</TableHead>
   301→                    <TableHead className="w-10"></TableHead>
   302→                  </TableRow>
   303→                </TableHeader>
   304→                <TableBody>
   305→                  {customers.map((customer) => (
   306→                    <TableRow
   307→                      key={customer.id}
   308→                      className="cursor-pointer"
   309→                      onClick={() => setSelectedCustomer(customer)}
   310→                    >
   311→                      <TableCell className="font-medium">{customer.name}</TableCell>
   312→                      <TableCell>
   313→                        <div className="text-xs space-y-0.5">
   314→                          {customer.email && (
   315→                            <div className="flex items-center gap-1">
   316→                              <Mail className="size-3" /> {customer.email}
   317→                            </div>
   318→                          )}
   319→                          {customer.phone && (
   320→                            <div className="flex items-center gap-1">
   321→                              <Phone className="size-3" /> {customer.phone}
   322→                            </div>
   323→                          )}
   324→                        </div>
   325→                      </TableCell>
   326→                      <TableCell>
   327→                        <Badge className={`text-[10px] ${getTypeBadge(customer.type)}`}>
   328→                          {customer.type}
   329→                        </Badge>
   330→                      </TableCell>
   331→                      <TableCell className="text-right">
   332→                        <span className="text-teal-600 font-medium">{customer.loyaltyPoints}</span>
   333→                      </TableCell>
   334→                      <TableCell className="text-right">
   335→                        ${customer.totalSpent.toLocaleString()}
   336→                      </TableCell>
   337→                      <TableCell className="text-right">
   338→                        <span className={customer.outstanding > 0 ? 'text-amber-600 font-medium' : ''}>
   339→                          ${customer.outstanding.toLocaleString()}
   340→                        </span>
   341→                      </TableCell>
   342→                      <TableCell>
   343→                        <Button
   344→                          variant="ghost"
   345→                          size="sm"
   346→                          className="h-6 w-6 p-0 text-red-500 hover:text-red-700"
   347→                          onClick={(e) => {
   348→                            e.stopPropagation()
   349→                            handleDelete(customer.id)
   350→                          }}
   351→                        >
   352→                          <Trash2 className="size-3" />
   353→                        </Button>
   354→                      </TableCell>
   355→                    </TableRow>
   356→                  ))}
   357→                </TableBody>
   358→              </Table>
   359→            </ScrollArea>
   360→          </CardContent>
   361→        </Card>
   362→      )}
   363→
   364→      <AddCustomerDialog open={showAddDialog} onOpenChange={setShowAddDialog} onSuccess={fetchCustomers} />
   365→      <CustomerDetailDialog
   366→        customer={selectedCustomer}
   367→        onOpenChange={(open) => !open && setSelectedCustomer(null)}
   368→      />
   369→    </div>
   370→  )
   371→}

   526→function CampaignsTab() {
   527→  const [campaigns, setCampaigns] = React.useState<Campaign[]>([])
   528→  const [loading, setLoading] = React.useState(true)
   529→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   530→
   531→  const fetchCampaigns = React.useCallback(async () => {
   532→    setLoading(true)
   533→    try {
   534→      const res = await fetch('/api/crm/campaigns')
   535→      if (res.ok) {
   536→        const json = await res.json()
   537→        if (json.data) setCampaigns(json.data)
   538→      }
   539→    } catch (err) {
   540→      console.error('Failed to fetch campaigns:', err)
   541→    } finally {
   542→      setLoading(false)
   543→    }
   544→  }, [])
   545→
   546→  React.useEffect(() => {
   547→    fetchCampaigns()
   548→  }, [fetchCampaigns])
   549→
   550→  const getStatusColor = (status: string) => {
   551→    switch (status) {
   552→      case 'draft':
   553→        return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   554→      case 'scheduled':
   555→        return 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400'
   556→      case 'sent':
   557→        return 'bg-teal-100 text-teal-700 dark:bg-teal-950/40 dark:text-teal-400'
   558→      case 'completed':
   559→        return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400'
   560→      default:
   561→        return ''
   562→    }
   563→  }
