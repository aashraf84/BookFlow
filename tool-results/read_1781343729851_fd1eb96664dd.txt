   111→// ─── Component ────────────────────────────────────────────────────
   112→export function WholesaleModule() {
   113→  const [activeTab, setActiveTab] = React.useState('orders')
   114→  const [orders, setOrders] = React.useState<WholesaleOrder[]>([])
   115→  const [dealers, setDealers] = React.useState<Dealer[]>([])
   116→  const [loading, setLoading] = React.useState(true)
   117→  const [error, setError] = React.useState<string | null>(null)
   118→
   119→  // Status filter
   120→  const [statusFilter, setStatusFilter] = React.useState('all')
   121→
   122→  // New order dialog
   123→  const [orderDialogOpen, setOrderDialogOpen] = React.useState(false)
   124→  const [saving, setSaving] = React.useState(false)
   125→  const [selectedDealerId, setSelectedDealerId] = React.useState('')
   126→  const [dealerName, setDealerName] = React.useState('')
   127→  const [orderDiscount, setOrderDiscount] = React.useState('')
   128→  const [orderTax, setOrderTax] = React.useState('')
   129→  const [deliveryAddress, setDeliveryAddress] = React.useState('')
   130→  const [orderNotes, setOrderNotes] = React.useState('')
   131→  const [orderItems, setOrderItems] = React.useState<OrderFormItem[]>([])
   132→
   133→  // Product search for adding items
   134→  const [productSearch, setProductSearch] = React.useState('')
   135→  const [searchResults, setSearchResults] = React.useState<Array<{ id: string; type: 'book' | 'stationery'; name: string; price: number }>>([])
   136→  const [searchLoading, setSearchLoading] = React.useState(false)
   137→
   138→  // Order detail dialog
   139→  const [detailOrder, setDetailOrder] = React.useState<WholesaleOrder | null>(null)
   140→  const [detailDialogOpen, setDetailDialogOpen] = React.useState(false)
   141→
   142→  // Updating status
   143→  const [updatingStatus, setUpdatingStatus] = React.useState<string | null>(null)
   144→
   145→  // Fetch orders
   146→  const fetchOrders = React.useCallback(async () => {
   147→    setLoading(true)
   148→    setError(null)
   149→    try {
   150→      const params = new URLSearchParams()
   151→      params.set('limit', '50')
   152→      if (statusFilter && statusFilter !== 'all') params.set('status', statusFilter)
   153→      const res = await fetch(`/api/wholesale?${params.toString()}`)
   154→      if (!res.ok) throw new Error('Failed to fetch orders')
   155→      const json = await res.json()
   156→      setOrders(json.data || [])
   157→    } catch (err) {
   158→      setError(err instanceof Error ? err.message : 'Unknown error')
   159→    } finally {
   160→      setLoading(false)
   161→    }
   162→  }, [statusFilter])
   163→
   164→  // Fetch dealers (customers of type wholesale/dealer/distributor)
   165→  const fetchDealers = React.useCallback(async () => {
   166→    try {
   167→      const res = await fetch('/api/customers?type=wholesale&limit=100')
   168→      const res2 = await fetch('/api/customers?type=dealer&limit=100')
   169→      const json1 = res.ok ? await res.json() : { data: [] }
   170→      const json2 = res2.ok ? await res2.json() : { data: [] }
   171→      // Also get distributors
   172→      const res3 = await fetch('/api/customers?type=distributor&limit=100')
   173→      const json3 = res3.ok ? await res3.json() : { data: [] }
   174→      setDealers([...(json1.data || []), ...(json2.data || []), ...(json3.data || [])])
   175→    } catch {
   176→      // silent
   177→    }
   178→  }, [])
   179→
   180→  React.useEffect(() => {
   181→    fetchOrders()
   182→  }, [fetchOrders])
   183→
   184→  React.useEffect(() => {
   185→    fetchDealers()
   186→  }, [fetchDealers])
   187→
   188→  // Product search for order items
   189→  React.useEffect(() => {
   190→    if (!productSearch) {
   191→      setSearchResults([])
   192→      return
   193→    }
   194→    const searchProducts = async () => {
   195→      setSearchLoading(true)
   196→      try {
   197→        const params = new URLSearchParams()
   198→        params.set('search', productSearch)
   199→        params.set('limit', '10')
   200→
   201→        const [booksRes, stationeryRes] = await Promise.all([
   202→          fetch(`/api/books?${params.toString()}`),
   203→          fetch(`/api/stationery?${params.toString()}`),
   204→        ])
   205→
   206→        const bookData = booksRes.ok ? await booksRes.json() : { data: [] }
   207→        const stationeryData = stationeryRes.ok ? await stationeryRes.json() : { data: [] }
   208→
   209→        const results: Array<{ id: string; type: 'book' | 'stationery'; name: string; price: number }> = [
   210→          ...(bookData.data || []).map((b: Record<string, unknown>) => ({
   211→            id: b.id as string,
   212→            type: 'book' as const,
   213→            name: b.title as string,
   214→            price: (b.price as number) || 0,
   215→          })),
   216→          ...(stationeryData.data || []).map((s: Record<string, unknown>) => ({
   217→            id: s.id as string,
   218→            type: 'stationery' as const,
   219→            name: s.name as string,
   220→            price: (s.price as number) || 0,
   221→          })),
   222→        ]
   223→
   224→        setSearchResults(results)
   225→      } catch {
   226→        // silent
   227→      } finally {
   228→        setSearchLoading(false)
   229→      }
   230→    }
   231→    const debounce = setTimeout(searchProducts, 300)
   232→    return () => clearTimeout(debounce)
   233→  }, [productSearch])
   234→
   235→  // Add product to order items
   236→  const addItemToOrder = (product: { id: string; type: 'book' | 'stationery'; name: string; price: number }) => {
   237→    const existing = orderItems.find((i) => i.productId === product.id && i.productType === product.type)
   238→    if (existing) {
   239→      setOrderItems((prev) =>
   240→        prev.map((i) =>
   241→          i.productId === product.id && i.productType === product.type
   242→            ? { ...i, quantity: i.quantity + 1 }
   243→            : i
   244→        )
   245→      )
   246→    } else {
   247→      setOrderItems((prev) => [
   248→        ...prev,
   249→        {
   250→          productType: product.type,
   251→          productId: product.id,
   252→          productName: product.name,
   253→          quantity: 1,
   254→          unitPrice: product.price,
   255→          discount: 0,
   256→        },
   257→      ])
   258→    }
   259→    setProductSearch('')
   260→    setSearchResults([])
   261→  }
   262→
   263→  // Remove order item
   264→  const removeOrderItem = (index: number) => {
   265→    setOrderItems((prev) => prev.filter((_, i) => i !== index))
   266→  }
   267→
   268→  // Update order item
   269→  const updateOrderItem = (index: number, field: keyof OrderFormItem, value: number | string) => {
   270→    setOrderItems((prev) =>
   271→      prev.map((item, i) => (i === index ? { ...item, [field]: value } : item))
   272→    )
   273→  }
   274→
   275→  // Status badges
   276→  const getStatusBadge = (status: string) => {
   277→    const map: Record<string, { className: string; label: string }> = {
   278→      pending: { className: 'bg-macos-orange/10 text-macos-orange', label: 'Pending' },
   279→      confirmed: { className: 'bg-macos-blue/10 text-macos-blue', label: 'Confirmed' },
   280→      shipped: { className: 'bg-macos-teal/10 text-macos-teal', label: 'Shipped' },
   281→      delivered: { className: 'bg-macos-green/10 text-macos-green', label: 'Delivered' },
   282→      cancelled: { className: 'bg-macos-red/10 text-macos-red', label: 'Cancelled' },
   283→    }
   284→    const s = map[status] || { className: 'bg-secondary text-secondary-foreground', label: status }
   285→    return <span className={`rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 ${s.className}`}>{s.label}</span>
   286→  }
   287→
   288→  const getPaymentBadge = (status: string) => {
   289→    const map: Record<string, { className: string; label: string }> = {
   290→      unpaid: { className: 'bg-macos-red/10 text-macos-red', label: 'Unpaid' },
   291→      partial: { className: 'bg-macos-orange/10 text-macos-orange', label: 'Partial' },
   292→      paid: { className: 'bg-macos-green/10 text-macos-green', label: 'Paid' },
   293→    }
   294→    const s = map[status] || { className: 'bg-secondary text-secondary-foreground', label: status }
   295→    return <span className={`rounded-md text-[10px] font-semibold tracking-wide px-2 py-0.5 ${s.className}`}>{s.label}</span>
   296→  }
   297→
   298→  // Compute order total
   299→  const orderSubtotal = orderItems.reduce((s, i) => s + i.quantity * i.unitPrice * (1 - i.discount / 100), 0)
   300→  const orderDiscountAmt = parseFloat(orderDiscount) || 0
   301→  const orderTaxAmt = (orderSubtotal - orderDiscountAmt) * ((parseFloat(orderTax) || 0) / 100)
   302→  const orderTotal = orderSubtotal - orderDiscountAmt + orderTaxAmt
   303→
   304→  // Create wholesale order
   305→  const handleCreateOrder = async () => {
   306→    if (orderItems.length === 0) return
   307→    setSaving(true)
   308→    setError(null)
   309→    try {
   310→      const items = orderItems.map((item) => ({
   311→        bookId: item.productType === 'book' ? item.productId : undefined,
   312→        stationeryId: item.productType === 'stationery' ? item.productId : undefined,
   313→        quantity: item.quantity,
   314→        unitPrice: item.unitPrice,
   315→        discount: item.discount,
   316→      }))
   317→
   318→      const res = await fetch('/api/wholesale', {
   319→        method: 'POST',
   320→        headers: { 'Content-Type': 'application/json' },
   321→        body: JSON.stringify({
   322→          customerId: selectedDealerId || undefined,
   323→          dealerName: dealerName || undefined,
   324→          discount: orderDiscountAmt,
   325→          tax: parseFloat(orderTax) || 0,
   326→          paymentStatus: 'unpaid',
   327→          deliveryAddress: deliveryAddress || undefined,
   328→          notes: orderNotes || undefined,
   329→          items,
   330→        }),
   331→      })
   332→
   333→      if (!res.ok) {
   334→        const errData = await res.json()
   335→        throw new Error(errData.error || 'Failed to create order')
   336→      }
   337→
   338→      // Reset form
   339→      setOrderDialogOpen(false)
   340→      setSelectedDealerId('')
   341→      setDealerName('')
   342→      setOrderDiscount('')
   343→      setOrderTax('')
   344→      setDeliveryAddress('')
   345→      setOrderNotes('')
   346→      setOrderItems([])
   347→      fetchOrders()
   348→    } catch (err) {
   349→      setError(err instanceof Error ? err.message : 'Failed to create order')
   350→    } finally {
   351→      setSaving(false)
   352→    }
   353→  }
   354→
   355→  // Update order status
   356→  const handleStatusUpdate = async (orderId: string, newStatus: string) => {
   357→    setUpdatingStatus(orderId)
   358→    try {
   359→      const res = await fetch('/api/wholesale', {
   360→        method: 'PUT',
   361→        headers: { 'Content-Type': 'application/json' },
   362→        body: JSON.stringify({ id: orderId, status: newStatus }),
   363→      })
   364→      if (!res.ok) throw new Error('Failed to update status')
   365→      fetchOrders()
   366→      if (detailOrder?.id === orderId) {
   367→        setDetailOrder((prev) => prev ? { ...prev, status: newStatus } : null)
   368→      }
   369→    } catch (err) {
   370→      setError(err instanceof Error ? err.message : 'Failed to update status')
   371→    } finally {
   372→      setUpdatingStatus(null)
   373→    }
   374→  }
   375→
   376→  // View order detail
   377→  const handleViewOrder = (order: WholesaleOrder) => {
   378→    setDetailOrder(order)
   379→    setDetailDialogOpen(true)
   380→  }
   381→
   382→  // Analytics
   383→  const totalRevenue = orders.filter((o) => o.status === 'delivered').reduce((s, o) => s + o.total, 0)
   384→  const pendingAmount = orders.filter((o) => o.paymentStatus === 'unpaid' || o.paymentStatus === 'partial').reduce((s, o) => s + o.total, 0)
   385→  const topDealers = dealers.sort((a, b) => b.totalSpent - a.totalSpent).slice(0, 5)
   386→  const outstandingReceivables = dealers.reduce((s, d) => s + d.outstanding, 0)
   387→
   388→  // Get item name
   389→  const getItemName = (item: OrderItem): string => {
   390→    if (item.book) return item.book.title
   391→    if (item.stationery) return item.stationery.name
   392→    return 'Unknown Item'
   393→  }
   394→
   395→  // Stats
   396→  const pendingOrders = orders.filter((o) => o.status === 'pending').length
   397→  const shippedOrders = orders.filter((o) => o.status === 'shipped').length
   398→
   399→  // Segmented tabs
   400→  const tabs = [
   401→    { id: 'orders', label: 'Orders' },
   402→    { id: 'dealers', label: 'Dealers' },
   403→    { id: 'analytics', label: 'Analytics' },
   404→  ]
   405→
   406→  // Status filter tabs
   407→  const statusTabs = [
   408→    { id: 'all', label: 'All' },
   409→    { id: 'pending', label: 'Pending' },
   410→    { id: 'confirmed', label: 'Confirmed' },
   411→    { id: 'shipped', label: 'Shipped' },
   412→    { id: 'delivered', label: 'Delivered' },
   413→    { id: 'cancelled', label: 'Cancelled' },
   414→  ]
   415→
   416→  return (
   417→    <div className="space-y-4">
   418→      {/* Header */}
   419→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   420→        <div>
   421→          <h2 className="text-[13px] font-semibold tracking-tight text-foreground">Wholesale Management</h2>
   422→          <p className="text-[11px] text-muted-foreground tracking-tight">Manage bulk orders, dealers, and distributor relationships</p>
   423→        </div>
   424→        <Button onClick={() => setOrderDialogOpen(true)} className="bg-primary text-primary-foreground rounded-lg font-medium text-[13px] h-8 shrink-0">
   425→          <Plus className="size-3.5 mr-1.5" />
   426→          New Order
   427→        </Button>
   428→      </div>
   429→
   430→      {/* Error Alert */}
   431→      {error && (
   432→        <div className="flex items-center gap-2 rounded-xl bg-macos-red/10 p-3 text-[13px] text-macos-red macos-shadow">
   433→          <AlertCircle className="size-3.5 shrink-0" />
   434→          {error}
   435→          <Button variant="ghost" size="sm" className="ml-auto h-6 text-[11px] text-macos-red hover:text-macos-red" onClick={() => setError(null)}>
   436→            Dismiss
   437→          </Button>
   438→        </div>
   439→      )}
   440→
   441→      {/* Stats Row */}
   442→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   443→        <div className="rounded-2xl macos-shadow p-4 bg-background">
   444→          <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Total Orders</p>
   445→          <p className="text-2xl font-bold tracking-tight">{orders.length}</p>
   446→        </div>
   447→        <div className="rounded-2xl macos-shadow p-4 bg-background">
   448→          <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Pending</p>
   449→          <p className="text-2xl font-bold tracking-tight text-macos-orange">{pendingOrders}</p>
   450→        </div>
   451→        <div className="rounded-2xl macos-shadow p-4 bg-background">
   452→          <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Shipped</p>
   453→          <p className="text-2xl font-bold tracking-tight text-macos-blue">{shippedOrders}</p>
   454→        </div>
   455→        <div className="rounded-2xl macos-shadow p-4 bg-background">
   456→          <p className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Outstanding</p>
   457→          <p className="text-2xl font-bold tracking-tight text-macos-red">${outstandingReceivables.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   458→        </div>
   459→      </div>
   460→
   461→      {/* Segmented Tabs */}
   462→      <div className="macos-segmented inline-flex p-0.5">
   463→        {tabs.map((tab) => (
   464→          <button
   465→            key={tab.id}
   466→            className={`px-3 py-1 text-[12px] font-medium rounded-md transition-all ${activeTab === tab.id ? 'macos-segmented-active' : 'text-muted-foreground hover:text-foreground'}`}
   467→            onClick={() => setActiveTab(tab.id)}
   468→          >
   469→            {tab.label}
   470→          </button>
   471→        ))}
   472→      </div>
   473→
   474→      {/* Orders Tab */}
   475→      {activeTab === 'orders' && (
   476→        <div className="space-y-4">
   477→          {/* Status Filter */}
   478→          <div className="macos-segmented inline-flex p-0.5">
   479→            {statusTabs.map((tab) => (
   480→              <button
   481→                key={tab.id}
   482→                className={`px-3 py-1 text-[12px] font-medium rounded-md transition-all ${statusFilter === tab.id ? 'macos-segmented-active' : 'text-muted-foreground hover:text-foreground'}`}
   483→                onClick={() => setStatusFilter(tab.id)}
   484→              >
   485→                {tab.label}
   486→              </button>
   487→            ))}
   488→          </div>
   489→
   490→          {/* Orders Table */}
   491→          <div className="rounded-2xl macos-shadow bg-background overflow-hidden">
   492→            {loading ? (
   493→              <div className="p-4 space-y-3">
   494→                {Array.from({ length: 5 }).map((_, i) => (
   495→                  <Skeleton key={i} className="h-10 w-full" />
   496→                ))}
   497→              </div>
   498→            ) : orders.length === 0 ? (
   499→              <div className="flex flex-col items-center justify-center py-16 text-center">
   500→                <Package className="size-12 text-muted-foreground/30 mb-3" />
   501→                <p className="text-[13px] text-muted-foreground tracking-tight">No wholesale orders found</p>
   502→                <p className="text-[11px] text-muted-foreground mt-1 tracking-tight">Create your first wholesale order</p>
   503→              </div>
   504→            ) : (
   505→              <div className="overflow-x-auto">
   506→                <Table>
   507→                  <TableHeader>
   508→                    <TableRow className="border-b border-border/50 hover:bg-transparent">
   509→                      <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Order #</TableHead>
   510→                      <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Dealer</TableHead>
   511→                      <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Status</TableHead>
   512→                      <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Total</TableHead>
   513→                      <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground">Payment</TableHead>
   514→                      <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground hidden sm:table-cell">Date</TableHead>
   515→                      <TableHead className="text-[11px] uppercase tracking-wider font-semibold text-muted-foreground text-right">Actions</TableHead>
   516→                    </TableRow>
   517→                  </TableHeader>
   518→                  <TableBody>
   519→                    {orders.map((order) => (
   520→                      <TableRow key={order.id} className="hover:bg-secondary/50 border-b border-border/30">
   521→                        <TableCell className="font-mono text-[13px] font-medium tracking-tight">{order.orderNumber}</TableCell>
   522→                        <TableCell className="text-[13px] tracking-tight">
   523→                          {order.customer?.name || order.dealerName || '—'}
