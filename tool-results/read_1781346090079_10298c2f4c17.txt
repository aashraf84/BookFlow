    99→    99→const getPublisherColor = (name: string) => {
   100→   100→  let hash = 0
   101→   101→  for (let i = 0; i < name.length; i++) {
   102→   102→    hash = name.charCodeAt(i) + ((hash << 5) - hash)
   103→   103→  }
   104→   104→  return publisherColors[Math.abs(hash) % publisherColors.length]
   105→   105→}
