   410→   410→function AddEmployeeDialog({
   411→   411→  open,
   412→   412→  onOpenChange,
   413→   413→  branches,
   414→   414→  onSuccess,
   415→   415→}: {
   416→   416→  open: boolean
   417→   417→  onOpenChange: (open: boolean) => void
   418→   418→  branches: Branch[]
   419→   419→  onSuccess: () => void
   420→   420→}) {
   421→   421→  const [saving, setSaving] = React.useState(false)
   422→   422→  const [form, setForm] = React.useState({
   423→   423→    name: '',
   424→   424→    email: '',
   425→   425→    phone: '',
   426→   426→    department: '',
   427→   427→    position: '',
   428→   428→    salary: '',
   429→   429→    hireDate: '',
   430→   430→    branchId: '',
   431→   431→  })
   432→   432→
   433→   433→  const handleSubmit = async (e: React.FormEvent) => {
   434→   434→    e.preventDefault()
   435→   435→    if (!form.name.trim() || !form.email.trim()) return
   436→   436→    setSaving(true)
   437→   437→    try {
   438→   438→      const res = await fetch('/api/employees', {
   439→   439→        method: 'POST',
   440→   440→        headers: { 'Content-Type': 'application/json' },
   441→   441→        body: JSON.stringify({
   442→   442→          ...form,
   443→   443→          salary: parseFloat(form.salary) || 0,
   444→   444→          hireDate: form.hireDate || null,
   445→   445→          branchId: form.branchId || null,
   446→   446→        }),
   447→   447→      })
   448→   448→      if (res.ok) {
   449→   449→        setForm({ name: '', email: '', phone: '', department: '', position: '', salary: '', hireDate: '', branchId: '' })
   450→   450→        onOpenChange(false)
   451→   451→        onSuccess()
   452→   452→      } else {
   453→   453→        const json = await res.json()
   454→   454→        alert(json.error || 'Failed to create employee')
   455→   455→      }
   456→   456→    } catch (err) {
   457→   457→      console.error('Failed to create employee:', err)
   458→   458→      alert('Failed to create employee')
   459→   459→    } finally {
   460→   460→      setSaving(false)
   461→   461→    }
   462→   462→  }
   463→   463→
   464→   464→  return (
   465→   465→    <Dialog open={open} onOpenChange={onOpenChange}>
   466→   466→      <DialogContent className="sm:max-w-md">
   467→   467→        <DialogHeader>
   468→   468→          <DialogTitle>Add Employee</DialogTitle>
   469→   469→          <DialogDescription>Register a new team member</DialogDescription>
   470→   470→        </DialogHeader>
   471→   471→        <form onSubmit={handleSubmit} className="space-y-4">
   472→   472→          <div className="space-y-2">
   473→   473→            <Label htmlFor="emp-name">Full Name *</Label>
   474→   474→            <Input
   475→   475→              id="emp-name"
   476→   476→              value={form.name}
   477→   477→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   478→   478→              placeholder="John Smith"
   479→   479→              required
   480→   480→            />
   481→   481→          </div>
   482→   482→          <div className="grid grid-cols-2 gap-3">
   483→   483→            <div className="space-y-2">
   484→   484→              <Label htmlFor="emp-email">Email *</Label>
   485→   485→              <Input
