   173→// ==================== OVERVIEW TAB ====================
   174→
   175→function OverviewTab() {
   176→  const [accounts, setAccounts] = React.useState<Account[]>([])
   177→  const [expenses, setExpenses] = React.useState<Expense[]>([])
   178→  const [loading, setLoading] = React.useState(true)
   179→
   180→  React.useEffect(() => {
   181→    async function fetchData() {
   182→      setLoading(true)
   183→      try {
   184→        const [accRes, expRes] = await Promise.all([
   185→          fetch('/api/finance?type=accounts&limit=100'),
   186→          fetch('/api/finance?type=expenses&limit=100'),
   187→        ])
   188→        const [accJson, expJson] = await Promise.all([accRes.json(), expRes.json()])
   189→        if (accJson.data) setAccounts(accJson.data)
   190→        if (expJson.data) setExpenses(expJson.data)
   191→      } catch (err) {
   192→        console.error('Failed to fetch finance data:', err)
   193→      } finally {
   194→        setLoading(false)
   195→      }
   196→    }
   197→    fetchData()
   198→  }, [])
   199→
   200→  const revenue = accounts.filter((a) => a.type === 'revenue').reduce((sum, a) => sum + a.balance, 0)
   201→  const expenseTotal = accounts.filter((a) => a.type === 'expense').reduce((sum, a) => sum + a.balance, 0)
   202→  const assets = accounts.filter((a) => a.type === 'asset').reduce((sum, a) => sum + a.balance, 0)
   203→  const liabilities = accounts.filter((a) => a.type === 'liability').reduce((sum, a) => sum + a.balance, 0)
   204→  const netIncome = revenue - expenseTotal
   205→  const pendingExpenses = expenses.filter((e) => e.status === 'pending').reduce((sum, e) => sum + e.amount, 0)
   206→
   207→  const accountTypes = ['asset', 'liability', 'equity', 'revenue', 'expense'] as const
   208→  const accountTypeLabels: Record<string, string> = {
   209→    asset: 'Assets',
   210→    liability: 'Liabilities',
   211→    equity: 'Equity',
   212→    revenue: 'Revenue',
   213→    expense: 'Expenses',
   214→  }
   215→  const accountTypeColors: Record<string, string> = {
   216→    asset: CHART_COLORS.green,
   217→    liability: CHART_COLORS.red,
   218→    equity: CHART_COLORS.teal,
   219→    revenue: CHART_COLORS.blue,
   220→    expense: CHART_COLORS.orange,
   221→  }
   222→
   223→  // Generate mock monthly data for charts
   224→  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']

   226→    month,
   227→    revenue: Math.round(revenue * (0.6 + Math.random() * 0.5 + i * 0.03)),
   228→    expenses: Math.round(expenseTotal * (0.5 + Math.random() * 0.5 + i * 0.02)),
   229→  }))
   230→
   231→  // Account breakdown for pie chart
   232→  const accountBreakdown = accountTypes.map((type) => {
   233→    const balance = accounts.filter((a) => a.type === type).reduce((sum, a) => sum + a.balance, 0)
   234→    return { name: accountTypeLabels[type], value: Math.abs(balance), color: accountTypeColors[type] }
   235→  }).filter((d) => d.value > 0)
   236→
   237→  // Expense categories for bar chart
   238→  const expenseByCategory = expenses.reduce((acc, e) => {
   239→    acc[e.category] = (acc[e.category] || 0) + e.amount
   240→    return acc
   241→  }, {} as Record<string, number>)
   242→
   243→  const expenseCategoryData = Object.entries(expenseByCategory)
   244→    .map(([category, amount]) => ({ category, amount }))
   245→    .sort((a, b) => b.amount - a.amount)
   246→    .slice(0, 8)
   247→
   248→  const categoryColors = [CHART_COLORS.blue, CHART_COLORS.green, CHART_COLORS.orange, CHART_COLORS.purple, CHART_COLORS.teal, CHART_COLORS.red, CHART_COLORS.golden, CHART_COLORS.sunset]
   249→
   250→  // Cash flow area chart data
   251→  const cashFlowData = months.map((month, i) => ({
   252→    month,
   253→    inflow: Math.round(revenue * (0.7 + Math.random() * 0.4 + i * 0.02)),
   254→    outflow: Math.round(expenseTotal * (0.6 + Math.random() * 0.4 + i * 0.015)),
   255→  }))
   256→
   257→  return (
   258→    <div className="space-y-5">
   259→      {loading ? (
   260→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   261→          {[1, 2, 3].map((i) => (
   262→            <div key={i} className="macos-card macos-shadow-card p-5 animate-pulse">
   263→              <div className="h-4 bg-muted/50 rounded w-2/3 mb-3" />
   264→              <div className="h-8 bg-muted/50 rounded w-1/2" />
   265→            </div>
   266→          ))}
   267→        </div>
   268→      ) : (
   269→        <>
   270→          {/* P&L Summary Cards */}
   271→          <MacosStatGrid className="grid-cols-1 sm:grid-cols-3">
   272→            <MacosStatCard
   273→              label="Revenue"
   274→              value={`$${revenue.toLocaleString()}`}
   275→              icon={TrendingUp}
   276→              iconColor="text-macos-green"
   277→              iconBg="bg-macos-green/10"
   278→              trend={{ value: 8.5, label: 'vs last month' }}
   279→            />
   280→            <MacosStatCard
   281→              label="Expenses"
   282→              value={`$${expenseTotal.toLocaleString()}`}
   283→              icon={TrendingDown}
   284→              iconColor="text-macos-red"
   285→              iconBg="bg-macos-red/10"
   286→              trend={{ value: -3.2, label: 'vs last month' }}
   287→            />
   288→            <MacosStatCard
   289→              label="Net Income"
   290→              value={`$${netIncome.toLocaleString()}`}
   291→              icon={DollarSign}
   292→              iconColor="text-macos-blue"
   293→              iconBg="bg-macos-blue/10"
   294→              trend={{ value: netIncome >= 0 ? 12.1 : -5.3, label: 'vs last month' }}
   295→            />
   296→          </MacosStatGrid>
   297→
   298→          {/* Revenue vs Expenses Area Chart */}
   299→          <div className="macos-card macos-shadow-card p-5">
   300→            <div className="macos-section-header mb-4">
   301→              <div>
   302→                <h3 className="macos-section-title">Revenue vs Expenses</h3>
   303→                <p className="macos-section-desc">Monthly financial performance</p>
   304→              </div>
   305→              <div className="flex items-center gap-3">
   306→                <div className="flex items-center gap-1.5">
   307→                  <div className="size-2.5 rounded-full bg-macos-green" />
   308→                  <span className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Revenue</span>
   309→                </div>
   310→                <div className="flex items-center gap-1.5">
   311→                  <div className="size-2.5 rounded-full bg-macos-red" />
   312→                  <span className="text-[10px] uppercase tracking-wider font-semibold text-muted-foreground">Expenses</span>
   313→                </div>
   314→              </div>
   315→            </div>
   316→            <div className="h-64">
   317→              <ResponsiveContainer width="100%" height="100%">
   318→                <AreaChart data={revenueExpenseData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   319→                  <defs>
   320→                    <linearGradient id="revenueGradient" x1="0" y1="0" x2="0" y2="1">
   321→                      <stop offset="5%" stopColor={CHART_COLORS.green} stopOpacity={0.3} />
   322→                      <stop offset="95%" stopColor={CHART_COLORS.green} stopOpacity={0} />
   323→                    </linearGradient>
   324→                    <linearGradient id="expenseGradient" x1="0" y1="0" x2="0" y2="1">
   325→                      <stop offset="5%" stopColor={CHART_COLORS.red} stopOpacity={0.3} />
   326→                      <stop offset="95%" stopColor={CHART_COLORS.red} stopOpacity={0} />
   327→                    </linearGradient>
   328→                  </defs>
   329→                  <CartesianGrid strokeDasharray="3 3" stroke="rgba(0,0,0,0.06)" />
   330→                  <XAxis dataKey="month" tick={{ fontSize: 10, fill: '#86868B' }} axisLine={false} tickLine={false} />
   331→                  <YAxis tick={{ fontSize: 10, fill: '#86868B' }} axisLine={false} tickLine={false} />
   332→                  <Tooltip
   333→                    contentStyle={{
   334→                      borderRadius: '10px',
   335→                      border: '0.5px solid rgba(0,0,0,0.06)',
   336→                      boxShadow: '0 4px 16px rgba(0,0,0,0.08)',
   337→                      fontSize: '12px',
   338→                      padding: '8px 12px',
   339→                    }}
   340→                    formatter={(value: number) => [`$${value.toLocaleString()}`, undefined]}
   341→                  />
   342→                  <Area type="monotone" dataKey="revenue" stroke={CHART_COLORS.green} strokeWidth={2} fill="url(#revenueGradient)" />
   343→                  <Area type="monotone" dataKey="expenses" stroke={CHART_COLORS.red} strokeWidth={2} fill="url(#expenseGradient)" />
   344→                </AreaChart>
   345→              </ResponsiveContainer>
   346→            </div>
   347→          </div>
   348→
   349→          {/* Two column: Cash Flow + Account Breakdown Pie */}
   350→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   351→            {/* Cash Flow Area Chart */}
   352→            <div className="macos-card macos-shadow-card p-5">
   353→              <div className="macos-section-header mb-4">
   354→                <div>
   355→                  <h3 className="macos-section-title">Cash Flow</h3>
   356→                  <p className="macos-section-desc">Monthly inflow vs outflow</p>
   357→                </div>
   358→              </div>
   359→              <div className="h-56">
   360→                <ResponsiveContainer width="100%" height="100%">
   361→                  <AreaChart data={cashFlowData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   362→                    <defs>
   363→                      <linearGradient id="inflowGradient" x1="0" y1="0" x2="0" y2="1">
   364→                        <stop offset="5%" stopColor={CHART_COLORS.blue} stopOpacity={0.3} />
   365→                        <stop offset="95%" stopColor={CHART_COLORS.blue} stopOpacity={0} />
   366→                      </linearGradient>
   367→                      <linearGradient id="outflowGradient" x1="0" y1="0" x2="0" y2="1">
   368→                        <stop offset="5%" stopColor={CHART_COLORS.orange} stopOpacity={0.3} />
   369→                        <stop offset="95%" stopColor={CHART_COLORS.orange} stopOpacity={0} />
   370→                      </linearGradient>
   371→                    </defs>
   372→                    <CartesianGrid strokeDasharray="3 3" stroke="rgba(0,0,0,0.06)" />
   373→                    <XAxis dataKey="month" tick={{ fontSize: 10, fill: '#86868B' }} axisLine={false} tickLine={false} />
   374→                    <YAxis tick={{ fontSize: 10, fill: '#86868B' }} axisLine={false} tickLine={false} />
   375→                    <Tooltip
   376→                      contentStyle={{
   377→                        borderRadius: '10px',
   378→                        border: '0.5px solid rgba(0,0,0,0.06)',
   379→                        boxShadow: '0 4px 16px rgba(0,0,0,0.08)',
   380→                        fontSize: '12px',
   381→                        padding: '8px 12px',
   382→                      }}
   383→                      formatter={(value: number) => [`$${value.toLocaleString()}`, undefined]}
   384→                    />
   385→                    <Area type="monotone" dataKey="inflow" stroke={CHART_COLORS.blue} strokeWidth={2} fill="url(#inflowGradient)" />
   386→                    <Area type="monotone" dataKey="outflow" stroke={CHART_COLORS.orange} strokeWidth={2} fill="url(#outflowGradient)" />
   387→                  </AreaChart>
   388→                </ResponsiveContainer>
   389→              </div>
   390→            </div>
   391→
   392→            {/* Account Breakdown Pie */}
   393→            <div className="macos-card macos-shadow-card p-5">
   394→              <div className="macos-section-header mb-4">
   395→                <div>
   396→                  <h3 className="macos-section-title">Account Breakdown</h3>
   397→                  <p className="macos-section-desc">Distribution by type</p>
   398→                </div>
   399→              </div>
   400→              {accountBreakdown.length > 0 ? (
   401→                <div className="h-56">
   402→                  <ResponsiveContainer width="100%" height="100%">
   403→                    <RechartsPieChart>
   404→                      <Pie
   405→                        data={accountBreakdown}
   406→                        cx="50%"
   407→                        cy="50%"
   408→                        innerRadius={55}
   409→                        outerRadius={80}
   410→                        paddingAngle={3}
   411→                        dataKey="value"
   412→                        strokeWidth={0}
   413→                      >
   414→                        {accountBreakdown.map((entry, index) => (
   415→                          <Cell key={`cell-${index}`} fill={entry.color} />
   416→                        ))}
   417→                      </Pie>
   418→                      <Tooltip
   419→                        contentStyle={{
   420→                          borderRadius: '10px',
   421→                          border: '0.5px solid rgba(0,0,0,0.06)',
   422→                          boxShadow: '0 4px 16px rgba(0,0,0,0.08)',
   423→                          fontSize: '12px',
   424→                          padding: '8px 12px',
   425→                        }}
   426→                        formatter={(value: number) => [`$${value.toLocaleString()}`, undefined]}
   427→                      />
   428→                      <Legend
   429→                        iconType="circle"
   430→                        iconSize={8}
   431→                        wrapperStyle={{ fontSize: '11px' }}
   432→                      />
   433→                    </RechartsPieChart>
   434→                  </ResponsiveContainer>
   435→                </div>
   436→              ) : (
   437→                <div className="h-56 flex items-center justify-center">
   438→                  <p className="text-[13px] text-muted-foreground">No account data</p>
   439→                </div>
   440→              )}
   441→            </div>
   442→          </div>
   443→
   444→          {/* Balance Sheet + Expense Categories */}
   445→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   446→            {/* Balance Sheet */}
   447→            <div className="macos-card macos-shadow-card p-5">
   448→              <div className="macos-section-header mb-4">
   449→                <div>
   450→                  <h3 className="macos-section-title">Balance Sheet</h3>
   451→                  <p className="macos-section-desc">Financial position summary</p>
   452→                </div>
   453→              </div>
   454→              <div className="space-y-3">
   455→                <div className="flex justify-between text-[13px]">
   456→                  <span className="text-muted-foreground">Total Assets</span>
   457→                  <span className="font-medium">${assets.toLocaleString()}</span>
   458→                </div>
   459→                <div className="h-2 rounded-full bg-secondary overflow-hidden">
   460→                  <div
   461→                    className="h-full bg-macos-green rounded-full transition-all"
   462→                    style={{ width: `${assets > 0 ? Math.min((assets / (assets + liabilities || 1)) * 100, 100) : 0}%` }}
   463→                  />
   464→                </div>
   465→                <div className="flex justify-between text-[13px]">
   466→                  <span className="text-muted-foreground">Total Liabilities</span>
   467→                  <span className="font-medium">${liabilities.toLocaleString()}</span>
   468→                </div>
   469→                <div className="h-2 rounded-full bg-secondary overflow-hidden">
   470→                  <div
   471→                    className="h-full bg-macos-red rounded-full transition-all"
   472→                    style={{ width: `${liabilities > 0 ? Math.min((liabilities / (assets + liabilities || 1)) * 100, 100) : 0}%` }}
   473→                  />
   474→                </div>
   475→                <Separator className="bg-border/50" />
   476→                <div className="flex justify-between text-[13px] font-semibold">
   477→                  <span>Net Worth</span>
   478→                  <span className={(assets - liabilities) >= 0 ? 'text-macos-green' : 'text-macos-red'}>
   479→                    ${(assets - liabilities).toLocaleString()}
   480→                  </span>
   481→                </div>
   482→              </div>
   483→            </div>
   484→
   485→            {/* Expense Categories Bar Chart */}
   486→            <div className="macos-card macos-shadow-card p-5">
   487→              <div className="macos-section-header mb-4">
   488→                <div>
   489→                  <h3 className="macos-section-title">Expense Categories</h3>
   490→                  <p className="macos-section-desc">Spending by category</p>
   491→                </div>
   492→              </div>
   493→              {expenseCategoryData.length > 0 ? (
   494→                <div className="h-56">
   495→                  <ResponsiveContainer width="100%" height="100%">
   496→                    <BarChart data={expenseCategoryData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }} layout="vertical">
   497→                      <CartesianGrid strokeDasharray="3 3" stroke="rgba(0,0,0,0.06)" horizontal={false} />
   498→                      <XAxis type="number" tick={{ fontSize: 10, fill: '#86868B' }} axisLine={false} tickLine={false} />
   499→                      <YAxis type="category" dataKey="category" tick={{ fontSize: 10, fill: '#86868B' }} axisLine={false} tickLine={false} width={80} />
   500→                      <Tooltip
   501→                        contentStyle={{
   502→                          borderRadius: '10px',
   503→                          border: '0.5px solid rgba(0,0,0,0.06)',
   504→                          boxShadow: '0 4px 16px rgba(0,0,0,0.08)',
   505→                          fontSize: '12px',
   506→                          padding: '8px 12px',
   507→                        }}
   508→                        formatter={(value: number) => [`$${value.toLocaleString()}`, 'Amount']}
   509→                      />
   510→                      <Bar dataKey="amount" radius={[0, 4, 4, 0]} maxBarSize={24}>
   511→                        {expenseCategoryData.map((_, index) => (
   512→                          <Cell key={`cat-${index}`} fill={categoryColors[index % categoryColors.length]} />
   513→                        ))}
   514→                      </Bar>
   515→                    </BarChart>
   516→                  </ResponsiveContainer>
   517→                </div>
   518→              ) : (
   519→                <div className="h-56 flex items-center justify-center">
   520→                  <p className="text-[13px] text-muted-foreground">No expense data</p>
   521→                </div>
   522→              )}
   523→            </div>
   524→          </div>
   525→

   970→function CreateAccountDialog({
   971→  open,
   972→  onOpenChange,
   973→  onSuccess,
   974→}: {
   975→  open: boolean
   976→  onOpenChange: (open: boolean) => void
   977→  onSuccess: () => void
   978→}) {
   979→  const [saving, setSaving] = React.useState(false)
   980→  const [form, setForm] = React.useState({
   981→    code: '',
   982→    name: '',
   983→    type: 'asset',
   984→    parentCode: '',
   985→  })
   986→
   987→  const handleSubmit = async (e: React.FormEvent) => {
   988→    e.preventDefault()
   989→    if (!form.code.trim() || !form.name.trim()) return
   990→    setSaving(true)
   991→    try {
   992→      const res = await fetch('/api/finance?type=accounts', {
   993→        method: 'POST',
   994→        headers: { 'Content-Type': 'application/json' },
   995→        body: JSON.stringify({
   996→          ...form,
   997→          parentCode: form.parentCode || null,
   998→        }),
   999→      })
  1000→      if (res.ok) {
  1001→        setForm({ code: '', name: '', type: 'asset', parentCode: '' })
  1002→        onOpenChange(false)
  1003→        onSuccess()
  1004→      } else {
  1005→        const json = await res.json()
  1006→        alert(json.error || 'Failed to create account')
  1007→      }
  1008→    } catch (err) {
  1009→      console.error('Failed to create account:', err)
  1010→      alert('Failed to create account')
  1011→    } finally {
  1012→      setSaving(false)
  1013→    }
  1014→  }
  1015→
  1016→  return (
  1017→    <MacosDialog
  1018→      open={open}
  1019→      onOpenChange={onOpenChange}
  1020→      title="Create Account"
  1021→      description="Add a new account to your chart of accounts"
  1022→      footer={
  1023→        <>
  1024→          <MacosButton variant="secondary" onClick={() => onOpenChange(false)}>Cancel</MacosButton>
  1025→          <MacosButton variant="primary" onClick={() => handleSubmit({ preventDefault: () => {} } as React.FormEvent)} disabled={saving || !form.code.trim() || !form.name.trim()}>
  1026→            {saving ? 'Creating...' : 'Create Account'}
  1027→          </MacosButton>
  1028→        </>
  1029→      }
  1030→    >
  1031→      <form onSubmit={handleSubmit} className="space-y-4">
