   145→const getCoverColor = (title: string) => {
   146→  let hash = 0
   147→  for (let i = 0; i < title.length; i++) {
   148→    hash = title.charCodeAt(i) + ((hash << 5) - hash)
   149→  }
   150→  return coverColors[Math.abs(hash) % coverColors.length]
   151→}

   172→export default function BookCatalogModule() {
   173→  const { authUser } = useAppStore()

   175→
   176→  const [books, setBooks] = useState<Book[]>([])

   178→  const [total, setTotal] = useState(0)
   179→  const [page, setPage] = useState(1)
   180→  const [loading, setLoading] = useState(true)
   181→  const [search, setSearch] = useState('')
   182→  const [filterCategory, setFilterCategory] = useState('')
   183→  const [filterAuthor, setFilterAuthor] = useState('')
   184→  const [filterPublisher, setFilterPublisher] = useState('')
   185→  const [dialogOpen, setDialogOpen] = useState(false)
   186→  const [editing, setEditing] = useState<Book | null>(null)
   187→  const [form, setForm] = useState<BookFormData>(emptyForm)
   188→  const [saving, setSaving] = useState(false)
   189→  const [deleting, setDeleting] = useState<string | null>(null)
   190→
   191→  // Dropdown data
   192→  const [authors, setAuthors] = useState<Author[]>([])
   193→  const [publishers, setPublishers] = useState<Publisher[]>([])
   194→  const [categories, setCategories] = useState<Category[]>([])
   195→
   196→  const limit = 10
   197→
   198→  // ── Fetch dropdown data ────────────────────────────────────
   199→  useEffect(() => {
   200→    Promise.all([
   201→      fetch('/api/authors?limit=200').then((r) => r.json()),
   202→      fetch('/api/publishers?limit=200').then((r) => r.json()),
   203→      fetch('/api/categories').then((r) => r.json()),
   204→    ]).then(([authorsJson, publishersJson, categoriesJson]) => {
   205→      setAuthors(authorsJson.data || [])
   206→      setPublishers(publishersJson.data || [])
   207→      setCategories(categoriesJson.data || [])
   208→    })
   209→  }, [])
   210→
   211→  // ── Fetch books ────────────────────────────────────────────
   212→  const fetchBooks = useCallback(async () => {
   213→    setLoading(true)
   214→    try {
   215→      const params = new URLSearchParams({
   216→        page: String(page),
   217→        limit: String(limit),
   218→      })
   219→      if (search) params.set('search', search)
   220→      if (filterCategory && filterCategory !== 'all') params.set('categoryId', filterCategory)
   221→      if (filterAuthor && filterAuthor !== 'all') params.set('authorId', filterAuthor)
   222→      if (filterPublisher && filterPublisher !== 'all') params.set('publisherId', filterPublisher)
   223→
   224→      const res = await fetch(`/api/books?${params}`)
   225→      if (!res.ok) throw new Error('Failed to fetch books')
   226→      const json = await res.json()
   227→      setBooks(json.data || [])
   228→      setTotalPages(json.pagination?.totalPages || 1)
   229→      setTotal(json.pagination?.total || 0)
   230→    } catch {
   231→      setBooks([])
   232→    } finally {
   233→      setLoading(false)
   234→    }
   235→  }, [page, search, filterCategory, filterAuthor, filterPublisher])
   236→
   237→  useEffect(() => {
   238→    fetchBooks()
   239→  }, [fetchBooks])
   240→
   241→  // Reset to page 1 when filters change
   242→  useEffect(() => {
   243→    setPage(1)
   244→  }, [search, filterCategory, filterAuthor, filterPublisher])
   245→
   246→  // ── Save book (create or update) ───────────────────────────
   247→  const handleSave = async () => {
   248→    if (!form.title.trim()) return
   249→    setSaving(true)
   250→    try {
   251→      const payload: Record<string, unknown> = {
   252→        title: form.title.trim(),
   253→        subtitle: form.subtitle || null,
   254→        isbn: form.isbn || null,
   255→        authorId: form.authorId || null,
   256→        publisherId: form.publisherId || null,
   257→        categoryId: form.categoryId || null,
   258→        edition: form.edition || null,
   259→        language: form.language || 'English',
   260→        pages: form.pages ? parseInt(form.pages) : null,
   261→        publishYear: form.publishYear ? parseInt(form.publishYear) : null,
   262→        description: form.description || null,
   263→        price: form.price ? parseFloat(form.price) : 0,
   264→        costPrice: form.costPrice ? parseFloat(form.costPrice) : 0,
   265→        discount: form.discount ? parseFloat(form.discount) : 0,
   266→        academicLevel: form.academicLevel || null,
   267→        subject: form.subject || null,
   268→        barcode: form.barcode || null,
   269→      }
   270→
   271→      let res: Response
   272→      if (editing) {
   273→        payload.id = editing.id
   274→        res = await fetch('/api/books', {
   275→          method: 'PUT',
   276→          headers: { 'Content-Type': 'application/json' },
   277→          body: JSON.stringify(payload),
   278→        })
   279→      } else {
   280→        res = await fetch('/api/books', {
   281→          method: 'POST',
   282→          headers: { 'Content-Type': 'application/json' },
   283→          body: JSON.stringify(payload),
   284→        })
   285→      }
   286→
   287→      if (!res.ok) {
   288→        const err = await res.json()
   289→        throw new Error(err.error || 'Failed to save book')
   290→      }
   291→
   292→      setDialogOpen(false)
   293→      setEditing(null)
   294→      setForm(emptyForm)
   295→      fetchBooks()
   296→    } catch (err) {
   297→      alert(err instanceof Error ? err.message : 'Failed to save book')
   298→    } finally {
   299→      setSaving(false)
   300→    }
   301→  }
   302→
   303→  // ── Delete book ────────────────────────────────────────────
   304→  const handleDelete = async (id: string) => {
   305→    if (!confirm('Are you sure you want to delete this book?')) return

   307→    try {
   308→      const res = await fetch(`/api/books?id=${id}`, { method: 'DELETE' })
   309→      if (!res.ok) {
   310→        const err = await res.json()
   311→        throw new Error(err.error || 'Failed to delete')
   312→      }
   313→      fetchBooks()
   314→    } catch (err) {
   315→      alert(err instanceof Error ? err.message : 'Failed to delete book')
   316→    } finally {
   317→      setDeleting(null)
   318→    }
   319→  }

   329→      publisherId: book.publisherId || '',
   330→      categoryId: book.categoryId || '',
   331→      edition: book.edition || '',
   332→      language: book.language || 'English',
   333→      pages: book.pages?.toString() || '',
   334→      publishYear: book.publishYear?.toString() || '',
   335→      description: book.description || '',
   336→      price: book.price?.toString() || '',

   338→      discount: book.discount?.toString() || '',
   339→      academicLevel: book.academicLevel || '',
   340→      subject: book.subject || '',
   341→      barcode: book.barcode || '',
   342→    })
   343→    setDialogOpen(true)
   344→  }
   345→
   346→  // ── Open dialog for add ────────────────────────────────────
   347→  const openAdd = () => {
   348→    setEditing(null)
   349→    setForm(emptyForm)
   350→    setDialogOpen(true)
   351→  }
   352→
   353→  const formatCurrency = (value: number) =>
   354→    new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(value)
   355→
   356→  const hasActiveFilters = search || (filterCategory && filterCategory !== 'all') || (filterAuthor && filterAuthor !== 'all') || (filterPublisher && filterPublisher !== 'all')
   357→
   358→  // ── Computed stats ─────────────────────────────────────────

   362→  return (

   367→        subtitle={`${total} ${total === 1 ? 'book' : 'books'} in your library`}
   368→        icon={Library}
   369→        iconColor="text-macos-blue"
   370→        iconBg="bg-macos-blue/10"
   371→      >

   373→          <MacosButton variant="primary" icon={Plus} onClick={openAdd}>
   374→            Add Book
   375→          </MacosButton>
   376→        )}
   377→      </MacosPageHeader>
   378→
   379→      {/* ── Stat Cards ─────────────────────────────────────────── */}
   380→      <MacosStatGrid>
   381→        <MacosStatCard
   382→          label="Total Books"
   383→          value={total}
   384→          icon={BookCopy}
   385→          iconColor="text-macos-blue"
   386→          iconBg="bg-macos-blue/10"
   387→        />
   388→        <MacosStatCard
   389→          label="Active"
   390→          value={activeBooks}
   391→          icon={BookOpen}
   392→          iconColor="text-macos-green"
   393→          iconBg="bg-macos-green/10"
   394→        />
   395→        <MacosStatCard
   396→          label="Categories"
   397→          value={categories.length}
   398→          icon={Tag}
   399→          iconColor="text-macos-purple"
   400→          iconBg="bg-macos-purple/10"
   401→        />
   402→        <MacosStatCard
   403→          label="Page Value"
   404→          value={formatCurrency(totalValue)}
   405→          icon={CircleDollarSign}
   406→          iconColor="text-macos-orange"
   407→          iconBg="bg-macos-orange/10"
   408→        />
   409→      </MacosStatGrid>
   410→
   411→      {/* ── Search & Filters Card ──────────────────────────────── */}
   412→      <div className="macos-card macos-shadow-card p-4">

   414→        <div className="mb-3">

   417→            onChange={setSearch}

   421→
   422→        {/* Filter bar with segmented control style */}
   423→        <div className="flex flex-wrap items-center gap-2">
   424→          <div className="macos-filter-bar">
   425→            <Select value={filterCategory} onValueChange={setFilterCategory}>
   426→              <SelectTrigger className="border-0 bg-transparent shadow-none text-[12px] font-medium h-6 w-auto min-w-[120px] p-0 pl-2 pr-6 focus:ring-0">
   427→                <Filter className="size-3 mr-1 text-muted-foreground" strokeWidth={1.5} />
   428→                <SelectValue placeholder="Category" />
   429→              </SelectTrigger>
   430→              <SelectContent>
   431→                <SelectItem value="all">All Categories</SelectItem>
   432→                {categories.map((cat) => (
   433→                  <SelectItem key={cat.id} value={cat.id}>
   434→                    {cat.name}
   435→                  </SelectItem>
   436→                ))}
   437→              </SelectContent>
   438→            </Select>
   439→          </div>
   440→          <div className="macos-filter-bar">
   441→            <Select value={filterAuthor} onValueChange={setFilterAuthor}>
   442→              <SelectTrigger className="border-0 bg-transparent shadow-none text-[12px] font-medium h-6 w-auto min-w-[110px] p-0 pl-2 pr-6 focus:ring-0">
   443→                <Filter className="size-3 mr-1 text-muted-foreground" strokeWidth={1.5} />
   444→                <SelectValue placeholder="Author" />
   445→              </SelectTrigger>
   446→              <SelectContent>
   447→                <SelectItem value="all">All Authors</SelectItem>

   449→                  <SelectItem key={a.id} value={a.id}>
   450→                    {a.name}
   451→                  </SelectItem>
   452→                ))}
   453→              </SelectContent>
   454→            </Select>
   455→          </div>
   456→          <div className="macos-filter-bar">
   457→            <Select value={filterPublisher} onValueChange={setFilterPublisher}>

   477→                setFilterAuthor('')
   478→                setFilterPublisher('')

   484→          )}

   487→
   488→      {/* ── Books Table Card ────────────────────────────────────── */}
   489→      <div className="macos-card macos-shadow-card overflow-hidden">
   490→        {loading ? (
   491→          <div className="p-5 space-y-3">
   492→            {Array.from({ length: 5 }).map((_, i) => (
   493→              <div key={i} className="flex items-center gap-3">
   494→                <Skeleton className="size-9 rounded-lg" />
   495→                <div className="flex-1 space-y-1.5">
   496→                  <Skeleton className="h-4 w-3/5" />
   497→                  <Skeleton className="h-3 w-2/5" />
   498→                </div>
   499→              </div>
   500→            ))}
   501→          </div>
   502→        ) : books.length === 0 ? (
   503→          <MacosEmptyState
   504→            icon={BookOpen}
   505→            title="No books found"
   506→            description="Try adjusting your search or filters"
   507→            action={
   508→              !isViewOnly ? (
   509→                <MacosButton variant="primary" icon={Plus} onClick={openAdd}>
   510→                  Add Your First Book
   511→                </MacosButton>
   512→              ) : undefined
   513→            }
   514→          />
   515→        ) : (
   516→          <div className="overflow-x-auto custom-scrollbar">
   517→            <table className="w-full">
   518→              <thead>
   519→                <tr className="border-b border-border/50">
   520→                  <th className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9 px-5 text-left font-medium">Title</th>
   521→                  <th className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9 px-3 text-left font-medium">Author</th>
   522→                  <th className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9 px-3 text-left font-medium">ISBN</th>
   523→                  <th className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/60 h-9 px-3 text-left font-medium">Category</th>

   531→              <tbody>
