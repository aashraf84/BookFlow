   174→function SchoolsTab() {
   175→  const [schools, setSchools] = React.useState<School[]>([])
   176→  const [loading, setLoading] = React.useState(true)
   177→  const [search, setSearch] = React.useState('')
   178→  const [showAddDialog, setShowAddDialog] = React.useState(false)
   179→  const [selectedSchool, setSelectedSchool] = React.useState<School | null>(null)
   180→
   181→  const fetchSchools = React.useCallback(async () => {
   182→    setLoading(true)
   183→    try {
   184→      const res = await fetch(`/api/schools?search=${encodeURIComponent(search)}&limit=50`)
   185→      const json = await res.json()
   186→      if (json.data) setSchools(json.data)
   187→    } catch (err) {
   188→      console.error('Failed to fetch schools:', err)
   189→    } finally {
   190→      setLoading(false)
   191→    }
   192→  }, [search])
   193→
   194→  React.useEffect(() => {
   195→    fetchSchools()
   196→  }, [fetchSchools])
   197→
   198→  const handleDelete = async (id: string) => {
   199→    if (!confirm('Are you sure you want to deactivate this school?')) return
   200→    try {
   201→      await fetch(`/api/schools?id=${id}`, { method: 'DELETE' })
   202→      fetchSchools()
   203→    } catch (err) {
   204→      console.error('Failed to delete school:', err)
   205→    }
   206→  }
   207→
   208→  return (
   209→    <div className="space-y-4">
   210→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   211→        <div className="relative flex-1 w-full sm:max-w-sm">
   212→          <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   213→          <Input
   214→            placeholder="Search schools..."
   215→            value={search}
   216→            onChange={(e) => setSearch(e.target.value)}
   217→            className="pl-8"
   218→          />
   219→        </div>
   220→        <Button onClick={() => setShowAddDialog(true)} className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   221→          <Plus className="size-4" /> Add School
   222→        </Button>
   223→      </div>
   224→
   225→      {loading ? (
   226→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   227→          {[1, 2, 3].map((i) => (
   228→            <Card key={i} className="animate-pulse">
   229→              <CardContent className="p-4">
   230→                <div className="h-4 bg-muted rounded w-2/3 mb-3" />
   231→                <div className="h-3 bg-muted rounded w-1/2 mb-2" />
   232→                <div className="h-3 bg-muted rounded w-3/4" />
   233→              </CardContent>
   234→            </Card>
   235→          ))}
   236→        </div>
   237→      ) : schools.length === 0 ? (
   238→        <Card>
   239→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   240→            <GraduationCap className="size-10 text-muted-foreground mb-3" />
   241→            <p className="text-muted-foreground">No schools found</p>
   242→            <p className="text-sm text-muted-foreground">Add your first school to get started</p>
   243→          </CardContent>
   244→        </Card>
   245→      ) : (
   246→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   247→          {schools.map((school) => (
   248→            <Card
   249→              key={school.id}
   250→              className="hover:shadow-md transition-shadow cursor-pointer group"
   251→              onClick={() => setSelectedSchool(school)}
   252→            >
   253→              <CardContent className="p-4">
   254→                <div className="flex items-start justify-between mb-3">
   255→                  <div className="flex items-center gap-2">
   256→                    <div className="rounded-lg bg-emerald-100 p-1.5 dark:bg-emerald-950/40">
   257→                      <GraduationCap className="size-4 text-emerald-600" />
   258→                    </div>
   259→                    <div>
   260→                      <h3 className="font-semibold text-sm leading-tight">{school.name}</h3>
   261→                      {school.academicSession && (
   262→                        <p className="text-xs text-muted-foreground">{school.academicSession}</p>
   263→                      )}
   264→                    </div>
   265→                  </div>
   266→                  <Badge
   267→                    variant={school.isActive ? 'default' : 'secondary'}
   268→                    className={
   269→                      school.isActive
   270→                        ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 text-[10px]'
   271→                        : 'text-[10px]'
   272→                    }
   273→                  >
   274→                    {school.isActive ? 'Active' : 'Inactive'}
   275→                  </Badge>
   276→                </div>
   277→                <div className="space-y-1.5 text-xs text-muted-foreground">
   278→                  {school.address && (
   279→                    <div className="flex items-center gap-1.5">
   280→                      <MapPin className="size-3 shrink-0" />
   281→                      <span className="truncate">{school.address}</span>
   282→                    </div>
   283→                  )}
   284→                  {school.phone && (
   285→                    <div className="flex items-center gap-1.5">
   286→                      <Phone className="size-3 shrink-0" />
   287→                      {school.phone}
   288→                    </div>
   289→                  )}
   290→                  {school.contactPerson && (
   291→                    <div className="flex items-center gap-1.5">
   292→                      <User className="size-3 shrink-0" />
   293→                      {school.contactPerson}
   294→                    </div>
   295→                  )}
   296→                </div>
   297→                <Separator className="my-3" />
   298→                <div className="flex items-center justify-between text-xs">
   299→                  <span className="text-muted-foreground">
   300→                    {school._count?.contracts || 0} contracts
   301→                  </span>
   302→                  <span className="text-muted-foreground">
   303→                    {school._count?.packages || 0} packages
   304→                  </span>
   305→                  <Button
   306→                    variant="ghost"
   307→                    size="sm"
   308→                    className="h-6 px-2 text-xs opacity-0 group-hover:opacity-100 text-red-600 hover:text-red-700 hover:bg-red-50 dark:hover:bg-red-950/30"
   309→                    onClick={(e) => {
   310→                      e.stopPropagation()
   311→                      handleDelete(school.id)
   312→                    }}
   313→                  >
   314→                    <Trash2 className="size-3" />
   315→                  </Button>
   316→                </div>
   317→              </CardContent>
   318→            </Card>
   319→          ))}
   320→        </div>
   321→      )}
   322→
   323→      <AddSchoolDialog open={showAddDialog} onOpenChange={setShowAddDialog} onSuccess={fetchSchools} />
   324→      <SchoolDetailDialog school={selectedSchool} onOpenChange={(open) => !open && setSelectedSchool(null)} onSuccess={fetchSchools} />
   325→    </div>
   326→  )
   327→}

   331→function ContractsTab() {
   332→  const [contracts, setContracts] = React.useState<SchoolContract[]>([])
   333→  const [schools, setSchools] = React.useState<School[]>([])
   334→  const [loading, setLoading] = React.useState(true)
   335→  const [showCreateDialog, setShowCreateDialog] = React.useState(false)
   336→
   337→  const fetchContracts = React.useCallback(async () => {
   338→    setLoading(true)
   339→    try {
   340→      const res = await fetch('/api/schools/contracts')
   341→      if (res.ok) {
   342→        const json = await res.json()
   343→        if (json.data) setContracts(json.data)
   344→      } else {
   345→        // Fallback: fetch schools with contracts
   346→        const schoolRes = await fetch('/api/schools?limit=50')
   347→        const schoolJson = await schoolRes.json()
   348→        if (schoolJson.data) {
   349→          const allContracts: SchoolContract[] = []
   350→          schoolJson.data.forEach((s: School & { contracts?: SchoolContract[] }) => {
   351→            if (s.contracts) {
   352→              s.contracts.forEach((c) => allContracts.push({ ...c, school: { id: s.id, name: s.name } }))
   353→            }
   354→          })
   355→          setContracts(allContracts)
   356→        }
   357→      }
   358→    } catch (err) {
   359→      console.error('Failed to fetch contracts:', err)
   360→    } finally {
   361→      setLoading(false)
   362→    }
   363→  }, [])
   364→
   365→  const fetchSchools = React.useCallback(async () => {
   366→    try {
   367→      const res = await fetch('/api/schools?limit=50')
   368→      const json = await res.json()
   369→      if (json.data) setSchools(json.data)
   370→    } catch (err) {
   371→      console.error('Failed to fetch schools:', err)
   372→    }
   373→  }, [])
   374→
   375→  React.useEffect(() => {
   376→    fetchContracts()
   377→    fetchSchools()
   378→  }, [fetchContracts, fetchSchools])
   379→
   380→  const getStatusColor = (status: string) => {
   381→    switch (status) {
   382→      case 'active':
   383→        return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400'
   384→      case 'expired':
   385→        return 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400'
   386→      case 'cancelled':
   387→        return 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400'
   388→      default:
   389→        return ''
   390→    }
   391→  }
   392→
   393→  return (
   394→    <div className="space-y-4">
   395→      <div className="flex items-center justify-between">
   396→        <p className="text-sm text-muted-foreground">{contracts.length} contracts</p>
   397→        <Button onClick={() => setShowCreateDialog(true)} className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   398→          <Plus className="size-4" /> Create Contract
   399→        </Button>
   400→      </div>
   401→
   402→      {loading ? (
   403→        <Card>
   404→          <CardContent className="p-6">
   405→            <div className="space-y-3">
   406→              {[1, 2, 3].map((i) => (
   407→                <div key={i} className="h-12 bg-muted rounded animate-pulse" />
   408→              ))}
   409→            </div>
   410→          </CardContent>
   411→        </Card>
   412→      ) : contracts.length === 0 ? (
   413→        <Card>
   414→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   415→            <FileText className="size-10 text-muted-foreground mb-3" />
   416→            <p className="text-muted-foreground">No contracts found</p>
   417→            <p className="text-sm text-muted-foreground">Create your first school contract</p>
   418→          </CardContent>
   419→        </Card>
   420→      ) : (
   421→        <Card>
   422→          <CardContent className="p-0">
   423→            <ScrollArea className="max-h-[500px]">
   424→              <Table>
   425→                <TableHeader>
   426→                  <TableRow>
   427→                    <TableHead>Title</TableHead>
   428→                    <TableHead>School</TableHead>
   429→                    <TableHead>Session</TableHead>
   430→                    <TableHead>Value</TableHead>
   431→                    <TableHead>Status</TableHead>
   432→                    <TableHead>Period</TableHead>
   433→                  </TableRow>
   434→                </TableHeader>
   435→                <TableBody>
   436→                  {contracts.map((contract) => (
   437→                    <TableRow key={contract.id}>
   438→                      <TableCell className="font-medium">{contract.title}</TableCell>
   439→                      <TableCell>{contract.school?.name || '—'}</TableCell>
   440→                      <TableCell>{contract.academicSession || '—'}</TableCell>
   441→                      <TableCell>${contract.totalValue.toLocaleString()}</TableCell>
   442→                      <TableCell>
   443→                        <Badge className={`text-[10px] ${getStatusColor(contract.status)}`}>
   444→                          {contract.status}
   445→                        </Badge>
   446→                      </TableCell>
   447→                      <TableCell className="text-xs text-muted-foreground">
   448→                        {contract.startDate
   449→                          ? `${new Date(contract.startDate).toLocaleDateString()} - ${contract.endDate ? new Date(contract.endDate).toLocaleDateString() : '...'}`
   450→                          : '—'}
   451→                      </TableCell>
   452→                    </TableRow>
   453→                  ))}
   454→                </TableBody>
   455→              </Table>
   456→            </ScrollArea>
   457→          </CardContent>
   458→        </Card>
   459→      )}
   460→
   461→      <CreateContractDialog
   462→        open={showCreateDialog}
   463→        onOpenChange={setShowCreateDialog}
   464→        schools={schools}
   465→        onSuccess={fetchContracts}
   466→      />
   467→    </div>
   468→  )
   469→}

  1065→function CreatePackageDialog({
  1066→  open,
  1067→  onOpenChange,
  1068→  schools,

  1070→  stationery,
  1071→  onSuccess,
  1072→}: {

  1074→  onOpenChange: (open: boolean) => void

  1082→    schoolId: '',
  1083→    name: '',
  1084→    grade: '',
  1085→    academicLevel: '',
  1086→    description: '',
  1087→    discount: '',
  1088→  })
  1089→  const [items, setItems] = React.useState<PackageItemForm[]>([
  1090→    { key: crypto.randomUUID(), itemType: 'book', itemId: '', quantity: 1, unitPrice: 0, name: '' },
  1091→  ])
  1092→
  1093→  const totalPrice = items.reduce((sum, item) => sum + item.quantity * item.unitPrice, 0)
  1094→
  1095→  const addItem = () => {
  1096→    setItems([
  1097→      ...items,
  1098→      { key: crypto.randomUUID(), itemType: 'book', itemId: '', quantity: 1, unitPrice: 0, name: '' },
  1099→    ])
  1100→  }
  1101→
  1102→  const removeItem = (key: string) => {
  1103→    if (items.length > 1) {
  1104→      setItems(items.filter((i) => i.key !== key))
  1105→    }
  1106→  }
  1107→
  1108→  const updateItem = (key: string, updates: Partial<PackageItemForm>) => {
  1109→    setItems(
  1110→      items.map((item) => {
  1111→        if (item.key !== key) return item
  1112→        const updated = { ...item, ...updates }
  1113→
  1114→        // Auto-fill price and name when selecting a book/stationery
  1115→        if (updates.itemId) {
  1116→          if (updated.itemType === 'book') {
  1117→            const book = books.find((b) => b.id === updates.itemId)
  1118→            if (book) {
  1119→              updated.unitPrice = book.price
  1120→              updated.name = book.title
  1121→            }
  1122→          } else {
  1123→            const product = stationery.find((s) => s.id === updates.itemId)
  1124→            if (product) {
  1125→              updated.unitPrice = product.price
  1126→              updated.name = product.name
  1127→            }
  1128→          }
  1129→        }
  1130→
  1131→        return updated
  1132→      })
  1133→    )
  1134→  }
  1135→
  1136→  const handleSubmit = async (e: React.FormEvent) => {
  1137→    e.preventDefault()
  1138→    if (!form.name.trim()) return
  1139→    setSaving(true)
  1140→    try {
  1141→      const res = await fetch('/api/schools/packages', {
  1142→        method: 'POST',
  1143→        headers: { 'Content-Type': 'application/json' },
  1144→        body: JSON.stringify({
  1145→          schoolId: form.schoolId || null,
  1146→          name: form.name,
  1147→          grade: form.grade,
  1148→          academicLevel: form.academicLevel,
  1149→          description: form.description,
  1150→          totalPrice,
  1151→          discount: parseFloat(form.discount) || 0,
  1152→          items: items
  1153→            .filter((i) => i.itemId)
  1154→            .map((i) => ({
  1155→              bookId: i.itemType === 'book' ? i.itemId : null,
  1156→              stationeryId: i.itemType === 'stationery' ? i.itemId : null,
  1157→              quantity: i.quantity,
  1158→              unitPrice: i.unitPrice,
  1159→            })),
  1160→        }),
  1161→      })
  1162→      if (res.ok) {
  1163→        setForm({ schoolId: '', name: '', grade: '', academicLevel: '', description: '', discount: '' })
  1164→        setItems([{ key: crypto.randomUUID(), itemType: 'book', itemId: '', quantity: 1, unitPrice: 0, name: '' }])
  1165→        onOpenChange(false)
  1166→        onSuccess()
  1167→      } else {
  1168→        const json = await res.json()
  1169→        alert(json.error || 'Failed to create package')
  1170→      }
  1171→    } catch (err) {
  1172→      console.error('Failed to create package:', err)
  1173→      alert('Failed to create package')
  1174→    } finally {
  1175→      setSaving(false)
  1176→    }
  1177→  }
  1178→

  1180→    <Dialog open={open} onOpenChange={onOpenChange}>
  1181→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
  1182→        <DialogHeader>
