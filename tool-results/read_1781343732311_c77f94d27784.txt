    80→const departmentColors: Record<string, { bg: string; text: string; icon: string }> = {

    87→  HR: { bg: 'bg-macos-yellow/10', text: 'text-macos-yellow', icon: 'bg-macos-yellow/10' },

    90→
    91→function getDeptBadge(dept: string) {
    92→  const colors = departmentColors[dept]
    93→  if (colors) return `${colors.bg} ${colors.text}`
    94→  return 'bg-secondary text-secondary-foreground'
    95→}
    96→
    97→function getDeptInitialsColor(dept: string) {
    98→  const colors = departmentColors[dept]
    99→  if (colors) return colors
   100→  return { bg: 'bg-secondary', text: 'text-muted-foreground', icon: 'bg-secondary' }
   101→}

   373→                          </div>
   374→                        </TableCell>
   375→                        <TableCell>
   376→                          <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
   377→                            <Button

   386→                              variant="ghost"
   387→                              size="sm"

   425→// ==================== ADD EMPLOYEE DIALOG ====================

   427→function AddEmployeeDialog({
   428→  open,
   429→  onOpenChange,
   430→  branches,
   431→  onSuccess,
   432→}: {
   433→  open: boolean
   434→  onOpenChange: (open: boolean) => void
   435→  branches: Branch[]
   436→  onSuccess: () => void
   437→}) {
   438→  const [saving, setSaving] = React.useState(false)
   439→  const [form, setForm] = React.useState({
   440→    name: '',
   441→    email: '',
   442→    phone: '',
   443→    department: '',
   444→    position: '',
   445→    salary: '',
   446→    hireDate: '',
   447→    branchId: '',
   448→  })
   449→
   450→  const handleSubmit = async (e: React.FormEvent) => {
   451→    e.preventDefault()
   452→    if (!form.name.trim() || !form.email.trim()) return
   453→    setSaving(true)
   454→    try {
   455→      const res = await fetch('/api/employees', {
   456→        method: 'POST',
   457→        headers: { 'Content-Type': 'application/json' },
   458→        body: JSON.stringify({
   459→          ...form,
   460→          salary: parseFloat(form.salary) || 0,
   461→          hireDate: form.hireDate || null,
   462→          branchId: form.branchId || null,
   463→        }),
   464→      })
   465→      if (res.ok) {
   466→        setForm({ name: '', email: '', phone: '', department: '', position: '', salary: '', hireDate: '', branchId: '' })
   467→        onOpenChange(false)
   468→        onSuccess()
   469→      } else {
   470→        const json = await res.json()
   471→        alert(json.error || 'Failed to create employee')
   472→      }
   473→    } catch (err) {
   474→      console.error('Failed to create employee:', err)
   475→      alert('Failed to create employee')
   476→    } finally {
   477→      setSaving(false)
   478→    }
   479→  }
   480→
   481→  return (
   482→    <Dialog open={open} onOpenChange={onOpenChange}>
   483→      <DialogContent className="sm:max-w-md rounded-2xl">
   484→        <DialogHeader>
   485→          <DialogTitle className="text-[13px] font-semibold tracking-tight">Add Employee</DialogTitle>
   486→          <DialogDescription className="text-[11px]">Register a new team member</DialogDescription>
   487→        </DialogHeader>
   488→        <form onSubmit={handleSubmit} className="space-y-4">
   489→          <div className="space-y-2">
   490→            <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Full Name *</Label>
   491→            <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="John Smith" className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" required />
   492→          </div>
   493→          <div className="grid grid-cols-2 gap-3">
   494→            <div className="space-y-2">
   495→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Email *</Label>
   496→              <Input type="email" value={form.email} onChange={(e) => setForm({ ...form, email: e.target.value })} placeholder="john@bookflow.com" className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" required />
   497→            </div>
   498→            <div className="space-y-2">
   499→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Phone</Label>
   500→              <Input value={form.phone} onChange={(e) => setForm({ ...form, phone: e.target.value })} placeholder="+1-555-0100" className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" />
   501→            </div>
   502→          </div>
   503→          <div className="grid grid-cols-2 gap-3">

   566→function EditEmployeeDialog({
   567→  employee,
   568→  onOpenChange,
   569→  branches,
   570→  onSuccess,
   571→}: {
   572→  employee: Employee | null
   573→  onOpenChange: (open: boolean) => void
   574→  branches: Branch[]
   575→  onSuccess: () => void
   576→}) {
   577→  const [saving, setSaving] = React.useState(false)
   578→  const [form, setForm] = React.useState({
   579→    name: '',
   580→    email: '',
   581→    phone: '',
   582→    department: '',
   583→    position: '',
   584→    salary: '',
   585→    hireDate: '',
   586→    branchId: '',
   587→    isActive: true,
   588→  })
   589→
   590→  React.useEffect(() => {
   591→    if (employee) {
   592→      setForm({
   593→        name: employee.name || '',
   594→        email: employee.email || '',
   595→        phone: employee.phone || '',
   596→        department: employee.department || '',
   597→        position: employee.position || '',
   598→        salary: employee.salary?.toString() || '',
   599→        hireDate: employee.hireDate ? new Date(employee.hireDate).toISOString().split('T')[0] : '',
   600→        branchId: employee.branchId || '',
   601→        isActive: employee.isActive,
   602→      })
   603→    }
   604→  }, [employee])
   605→
   606→  const handleSubmit = async (e: React.FormEvent) => {
   607→    e.preventDefault()
   608→    if (!employee) return
   609→    setSaving(true)
   610→    try {
   611→      const res = await fetch('/api/employees', {
   612→        method: 'PUT',
   613→        headers: { 'Content-Type': 'application/json' },
   614→        body: JSON.stringify({
   615→          id: employee.id,
   616→          name: form.name,
   617→          email: form.email,
   618→          phone: form.phone,
   619→          department: form.department,
   620→          position: form.position,
   621→          salary: parseFloat(form.salary) || 0,
   622→          hireDate: form.hireDate || null,
   623→          branchId: form.branchId || null,
   624→          isActive: form.isActive,
   625→        }),
   626→      })
   627→      if (res.ok) {
   628→        onOpenChange(false)
   629→        onSuccess()
   630→      } else {
   631→        const json = await res.json()
   632→        alert(json.error || 'Failed to update employee')
   633→      }
   634→    } catch (err) {
   635→      console.error('Failed to update employee:', err)
   636→      alert('Failed to update employee')
   637→    } finally {
   638→      setSaving(false)
   639→    }
   640→  }
   641→
   642→  if (!employee) return null
   643→
   644→  return (
   645→    <Dialog open={!!employee} onOpenChange={onOpenChange}>
   646→      <DialogContent className="sm:max-w-md rounded-2xl">
   647→        <DialogHeader>
   648→          <DialogTitle className="text-[13px] font-semibold tracking-tight">Edit Employee</DialogTitle>
   649→          <DialogDescription className="text-[11px]">Update employee information</DialogDescription>
   650→        </DialogHeader>
   651→        <form onSubmit={handleSubmit} className="space-y-4">
   652→          <div className="space-y-2">
   653→            <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Full Name *</Label>
   654→            <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" required />
   655→          </div>
   656→          <div className="grid grid-cols-2 gap-3">
   657→            <div className="space-y-2">
   658→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Email *</Label>
   659→              <Input type="email" value={form.email} onChange={(e) => setForm({ ...form, email: e.target.value })} className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" required />
   660→            </div>
   661→            <div className="space-y-2">
   662→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Phone</Label>
   663→              <Input value={form.phone} onChange={(e) => setForm({ ...form, phone: e.target.value })} className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8" />
   664→            </div>
   665→          </div>
   666→          <div className="grid grid-cols-2 gap-3">
   667→            <div className="space-y-2">
   668→              <Label className="text-[11px] font-semibold tracking-wide text-muted-foreground">Department</Label>
   669→              <Select value={form.department} onValueChange={(v) => setForm({ ...form, department: v })}>
   670→                <SelectTrigger className="rounded-lg bg-secondary/80 border-0 text-[13px] h-8">
   671→                  <SelectValue placeholder="Select..." />
   672→                </SelectTrigger>
   673→                <SelectContent>
   674→                  <SelectItem value="Sales">Sales</SelectItem>
   675→                  <SelectItem value="Management">Management</SelectItem>
   676→                  <SelectItem value="Finance">Finance</SelectItem>
